// 96-well PCR tube rack (0.2 mL, microplate footprint)
// Units: mm. Print orientation: base on XY plane, building up +Z.

// ---------- Block ----------
plate_len   = 127.76;   // block length along X
plate_wid   = 85.48;    // block width along Y
block_h     = 20;       // block height along Z

// ---------- Holes ----------
hole_count  = 96;       // total number of tube holes
hole_d      = 6.2;      // hole diameter
hole_rows   = 8;        // number of rows along Y
hole_cols   = 12;       // number of columns along X
hole_pitch  = 9.0;      // centre-to-centre spacing (X and Y)

// ---------- Position ----------
x0          = 14.38;    // first hole (A1) distance from left edge
y0          = 11.24;    // first hole (A1) distance from back edge

// ---------- Chamfer ----------
chamf_len   = 5;        // leg length of 45-degree chamfer at A1 corner

// ---------- Misc ----------
eps         = 0.5;      // overshoot for clean boolean cuts
$fn         = 96;       // curve resolution

difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-plate_len/2, -plate_wid/2, 0])
        cube([plate_len, plate_wid, block_h]);

    // Hole grid (A1 = row 0, col 0)
    for (i = [0 : hole_rows - 1])
        for (j = [0 : hole_cols - 1])
            translate([j * hole_pitch - x0, i * hole_pitch - y0, -eps])
                cylinder(d = hole_d, h = block_h + 2 * eps);

    // Orientation chamfer: 45-degree slope from top edge down to the A1 corner
    translate([0, plate_wid/2 - chamf_len, block_h - chamf_len])
        rotate([90, 0, 0])
            linear_extrude(height = chamf_len)
                polygon([[-chamf_len, -chamf_len], [chamf_len, -chamf_len],
                         [chamf_len, 0], [-chamf_len, 0]]);
}
