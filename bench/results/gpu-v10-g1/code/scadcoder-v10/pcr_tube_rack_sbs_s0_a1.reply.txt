// 96-well PCR tube rack (0.2 mL) - microplate footprint
// Units: mm. Print orientation: base on XY plane, building up +Z.

// ---------- Parameters ----------
plate_length    = 127.76;   // block length along X
plate_width     = 85.48;    // block width along Y
plate_height    = 20;       // block height along Z

hole_count      = 96;       // total number of tube holes
hole_diameter   = 6.2;      // hole diameter
hole_pitch      = 9.0;      // centre-to-centre spacing (X and Y)
hole_rows       = 8;        // number of rows along Y
hole_columns    = 12;       // number of columns along X

corner_x        = 14.38;    // distance of A1 from left edge
corner_y        = 11.24;    // distance of A1 from back edge
chamfer_leg     = 5;        // 45-degree chamfer leg length at A1

eps             = 0.5;      // overshoot for clean boolean cuts
$fn             = 96;       // curve resolution

// ---------- Derived ----------
first_x = corner_x - (hole_columns - 1) * hole_pitch / 2;
first_y = corner_y - (hole_rows - 1) * hole_pitch / 2;

// ---------- Model ----------
difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-plate_length / 2, -plate_width / 2, 0])
        cube([plate_length, plate_width, plate_height]);

    // Tube holes (through)
    for (i = [0 : hole_columns - 1])
        for (j = [0 : hole_rows - 1])
            translate([first_x + i * hole_pitch,
                       first_y + j * hole_pitch,
                       -eps])
                cylinder(d = hole_diameter, h = plate_height + 2 * eps);

    // Orientation chamfer: 45-degree cut off the vertical edge at A1
    translate([corner_x, corner_y, 0])
        rotate([90, 0, 0])
            linear_extrude(height = plate_width, center = true)
                polygon([
                    [0, 0],
                    [plate_height - corner_y + eps, eps],
                    [plate_height - corner_y + eps, plate_height - corner_x + eps]
                ]);
}
