// 10-well agarose gel comb, printed lying flat
// Units: mm. Comb plate thickness (Z) = 1.5 mm.

// ---- Spine (back bar) ----
spine_length = 70;    // X length of spine
spine_width  = 12;    // Y width of spine

// ---- Teeth ----
teeth_count   = 10;   // number of wells/teeth
teeth_width   = 5.0;  // tooth width along X
teeth_length  = 12;   // tooth length along Y (hanging from spine)
teeth_gap     = 1.5;  // gap between neighbouring teeth along X

// ---- Derived ----
plate_thick   = 1.5;  // plate thickness (Z)
row_start_x   = -(teeth_count * teeth_width + (teeth_count - 1) * teeth_gap) / 2;
overcut       = 0.5;  // cutter extension past faces

$fn = 64;

union() {
    // Spine
    translate([-spine_length / 2, 0, 0])
        cube([spine_length, spine_width, plate_thick]);

    // Teeth, centred on the spine in X, hanging in -Y
    for (i = [0 : teeth_count - 1])
        translate([row_start_x + i * (teeth_width + teeth_gap),
                   teeth_length - 0.5,
                   0])
            cube([teeth_width, teeth_length + 0.5, plate_thick]);
}
