```openscad
// 10-well agarose gel comb - printed lying flat
// Units: mm

plate_t     = 1.5;   // plate thickness (Z)
spine_len   = 70;    // spine length (X)
spine_w     = 12;    // spine width (Y)
tooth_w     = 5.0;   // tooth width (X)
tooth_l     = 12;    // tooth length (Y)
tooth_n     = 10;    // number of teeth
gap         = 1.5;   // gap between teeth (X)
overcut     = 0.5;   // cutter overshoot past faces

$fn = 64;

pitch = tooth_w + gap;          // centre-to-centre spacing
x0    = -(tooth_n - 1) * pitch / 2;

union() {
    // Spine
    translate([-spine_len / 2, 0, 0])
        cube([spine_len, spine_w, plate_t]);

    // Teeth, centred on spine in X
    for (i = [0 : tooth_n - 1])
        translate([x0 + i * pitch - tooth_w / 2,
                   -tooth_l,
                   0])
            cube([tooth_w, tooth_l + tooth_w / 2, plate_t]);
}

difference() {
    // Combine spine and teeth
    union() {
        translate([-spine_len / 2, 0, 0])
            cube([spine_len, spine_w, plate_t]);

        for (i = [0 : tooth_n - 1])
            translate([x0 + i * pitch - tooth_w / 2,
                       -tooth_l,
                       0])
                cube([tooth_w, tooth_l + tooth_w / 2, plate_t]);
    }

    // Open the well gaps from above
    for (i = [1 : tooth_n - 1])
        translate([x0 + i * pitch - tooth_w / 2 - overcut,
                   -tooth_l - overcut,
                   -overcut])
            cube([tooth_w + 2 * overcut,
                  tooth_l + tooth_w / 2 + 2 * overcut,
                  plate_t + 2 * overcut]);
}
```
