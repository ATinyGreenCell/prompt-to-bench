```openscad
// 10-well agarose gel comb, printed lying flat
// Units: mm. Print orientation: plate flat on bed, teeth pointing down (-Z).

// ---------------- Parameters ----------------
plate_t      = 1.5;    // plate thickness (Z)
spine_len    = 70;     // spine length (X)
spine_w      = 12;     // spine width (Y)
teeth_count  = 10;     // number of wells
teeth_w      = 5.0;    // tooth width (X)
teeth_l      = 12;     // tooth length (Y)
gap          = 1.5;    // gap between neighbouring teeth (X)
overcut      = 0.5;    // extra length so cutters pass through faces

$fn = 64;

// Derived
pitch   = teeth_w + gap;              // centre-to-centre spacing
x0      = -(teeth_count - 1) * pitch / 2; // first tooth centre (row centred)

// ---------------- Model ----------------
difference() {
    // Spine (back bar)
    translate([-spine_len / 2, -spine_w / 2, 0])
        cube([spine_len, spine_w, plate_t]);

    // Teeth cut out from the top face
    for (i = [0 : teeth_count - 1])
        translate([x0 + i * pitch - teeth_w / 2,
                   -teeth_l / 2,
                   plate_t - overcut])
            cube([teeth_w, teeth_l, plate_t + overcut]);
}
```
