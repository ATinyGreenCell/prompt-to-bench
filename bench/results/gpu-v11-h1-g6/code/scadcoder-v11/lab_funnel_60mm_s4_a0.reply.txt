```openscad
// Lab funnel - printed upright with wide opening at the top
// Units: mm

// ---- Parameters ----
spout_od        = 10;    // spout outer diameter
spout_h         = 20;    // spout height (z = 0 to 20)
cone_od_top     = 60;    // cone outer diameter at rim
cone_h          = 30;    // cone height (z = 20 to 50)
wall            = 1.6;   // wall thickness everywhere
eps             = 0.5;   // overlap for clean boolean cuts
$fn             = 128;   // curve resolution

// Derived
spout_id = spout_od - 2 * wall;   // 6.8 mm bore
cone_od_bottom = spout_od;       // cone starts at spout diameter

// ---- Model ----
difference() {
    union() {
        // Spout
        cylinder(d = spout_od, h = spout_h);

        // Cone (expanded from bottom diameter at z = spout_h)
        translate([0, 0, spout_h - eps])
            cylinder(d1 = cone_od_bottom, d2 = cone_od_top, h = cone_h + eps);
    }

    // Bottom cut (open bottom)
    translate([0, 0, -eps])
        cylinder(d = spout_id, h = spout_h + 2 * eps);
}
```
