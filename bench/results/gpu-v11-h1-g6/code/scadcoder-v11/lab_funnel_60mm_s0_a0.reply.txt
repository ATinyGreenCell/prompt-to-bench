```openscad
// Lab funnel - printed upright, wide opening at the top
// Straight spout from z = 0 to z = 20, conical body from z = 20 to z = 50

// ---- Dimensions (mm) ----
spout_od      = 10;    // outer diameter of spout
spout_h       = 20;    // spout height (z = 0 to z = 20)
cone_od_top   = 60;    // outer diameter of cone at rim (z = 50)
cone_od_base  = 10;    // outer diameter of cone at base (z = 20)
cone_h        = 30;    // cone height (z = 20 to z = 50)
wall          = 1.6;   // wall thickness everywhere
eps           = 0.5;   // overlap for clean boolean cuts

$fn = 128;

// Derived
bore = spout_od - 2 * wall;   // 6.8 mm bore in spout

difference() {
    union() {
        // Spout
        cylinder(d = spout_od, h = spout_h);

        // Cone, slightly overlapping the spout for a solid union
        translate([0, 0, spout_h - eps])
            cylinder(d1 = cone_od_base, d2 = cone_od_top, h = cone_h + eps);
    }

    // Bottom cut - open bottom at z = 0
    translate([0, 0, -eps])
        cylinder(d = spout_od, h = eps + 1);

    // Axial bore through the spout (extends into cone to keep a clear channel)
    translate([0, 0, -eps])
        cylinder(d = bore, h = spout_h + eps + 10);
}
```
