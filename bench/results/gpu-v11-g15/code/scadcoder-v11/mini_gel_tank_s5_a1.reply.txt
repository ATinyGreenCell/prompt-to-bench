```openscad
// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: base on XY plane, open top facing +Z

// ---- Tank dimensions ----
outer_x = 120;   // outer length in X
outer_y = 70;    // outer width in Y
height  = 40;    // overall height in Z
wall    = 3;     // wall thickness
floor   = 3;     // floor thickness (bottom solid)

// ---- Platform dimensions ----
plat_len   = 60; // platform length in X (centred)
plat_w     = outer_y - 2*wall; // platform spans full inner width
plat_top_z = 13; // top of platform (inner floor + 10 mm)

// ---- Electrode holes ----
hole_d     = 2;  // hole diameter
hole_y     = 0;  // hole Y position
hole_z     = 30; // hole Z position
eps        = 0.5; // cutter overshoot

$fn = 64;

difference() {
    // Tank body
    union() {
        // Floor
        translate([-outer_x/2, -outer_y/2, 0])
            cube([outer_x, outer_y, floor]);
        // Walls (empty interior)
        translate([-outer_x/2, -outer_y/2, floor])
            cube([outer_x, outer_y, height - floor]);
    }

    // Raised platform
    translate([-plat_len/2, -plat_w/2, floor])
        cube([plat_len, plat_w, plat_top_z - floor + 1]); // overlaps floor for a clean union

    // Electrode holes through end walls
    for (sx = [-1, 1])
        translate([sx * (outer_x/2 - eps), hole_y, hole_z])
            rotate([90, 0, 0])
                cylinder(d = hole_d, h = outer_y + 2*eps);
}
```
