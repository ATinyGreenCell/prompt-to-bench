```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate on bed, motor plate building up in +Z

$fn = 64;

// ---- Base plate ----
base_x = 50;            // X span of base plate (-25 to 25)
base_y = 45;            // Y span of base plate (0 to 45)
base_t = 5;             // base plate thickness (Z 0 to 5)

// ---- Motor plate ----
motor_x = 50;           // X span of motor plate (-25 to 25)
motor_y = 5;            // motor plate thickness (Z 0 to 5)
motor_h = 50;           // total height of motor plate (Z 0 to 50)

// ---- Boss hole ----
boss_d = 23;            // motor centring boss hole diameter
boss_x = 0;             // boss hole centre X
boss_z = 28;            // boss hole centre Z

// ---- Screw holes ----
screw_d = 3.4;          // M3 screw hole diameter
screw_p = 31;           // square pattern pitch (31 mm x 31 mm)

// ---- Gussets ----
gusset_t = 5;           // gusset thickness (X)
gusset_l = 20;          // leg length along Y and Z
gusset_x = [-25, 20];   // gusset positions (left and right edges)

// ---- Misc ----
eps = 0.5;              // cutter overshoot past faces

difference() {
    union() {
        // Base plate
        translate([-base_x/2, 0, 0])
            cube([base_x, base_y, base_t]);

        // Motor plate
        translate([-motor_x/2, 0, 0])
            cube([motor_x, motor_y, motor_h]);

        // Triangular gussets, flush with plate edges
        for (x = gusset_x)
            translate([x, 0, 0])
                rotate([90, 0, 0])
                    linear_extrude(height = gusset_t)
                        polygon([[0, 5], [0, l] , [l, 5]]);
    }

    // Boss hole for motor centring boss
    translate([boss_x, 0, boss_z])
        cylinder(d = boss_d, h = motor_y + eps, $fn = 96);

    // Four M3 screw holes on a 31 mm square pattern
    for (a = [0, 90, 180, 270])
        rotate([0, 0, a])
            translate([screw_p/2, 0, -eps])
                cylinder(d = screw_d, h = motor_h + 2*eps, $fn = 64);
}
```
