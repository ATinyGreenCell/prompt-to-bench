```openscad
// 96-well PCR tube rack (0.2 mL, standard microplate footprint)
// Print orientation: base on XY plane, walls build upward (+Z)

// ---------------- Parameters ----------------
block_len    = 127.76;   // block length along X (mm)
block_wid    = 85.48;    // block width along Y (mm)
block_ht     = 20;       // block height along Z (mm)

hole_d       = 6.2;      // through-hole diameter (mm)
hole_count_r = 8;        // number of rows along Y
hole_count_c = 12;       // number of columns along X
hole_pitch   = 9.0;      // centre-to-centre spacing, X and Y (mm)

a1_x         = 14.38;    // X of first hole (A1) from left edge (mm)
a1_y         = 11.24;    // Y of first hole (A1) from back edge (mm)

chamf_leg    = 5;        // 45-degree chamfer leg length (mm)
eps          = 0.5;      // cutter overshoot past faces (mm)

$fn = 96;

// ---------------- Model ----------------
difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_ht]);

    // 96 through-holes, 8 rows x 12 columns
    for (i = [0 : hole_count_r - 1])
        for (j = [0 : hole_count_c - 1])
            translate([a1_x + j * hole_pitch,
                       a1_y + i * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = block_ht + 2 * eps);

    // 45-degree chamfer at the A1 corner (vertical edge), orientation mark
    translate([-block_len/2 + a1_x,
               -block_wid/2 + a1_y,
               0])
        linear_extrude([block_ht])
            polygon([[-chamf_leg, -chamf_leg],
                     [-chamf_leg, chamf_leg + 0.01],
                     [chamf_leg, chamf_leg + 0.01],
                     [chamf_leg, -chamf_leg]]);
}
```
