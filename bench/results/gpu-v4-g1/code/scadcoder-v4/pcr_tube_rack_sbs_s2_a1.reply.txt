```openscad
// 96-well PCR tube rack (0.2 mL, standard microplate footprint)
// Units: mm. Print orientation: base on XY plane, builds +Z.

// ---------------- Parameters ----------------
block_len    = 127.76;   // block length along X
block_wid    = 85.48;    // block width along Y
block_ht     = 20;       // block height along Z

hole_d       = 6.2;      // through-hole diameter
hole_count_r = 8;        // number of rows along Y
hole_count_c = 12;       // number of columns along X
hole_pitch   = 9.0;      // centre-to-centre spacing (X and Y)

a1_x         = 14.38;    // X of first hole (A1) from left edge
a1_y         = 11.24;    // Y of first hole (A1) from back edge

chamf        = 45;       // chamfer angle (degrees) off A1 vertical edge
chamf_leg    = 5;        // chamfer leg length (X and Y)

eps          = 0.5;      // cutter overshoot past faces
$fn          = 96;       // curve resolution

// ---------------- Derived ----------------
x0 = a1_x - (hole_count_c - 1) * hole_pitch / 2;  // first hole X (grid centred)
y0 = a1_y - (hole_count_r - 1) * hole_pitch / 2;  // first hole Y (grid centred)

// ---------------- Model ----------------
difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_len / 2, -block_wid / 2, 0])
        cube([block_len, block_wid, block_ht]);

    // 96 vertical through-holes
    for (i = [0 : hole_count_c - 1])
        for (j = [0 : hole_count_r - 1])
            translate([x0 + i * hole_pitch, y0 + j * hole_pitch, -eps])
                cylinder(d = hole_d, h = block_ht + 2 * eps);

    // Orientation mark: 45-degree chamfer off the A1 corner vertical edge
    translate([x0 - chamf_leg, y0 - chamf_leg, 0])
        linear_extrude(height = block_ht + 1)
            polygon([[-chamf_leg, -chamf_leg],
                     [-chamf_leg,  a1_y - y0 + chamf_leg],
                     [ a1_x - x0 + chamf_leg,  a1_y - y0 + chamf_leg],
                     [ a1_x - x0 + chamf_leg, -chamf_leg]]);
}
```
