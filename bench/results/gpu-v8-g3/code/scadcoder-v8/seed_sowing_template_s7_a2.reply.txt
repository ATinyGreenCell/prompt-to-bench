// Petri dish sowing template - flat disc with a 7 x 7 hole grid and a rim notch for orientation
// Print orientation: flat on the bed, z = 0 up to disc thickness

// ---------------- Parameters ----------------
disc_d        = 85;    // disc outer diameter (mm)
disc_h        = 2;     // disc thickness (mm)

hole_count_x  = 7;     // number of holes along X
hole_count_y  = 7;     // number of holes along Y
hole_d        = 3;     // through-hole diameter (mm)
hole_pitch    = 9;     // centre-to-centre spacing (mm)

notch_w       = 10;    // orientation notch width along X (mm)
notch_depth   = 4;     // orientation notch depth along Y, from rim (mm)

eps           = 0.5;   // cutter overshoot past faces (mm)
$fn           = 128;   // curve resolution

// ---------------- Derived ----------------
grid_centre = (hole_count_x - 1) * hole_pitch / 2;   // 36 mm offset of first row/col from centre

// ---------------- Model ----------------
difference() {
    // Main disc
    cylinder(d = disc_d, h = disc_h);

    // Hole grid
    for (i = [0 : hole_count_x - 1])
        for (j = [0 : hole_count_y - 1])
            translate([i * hole_pitch - grid_centre, j * hole_pitch - grid_centre, -eps])
                cylinder(d = hole_d, h = disc_h + 2 * eps);

    // Rectangular orientation notch on the +Y side, centred on the Y axis
    translate([0, disc_d / 2 - notch_depth, -eps])
        cube([notch_w, notch_depth + eps, disc_h + eps]);
}
