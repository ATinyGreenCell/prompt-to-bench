// Petri dish sowing template - flat disc with 7x7 hole grid and rim notch
// Units: mm. Print orientation: flat on bed, z = 0 up to disc thickness.

// ---------- Parameters ----------
disc_d        = 90;    // disc outer diameter
disc_h        = 2;     // disc thickness (Z)
hole_d        = 3;     // through-hole diameter
hole_pitch    = 9;     // centre-to-centre spacing (X and Y)
hole_nx       = 7;     // holes along X
hole_ny       = 7;     // holes along Y
notch_w       = 10;    // orientation notch width (X)
notch_depth   = 4;     // orientation notch depth from rim (Y, +Y side)
eps           = 0.5;   // cutter overshoot past faces
$fn           = 128;   // circle resolution

// ---------- Derived ----------
grid_x0 = -(hole_nx - 1) * hole_pitch / 2;  // first hole X (-27)
grid_y0 = -(hole_ny - 1) * hole_pitch / 2;  // first hole Y (-27)
notch_cy = disc_d / 2 - notch_depth;        // notch centre Y (38)

// ---------- Model ----------
difference() {
    // Main disc
    cylinder(d = disc_d, h = disc_h);

    // 7 x 7 through-hole grid
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([grid_x0 + i * hole_pitch,
                       grid_y0 + j * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = disc_h + 2 * eps);

    // Rectangular orientation notch into rim on +Y side
    translate([0, notch_cy - notch_depth, 0])
        cube([notch_w, notch_depth, disc_h + eps]);
}
