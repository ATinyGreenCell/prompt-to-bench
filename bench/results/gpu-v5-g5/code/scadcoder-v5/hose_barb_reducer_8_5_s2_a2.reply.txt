// Hose-barb reducer: joins 8 mm ID silicone tubing to 5 mm ID silicone tubing
// Print orientation: standing upright along +Z, resting on z = 0

// ---------------- Parameters ----------------
// Tubing inner diameters
tube1_id   = 8;    // larger tubing (mm)
tube2_id   = 5;    // smaller tubing (mm)

// Bore that runs straight through the full part length
bore_d     = 3;    // through-bore diameter (mm)

// Barb 1 (8 mm tubing side): z = 0 to z = 20
barb1_h    = 10;   // height of each barb (mm)
barb1_d0   = 7.5;  // outer diameter at bottom of barb (mm)
barb1_d1   = 9.5;  // outer diameter at top of barb (mm)
barb1_n    = 2;    // number of stacked barbs

// Collar: z = 20 to z = 25
collar_d   = 12;   // collar outer diameter (mm)

// Barb 2 (5 mm tubing side): z = 25 to z = 41
barb2_h    = 8;    // height of each barb (mm)
barb2_d0   = 6.5;  // outer diameter at bottom of barb (mm)
barb2_d1   = 5.0;  // outer diameter at top of barb (mm)
barb2_n    = 2;    // number of stacked barbs

// ---------------- Helpers ----------------
// Sawtooth profile for a hose barb: diameter grows linearly from d0 to d1
// over height h, then steps straight back to d0 at the top.
// The step is extended by eps so no faces are coplanar.
module hose_barb(d0, d1, h, eps = 0.5) {
    $fn = 64;
    s = (d1 - d0) / h;                     // slope of the growth
    difference() {
        union() {
            // Growing cone (cylinder with a tiny radius overlap at the seam)
            cylinder(h = h + 0.01, d0 = d0, d1 = d1 + 0.01);
            // Step back from the top rim to the original diameter
            translate([0, 0, h - eps])
                cylinder(h = eps + 0.01, d1 = d0, d2 = d1);
        }
        // Through-bore, extended past both end faces
        translate([0, 0, -eps])
            cylinder(h = h + 2 * eps, d = bore_d);
    }
}

// ---------------- Model ----------------
union() {
    // Barb 1 stack: z = 0 to z = 20
    for (i = [0 : barb1_n - 1])
        translate([0, 0, i * barb1_h])
            hose_barb(barb1_d0, barb1_d1, barb1_h);

    // Collar: z = 20 to z = 25
    translate([0, 0, 20])
        cylinder(h = 5, d = collar_d);

    // Barb 2 stack: z = 25 to z = 41
    for (i = [0 : barb2_n - 1])
        translate([0, 0, 25 + i * barb2_h])
            hose_barb(barb2_d0, barb2_d1, barb2_h);
}
