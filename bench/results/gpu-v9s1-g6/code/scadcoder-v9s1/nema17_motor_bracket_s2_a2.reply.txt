```openscad
// L-shaped mounting bracket for a NEMA 17 stepper motor
// Print orientation: base plate on the bed (z = 0), motor plate builds upward (+Z)

// ---------------- Parameters ----------------
$fn = 96;                 // smoothness for curved shapes

// ---- Base plate ----
base_x0 = -25;            // base plate X start
base_x1 = 25;             // base plate X end
base_y0 = 0;              // base plate Y start
base_y1 = 45;             // base plate Y end
base_z0 = 0;              // base plate Z start
base_z1 = 5;              // base plate Z end (thickness)

// ---- Vertical motor plate ----
motor_x0 = -25;           // motor plate X start
motor_x1 = 25;            // motor plate X end
motor_y0 = 0;             // motor plate Y start
motor_y1 = 5;             // motor plate Y end (thickness)
motor_z0 = 0;             // motor plate Z start
motor_z1 = 50;            // motor plate Z end (height)

// ---- Base plate through-holes (M4) ----
mount_hole_d = 4.5;       // hole diameter
mount_hole_x = [-17, 17]; // hole X positions
mount_hole_y = 30;        // hole Y position

// ---- Motor boss hole ----
boss_d = 23;              // hole diameter for motor centring boss
boss_x = 0;               // hole X centre
boss_z = 28;              // hole Z centre

// ---- Boss hole screw pattern ----
screw_d = 3.4;            // M3 screw diameter
screw_s = 31;             // square pattern side length
screw_n = 4;              // number of screws

// ---- Gussets ----
gusset_t = 5;             // gusset thickness (along X)
gusset_w = 20;            // gusset leg along +Y
gusset_h = 20;            // gusset leg along +Z
gusset_x0 = -25;          // left gusset X start
gusset_x1 = -20;          // left gusset X end
gusset_x2 = 20;           // right gusset X start
gusset_x3 = 25;           // right gusset X end

// ---- Cutting overlap ----
eps = 0.5;                // extension of cutters past faces

// ---------------- Model ----------------
body = union() {
    // Base plate
    translate([base_x0, base_y0, base_z0])
        cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);

    // Vertical motor plate
    translate([motor_x0, motor_y0, motor_z0])
        cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1 - motor_z0]);

    // Left gusset: right triangle in YZ, flush with left edge
    translate([gusset_x0, 0, 0])
        rotate([90, 0, 0])
            linear_extrude(height = gusset_t)
                polygon(points = [[0, 0], [w, 0], [0, h]]);

    // Right gusset: right triangle in YZ, flush with right edge
    translate([gusset_x2, 0, 0])
        rotate([90, 0, 0])
            linear_extrude(height = gusset_t)
                polygon(points = [[0, 0], [w, 0], [0, h]]);
}

cutters = union() {
    // Base plate through-holes
    for (x = mount_hole_x)
        translate([x, mount_hole_y, -eps])
            cylinder(d = mount_hole_d, h = base_z1 + 2 * eps);

    // Boss hole
    translate([boss_x, 0, boss_z])
        rotate([90, 0, 0])
            cylinder(d = boss_d, h = motor_y1 - motor_y0 + eps);

    // Screw holes on square pattern
    for (i = [0 : screw_n - 1]) {
        x = (i % 2 == 0) ? -screw_s / 2 : screw_s / 2;
        z = (i < 2) ? -screw_s / 2 : screw_s / 2;
        translate([x, 0, z])
            rotate([0, 90, 0])
                cylinder(d = screw_d, h = motor_z1 - motor_z0 + eps);
    }
}

difference() {
    body;
    cutters;
}
```
