```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate on bed, motor plate builds upward (+Z)

$fn = 64;

// ---- Base plate ----
base_x0 = -25; base_x1 = 25;   // X span 50
base_y0 = 0; base_y1 = 45;     // Y span 45
base_z0 = 0; base_z1 = 5;      // Z thickness 5
mount_hole_d = 4.5;            // M4 through-holes
mount_hole_x = [-17, 17];      // hole X positions
mount_hole_y = 30;             // hole Y position

// ---- Motor plate ----
motor_x0 = -25; motor_x1 = 25; // X span 50
motor_y0 = 0; motor_y1 = 5;    // Y span 5 (sits on back edge of base)
motor_z0 = 0; motor_z1 = 50;   // Z height 50

// ---- Boss hole (motor centring boss) ----
boss_d = 23;                   // hole diameter
boss_x = 0; boss_z = 28;       // hole centre

// ---- M3 screw holes on square pattern ----
screw_d = 3.4;                 // screw diameter
screw_s = 31;                  // square pattern side length
screw_h = 10;                  // depth into motor plate

// ---- Gussets ----
gusset_t = 5;                  // thickness along X
gusset_w = 20;                 // leg along +Y
gusset_h = 20;                 // leg along +Z
gusset_x0 = -25; gusset_x1 = -20; // left gusset X range
gusset_x2 = 20; gusset_x3 = 25;  // right gusset X range

// ---- Cutting overlap ----
eps = 0.5;                     // extension past cut faces

difference() {
    union() {
        // Base plate
        translate([base_x0, base_y0, base_z0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);

        // Motor plate
        translate([motor_x0, motor_y0, motor_z0])
            cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1 - motor_z0]);

        // Gussets (right-angle corner at y = 5, z = 5)
        for (sx = [gusset_x0, gusset_x2])
            translate([sx, 0, 0])
                cube([gusset_t, gusset_w, gusset_h]);
    }

    // Base plate through-holes
    for (x = mount_hole_x)
        translate([x, mount_hole_y, -eps])
            cylinder(d = mount_hole_d, h = base_z1 + 2 * eps);

    // Boss hole
    translate([boss_x, 0, boss_z])
        rotate([90, 0, 0])
            cylinder(d = boss_d, h = motor_y1 - motor_y0 + eps, $fn = 96);

    // Screw holes
    for (i = [0 : 3]) {
        a = (i < 2) ? 0 : 90;
        translate([boss_x - screw_s / 2 + (i < 2 ? 0 : screw_s),
                   boss_z - screw_s / 2 + (i >= 2 ? 0 : -screw_s)])
            rotate([0, 0, a])
                cylinder(d = screw_d, h = screw_h + eps);
    }
}
```
