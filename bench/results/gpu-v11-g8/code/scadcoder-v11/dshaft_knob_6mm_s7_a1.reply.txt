// Hotplate stirrer knob - D-shaft 6 mm
// Print orientation: bottom face (with D-bore opening) on the bed, z = 0

$fn = 96;

// ---- Dimensions (mm) ----
shaft_d      = 6;    // D-shaft diameter
knob_d       = 30;   // knob outer diameter
knob_h       = 15;   // knob height
bore_depth   = 12;   // D-blind bore depth from bottom face
bore_w       = 4.7;  // D-bore width (flat to opposite side of circle)
groove_count = 18;   // number of vertical grip grooves
groove_d     = 2;    // grip groove cylinder diameter
pointer_w    = 1.5;  // pointer groove width
pointer_d    = 1;    // pointer groove depth
eps          = 0.5;  // overcut for clean boolean differences

// Derived
bore_r = bore_w / 2 + 0.01;  // D-bore radius (flat offset by 0.01)

difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-blind bore in bottom face
    translate([0, 0, -eps])
        cylinder(r = bore_r, h = bore_depth + eps, $fn = 64);

    // 18 evenly spaced vertical grip grooves along full height
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 64);

    // Pointer groove on top face, from centre to outer edge
    translate([0, 0, knob_h - pointer_d])
        cylinder(r = knob_d / 2, h = pointer_d + eps, $fn = 64);

    // Keep pointer groove clear of any grip grooves
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, knob_h - pointer_d])
                cube([groove_d / 2 + eps, knob_d / 2 + eps, pointer_d + eps]);
}
