// Hotplate stirrer knob - replacement part for D-shaft (6 mm)
// Print orientation: bottom face (with D-bore opening) on the bed at z = 0

$fn = 96;

// ---- Dimensions (mm) ----
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
shaft_d       = 6;     // D-shaft diameter
bore_d        = 6.2;   // D-blind bore diameter
bore_depth    = 12;    // bore depth from bottom face
bore_flat     = (bore_d - 4.7) / 2;  // offset of circle centre from flat side
groove_d      = 2;     // grip groove diameter (half-cylinder)
groove_count  = 18;    // number of evenly spaced grooves
pointer_w     = 1.5;   // pointer groove width
pointer_depth = 1;     // pointer groove depth
eps           = 0.5;   // overcut for clean differences

difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore in bottom face
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = bore_depth + eps);
    translate([0, bore_flat, -eps])
        rotate([90, 0, 0])
            cylinder(d = bore_d, h = bore_depth + eps);

    // Vertical grip grooves along full height
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 64);

    // Pointer groove from centre to outer edge (top face)
    translate([0, 0, knob_h - pointer_depth])
        rotate([90, 0, 0])
            cylinder(d = pointer_w, h = pointer_depth + eps, $fn = 64);
}
