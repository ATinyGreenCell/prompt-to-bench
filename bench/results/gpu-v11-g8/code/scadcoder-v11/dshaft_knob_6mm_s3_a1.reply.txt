```openscad
// Hotplate stirrer knob - D-shaft adapter
// Print orientation: flat bottom face on bed (z = 0)

$fn = 96;

// ---- Dimensions (mm) ----
shaft_d      = 6;     // D-shaft diameter
knob_d       = 30;    // knob outer diameter
knob_h       = 15;    // knob height
bore_depth   = 12;    // blind bore depth from bottom face
bore_w       = 4.7;   // bore width (flat to opposite side)
groove_d     = 2;     // grip groove diameter
groove_n     = 18;    // grip groove count
ptr_w        = 1.5;   // pointer groove width
ptr_d        = 1;     // pointer groove depth
eps          = 0.5;   // cutter overshoot

// Derived
bore_r = bore_d(bore_w, sqrt(bore_w^2 + 4*eps^2))/2; // D-bore radius (flat on -y)
gd = groove_d/2;

difference() {
    // Body
    cylinder(d = knob_d, h = knob_h);

    // Shaft bore
    translate([0, 0, -eps])
        cylinder(d = shaft_d, h = knob_h + 2*eps);

    // D-blind bore
    translate([0, 0, -eps])
        rotate([90, 0, 0])
            cylinder(r = bore_r, h = bore_depth + eps);

    // Grip grooves
    for (i = [0 : groove_n - 1])
        rotate([0, 0, i*360/groove_n])
            translate([knob_d/2, 0, -eps])
                cylinder(r = gd, h = knob_h + 2*eps);

    // Pointer groove (top face, -y side)
    translate([0, -knob_d/2, knob_h - ptr_d])
        cube([ptr_w, knob_d/2 + eps, ptr_d + eps]);
}
```
