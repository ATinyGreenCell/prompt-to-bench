// Hotplate stirrer knob for a 6 mm D-shaft
// Print orientation: flat bottom face on the bed (z = 0)

// ---------- Dimensions (mm) ----------
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
bore_d        = 6.2;   // D-bore diameter
bore_depth    = 12;    // D-bore depth from bottom face
bore_flat     = (bore_d - 4.7) / 2;  // offset of D-flat from bore centre (4.7 mm wide flat)
groove_d      = 2;     // grip groove diameter (half-cylinder)
groove_count  = 18;    // number of grip grooves
pointer_w     = 1.5;   // pointer groove width
pointer_d     = 1;     // pointer groove depth
eps           = 0.5;   // overshoot for clean boolean cuts

$fn = 96;

// ---------- Model ----------
difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = bore_depth + eps);
    translate([0, -bore_flat, -eps])
        rotate([90, 0, 0])
            cylinder(d = bore_flat, h = bore_depth + eps);

    // Vertical grip grooves around the outside
    for (i = [0 : groove_count - 1]) {
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps);
    }

    // Pointer groove on top face, same side as the D-flat
    translate([0, -bore_flat, knob_h - pointer_d])
        cube([knob_d / 2, pointer_w, pointer_d + eps]);
}
