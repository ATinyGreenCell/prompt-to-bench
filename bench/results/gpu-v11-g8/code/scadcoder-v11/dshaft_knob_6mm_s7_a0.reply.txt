// Hotplate stirrer knob - replacement part for D-shaft (6 mm)
// Print orientation: flat D-side down on the bed (z = 0)

$fn = 96;

// ---- Dimensions (mm) ----
knob_d        = 30;    // knob diameter
knob_h        = 15;    // knob height
shaft_d       = 6;     // D-shaft diameter
bore_d        = 6.2;   // D-bore circle diameter
bore_depth    = 12;    // blind bore depth from bottom face
bore_flat     = (bore_d - 4.7) / 2;  // offset of flat from circle centre (1.95)
groove_d      = 2;     // grip groove cylinder diameter
groove_count  = 18;    // number of evenly spaced grooves
groove_w      = 1.5;   // pointer groove width
groove_depth  = 1;     // pointer groove depth
eps           = 0.5;   // overcut for clean differences

difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore
    translate([0, 0, -eps])
        intersection() {
            cylinder(d = bore_d, h = bore_depth + eps);
            translate([0, -bore_flat - eps, 0])
                cube([bore_flat + eps, knob_d + 2 * eps, bore_depth + eps]);
        }

    // Vertical grip grooves around the outside
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 64);

    // Pointer groove on top face, same side as the D-flat
    translate([0, knob_d / 2 - groove_w / 2, knob_h - groove_depth])
        cube([groove_w, knob_d / 2 + eps, groove_depth + eps]);
}
