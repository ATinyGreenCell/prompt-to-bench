```openscad
// 10-well comb for casting agarose gels
// Print orientation: flat on bed, teeth pointing down (-Z)

// ---------- Parameters ----------
spine_len      = 70;    // spine length along X (mm)
spine_wid      = 12;    // spine width along Y (mm)
plate_thk      = 1.5;   // plate thickness along Z (mm)

tooth_count    = 10;    // number of teeth
tooth_w        = 5.0;   // tooth width along X (mm)
tooth_len      = 12;    // tooth length along Y (mm)
tooth_depth    = 12;    // tooth height from spine into plate (mm)
tooth_gap      = 1.5;   // gap between neighbouring teeth (mm)

eps            = 0.5;   // cutter overshoot past faces (mm)

$fn = 64;

// ---------- Derived ----------
pitch   = tooth_w + tooth_gap;            // centre-to-centre spacing
x_start = -(tooth_count - 1) * pitch / 2; // first tooth centre (row centred)

// ---------- Model ----------
union() {
    // Spine (back bar)
    translate([-spine_len / 2, -spine_wid / 2, 0])
        cube([spine_len, spine_wid, plate_thk]);

    // Teeth hanging from the -Y edge of the spine
    for (i = [0 : tooth_count - 1])
        translate([x_start + i * pitch - tooth_w / 2,
                   spine_wid / 2 - tooth_len,
                   plate_thk - tooth_depth])
            cube([tooth_w, tooth_len, tooth_depth + eps]);
}
```
