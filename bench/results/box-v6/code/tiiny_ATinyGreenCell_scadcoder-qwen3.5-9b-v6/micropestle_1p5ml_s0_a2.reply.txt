// Pestle for grinding tissue inside a 1.5 mL microcentrifuge tube
// Print orientation: standing upright, handle on the bed (z = 0)

// ---------------- Parameters ----------------
handle_d      = 8;    // handle diameter (mm)
handle_h      = 45;   // handle height, z = 0 to z = 45 (mm)
cone_top_d    = 8;    // cone diameter at z = 45 (mm)
cone_bot_d    = 3;    // cone diameter at z = 62 (mm)
cone_top_z    = 45;   // cone top height (mm)
cone_bot_z    = 62;   // cone bottom height (mm)
tip_d         = 3;    // tip sphere diameter (mm)
tip_z         = 62;   // tip sphere centre height (mm)
groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width along Z (mm)
groove_depth  = 1;    // groove depth into handle (mm)
groove_zs     = [10, 15, 20, 25]; // groove centre heights (mm)
eps           = 0.5;  // cutter overshoot past faces (mm)

$fn = 96;

// ---------------- Model ----------------
difference() {
    union() {
        // (1) Cylindrical handle
        cylinder(d = handle_d, h = handle_h);

        // (2) Tapering cone
        translate([0, 0, cone_top_z])
            cylinder(d1 = cone_top_d, d2 = cone_bot_d, h = cone_bot_z - cone_top_z);

        // (3) Rounded tip: sphere centred at z = 62
        translate([0, 0, tip_z - tip_d / 2])
            sphere(d = tip_d);
    }

    // Grip grooves, cut through the full diameter, centred at specified z
    for (gz = groove_zs)
        translate([0, 0, gz])
            rotate([0, 90, 0])
                cylinder(d = groove_w, h = handle_d + 2 * eps, center = true);
}
