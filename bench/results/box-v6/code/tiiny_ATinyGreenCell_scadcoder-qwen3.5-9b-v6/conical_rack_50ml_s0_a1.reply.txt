```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, walls up. Tubes pass through holes.
// After printing, flip the part so it stands on the two walls.

// ---------- Parameters ----------
plate_x      = 130;   // plate length (X)
plate_y      = 90;    // plate width (Y)
plate_t      = 4;     // plate thickness (Z)

hole_d       = 30.5;  // through-hole diameter
hole_nx      = 3;     // holes per row (along X)
hole_ny      = 2;     // number of rows (along Y)
hole_pitch   = 40;    // centre-to-centre spacing in X and Y

wall_t       = 4;     // wall thickness (Y)
wall_top_z   = 70;    // top of walls (Z)

eps          = 0.5;   // cutter overshoot past faces
$fn          = 96;    // circle resolution

// ---------- Model ----------
difference() {
    union() {
        // Base plate
        translate([-plate_x/2, -plate_y/2, 0])
            cube([plate_x, plate_y, plate_t]);

        // Two solid walls along the long edges
        for (sx = [-1, 1])
            translate([sx > 0 ? plate_x/2 - wall_t : -plate_x/2,
                       -plate_y/2, 0])
                cube([wall_t, plate_y, wall_top_z]);
    }

    // Six through-holes
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([(i - (hole_nx - 1)/2) * hole_pitch,
                       (j - (hole_ny - 1)/2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2*eps);
}
```
