```openscad
// PCR tube adapter sleeve - print upside down
// Units: mm

$fn = 96;

// --- Dimensions ---
collar_d      = 13;    // collar outer diameter
collar_h      = 1.5;   // collar thickness (z = 0 to 1.5)
sleeve_od     = 10.8;  // sleeve outer diameter
sleeve_top_z  = 21.5;  // top of sleeve body
bore_d        = 6.2;   // through-bore diameter
eps           = 0.5;   // cutter overshoot

// --- Model ---
difference() {
    union() {
        // Collar on the bed
        cylinder(d = collar_d, h = collar_h);
        // Sleeve body rising from the collar
        translate([0, 0, collar_h])
            cylinder(d = sleeve_od, h = sleeve_top_z - collar_h);
    }
    // Bore through full height
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = sleeve_top_z + 2 * eps);
}
```
