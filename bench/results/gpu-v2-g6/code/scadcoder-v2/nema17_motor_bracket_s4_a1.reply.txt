```openscad
// L-shaped mounting bracket for a NEMA 17 stepper motor (e.g. a syringe pump).
// Base plate with two vertical through-holes; vertical motor plate with a centring
// boss hole and a screw pattern; two triangular gussets at the edges.
//

base_x  = 25;
base_y  = 45;
base_z  = 5;
screw_h = 4.5;

motor_x = 25;
motor_y = 5;
motor_z = 50;
boss_h  = 23;
screw_p = 31;
screw_d = 3.4;

gusset_t = 5;
gusset_l = 20;
gusset_h = 20;

$fn = 48;

base_holes = [
    [-17, 30],
    [17, 30]
];

motor_hole = [0, 25, 28];
screw_grid = [
    [-15.5, -15.5],
    [-15.5, 15.5],
    [15.5, -15.5],
    [15.5, 15.5]
];

difference() {
    union() {
        translate([-base_x / 2, 0, 0]) cube([base_x, base_y, base_z]);
        translate([-motor_x / 2, 0, 0]) cube([motor_x, motor_y, motor_z]);
        for (x = [-base_x / 2, base_x / 2])
            translate([x, 0, 0]) linear_extrude(gusset_h) polygon([[0, 0], [gusset_l, 0], [0, gusset_l]]);
    }
    for (p = base_holes)
        translate(p) cylinder(h = base_z + 1, d = screw_h);
    translate(motor_hole) cylinder(h = motor_y + 1, d = boss_h);
    for (p = screw_grid)
        translate([p[0], p[1], -1]) cylinder(h = motor_y + 2, d = screw_d);
}
```
