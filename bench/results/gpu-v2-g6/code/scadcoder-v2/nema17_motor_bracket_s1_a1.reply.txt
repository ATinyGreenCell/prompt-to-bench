```openscad
// L-shaped mounting bracket for a NEMA 17 stepper motor (e.g. syringe pump).
// Base plate with two vertical through-holes; vertical motor plate with a
// motor-centring boss hole and a 3 mm screw pattern; two triangular gussets
// at the edges. NOVEL general part.
//

base_x   = 25;
base_y   = 45;
base_z   = 5;
screw_h  = 4.5;
motor_y  = 5;
motor_z  = 50;
boss_h   = 23;
screw_p  = 31;
gusset_t = 5;
gusset_l = 20;
gusset_w = 20;

$fn = 48;

base_holes = [
    [-17, 30],
    [17, 30]
];

motor_holes = [
    [0, 28],                 // boss - a hole for the motor's centring boss
    [-15.5, 15.5],
    [-15.5, -15.5],
    [15.5, 15.5],
    [15.5, -15.5]
];

difference() {
    union() {
        translate([-base_x / 2, 0, 0]) cube([base_x, base_y, base_z]);                 // base plate
        translate([-base_x / 2, 0, 0]) cube([base_x, motor_y, motor_z]);               // motor plate
        for (x = [-base_x / 2, base_x / 2])
            translate([x, 0, 0])
                linear_extrude(motor_y)
                    polygon([[-gusset_l, gusset_w], [-gusset_l, 0], [0, 0], [0, gusset_w]]);  // gussets
    }
    for (p = base_holes)
        translate(p) cylinder(h = base_z + 1, d = screw_h);                      // base through-holes
    for (p = motor_holes)
        translate([p[0], -1, p[1]]) rotate([0, 90, 0]) cylinder(h = motor_y + 1, d = boss_h);   // motor holes
    for (p = motor_holes)
        translate(p) rotate([-90, 0, 0]) cylinder(h = motor_y + 1, d = 3);       // screw holes
}
```
