// 96-well PCR tube rack (0.2 mL) - microplate footprint
// Print orientation: base on XY plane, holes through full height

// ---- Dimensions (mm) ----
plate_length   = 127.76;  // X
plate_width    = 85.48;   // Y
plate_height   = 20;      // Z
hole_d         = 6.2;     // tube hole diameter
hole_pitch     = 9.0;     // centre-to-centre spacing
hole_count_x   = 12;      // columns along X
hole_count_y   = 8;       // rows along Y
first_hole_x   = 14.38;   // A1 distance from left edge
first_hole_y   = 11.24;   // A1 distance from back edge
chamfer_leg    = 5;       // 45-degree chamfer leg length
cut_extra      = 0.5;     // overshoot for clean cuts

$fn = 64;

difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-plate_length/2, -plate_width/2, 0])
        cube([plate_length, plate_width, plate_height]);

    // Tube holes
    for (i = [0 : hole_count_x - 1])
        for (j = [0 : hole_count_y - 1])
            translate([first_hole_x + i * hole_pitch - hole_d/2,
                       first_hole_y + j * hole_pitch - hole_d/2,
                       -cut_extra])
                cylinder(d = hole_d, h = plate_height + 2 * cut_extra);

    // Chamfer at A1 corner (extends past vertical edge)
    translate([plate_length/2 - chamfer_leg,
               plate_width/2 - chamfer_leg,
               plate_height - chamfer_leg])
        cube([chamfer_leg, chamfer_leg, chamfer_leg]);
}
