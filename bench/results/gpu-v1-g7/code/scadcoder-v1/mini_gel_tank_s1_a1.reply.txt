```openscad
// Open-top rectangular buffer tank for a mini horizontal gel-electrophoresis box. NOVEL general part.
//

tank_L = 120;
tank_W = 70;
tank_H = 40;
wall   = 3;
floor  = 3;

$fn = 48;

inner_L = tank_L - 2 * wall;
inner_W = tank_W - 2 * wall;
inner_H = tank_H - floor;

platform_x = 60;
platform_t = 10;

difference() {
    cube([tank_L, tank_W, tank_H]);
    translate([0, 0, floor]) cube([tank_L, tank_W, tank_H - floor]);       // hollow, open top
    translate([-(tank_L / 2 - platform_x / 2), 0, floor + wall])
        cube([platform_x, tank_W, platform_t]);                           // central raised platform
    for (y = [-1, 1])
        translate([0, y * tank_W / 2 - wall, 0])
            rotate([-90, 0, 0]) cylinder(h = tank_H + 1, d = 2);          // electrode holes (X-directed)
}
```
