```openscad
// 96-well PCR tube rack (0.2 mL) - microplate footprint
// Print orientation: base on XY plane, holes through the full height

// ---------------- Parameters ----------------
plate_len    = 127.76;   // block length along X (mm)
plate_wid    = 85.48;    // block width along Y (mm)
plate_h      = 20;       // block height along Z (mm)

hole_d       = 6.2;      // tube hole diameter (mm)
hole_pitch   = 9.0;      // centre-to-centre spacing of holes (mm)
hole_n_rows  = 8;        // number of holes along Y
hole_n_cols  = 12;       // number of holes along X

chamfer_leg  = 5;        // 45-degree chamfer leg length at A1 corner (mm)

a1_x_offset  = 14.38;    // centre of A1 from left edge (mm)
a1_y_offset  = 11.24;    // centre of A1 from back edge (mm)

eps          = 0.5;      // cutter overshoot past faces (mm)
$fn          = 64;       // circle resolution

// ---------------- Derived ----------------
first_x = a1_x_offset - (hole_n_cols - 1) * hole_pitch / 2;   // leftmost hole X
first_y = a1_y_offset - (hole_n_rows - 1) * hole_pitch / 2;   // rearmost hole Y

// ---------------- Model ----------------
difference() {
    // Main block, centred in X/Y, sitting on z = 0
    translate([-plate_len / 2, -plate_wid / 2, 0])
        cube([plate_len, plate_wid, plate_h]);

    // 96 vertical through-holes
    for (i = [0 : hole_n_cols - 1])
        for (j = [0 : hole_n_rows - 1])
            translate([first_x + i * hole_pitch,
                       first_y + j * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_h + 2 * eps);

    // 45-degree chamfer at the A1 corner (minimum of the block)
    translate([0, 0, 0])
        linear_extrude(height = plate_h)
            polygon(points = [
                [0, 0],
                [chamfer_leg, 0],
                [0, chamfer_leg]
            ]);
}
```
