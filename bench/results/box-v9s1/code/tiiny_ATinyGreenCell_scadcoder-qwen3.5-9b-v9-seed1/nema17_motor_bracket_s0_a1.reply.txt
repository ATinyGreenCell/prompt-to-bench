```openscad
// L-shaped mounting bracket for a NEMA 17 stepper motor (e.g. for a syringe pump)
// Print orientation: base plate lies on the bed (z = 0), motor plate builds upward.

// ---------------- Parameters ----------------
$fn = 64;                 // curve resolution

// Base plate
base_x0 = -25;            // plate left edge (X)
base_x1 = 25;             // plate right edge (X)
base_y0 = 0;              // plate front edge (Y)
base_y1 = 45;             // plate back edge (Y)
base_z1 = 5;              // plate top height (Z)
bolt_d = 4.5;             // M4 through-hole diameter
bolt_x = -17;             // hole X offset from plate left edge
bolt_y = 30;              // hole Y offset from plate front edge

// Motor plate
motor_x0 = -25;           // plate left edge (X)
motor_x1 = 25;            // plate right edge (X)
motor_y0 = 0;             // plate front edge (Y)
motor_y1 = 5;             // plate depth (Y), sits on base back edge
motor_z1 = 50;            // plate top height (Z)
boss_d = 23;              // motor centring boss hole diameter
boss_x = 0;               // boss hole centre X
boss_z = 28;              // boss hole centre Z
screw_d = 3.4;            // M3 screw hole diameter
screw_pitch = 31;         // square pattern side length
screw_count = 4;          // number of screw holes

// Gussets
gusset_t = 5;             // gusset thickness (X)
gusset_y = 20;            // gusset leg along +Y
gusset_z = 20;            // gusset leg along +Z

eps = 0.5;                // cutter overshoot past faces

// ---------------- Model ----------------
union() {
    // Base plate with two vertical M4 through-holes
    translate([base_x0, base_y0, 0])
        difference() {
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1]);
            for (sx = [-1, 1])
                translate([sx * bolt_x, bolt_y, -eps])
                    cylinder(d = bolt_d, h = base_z1 + 2 * eps);
        }

    // Motor plate standing on the back edge of the base
    translate([motor_x0, motor_y1 - motor_y0, 0])
        difference() {
            cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1]);
            // Boss hole (along Y), centred at given X and Z
            translate([boss_x, 0, boss_z])
                rotate([0, 90, 0])
                    cylinder(d = boss_d, h = motor_y1 - motor_y0 + 2 * eps);
            // M3 screw holes on a square pattern centred on the boss hole
            for (i = [0 : screw_count - 1]) {
                a = (i - (screw_count - 1) / 2) * screw_pitch;
                translate([boss_x + a, -eps, boss_z + a])
                    cylinder(d = screw_d, h = motor_y1 - motor_y0 + 2 * eps);
            }
        }

    // Triangular gussets, flush with the left and right edges
    for (gx = [motor_x0, motor_x1 - gusset_t]) {
        translate([gx, motor_y0, 0])
            rotate([90, 0, 0])
                linear_extrude(height = gusset_t)
                    polygon(points = [
                        [gusset_y, gusset_z],   // right-angle corner at y = 5, z = 5
                        [gusset_y, 0],
                        [0, gusset_z]
                    ]);
    }
}
```
