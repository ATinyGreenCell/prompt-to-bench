```openscad
// Replacement knob for a hotplate stirrer with a 6 mm D-shaft
// Print orientation: bottom face (with shaft bore) on the bed, z = 0

$fn = 96;

// --- Knob dimensions ---
knob_d      = 30;    // knob outer diameter
knob_h      = 15;    // knob height

// --- Shaft bore ---
shaft_d     = 6.2;   // D-shaft bore diameter (circle)
shaft_flat  = 4.7;   // distance from flat side to opposite side of circle
bore_depth  = 12;    // blind bore depth from bottom face

// --- Grip grooves ---
groove_count = 18;   // number of vertical grip grooves
groove_d     = 2;    // groove diameter (half-cylinder)

// --- Pointer groove ---
pointer_w   = 1.5;   // pointer groove width
pointer_d   = 1;     // pointer groove depth

// --- Cutting overlap ---
eps         = 0.5;   // extension past cut faces

difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, -eps]) {
        cylinder(d = shaft_d, h = bore_depth + eps);
        // Flat side along -x, so the distance from flat to circle edge is shaft_d/2
        translate([-shaft_flat, 0, 0]) cube([shaft_flat, knob_d + 2*eps, bore_depth + 2*eps]);
    }

    // Vertical grip grooves along full height, centred on outer surface
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2*eps, $fn = 16);

    // Pointer groove from centre to outer edge on top face, same side as flat (-x)
    translate([0, 0, knob_h - pointer_d])
        linear_extrude(height = pointer_d)
            polygon([[-pointer_w / 2, 0], [pointer_w / 2, 0], [pointer_w / 2, knob_d / 2]]);
}
```
