```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate on bed (z = 0), motor plate builds upward (+Z)

// ---------------- Parameters ----------------
base_x0      = -25;    // base plate: x start
base_x1      = 25;     // base plate: x end
base_y0      = 0;      // base plate: y start
base_y1      = 45;     // base plate: y end
base_z1      = 5;      // base plate: z end (thickness)

hole_x       = -17;    // M4 through-hole x position (both holes)
hole_y       = 30;     // M4 through-hole y position (both holes)
hole_d       = 4.5;    // M4 hole diameter

plate_x0     = -25;    // motor plate: x start
plate_x1     = 25;     // motor plate: x end
plate_y0     = 0;      // motor plate: y start
plate_y1     = 5;      // motor plate: y end (thickness)
plate_z1     = 50;     // motor plate: z end (height)

boss_d       = 23;     // motor centring boss hole diameter
boss_x       = 0;      // boss hole centre x
boss_z       = 28;     // boss hole centre z

screw_d      = 3.4;    // M3 screw hole diameter
screw_pitch  = 31;     // square pattern side length

gusset_t     = 5;      // gusset thickness (X direction)
gusset_leg   = 20;     // right triangle leg length (Y and Z)
gusset_x0    = -25;    // left gusset x start
gusset_x1    = 25;     // right gusset x end

eps          = 0.5;    // cutter overshoot past faces

$fn = 96;

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([base_x0, base_y0, 0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1]);

        // Motor plate
        translate([plate_x0, plate_y0, 0])
            cube([plate_x1 - plate_x0, plate_y1 - plate_y0, plate_z1]);

        // Left gusset (flush with left edge)
        translate([gusset_x0, 0, 5])
            linear_extrude(height = gusset_t)
                polygon(points = [
                    [0, 5],
                    [gusset_leg, 5],
                    [0, gusset_leg + 5]
                ]);

        // Right gusset (flush with right edge)
        translate([gusset_x1, 0, 5])
            linear_extrude(height = gusset_t)
                polygon(points = [
                    [0, 5],
                    [-gusset_leg, 5],
                    [0, gusset_leg + 5]
                ]);
    }

    // Base plate through-holes (M4)
    translate([hole_x, hole_y, -eps])
        cylinder(d = hole_d, h = base_z1 + 2 * eps);

    // Motor boss hole (through along Y)
    translate([boss_x, plate_y0 - eps, boss_z])
        cylinder(d = boss_d, h = plate_y1 - plate_y0 + 2 * eps);

    // M3 screw holes on square pattern centred on boss hole
    for (a = [0, 90, 180, 270])
        rotate([0, 0, a])
            translate([screw_pitch / 2, 0, -eps])
                cylinder(d = screw_d, h = plate_z1 + 2 * eps);
}
```
