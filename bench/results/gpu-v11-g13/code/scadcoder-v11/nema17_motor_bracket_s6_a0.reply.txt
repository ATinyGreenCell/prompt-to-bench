```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate on bed (z = 0), motor plate builds upward (+Z)

// ---------------- Parameters ----------------
base_x0      = -25;    // base plate: x start
base_x1      = 25;     // base plate: x end
base_y0      = 0;      // base plate: y start
base_y1      = 45;     // base plate: y end
base_z0      = 0;      // base plate: z start
base_z1      = 5;      // base plate: z end

hole_x       = -17;    // M4 through-hole x position
hole_y       = 30;     // M4 through-hole y position
hole_d       = 4.5;    // M4 hole diameter

motor_x0     = -25;    // motor plate: x start
motor_x1     = 25;     // motor plate: x end
motor_y0     = 0;      // motor plate: y start
motor_y1     = 5;      // motor plate: y end
motor_z0     = 0;      // motor plate: z start
motor_z1     = 50;     // motor plate: z end

boss_hole_d  = 23;     // motor centring boss hole diameter
boss_x       = 0;      // boss hole centre x
boss_z       = 28;     // boss hole centre z

screw_d      = 3.4;    // M3 screw hole diameter
screw_n      = 4;      // number of screw holes
screw_p      = 31;     // square pattern side length

gusset_t     = 5;      // gusset thickness (X direction)
gusset_y     = 20;     // gusset leg length along +Y
gusset_z     = 20;     // gusset leg length along +Z

eps          = 0.5;    // cutter overshoot past faces

$fn = 96;

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([base_x0, base_y0, base_z0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);

        // Vertical motor plate
        translate([motor_x0, motor_y0, motor_z0])
            cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1 - motor_z0]);

        // Left gusset (x = -25 to -20)
        translate([-25, 0, 0])
            linear_extrude(height = gusset_t)
                polygon(points = [
                    [-25, 5],
                    [-25, gusset_y],
                    [0, gusset_y]
                ]);

        // Right gusset (x = 20 to 25)
        translate([20, 0, 0])
            linear_extrude(height = gusset_t)
                polygon(points = [
                    [20, 5],
                    [20, gusset_y],
                    [0, gusset_y]
                ]);
    }

    // M4 through-holes in base plate
    translate([hole_x, hole_y, -eps])
        cylinder(d = hole_d, h = base_z1 - base_z0 + 2 * eps);

    // Boss hole through motor plate (along Y)
    translate([boss_x, 0, boss_z])
        rotate([90, 0, 0])
            cylinder(d = boss_hole_d, h = motor_y1 - motor_y0 + 2 * eps);

    // M3 screw holes on square pattern, centred on boss hole
    for (i = [0 : screw_n - 1]) {
        for (j = [0 : screw_n - 1]) {
            x = (i - (screw_n - 1) / 2) * screw_p;
            y = (j - (screw_n - 1) / 2) * screw_p;
            translate([x, y, -eps])
                rotate([90, 0, 0])
                    cylinder(d = screw_d, h = motor_y1 - motor_y0 + 2 * eps);
        }
    }
}
```
