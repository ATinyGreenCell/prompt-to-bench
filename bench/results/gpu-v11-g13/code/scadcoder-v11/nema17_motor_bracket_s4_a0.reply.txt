```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate on bed (z = 0), vertical plate builds upward (+Z)

// ---------------- Parameters ----------------
base_x0      = -25;    // base plate: x start
base_x1      = 25;     // base plate: x end
base_y0      = 0;      // base plate: y start
base_y1      = 45;     // base plate: y end
base_z0      = 0;      // base plate: z start
base_z1      = 5;      // base plate: z end

motor_plate_x0 = -25;  // motor plate: x start
motor_plate_x1 = 25;   // motor plate: x end
motor_plate_y0 = 0;    // motor plate: y start
motor_plate_y1 = 5;    // motor plate: y end
motor_plate_z0 = 0;    // motor plate: z start
motor_plate_z1 = 50;   // motor plate: z end

boss_hole_d  = 23;     // motor centring boss hole diameter
boss_x       = 0;      // boss hole centre X
boss_z       = 28;     // boss hole centre Z

screw_hole_d = 3.4;    // M3 screw hole diameter
screw_pat_h  = 31;     // square pattern side length

gusset_t     = 5;      // gusset thickness (X)
gusset_leg   = 20;     // right triangle leg length (Y and Z)
gusset_x0    = -25;    // left gusset X start
gusset_x1    = -20;    // left gusset X end

eps          = 0.5;    // cutter overshoot past faces

$fn = 96;

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([base_x0, base_y0, base_z0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);

        // Vertical motor plate
        translate([motor_plate_x0, motor_plate_y0, motor_plate_z0])
            cube([motor_plate_x1 - motor_plate_x0, motor_plate_y1 - motor_plate_y0, motor_plate_z1 - motor_plate_z0]);

        // Left gusset (flush with left edge)
        translate([gusset_x0, 0, 0])
            linear_extrude(height = gusset_leg, rotate_90 = true)
                polygon([[gusset_leg, 0], [0, gusset_leg]]);

        // Right gusset (flush with right edge)
        translate([gusset_x1, 0, 0])
            linear_extrude(height = gusset_leg, rotate_90 = true)
                polygon([[0, gusset_leg], [gusset_leg, 0]]);
    }

    // Motor boss hole (along Y)
    translate([boss_x, motor_plate_y0 - eps, boss_z])
        rotate([90, 0, 0])
            cylinder(d = boss_hole_d, h = motor_plate_y1 - motor_plate_y0 + 2 * eps);

    // M3 screw holes on square pattern
    for (i = [0 : 3]) {
        for (j = [0 : 3]) {
            x = -screw_pat_h / 2 + (i - 1.5) * screw_pat_h / 3;
            y = -screw_pat_h / 2 + (j - 1.5) * screw_pat_h / 3;
            translate([x, y, -eps])
                cylinder(d = screw_hole_d, h = base_z1 + motor_plate_z1 + 2 * eps);
        }
    }
}
```
