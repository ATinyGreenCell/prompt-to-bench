```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate flat on the bed (z = 0), motor plate building up in +Z

// ---------------- Parameters ----------------
// Base plate
base_x_min   = -25;   // base plate X extent (lower end)
base_x_max   = 25;    // base plate X extent (upper end)
base_y_min   = 0;     // base plate Y extent (front)
base_y_max   = 45;    // base plate Y extent (back)
base_z       = 5;     // base plate thickness

// Motor plate
motor_x_min  = -25;   // motor plate X extent (lower end)
motor_x_max  = 25;    // motor plate X extent (upper end)
motor_y_min  = 0;     // motor plate Y extent (front)
motor_y_max  = 5;     // motor plate Y extent (back)
motor_z      = 50;    // motor plate height

// Mounting holes
hole_m4_x    = -17;   // M4 through-hole X position (left side)
hole_m4_y    = 30;    // M4 through-hole Y position
hole_m4_d    = 4.5;   // M4 through-hole diameter
hole_m3_x    = 17;    // M3 through-hole X position (right side)
hole_m3_y    = 30;    // M3 through-hole Y position
hole_m3_d    = 3.4;   // M3 through-hole diameter

// Boss hole for motor centring
boss_d       = 23;    // boss hole diameter (along Y)
boss_x       = 0;     // boss hole centre X
boss_z       = 28;    // boss hole centre Z

// Screw holes for boss
screw_pat    = 31;    // square pattern size for M3 screws
screw_count  = 4;     // number of screw holes

// Gussets
gusset_t     = 5;     // gusset thickness (X)
gusset_y     = 20;    // gusset leg length along +Y
gusset_z     = 20;    // gusset leg length along +Z

// Misc
eps          = 0.5;   // cutter overshoot past faces
$fn          = 96;    // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([base_x_min, base_y_min, 0])
            cube([base_x_max - base_x_min, base_y_max - base_y_min, base_z]);

        // Vertical motor plate, standing on the back edge of the base
        translate([motor_x_min, motor_y_min, 0])
            cube([motor_x_max - motor_x_min, motor_y_max - motor_y_min, motor_z]);

        // Left gusset: right triangle in YZ, flush with left edge (x = -25)
        translate([base_x_min, 0, 0])
            rotate([90, 0, 0])
                linear_extrude(height = gusset_t)
                    polygon(points = [
                        [0, 0],
                        [gusset_y, 0],
                        [0, gusset_z]
                    ]);

        // Right gusset: right triangle in YZ, flush with right edge (x = 25)
        translate([base_x_max - gusset_t, 0, 0])
            rotate([90, 0, 0])
                linear_extrude(height = gusset_t)
                    polygon(points = [
                        [0, 0],
                        [gusset_y, 0],
                        [0, gusset_z]
                    ]);
    }

    // Through-cut for motor centring boss (along Y)
    translate([boss_x, 0, boss_z])
        rotate([90, 0, 0])
            cylinder(d = boss_d, h = motor_y_max - motor_y_min + 2 * eps, center = true);

    // M4 vertical through-hole on the left
    translate([hole_m4_x, hole_m4_y, -eps])
        cylinder(d = hole_m4_d, h = base_z + 2 * eps);

    // M3 vertical through-hole on the right
    translate([hole_m3_x, hole_m3_y, -eps])
        cylinder(d = hole_m3_d, h = base_z + 2 * eps);

    // Four M3 screw holes on a 31 mm square pattern, centred on the boss
    for (i = [0 : 3]) {
        a = [0, 90, 180, 270][i];
        translate([screw_pat / 2 * cos(a), screw_pat / 2 * sin(a), -eps])
            cylinder(d = hole_m3_d, h = motor_z + 2 * eps);
    }
}
```
