```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate on bed (z = 0), motor plate builds upward

// ---------- Parameters ----------
base_x      = 25;     // base plate half-width (X)
base_y      = 45;     // base plate length (Y)
base_z      = 5;      // base plate thickness (Z)

hole_x      = 17;     // M4 through-hole offset from centre (X)
hole_y      = 30;     // M4 through-hole offset from centre (Y)
hole_d      = 4.5;    // M4 hole diameter

motor_x     = 25;     // motor plate half-width (X)
motor_y     = 5;      // motor plate thickness (Y)
motor_z     = 50;     // motor plate height (Z)

boss_d      = 23;     // motor centring boss hole diameter
boss_x      = 0;      // boss hole centre X
boss_z      = 28;     // boss hole centre Z

screw_d     = 3.4;    // M3 screw hole diameter
screw_pat   = 31;     // square pattern size (X and Y)

gusset_t    = 5;      // gusset thickness (X)
gusset_l    = 20;     // gusset leg length (Y and Z)

eps         = 0.5;    // cutter overshoot past faces

$fn = 96;

// ---------- Model ----------
difference() {
    union() {
        // Base plate
        translate([-base_x, 0, 0])
            cube([2*base_x, base_y, base_z]);

        // Motor plate
        translate([-motor_x, 0, 0])
            cube([2*motor_x, motor_y, motor_z]);

        // Gussets (right triangle in YZ, flush with outer edges)
        for (sx = [-1, 1])
            translate([sx > 0 ? 20 : -25, 0, 0])
                rotate([90, 0, 0])
                    linear_extrude(height = gusset_t)
                        polygon([
                            [0, 5],
                            [0, 5 + gusset_l],
                            [gusset_l, 5 + gusset_l]
                        ]);
    }

    // Base plate through-holes
    for (sx = [-1, 1])
        translate([sx * hole_x, hole_y, -eps])
            cylinder(d = hole_d, h = base_z + 2*eps);

    // Motor plate boss hole (along Y)
    translate([boss_x, 0, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_y + 2*eps);

    // Screw holes on square pattern centred on boss
    for (a = [0, 90, 180, 270])
        rotate([0, 0, a])
            translate([screw_pat / 2, 0, 0])
                cylinder(d = screw_d, h = motor_z + 2*eps);
}
```
