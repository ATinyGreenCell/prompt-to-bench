```openscad
// DIY Magnetic Stirrer Housing
// Based on an 80mm PC case fan
// Print orientation: Upside down (Top plate on bed, walls rising up)

// --- Dimensions ---
// Outer dimensions of the box
W = 90;      // Width (X)
D = 90;      // Depth (Y)
H = 40;      // Height of walls (Z)

// Wall thickness
T_WALL = 2.5;

// Top plate thickness
T_TOP = 2;

// Fan mounting hole pattern
PATTERN_SIZE = 71.5; // Distance between opposite holes
HOLE_SIZE = 4.5;      // Diameter of M4 screw holes

// Cable notch dimensions
NOTCH_WIDTH = 12;
NOTCH_DEPTH = 8;      // Depth from the open top edge (Z = H)

// --- Configuration ---
$fn = 64; // Smooth curves

// --- Helper Functions ---
// Create a rectangular hole through a solid
module hole(x, y, z, dx, dy, dz, d) {
    difference() {
        translate([x, y, z]) cube([dx, dy, dz]);
        translate([x, y, z]) cylinder(h=d, r=d/2);
    }
}

// Create a rectangular notch on a face
module notch(x, y, z, dx, dy, dz, d) {
    difference() {
        translate([x, y, z]) cube([dx, dy, dz]);
        translate([x, y, z]) cube([dx, dy, NOTCH_DEPTH]);
    }
}

// --- Main Body Construction ---
// We build the solid from the bottom up.
// The base is the top plate (z=0 to z=2).
// The walls rise from z=2 to z=42 (2mm extra to ensure clean cut in difference).

difference() {
    union() {
        // 1. The Top Plate (Base)
        // Outer size W x D, thickness T_TOP
        translate([0, 0, 0]) 
            cube([W, D, T_TOP]);

        // 2. The Walls
        // Rise from z = T_TOP to z = H + T_WALL
        // Outer dimensions W x D
        translate([0, 0, T_TOP]) 
            cube([W, D, H + T_WALL]);
    }

    // --- Cutouts ---

    // 1. Screw Holes in the Top Plate
    // Pattern: 71.5mm square centered on the plate.
    // Plate is 90x90. Center is (45, 45).
    // Pattern center is (45, 45).
    // Hole positions: +/- 35.75 from center.
    // Coordinates: 9.25, 80.75
    
    for (i = [-1, 1]) {
        for (j = [-1, 1]) {
            // Calculate position
            px = 45 + (PATTERN_SIZE / 2) * i;
            py = 45 + (PATTERN_SIZE / 2) * j;
            
            // Cut hole through the top plate only (z=0 to z=2)
            // Extend slightly past the plate to ensure clean cut
            hole(px, py, 0, HOLE_SIZE, HOLE_SIZE, T_TOP + 0.1, HOLE_SIZE);
        }
    }

    // 2. Cable Notch in One Wall
    // Location: One of the walls. Let's choose the front wall (Y = D/2).
    // The wall is at Y = 45 (center) to Y = 90 (edge)? No, the box is 90x90.
    // Walls are at X=0..90, Y=0..90.
    // Let's pick the wall at Y = 0 (Front) or Y = 90 (Back).
    // Let's use the wall at Y = 0 (Front face looking from +Y).
    // Actually, standard orientation: X is width, Y is depth.
    // Let's cut the notch in the wall at Y = 0 (the "front" wall if looking from +Y).
    // Wait, if the box is 90x90, the walls are at X=0, X=90, Y=0, Y=90.
    // Let's cut the notch in the wall at Y = 0.
    // The notch is 12mm wide, centered along the wall.
    // Wall length is 90mm. Center is 45mm.
    // Notch X range: 45 - 6 = 39 to 45 + 6 = 51.
    // Notch Z range: From open edge (Z=40) down 8mm -> Z=32 to Z=40.
    // Notch Y range: Thickness of wall (2.5mm). Centered at Y=0? 
    // The wall exists from Y=-1.25 to Y=1.25? No, the solid is 0 to 90.
    // The wall at Y=0 is the face at Y=0. The material is from Y=0 to Y=2.5.
    // So the notch should be in the material Y=0 to Y=2.5.
    // Center Y = 1.25.
    
    // Notch Position
    nx = 45;
    ny = 1.25; // Center of the wall thickness at Y=0
    nz = 32;   // Start of the cut (8mm down from Z=40)
    
    // Dimensions
    ndx = NOTCH_WIDTH;
    ndy = T_WALL;
    ndz = NOTCH_DEPTH;
    
    // Cut the notch
    // We cut a block that removes the material from Z=32 to Z=40
    notch(nx, ny, nz, ndx, ndy, ndz, ndz);
    
    // Also need to cut the "back" of the notch if we consider the wall thickness properly
    // The notch goes through the wall thickness.
    // The wall is at Y=0 to Y=2.5.
    // The notch is centered at Y=1.25.
    // The cut shape is a cube of size [12, 2.5, 8].
    // This removes the material correctly.
}
```
