// 96-well PCR tube rack (0.2 mL, standard microplate footprint)
// Print orientation: base on XY plane, building up in +Z

// ---- Block ----
block_l = 127.76;   // length along X
block_w = 85.48;    // width along Y
block_h = 20;       // height along Z

// ---- Holes ----
hole_d     = 6.2;   // through-hole diameter
hole_nx    = 12;    // columns along X
hole_ny    = 8;     // rows along Y
hole_pitch = 9.0;   // centre-to-centre spacing
hole_x0    = 14.38; // first hole (A1) offset from left edge
hole_y0    = 11.24; // first hole (A1) offset from back edge

// ---- Chamfer ----
chamf_leg  = 5;     // chamfer leg length
chamf_ang  = 45;    // chamfer angle (degrees)

// ---- Misc ----
eps = 0.5;          // cutter overshoot past faces
$fn = 64;

// Derived
chamf_r = chamf_leg / cos(chamf_ang);  // end radius of cone
grid_x  = (hole_nx - 1) * hole_pitch;
grid_y  = (hole_ny - 1) * hole_pitch;

difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_l/2, -block_w/2, 0])
        cube([block_l, block_w, block_h]);

    // Through-holes
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([hole_x0 + i * hole_pitch - block_l/2,
                       hole_y0 + j * hole_pitch - block_w/2,
                       -eps])
                cylinder(d = hole_d, h = block_h + 2*eps);

    // Orientation chamfer at A1 corner, extended above the top face
    translate([hole_x0 - block_l/2, hole_y0 - block_w/2, block_h - eps])
        translate([0, 0, -eps])
            cylinder(r1 = block_l/2 + eps, r2 = chamf_r, h = block_h + 2*eps);
}
