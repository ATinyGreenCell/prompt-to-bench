// 96-well PCR tube rack (0.2 mL, standard microplate footprint)
// Print orientation: base on XY plane, building up in +Z

// ---- Block ----
block_l = 127.76;   // length along X
block_w = 85.48;    // width along Y
block_h = 20;       // height along Z

// ---- Holes ----
hole_d     = 6.2;   // through-hole diameter
hole_rows  = 8;     // rows along Y
hole_cols  = 12;    // columns along X
hole_pitch = 9.0;   // centre-to-centre spacing (X and Y)
hole_z0    = -1;    // cutter start height (extends past bottom face)

// ---- Corner chamfer (orientation mark) ----
chamfer_leg = 5;    // leg length along X and Y
chamfer_ang = 45;   // angle from X axis (degrees)

// ---- Misc ----
eps = 0.5;          // cutter overshoot past top face
$fn = 64;

// First hole (A1) offsets from block edges
a1_x = 14.38;
a1_y = 11.24;

difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_l/2, -block_w/2, 0])
        cube([block_l, block_w, block_h]);

    // 96 through-holes
    for (i = [0 : hole_rows - 1])
        for (j = [0 : hole_cols - 1])
            translate([a1_x + j * hole_pitch - block_l/2,
                       a1_y + i * hole_pitch - block_w/2,
                       hole_z0])
                cylinder(d = hole_d, h = block_h + 2 * eps);

    // 45-degree chamfer at the A1 corner (vertical edge)
    translate([-block_l/2 - eps, -block_w/2 - eps, 0])
        linear_extrude(height = block_h + 2 * eps)
            polygon([[-chamfer_leg, -chamfer_leg],
                     [-chamfer_leg, chamfer_leg * tan(chamfer_ang)],
                     [chamfer_leg * tan(chamfer_ang), chamfer_leg],
                     [chamfer_leg, -chamfer_leg]]);
}
