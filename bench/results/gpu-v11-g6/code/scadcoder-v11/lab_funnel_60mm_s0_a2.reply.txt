```openscad
// Lab funnel - printed upright, wide opening at the top
// Units: mm

$fn = 64;

// ---- Dimensions ----
spout_od      = 10;   // spout outer diameter
spout_h       = 20;   // spout height (z = 0 to 20)
cone_bottom_d = 10;   // cone outer diameter at z = 20
cone_top_d    = 60;   // cone outer diameter at the rim
cone_h        = 30;   // cone height (z = 20 to 50)
wall          = 1.6;  // wall thickness everywhere
eps           = 0.5;  // overlap / cut extension

// Derived
bore = spout_od - 2 * wall;  // spout bore = 6.8 mm

// ---- Model ----
difference() {
    union() {
        // Spout
        translate([0, 0, 0])
            cylinder(d = spout_od, h = spout_h);

        // Cone, slightly overlapping the spout for a clean union
        translate([0, 0, spout_h - eps])
            cylinder(d1 = cone_bottom_d, d2 = cone_top_d, h = cone_h + eps);
    }

    // Through-cut for the bore (extends past both ends)
    translate([0, 0, -eps])
        cylinder(d = bore, h = spout_h + cone_h + 2 * eps);
}
```
