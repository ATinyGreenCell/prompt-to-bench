```openscad
// Lab funnel - printed upright with wide opening at the top
// Units: mm. Wall thickness 1.6 mm throughout.

$fn = 64;

// ---- Dimensions ----
spout_od        = 10;    // outer diameter of straight spout (z = 0..20)
spout_id        = spout_od - 2*1.6;  // bore diameter = 6.8 mm
spout_h         = 20;    // spout height (z = 0 to 20)

cone_bottom_od  = spout_od;  // cone starts at spout diameter (z = 20)
cone_top_od     = 60;    // cone rim diameter (z = 50)
cone_h          = cone_top_od - cone_bottom_od;  // 40 mm cone height

total_h         = spout_h + cone_h;  // 50 mm overall

eps             = 0.5;   // overlap for clean boolean cuts

// ---- Model ----
difference() {
    union() {
        // Spout
        cylinder(h = spout_h, d = spout_od);

        // Cone
        translate([0, 0, spout_h - eps])
            cylinder(h = cone_h + eps, d1 = cone_bottom_od, d2 = cone_top_od);
    }

    // Bottom cut (open bottom)
    translate([0, 0, -eps])
        cylinder(h = eps + 1, d = spout_id);

    // Wall thinning helper: extends above top rim, keeps full bore
    translate([0, 0, spout_h])
        cylinder(h = cone_h + eps + 1, d = spout_id);
}
```
