```openscad
// Lab funnel - printed upright with wide opening at the top
// Units: mm

// ---- Parameters ----
spout_od        = 10;    // spout outer diameter
spout_h         = 20;    // spout height (z = 0 to 20)
cone_od_bottom  = 10;    // cone outer diameter at z = 20
cone_od_top     = 60;    // cone outer diameter at rim (z = 50)
cone_top_z      = 50;    // rim height
wall            = 1.6;   // wall thickness everywhere
eps             = 0.5;   // cutter overshoot

$fn = 64;

// ---- Derived ----
spout_id = spout_od - 2*wall;   // 6.8 mm bore
cone_h   = cone_top_z - spout_h;

// ---- Model ----
difference() {
    union() {
        // Spout
        cylinder(d = spout_od, h = spout_h);
        // Cone (slightly overlapping spout for a clean union)
        translate([0, 0, spout_h - 0.01])
            linear_extrude(height = cone_h + 0.01)
                circle(d = (cone_od_bottom + cone_od_top) / 2);
    }
    // Through-cut: straight axis from bottom to top
    translate([0, 0, -eps])
        cylinder(h = cone_top_z + 2*eps, d = spout_id);
}
```
