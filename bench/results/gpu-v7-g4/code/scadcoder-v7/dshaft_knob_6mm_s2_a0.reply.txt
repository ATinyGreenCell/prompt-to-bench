```openscad
// Replacement knob for hotplate stirrer (D-shaft)
// Print orientation: D-flat side down on bed (z = 0)

// ---------------- Parameters ----------------
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
shaft_d       = 6.2;   // D-shaft diameter
bore_depth    = 12;    // blind bore depth from bottom face
bore_flat     = 4.7;   // D-bore flat-to-flat distance (across circle)
groove_count  = 18;    // number of grip grooves
groove_d      = 2;     // grip groove diameter (half-cylinder)
pointer_w     = 1.5;   // pointer groove width
pointer_d    = pointer_w; // pointer groove depth (same as width)
eps           = 0.5;   // cutter overshoot past faces

// Derived
knob_r = knob_d / 2;
// D-bore radius from flat-to-flat: 2*R - flat = bore_flat
bore_r = (bore_flat + shaft_d) / 2;

// ---------------- Model ----------------
difference() {
    // Main cylinder
    cylinder(d = knob_d, h = knob_h, $fn = 96);

    // D-shaped blind bore, centred on axis, flat on +x side
    translate([0, 0, -eps])
        cylinder(r = bore_r, h = bore_depth + 2 * eps, $fn = 64);
    translate([bore_r, 0, -eps])
        cube([bore_r, 2 * knob_r, bore_depth + 2 * eps]);

    // Grip grooves: vertical half-cylinders centred on outer surface
    for (i = [0 : groove_count - 1]) {
        a = i * 360 / groove_count;
        translate([knob_r * cos(a), knob_r * sin(a), -eps])
            cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 32);
    }

    // Pointer groove: runs from centre to outer edge on +x side
    translate([0, 0, -eps])
        linear_extrude(height = knob_h + 2 * eps)
            circle(d = pointer_w, $fn = 72);
}
```
