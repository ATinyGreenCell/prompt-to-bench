```openscad
// Magnetic stirrer housing - upside-down open box over 80 mm PC fan
// Print orientation: top plate on bed (z = 0), walls rise to z = 40

$fn = 64;

// ---- Dimensions (mm) ----
plate_t      = 2;     // top plate thickness
wall_t       = 2.5;   // wall thickness
box_h        = 40;    // total height of box (open top)
outer_x      = 90;    // outer size in X
outer_y      = 90;    // outer size in Y
hole_d       = 4.5;   // M4 screw hole diameter
hole_p       = 71.5;  // square pitch of hole pattern
notch_w      = 12;    // cable notch width
notch_depth  = 8;     // cable notch depth from open edge
eps          = 0.5;   // cutter overshoot past faces

// ---- Derived ----
notch_z0 = box_h - notch_depth;  // notch runs from here to open edge

difference() {
    // Outer block
    translate([-outer_x/2, -outer_y/2, 0])
        cube([outer_x, outer_y, box_h]);

    // Open top: cut down to notch_z0, leaving walls above
    translate([-outer_x/2, -outer_y/2, notch_z0])
        cube([outer_x, outer_y, notch_depth + eps]);

    // Screw holes through top plate
    for (i = [0 : 3]) {
        for (j = [0 : 3]) {
            if ((i + j) % 2 == 0) {  // 4 holes on square pattern
                translate([(i - 1.5) * hole_p, (j - 1.5) * hole_p, -eps])
                    cylinder(d = hole_d, h = plate_t + 2 * eps);
            }
        }
    }

    // Cable notch in one wall (along X, centred)
    translate([-outer_y/2, box_h - notch_depth, 0])
        cube([notch_w, outer_y, notch_depth + eps]);
}
```
