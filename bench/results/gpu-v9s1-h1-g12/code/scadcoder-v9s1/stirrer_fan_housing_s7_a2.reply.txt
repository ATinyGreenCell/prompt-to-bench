```openscad
// DIY magnetic stirrer housing - upside-down open box over 80 mm PC fan
// Print orientation: top plate on bed (z = 0), walls rise to open top at z = 40

// ---- Dimensions (mm) ----
plate_t      = 2;     // top plate thickness
wall_t       = 2.5;   // wall thickness
box_h        = 40;    // total height
outer_x      = 90;    // outer size in X
outer_y      = 90;    // outer size in Y
hole_d       = 4.5;   // M4 screw hole diameter
hole_p       = 71.5;  // square pitch of hole centres
notch_w      = 12;    // cable notch width
notch_d      = 8;     // cable notch depth from open edge (z = 32 to z = 40)
eps          = 0.5;   // cutter overshoot past faces

$fn = 64;

// ---- Model ----
difference() {
    // Outer block
    translate([-outer_x/2, -outer_y/2, 0])
        cube([outer_x, outer_y, box_h]);

    // Cut out the fan opening (under the open top edge)
    translate([-outer_x/2, -outer_y/2, box_h - wall_t - eps])
        cube([outer_x, outer_y, wall_t + 2*eps]);

    // Four M4 screw holes through the top plate, square pattern
    for (i = [0 : 3])
        for (j = [0 : 3])
            if ((i + j) % 2 == 1)
                translate([(i - 1.5) * hole_p / 2, (j - 1.5) * hole_p / 2, -eps])
                    cylinder(d = hole_d, h = plate_t + 2*eps);

    // Cable notch in one wall, centred along X
    translate([-outer_x/2 + notch_w/2, 0, box_h - notch_d])
        cube([notch_w, outer_y, notch_d + eps]);
}
```
