// Hose-barb reducer: joins 8 mm ID tubing to 5 mm ID tubing
// Print orientation: standing upright along +Z, centred on origin in X/Y

$fn = 96;

// ---- Dimensions (mm) ----
eps      = 0.5;         // overcut for clean differences
barb_h8  = 10;          // height of each 8 mm barb
barb_od8_bottom = 7.5;  // outer diameter of 8 mm barb at bottom
barb_od8_top    = 9.5;  // outer diameter of 8 mm barb at top
collar_d      = 12;     // diameter of the collar
barb_h5  = 8;           // height of each 5 mm barb
barb_od5_bottom = 6.5;  // outer diameter of 5 mm barb at bottom
barb_od5_top    = 5.0;  // outer diameter of 5 mm barb at top
bore      = 3;          // central through-bore diameter
length8   = 20;         // z extent of the 8 mm tubing section
length5   = 16;         // z extent of the 5 mm tubing section (z = 25 to 41)
overlap   = 0.01;       // small overlap between stacked barbs

// ---- Model ----
union() {
    // 8 mm tubing section: two stacked barbs with sawtooth profile
    for (i = [0, 1]) {
        translate([0, 0, i * barb_h8])
            difference() {
                linear_extrude(height = bore + 2 * eps)
                    circle(d = bore + 2 * eps);
                // Barb body: OD grows linearly, then steps back to bottom OD
                hull() {
                    translate([0, 0, i * barb_h8 - overlap])
                        cylinder(d = barb_od8_bottom, h = overlap + 0.01);
                    translate([0, 0, i * barb_h8 + barb_h8 - overlap])
                        cylinder(d = barb_od8_top, h = overlap + 0.01);
                }
                // Step down from top of barb back to 7.5 mm OD
                translate([0, 0, i * barb_h8 + barb_h8 - overlap])
                    cylinder(d = barb_od8_bottom, h = overlap + 0.01);
            }
    }
    // Collar
    translate([0, 0, length8])
        cylinder(d = collar_d, h = 5);
    // 5 mm tubing section: two stacked barbs, shrinking OD
    for (i = [0, 1]) {
        translate([0, 0, length8 + 5 + i * barb_h5])
            difference() {
                linear_extrude(height = bore + 2 * eps)
                    circle(d = bore + 2 * eps);
                // Barb body: OD shrinks linearly from bottom to top
                hull() {
                    translate([0, 0, length8 + 5 + i * barb_h5 - overlap])
                        cylinder(d = barb_od5_bottom, h = overlap + 0.01);
                    translate([0, 0, length8 + 5 + i * barb_h5 + barb_h5 - overlap])
                        cylinder(d = barb_od5_top, h = overlap + 0.01);
                }
                // Step back out from top of barb to bottom OD
                translate([0, 0, length8 + 5 + i * barb_h5 + barb_h5 - overlap])
                    cylinder(d = barb_od5_bottom, h = overlap + 0.01);
            }
    }
}
