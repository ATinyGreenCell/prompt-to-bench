// Hose-barb reducer: 8 mm ID -> 5 mm ID
// Print orientation: standing upright along +Z

// ---- Parameters ----
eps      = 0.5;   // overcut for clean differences
$fn      = 96;    // smoothness of curved surfaces

tube_d1  = 8;     // inner diameter of large tubing
tube_d2  = 5;     // inner diameter of small tubing
bore     = 3;     // through-bore diameter

base_h   = 20;    // height of first (8 mm) barb section
collar_d = 12;    // diameter of joining collar
collar_h = 5;     // height of collar

top_h    = 8;     // height of second (5 mm) barb section
total_l  = base_h + collar_h + top_h;  // 41 mm overall length

// ---- Derived ----
r_bore = bore / 2;
// keep the sawtooth teeth tall enough to grip the tubing
teeth_h = max(1, tube_d1 / 2 - 1.5);

// ---- Profile (radius vs height), counter-clockwise ----
profile = [
    [r_bore, 0],
    [collar_d / 2, base_h],
    [collar_d / 2, base_h + collar_h],
    [r_bore, total_l]
];

// ---- Model ----
difference() {
    union() {
        // 8 mm tubing barbs: sawtooth outer profile
        for (z = [0, base_h])
            translate([0, 0, z])
                linear_extrude(height = teeth_h)
                    circle(r = [7.5, 9.5]);   // linear growth 7.5 -> 9.5

        // collar
        translate([0, 0, base_h])
            cylinder(h = collar_h, d = collar_d);

        // 5 mm tubing barbs: tapered outer profile
        for (z = [base_h + collar_h, total_l])
            translate([0, 0, z])
                cylinder(h = 8, d1 = 6.5, d2 = 5);   // linear shrink 6.5 -> 5.0
    }

    // through-bore
    translate([0, 0, -eps])
        cylinder(h = total_l + 2 * eps, d = bore);
}
