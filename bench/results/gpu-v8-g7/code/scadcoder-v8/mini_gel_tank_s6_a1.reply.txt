```openscad
// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: tank rests on XY plane at z = 0, builds upward (+Z)

// ---------------- Parameters ----------------
tank_x      = 120;   // outer length in X (mm)
tank_y      = 70;    // outer width in Y (mm)
tank_h      = 40;    // outer height in Z (mm)
wall        = 3;     // wall thickness (mm)
floor       = 3;     // floor thickness (mm)

platform_x  = 60;    // platform length in X (mm), centred
platform_y  = tank_y - 2 * wall;  // platform spans full inner width (mm)
platform_z  = 10;    // platform top height above inner floor (mm)

hole_d      = 2;     // electrode hole diameter (mm)
hole_y      = 0;     // hole centre Y (mm)
hole_z      = 30;    // hole centre Z (mm)

eps         = 0.5;   // cutter overshoot past faces (mm)
$fn         = 96;    // curve resolution

// ---------------- Derived ----------------
inner_h = tank_h - floor;              // 37 mm inner floor-to-top height
// Platform sits on the floor, so its bottom is at z = floor = 3

// ---------------- Model ----------------
difference() {
    // Tank body: open-top rectangular box with floor
    translate([-tank_x / 2, -tank_y / 2, 0])
        cube([tank_x, tank_y, tank_h]);

    // Gel-tray platform, centred in X, spanning full inner width
    translate([-platform_x / 2, -platform_y / 2, floor])
        cube([platform_x, platform_y, platform_z]);

    // Horizontal electrode holes through each end wall (at X = +/- 60)
    for (sx = [-1, 1])
        translate([sx * (tank_x / 2 - eps), hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = tank_y + 2 * eps);
}
```
