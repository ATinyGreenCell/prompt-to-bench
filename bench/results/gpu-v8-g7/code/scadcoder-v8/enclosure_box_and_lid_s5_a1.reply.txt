// Two-part enclosure for a small electronics board (e.g. Arduino-based temperature logger)
// Print orientation: both parts rest on the bed (z = 0), build upward, side by side.
// The lid is printed upside down: a flat plate with a lip that slides into the box opening.

// ---------------- Parameters ----------------
box_x      = 70;    // box outer length (X)
box_y      = 50;    // box outer width (Y)
box_z      = 30;    // box outer height (Z)
box_wall   = 2;     // box wall thickness (sides and floor)
box_floor  = 2;     // box floor thickness (separate from side walls per the given dimensions)

lip_t      = 1.5;   // lip thickness (Z), printed on top of the lid plate
lip_h      = 4;     // lip height (Z), printed above the lid plate
lip_x      = 65.6;  // lip outer length (X)
lip_y      = 45.6;  // lip outer width (Y)
lip_clear  = 0.2;   // clearance between lip outer size and box opening on each side

gap        = 10;    // gap between the two parts on the bed (X direction)

eps        = 0.5;   // cutter overshoot past faces

$fn = 64;

// ---------------- Derived ----------------
box_in_x = box_x - 2 * box_wall;   // opening length (X)
box_in_y = box_y - 2 * box_wall;   // opening width (Y)
lip_h2   = lip_h + lip_t;          // total height of the lip above the bed

// Check: opening 66.0 x 46.0 vs lip outer 65.6 x 45.6 -> 0.2 mm clearance each side
assert(abs((box_in_x) - lip_x) < 0.01, "X dimensions do not give the expected clearance");
assert(abs((box_in_y) - lip_y) < 0.01, "Y dimensions do not give the expected clearance");

// ---------------- Part 1: Box ----------------
module box() {
    difference() {
        // Main block: side walls + floor (the given dimensions treat the floor as separate)
        translate([-box_x / 2, -box_y / 2, 0])
            cube([box_x, box_y, box_z - box_floor]);
        // Floor plate
        translate([-box_x / 2, -box_y / 2, 0])
            cube([box_x, box_y, box_floor]);
        // Open the top: cut through the side walls above the floor
        translate([-box_in_x / 2, -box_in_y / 2, box_floor])
            cube([box_in_x, box_in_y, box_z - box_floor + eps]);
    }
}

// ---------------- Part 2: Lid ----------------
module lid() {
    union() {
        // Base plate, printed flat on the bed
        translate([-box_x / 2 + gap, -box_y / 2, 0])
            cube([box_x - 2 * gap, box_y, lip_t]);
        // Lip that slides into the box opening (centred on the plate)
        translate([-lip_x / 2, -lip_y / 2, lip_t])
            cube([lip_x, lip_y, lip_h]);
    }
}

// ---------------- Assembly ----------------
box();
translate([0, 0, 0])
    lid();
