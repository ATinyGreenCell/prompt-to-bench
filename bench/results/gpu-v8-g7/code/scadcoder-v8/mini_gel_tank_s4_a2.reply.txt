```openscad
// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: open top, resting on XY plane at z = 0

$fn = 64;

// ---- Tank ----
tank_x      = 120;   // outer length (X)
tank_y      = 70;    // outer width (Y)
tank_h      = 40;    // outer height (Z)
wall        = 3;     // wall thickness
floor       = 3;     // floor thickness

// ---- Platform ----
plat_x      = 60;    // platform length in X (centred)
plat_top_z  = 13;    // platform top height above outer base

// ---- Electrode holes ----
hole_d      = 2;     // horizontal hole diameter
hole_y      = 0;     // hole Y position
hole_z      = 30;    // hole Z position

// ---- Misc ----
eps         = 0.5;   // cutter overshoot

// Derived
inner_h = tank_h - floor;   // inner floor height (z = 3)
plat_h  = plat_top_z - floor; // platform height above inner floor (10)

difference() {
    // Tank body
    translate([-tank_x/2, -tank_y/2, 0])
        cube([tank_x, tank_y, tank_h]);

    // Open top
    translate([-tank_x/2, -tank_y/2, tank_h - eps])
        cube([tank_x, tank_y, eps]);

    // Raised platform (spans full inner width)
    translate([-plat_x/2, -tank_y/2, floor])
        cube([plat_x, tank_y, plat_h]);

    // Electrode holes through end walls, along X
    for (sx = [-1, 1])
        translate([sx * (tank_x/2 - hole_d/2), hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = tank_y + 2*eps, center = true);
}
```
