// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: tank rests on XY plane, builds upward (+Z)

$fn = 64;

// ---- Tank dimensions ----
tank_x      = 120;   // outer length in X (mm)
tank_y      = 70;    // outer width in Y (mm)
tank_h      = 40;    // outer height in Z (mm)
wall        = 3;     // wall thickness (mm)
floor       = 3;     // floor thickness (mm)

// ---- Platform dimensions ----
plat_x      = 60;    // platform length in X (mm), centred
plat_top_z  = 10;    // platform top height above tank bottom (mm)

// ---- Electrode holes ----
hole_d      = 2;     // horizontal hole diameter (mm)
hole_y      = 0;     // hole centre Y (mm)
hole_z      = 30;    // hole centre Z (mm)

// ---- Derived ----
inner_h = tank_h - floor;   // inner floor height above tank bottom
eps     = 0.5;              // cutter overshoot past faces

difference() {
    // Tank body (open top)
    translate([-tank_x/2, -tank_y/2, 0])
        cube([tank_x, tank_y, tank_h]);

    // Raised platform, full inner width (Y), centred in X
    translate([-plat_x/2, -tank_y/2, floor])
        cube([plat_x, tank_y, plat_top_z - floor]);

    // Horizontal electrode holes through each end wall (along X)
    for (sx = [-1, 1])
        translate([sx > 0 ? tank_x/2 - eps : -tank_x/2, hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = tank_y + 2*eps);
}
