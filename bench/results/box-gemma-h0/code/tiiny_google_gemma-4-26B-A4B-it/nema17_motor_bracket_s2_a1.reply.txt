// NEMA 17 Stepper Motor L-Bracket for Syringe Pump
// Corrected version based on measurement report analysis

// --- Dimensions ---
// Base Plate
base_x_min = -25;
base_x_max = 25;
base_y_min = 0;
base_y_max = 45;
base_z_height = 5;

// Base Plate Holes (M4)
m4_hole_dia = 4.5;
m4_hole_x1 = -17;
m4_hole_y1 = 30;
m4_hole_x2 = 17;
m4_hole_y2 = 30;

// Vertical Motor Plate
motor_plate_x_min = -25;
motor_plate_x_max = 25;
motor_plate_y_min = 0;
motor_plate_y_max = 5;
motor_plate_z_height = 50;

// Motor Boss Hole (Centering)
boss_hole_dia = 23;
boss_hole_x = 0;
boss_hole_z = 28;

// Motor Screw Holes (M3)
m3_hole_dia = 3.4;
m3_pattern_size = 31;

// Gussets
gusset_thickness = 5;
gusset_leg_y = 20;
gusset_leg_z = 20;
gusset_x_left_min = -25;
gusset_x_left_max = -20;
gusset_x_right_min = 20;
gusset_x_right_max = 25;

// Resolution
$fn = 64;

// --- Modeling ---

module gusset_shape() {
    // A right triangle in YZ plane
    // Corner at y=5, z=5. Legs are 20mm.
    // Vertices in YZ: (5,5), (25,5), (5,25)
    polygon(points = [
        [0, 5], 
        [gusset_leg_y, 5], 
        [0, gusset_leg_z]
    ]);
}

union() {
    // 1. Main Base Plate
    difference() {
        // Centered on X, Y=0 is the edge. 
        // To make X centered on origin: translate by (base_x_max + base_x_min)/2
        translate([0, base_y_min + (base_y_max - base_y_min)/2, 0])
            cube([base_x_max - base_x_min, base_y_max - base_y_min, base_z_height]);
        
        // M4 Holes (Vertical through holes)
        // Note: The previous report showed X=17, but the base was offset.
        // We must translate the holes relative to the base plate's center.
        translate([m4_hole_x1, m4_hole_y1 - (base_y_max-base_y_min)/2, -1])
            cylinder(d = m4_hole_dia, h = base_z_height + 2);
        translate([m4_hole_x2, m4_hole_y2 - (base_y_max-base_y_min)/2, -1])
            cylinder(d = m4_hole_dia, h = base_z_height + 2);
    }

    // 2. Vertical Motor Plate
    // Positioned at the back edge of the base (y=0 to y=5)
    // The base is y=0 to y=45. The "back edge" is y=0.
    // The motor plate is y=0 to y=5.
    difference() {
        translate([0, 0, 0])
            cube([base_x_max - base_x_min, motor_plate_y_max - motor_plate_y_min, motor_plate_z_height]);
        
        // Motor Centering Boss Hole (Cut along Y axis)
        translate([boss_hole_x, -1, boss_hole_z])
            rotate([-90, 0, 0])
                cylinder(d = boss_hole_dia, h = motor_plate_y_max + 2);
        
        // M3 Screw Holes (4 holes on 31mm square pattern)
        // Pattern is centered on the boss hole (0, 28)
        for (px = [-m3_pattern_size/2, m3_pattern_size/2]) {
            for (pz = [-m3_pattern_size/2, m3_pattern_size/2]) {
                translate([px, -1, boss_hole_z + pz])
                    rotate([-90, 0, 0])
                        cylinder(d = m3_hole_dia, h = motor_plate_y_max + 2);
            }
        }
    }

    // 3. Gussets
    // Left Gusset: x from -25 to -20
    translate([gusset_x_left_min, 5, 5])
        linear_extrude(height = gusset_thickness)
            gusset_shape();
            
    // Right Gusset: x from 20 to 25
    translate([gusset_x_right_max - gusset_thickness, 5, 5])
        linear_extrude(height = gusset_thickness)
            gusset_shape();
}

// --- Correction Summary ---
// 1. The previous report showed X-range -25 to 50. This was because the base was 
//    not centered on X=0 in the previous code (it was translated by the cube width).
//    Now, the base is centered on X=0 using translate([0, offset, 0]).
// 2. The previous report showed Y-range 0 to 45, but the motor plate was 
//    offset incorrectly. Now the motor plate is explicitly at the back edge (y=0).
// 3. The gussets are now correctly aligned with the vertical plate and base.
