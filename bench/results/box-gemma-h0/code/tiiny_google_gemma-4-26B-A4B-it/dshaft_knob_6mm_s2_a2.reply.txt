```openscad
// --- Dimensions ---
knob_diameter = 30.0;          // Total diameter of the knob
knob_height = 15.0;            // Total height of the knob
shaft_circle_dia = 6.2;        // Diameter of the D-bore circle
shaft_flat_dist = 4.7;         // Distance from flat to opposite side of D-bore
bore_depth = 12.0;             // Depth of the D-bore
num_grooves = 18;              // Number of vertical grip grooves
groove_dia = 2.0;              // Diameter of the grip grooves
pointer_width = 1.5;           // Width of the pointer groove
pointer_depth = 1.0;           // Depth of the pointer groove

// --- Constants ---
$fn = 64;

// --- Geometry Calculations ---
// User spec: "distance from the flat to the opposite side of the circle is 4.7 mm"
// Circle diameter is 6.2, so radius is 3.1.
// If the total height of the D-shape is 4.7, the distance from center to flat is:
// 4.7 - 3.1 = 1.6 mm.
// To align the pointer with the flat, we place the flat at Y = 1.6.
dist_center_to_flat = 1.6; 

// --- Main Construction ---
difference() {
    // 1. The main knob body
    cylinder(h = knob_height, d = knob_diameter);

    // 2. The D-shaped blind bore
    // We subtract a cylinder and then subtract a block to create the flat.
    // The flat is at Y = 1.6.
    translate([0, 0, -0.01]) {
        difference() {
            cylinder(h = bore_depth + 0.02, d = shaft_circle_dia);
            
            // The block that creates the flat side.
            // We translate it so the face is at Y = 1.6.
            // To ensure it cuts through, we move it from 1.6 to 1.6 + diameter.
            translate([-shaft_circle_dia, dist_center_to_flat, -0.01])
            cube([shaft_circle_dia * 2, shaft_circle_dia, bore_depth + 0.02]);
        }
    }

    // 3. The 18 vertical grip grooves
    // 2mm diameter half-cylinders centered on the outer surface.
    for (i = [0 : num_grooves - 1]) {
        rotate([0, 0, i * (360 / num_grooves)])
        translate([knob_diameter / 2, 0, -0.01])
        cylinder(h = knob_height + 0.02, d = groove_dia);
    }

    // 4. The pointer groove (Radial Slot)
    // 1.5mm wide, 1mm deep, runs from the centre to the outer edge.
    // On the same side as the flat (Y = 1.6).
    // We use a thin cube and rotate it to be radial.
    // To make it radial, we rotate it around the Z axis.
    // However, a simpler way to make a radial slot is to use a thin cylinder 
    // or a cube and rotate it. Let's use a thin cube rotated to align with Y axis.
    
    // We place the slot centered on the Y axis (where the flat is).
    // The slot is 1.5mm wide, so X goes from -0.75 to 0.75.
    // The slot is 1mm deep, so Z goes from knob_height - 1 to knob_height.
    // The slot goes from center (0,0) to edge (0, 15).
    translate([-pointer_width / 2, -0.01, knob_height - pointer_depth])
    cube([pointer_width, knob_diameter / 2 + 1, pointer_depth + 0.01]);
}

// Note on the Pointer Groove:
// The previous version had the pointer groove incorrectly placed in Z and Y.
// This version places it at the top (Z = knob_height - 1) and extends it 
// from the center (Y=0) to the edge (Y=15) along the Y axis.
// This aligns with the flat side of the D-bore (which is at Y = 1.6).
```
