// --- Dimensions ---
knob_diameter = 30.0;          // Total diameter of the knob
knob_height = 15.0;            // Total height of the knob
shaft_circle_dia = 6.2;        // Diameter of the D-bore circle
shaft_flat_dist = 4.7;         // Distance from flat to opposite side of D-bore
bore_depth = 12.0;             // Depth of the D-bore
num_grooves = 18;              // Number of vertical grip grooves
groove_dia = 2.0;              // Diameter of the grip grooves
pointer_width = 1.5;           // Width of the pointer groove
pointer_depth = 1.0;           // Depth of the pointer groove

// --- Constants ---
$fn = 64;

// --- Geometry Calculations ---
// The user specified: distance from flat to opposite side is 4.7mm.
// Circle radius = 6.2 / 2 = 3.1mm.
// Distance from center to flat = 4.7 - 3.1 = 1.6mm.
dist_center_to_flat = shaft_flat_dist - (shaft_circle_dia / 2);

// --- Main Construction ---
difference() {
    // 1. The main knob body
    cylinder(h = knob_height, d = knob_diameter);

    // 2. The D-shaped blind bore
    // We subtract a cylinder and then subtract a block to create the flat.
    // The flat is located at y = dist_center_to_flat.
    // To ensure the flat is "cut", we move the block slightly.
    translate([0, 0, -0.01]) {
        difference() {
            cylinder(h = bore_depth + 0.01, d = shaft_circle_dia);
            
            // The block that creates the flat side
            // We place the flat face at y = dist_center_to_flat
            translate([-shaft_circle_dia, dist_center_to_flat, -0.01])
            cube([shaft_circle_dia * 2, shaft_circle_dia, bore_depth + 0.02]);
        }
    }

    // 3. The 18 vertical grip grooves
    // These are 2mm diameter half-cylinders centered on the outer surface.
    for (i = [0 : num_grooves - 1]) {
        rotate([0, 0, i * (360 / num_grooves)])
        translate([knob_diameter / 2, 0, -0.01])
        cylinder(h = knob_height + 0.02, d = groove_dia);
    }

    // 4. The pointer groove
    // 1.5mm wide, 1mm deep, runs from center to outer edge.
    // Located on the same side as the flat.
    // The flat is at y = 1.6. We center the pointer on the Y axis.
    translate([-pointer_width / 2, knob_height - pointer_depth, 0])
    cube([pointer_width, knob_diameter, pointer_depth + 0.01]);
}
