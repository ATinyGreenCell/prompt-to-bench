// --- Dimensions (mm) ---
collar_diameter = 13.0;      // Diameter of the base collar
collar_thickness = 1.5;      // Height of the base collar
sleeve_outer_diameter = 10.8; // Diameter of the main sleeve body
sleeve_height = 21.5;        // Total height from bottom of collar to top of sleeve
bore_diameter = 6.2;         // Diameter of the central hole

// --- Constants ---
$fn = 64; // Smoothness for cylinders

// --- Implementation ---

// The part is printed "upside down" relative to the user's description 
// of the collar being at z=0.
// The collar rests on the bed (z=0 to 1.5).
// The sleeve body rises from the collar to z=21.5.

module adapter_sleeve() {
    difference() {
        // Main Body (Union of collar and sleeve)
        union() {
            // Collar (Base)
            cylinder(h = collar_thickness, d = collar_diameter);
            
            // Sleeve Body
            // Note: The sleeve rises from the collar. 
            // Total height is 21.5. Since collar is 1.5, sleeve body height is 20.0.
            translate([0, 0, collar_thickness])
                cylinder(h = sleeve_height - collar_thickness, d = sleeve_outer_diameter);
        }

        // Central Bore (The hole running through the full height)
        // We extend the bore slightly beyond the top and bottom to ensure a clean cut.
        translate([0, 0, -1])
            cylinder(h = sleeve_height + 2, d = bore_diameter);
    }
}

// Render the part
adapter_sleeve();
