// Replacement knob for hotplate stirrer - D-shaft fit, grip grooves, and pointer.

// Dimensions
shaft_d = 6.0;
knob_d  = 30.0;
knob_h  = 15.0;
bore_h  = 12.0;
bore_d  = 6.2;
flat_w  = 4.7; // Width of the flat side of the D-bore
groove_n = 18;
groove_d = 2.0;
pointer_w = 1.5;
pointer_d = 1.0; // Depth of pointer groove

$fn = 96;

// Calculate D-bore geometry
// The D-bore is a circle of diameter bore_d, but one side is cut flat.
// The distance from the flat to the opposite arc is flat_w.
// We define the flat side as being at x = -flat_w/2 relative to the center of the circle.
// The circle center is at x = 0. The flat plane is at x = -flat_w/2.
// The circle radius is bore_d / 2.
// The flat width is the chord length at that x position.
// We will model the D-bore as a cylinder with a flat side removed.

module d_bore() {
    // Create the cylindrical bore
    translate([0, 0, -1]) cylinder(h = bore_h + 2, d = bore_d);
    
    // Cut the flat side of the D-bore
    // The flat is a rectangle extending from the bottom of the bore to the top,
    // located at the specific x position where the chord is flat_w wide.
    // We need to find the x-coordinate of the flat plane.
    // Let r = bore_d / 2.
    // Let x_flat be the coordinate of the flat plane.
    // The chord length at x_flat is 2 * sqrt(r^2 - x_flat^2).
    // We want 2 * sqrt(r^2 - x_flat^2) = flat_w.
    // sqrt(r^2 - x_flat^2) = flat_w / 2.
    // r^2 - x_flat^2 = (flat_w / 2)^2.
    // x_flat^2 = r^2 - (flat_w / 2)^2.
    // x_flat = -sqrt(r^2 - (flat_w / 2)^2) (negative because it's on the left side).
    
    r = bore_d / 2;
    half_flat = flat_w / 2;
    x_flat = -sqrt(r * r - half_flat * half_flat);
    
    // The flat cut is a rectangle of width flat_w and height bore_h.
    // It is located at x = x_flat.
    // We extrude a rectangle in the XY plane and rotate it 90 deg around Y to cut along Z.
    // Actually, simpler: translate to x_flat, then difference with a cylinder or box?
    // A box cut is easier.
    translate([x_flat, -flat_w / 2, 0]) cube([flat_w, knob_h + 2, bore_h + 2]);
}

module grip_grooves() {
    // 18 evenly spaced vertical grooves along the full height.
    // Each groove is a 2 mm diameter half-cylinder centred on the outer surface.
    // The outer surface is at radius knob_d / 2.
    // The groove is a half-cylinder, so it cuts into the knob.
    // We model this by rotating a half-cylinder shape around the Z axis.
    
    angle_step = 360 / groove_n;
    
    for (i = [0 : groove_n - 1]) {
        angle = i * angle_step;
        translate([0, 0, -1]) {
            rotate([0, 0, angle]) {
                // Half cylinder: radius = groove_d / 2, height = knob_h + 2
                // The flat face of the half-cylinder should face inward towards the knob center.
                // So the cylinder axis is along Y, and the flat face is at x = -groove_d/2 relative to the cylinder center.
                // Wait, rotate_extrude is better for the groove shape if we want a perfect half-cylinder.
                // But we are inside a difference block.
                // Let's use a cylinder and cut it? No, let's use rotate_extrude on a 2D shape.
                // 2D shape: a rectangle of width groove_d and height knob_h, but only half of it?
                // No, a half-cylinder is generated by rotating a rectangle of width groove_d/2 and height knob_h around one of its edges.
                
                // Let's define the 2D profile for rotate_extrude.
                // The profile is a rectangle of width groove_d/2 and height knob_h.
                // We rotate this around the edge that is at radius knob_d/2.
                
                // Profile points:
                // Start at radius knob_d/2, z = -knob_h/2
                // Go to radius knob_d/2 + groove_d/2, z = -knob_h/2 (outer edge)
                // Go to radius knob_d/2 + groove_d/2, z = knob_h/2
                // Go to radius knob_d/2, z = knob_h/2
                // Close loop.
                
                // Wait, the groove is a "half-cylinder". This means the cross section is a semi-circle.
                // So the profile is a semi-circle of diameter groove_d.
                // The straight edge of the semi-circle is at radius knob_d/2.
                // The curved part goes out to radius knob_d/2 + groove_d/2.
                
                // Let's use rotate_extrude.
                // Profile:
                // [knob_d/2, -knob_h/2]
                // [knob_d/2 + groove_d/2, -knob_h/2]
                // [knob_d/2 + groove_d/2, knob_h/2]
                // [knob_d/2, knob_h/2]
                // This creates a cylinder of diameter groove_d.
                // To make it a half-cylinder, we need the profile to be a semi-circle?
                // No, the user said "2 mm diameter half-cylinder".
                // This usually means the shape of the cut is a semi-cylinder.
                // So the profile is a semi-circle of diameter 2mm.
                // The straight edge is at the surface.
                
                // Let's try a different approach:
                // Create a full cylinder of diameter groove_d, centered at the surface.
                // Then cut it in half? No, that's complex.
                // Just create the half-cylinder directly.
                // A half-cylinder can be made by rotating a rectangle of width groove_d/2 and height knob_h
                // around the edge at radius knob_d/2.
                
                // Profile for rotate_extrude:
                // x is radius, y is height.
                // [knob_d/2, -knob_h/2]
                // [knob_d/2 + groove_d/2, -knob_h/2]
                // [knob_d/2 + groove_d/2, knob_h/2]
                // [knob_d/2, knob_h/2]
                // This is a full cylinder.
                
                // To get a half-cylinder, the profile should be:
                // [knob_d/2, -knob_h/2]
                // [knob_d/2 + groove_d/2, -knob_h/2]
                // [knob_d/2 + groove_d/2, knob_h/2]
                // [knob_d/2, knob_h/2]
