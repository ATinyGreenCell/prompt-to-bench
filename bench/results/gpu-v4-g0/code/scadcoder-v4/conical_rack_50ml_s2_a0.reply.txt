```openscad
// Tube rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, walls rising up (+Z)
// After printing, flip the part over so the walls become the base

// ---------- Parameters ----------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm), z = 0 to 4

hole_d       = 30.5;  // through-hole diameter (mm)
hole_count   = 6;     // total number of holes
hole_row     = hole_count / 2;  // holes per row along Y
hole_pitch   = 40;    // centre-to-centre spacing, both X and Y (mm)

wall_len     = 130;   // wall length along X (mm)
wall_t       = 4;     // wall thickness along Y (mm)
wall_h       = 70;    // wall height from top of plate (mm), z = 4 to 74

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 96;    // circle resolution

// ---------- Derived ----------
wall_top_z   = plate_t + wall_h;   // 74 mm

// ---------- Model ----------
union() {
    // Base plate
    translate([-plate_x/2, -plate_y/2, 0])
        cube([plate_x, plate_y, plate_t]);

    // Two solid walls along the long edges, flush with the plate
    for (s = [-1, 1])
        translate([
            -plate_x/2 + (s > 0 ? plate_y/2 - wall_t : -plate_y/2),
            s > 0 ? -wall_t : plate_y/2 - wall_t,
            0
        ])
            cube([wall_len, wall_t, wall_top_z]);

    // Six through-holes, 2 rows along Y, 3 along X
    for (i = [0 : hole_row - 1])
        for (j = [0 : (hole_count - 1) / 2 - 1]) {
            x = j * hole_pitch - (hole_count - 1) * hole_pitch / 2;
            y = i * hole_pitch - (hole_row - 1) * hole_pitch / 2;
            translate([x, y, -eps])
                cylinder(d = hole_d, h = plate_t + 2 * eps);
        }
}
```
