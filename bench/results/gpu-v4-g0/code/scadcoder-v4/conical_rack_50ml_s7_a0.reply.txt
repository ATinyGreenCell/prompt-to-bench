```openscad
// Tube rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, walls rising up (+Z)
// After printing, flip over so the rack stands on the walls

// ---------- Parameters ----------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm), z = 0 to 4

hole_d       = 30.5;  // through-hole diameter (mm)
hole_nx      = 3;     // holes along X per row
hole_ny      = 2;     // holes along Y (rows)
hole_pitch   = 40;    // centre-to-centre spacing both directions (mm)

wall_t       = 4;     // wall thickness (mm), same as Y thickness
wall_h       = 70;    // wall total height from bed (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 96;    // circle resolution

// ---------- Derived ----------
x0 = -(hole_nx - 1) * hole_pitch / 2;  // first hole X (centres row)
y0 = -(hole_ny - 1) * hole_pitch / 2;  // first hole Y (centres row)

// ---------- Model ----------
union() {
    // Base plate
    translate([-plate_x / 2, -plate_y / 2, 0])
        cube([plate_x, plate_y, plate_t]);

    // Two solid walls along the long edges, flush with plate
    for (s = [-1, 1])
        translate([s > 0 ? plate_x / 2 - wall_t : -plate_x / 2,
                   -plate_y / 2, 0])
            cube([wall_t, plate_y, wall_h]);

    // Through-holes in the plate (walls remain solid)
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([x0 + i * hole_pitch, y0 + j * hole_pitch, -eps])
                cylinder(d = hole_d, h = plate_t + 2 * eps);
}
```
