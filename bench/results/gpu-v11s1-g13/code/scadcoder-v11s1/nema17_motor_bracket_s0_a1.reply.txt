```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate on bed (z = 0), motor plate builds upward (+Z)

// ---------------- Parameters ----------------
base_x0      = -25;   // base plate x start
base_x1      = 25;    // base plate x end
base_y0      = 0;     // base plate y start
base_y1      = 45;    // base plate y end
base_z0      = 0;     // base plate z start
base_z1      = 5;     // base plate z end

motor_x0     = -25;   // motor plate x start
motor_x1     = 25;    // motor plate x end
motor_y0     = 0;     // motor plate y start
motor_y1     = 5;     // motor plate y end
motor_z0     = 0;     // motor plate z start
motor_z1     = 50;    // motor plate z end

hole_m4_x1   = -17;   // M4 hole 1 x
hole_m4_y1   = 30;    // M4 hole 1 y
hole_m4_x2   = 17;    // M4 hole 2 x
hole_m4_y2   = 30;    // M4 hole 2 y
hole_m4      = 4.5;   // M4 hole diameter

boss_d       = 23;    // motor centring boss hole diameter
boss_x       = 0;     // boss hole centre x
boss_z       = 28;    // boss hole centre z

screw_d      = 3.4;   // M3 screw hole diameter
screw_pitch  = 31;    // square pattern side length
screw_count  = 4;     // number of screw holes

gusset_x0    = -25;   // gusset left edge x
gusset_x1    = -20;   // gusset right edge x (left)
gusset_l     = 20;    // triangle leg along +Y
gusset_h     = 20;    // triangle leg along +Z
gusset_t     = 5;     // gusset thickness

overlap      = 0.01;  // small overlap to avoid coplanar faces

$fn = 64;

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([base_x0, base_y0, base_z0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);

        // Motor plate
        translate([motor_x0, motor_y0, motor_z0])
            cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1 - motor_z0]);

        // Gussets (right triangle in YZ, flush with left/right edges)
        for (sx = [-1, 1])
            translate([sx > 0 ? motor_x1 - gusset_t : gusset_x0, 0, 0])
                rotate([90, 0, 0])
                    linear_extrude(height = gusset_t)
                        polygon([[0, 0], [gusset_l, 0], [0, gusset_h]]);
    }

    // M4 through-holes
    for (i = [1, 2])
        translate([i == 1 ? hole_m4_x1 : hole_m4_x2,
                   i == 1 ? hole_m4_y1 : hole_m4_y2,
                   -overlap])
            cylinder(d = hole_m4, h = base_z1 - base_z0 + motor_z1 - motor_z0 + 2 * overlap);

    // Boss hole
    translate([boss_x, 0, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_y1 - motor_y0 + 2 * overlap, center = true);

    // Screw holes (square pattern centred on boss hole)
    for (j = [0 : screw_count - 1]) {
        angle = j * 90;
        translate([boss_x + (j % 2) * screw_pitch / 2,
                   boss_y - (j / 2) * screw_pitch / 2,
                   motor_z0 - overlap])
            rotate([0, 0, angle])
                cylinder(d = screw_d, h = motor_z1 - motor_z0 + 2 * overlap);
    }
}
```
