// Adapter sleeve: 0.2 mL PCR tube for 1.5 mL rack hole
// Print orientation: collar on bed (z = 0), sleeve rises to z = 21.5

$fn = 64;

// --- Dimensions (mm) ---
collar_d      = 13;    // collar diameter
collar_h      = 1.5;   // collar thickness
sleeve_od     = 10.8;  // sleeve outer diameter
sleeve_h      = 20;    // sleeve body height (z = 1.5 to 21.5)
bore_d        = 6.2;   // through-bore diameter
eps           = 0.5;   // cutter overshoot

total_h = collar_h + sleeve_h; // 21.5

difference() {
    union() {
        // Collar
        cylinder(d = collar_d, h = collar_h);
        // Sleeve body
        translate([0, 0, collar_h])
            cylinder(d = sleeve_od, h = sleeve_h);
    }
    // Through bore
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = total_h + 2 * eps);
}
