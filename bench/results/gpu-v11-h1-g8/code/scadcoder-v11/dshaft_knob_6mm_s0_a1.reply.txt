```openscad
// Hotplate stirrer knob - D-shaft adapter
// Print orientation: bottom face (with D-bore) on the bed at z = 0

$fn = 96;

// ---- Dimensions (mm) ----
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
shaft_d       = 6;     // D-shaft diameter
bore_depth    = 12;    // depth of D-blind bore from bottom face
bore_d        = 6.2;   // bore circle diameter
bore_flat     = 4.7;   // distance from flat to opposite side of bore circle
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // groove half-cylinder diameter
pointer_w     = 1.5;   // pointer groove width
pointer_d     = 1;     // pointer groove depth
eps           = 0.5;   // overshoot for clean boolean cuts

// ---- Derived ----
bore_r = bore_d / 2;
// x is radius, y is height for the outline of rotated shapes
outline = [
    [bore_r, -bore_flat / 2],
    [bore_r, bore_flat / 2]
    // the circle arc between these points is drawn counter-clockwise,
    // so the flat lies on the -x side (90 deg / top)
];

difference() {
    // Main knob body
    cylinder(d = knob_d, h = knob_h);

    // D-blind bore
    translate([0, 0, -eps])
        rotate_extrude($fn = 64)
            outline;

    // Vertical grip grooves, evenly spaced along the full height
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 64);

    // Pointer groove on the top face, same side as the bore flat (+y)
    translate([0, pointer_w, knob_h - pointer_d])
        cube([knob_d / 2, pointer_w, pointer_d + eps]);
}
```
