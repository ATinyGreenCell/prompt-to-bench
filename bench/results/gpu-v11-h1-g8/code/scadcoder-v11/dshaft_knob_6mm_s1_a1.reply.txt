// Hotplate stirrer knob for 6 mm D-shaft
// Print orientation: bottom face (with D-bore) on the bed at z = 0

$fn = 96;

// ---- Dimensions (mm) ----
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
shaft_d       = 6;     // D-shaft diameter
bore_depth    = 12;    // depth of D-blind bore from bottom face
bore_d        = 6.2;   // bore circle diameter
bore_flat     = bore_d - 4.7;  // chord length of flat (1.5 mm thick)
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // groove cylinder diameter
pointer_w     = 1.5;   // pointer groove width
pointer_d     = 1;     // pointer groove depth
eps           = 0.5;   // overcut for clean booleans

// ---- Model ----
difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-blind bore, centred on origin, flat on +y side
    translate([0, bore_flat / 2, -eps])
        intersection() {
            cylinder(d = bore_d, h = bore_depth + eps);
            translate([-bore_d / 2 - eps, -bore_d / 2 - eps, 0])
                cube([bore_d + 2 * eps, bore_flat / 2 + eps, bore_depth + eps]);
        };

    // Vertical grip grooves, evenly spaced, centred on outer surface
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 64);

    // Pointer groove from centre to outer edge, on +y side
    translate([0, pointer_w / 2, knob_h - pointer_d])
        rotate([90, 0, 0])
            cylinder(d = pointer_w, h = pointer_d + eps, $fn = 64);
}
