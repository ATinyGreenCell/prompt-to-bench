```openscad
// Slide drying rack - holds 10 microscope slides standing upright on their long edge
// Print orientation: base block on the bed, slots open upward (+Z)

// ---------------- Parameters ----------------
base_len    = 90;    // block length along X (mm)
base_wid    = 45;    // block width along Y (mm)
base_ht     = 20;    // block height along Z (mm)

slot_len    = 77;    // slot length along X (mm)
slot_wid    = 1.6;   // slot width along Y (mm)
slot_depth  = 10;    // slot depth from top face (mm)

slot_count  = 10;    // number of slots
slot_pitch  = 3.5;   // centre-to-centre spacing of slots in Y (mm)

eps         = 0.5;   // overcut so cutters pass through the top face (mm)

// Derived
first_y = -(slot_count - 1) * slot_pitch / 2;  // Y of first slot, groups centred on block

// ---------------- Model ----------------
difference() {
    // Base block, centred in X/Y, sitting on z = 0
    translate([-base_len / 2, -base_wid / 2, 0])
        cube([base_len, base_wid, base_ht]);

    // Slots, open at the top
    for (i = [0 : slot_count - 1]) {
        translate([0, first_y + i * slot_pitch, base_ht - slot_depth])
            cube([slot_len, slot_wid, slot_depth + eps]);
    }
}
```
