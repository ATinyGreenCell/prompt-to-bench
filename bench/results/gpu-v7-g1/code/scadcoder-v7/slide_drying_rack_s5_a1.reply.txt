// Slide drying rack for microscope slides (75 x 25 x 1 mm)
// Print orientation: base on XY plane, slots open upward (+Z)

// ---- Block dimensions ----
block_len   = 90;    // block length along X (mm)
block_wid   = 45;    // block width along Y (mm)
block_ht    = 20;    // block height along Z (mm)

// ---- Slot dimensions ----
slot_count  = 10;    // number of slide slots
slot_len    = 77;    // slot length along X (mm)
slot_wid    = 1.6;   // slot width along Y (mm)
slot_depth  = 10;    // slot depth from top face (mm)
slot_pitch  = 3.5;   // centre-to-centre spacing in Y (mm)

// ---- Misc ----
eps         = 0.5;   // overshoot for clean cuts (mm)
$fn         = 64;    // circle resolution

// X offset of the first slot so the row is centred on the block
first_slot_x = -(slot_count - 1) * slot_pitch / 2;

difference() {
    // Base block, centred in X/Y, resting on z = 0
    translate([-block_len / 2, -block_wid / 2, 0])
        cube([block_len, block_wid, block_ht]);

    // Slots, open at the top
    for (i = [0 : slot_count - 1]) {
        translate([first_slot_x + i * slot_pitch - slot_len / 2,
                   -block_wid / 2 + slot_pitch / 2,
                   block_ht - slot_depth])
            cube([slot_len, slot_wid, slot_depth + eps]);
    }
}
