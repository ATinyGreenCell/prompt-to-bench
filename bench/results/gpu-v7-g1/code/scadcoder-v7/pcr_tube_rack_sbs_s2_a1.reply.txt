```openscad
// 96-well PCR tube rack (0.2 mL, standard microplate footprint)
// Print orientation: base on XY plane, builds upward (+Z)

// ---------- Block ----------
block_len   = 127.76;   // block length along X (mm)
block_wid   = 85.48;    // block width along Y (mm)
block_ht    = 20;       // block height along Z (mm)

// ---------- Hole grid ----------
hole_d      = 6.2;      // through-hole diameter (mm)
hole_nx     = 12;       // number of columns along X
hole_ny     = 8;        // number of rows along Y
hole_pitch  = 9.0;      // centre-to-centre spacing (mm)
hole_start_x = 14.38;   // centre of A1 from left edge (mm)
hole_start_y = 11.24;   // centre of A1 from back edge (mm)

// ---------- Chamfer (orientation mark) ----------
chamfer_leg = 5;        // chamfer leg length along X and Y (mm)
chamfer_ang = 45;       // chamfer angle (degrees)

// ---------- Misc ----------
eps = 0.5;              // cutter overshoot past faces (mm)
$fn = 64;               // curve resolution

// Derived: grid centre offsets so the grid is centred on the block
grid_x0 = hole_start_x - (hole_nx - 1) * hole_pitch / 2;
grid_y0 = hole_start_y - (hole_ny - 1) * hole_pitch / 2;

difference() {
    // Main block
    translate([-block_len / 2, -block_wid / 2, 0])
        cube([block_len, block_wid, block_ht]);

    // Through-holes
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([grid_x0 + i * hole_pitch,
                       grid_y0 + j * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = block_ht + 2 * eps);

    // Chamfer off the A1 corner (vertical edge, 45 deg, 5 mm legs)
    translate([-block_len / 2 + chamfer_leg,
               -block_wid / 2 + chamfer_leg,
               0])
        linear_extrude(height = block_ht)
            polygon([[-chamfer_leg, -chamfer_leg],
                     [chamfer_leg, -chamfer_leg],
                     [chamfer_leg, chamfer_leg],
                     [-chamfer_leg, chamfer_leg]]);
}
```
