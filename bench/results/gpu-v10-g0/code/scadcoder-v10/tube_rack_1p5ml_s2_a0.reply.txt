```openscad
// Bench rack for 1.5 mL microcentrifuge tubes
// Print orientation: base on XY plane, holes open upward (+Z)

// ---------------- Parameters ----------------
block_len    = 106;   // block length along X (mm)
block_wid    = 72;    // block width along Y (mm)
block_h      = 30;    // block height along Z (mm)

hole_count   = 24;    // total number of tube holes
hole_rows    = 4;     // number of hole rows along Y
hole_cols    = 6;     // number of holes per row along X
hole_pitch   = 16;    // centre-to-centre hole spacing, both directions (mm)

hole_d       = 11.2;  // hole diameter (mm)
hole_depth   = 25;    // hole depth from top face (mm)
floor_t      = block_h - hole_depth; // remaining solid floor thickness = 5 mm

eps          = 0.5;   // overshoot for clean boolean cuts (mm)
$fn          = 96;    // circle resolution

// ---------------- Model ----------------
difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_h]);

    // Hole grid, centred on the block
    for (i = [0 : hole_rows - 1])
        for (j = [0 : hole_cols - 1])
            translate([(j - (hole_cols - 1)/2) * hole_pitch,
                       (i - (hole_rows - 1)/2) * hole_pitch,
                       floor_t])
                cylinder(d = hole_d, h = hole_depth + eps);
}
```
