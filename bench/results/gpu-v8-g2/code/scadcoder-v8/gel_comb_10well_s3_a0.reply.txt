```openscad
// 10-well agarose gel comb, printed lying flat
// Units: mm

// ---------- Parameters ----------
plate_t      = 1.5;    // comb plate thickness (Z)
spine_len    = 70;     // spine (back bar) length (X)
spine_w      = 12;     // spine width (Y)
tooth_count  = 10;     // number of teeth
tooth_w      = 5.0;    // tooth width (X)
tooth_len    = 12;     // tooth length from spine edge (Y)
tooth_gap    = 1.5;    // gap between neighbouring teeth (X)
eps          = 0.5;    // cutter overshoot past faces (Z)

$fn = 64;

// ---------- Derived ----------
pitch   = tooth_w + tooth_gap;            // centre-to-centre tooth spacing
x_start = -(tooth_count - 1) * pitch / 2; // first tooth centre X (row centred)

// ---------- Model ----------
difference() {
    // Base plate
    translate([-spine_len / 2, -spine_w / 2, 0])
        cube([spine_len, spine_w, plate_t]);

    // Teeth cut out from the -Y edge of the spine
    for (i = [0 : tooth_count - 1])
        translate([x_start + i * pitch - tooth_w / 2,
                   -spine_w / 2 - tooth_len,
                   -eps])
            cube([tooth_w, tooth_len + eps, plate_t + 2 * eps]);
}
```
