```openscad
// 10-well agarose gel comb, printed lying flat
// Units: mm

// ---------- Parameters ----------
plate_t      = 1.5;    // comb plate thickness (Z)
spine_len    = 70;     // spine length along X
spine_w      = 12;     // spine width along Y
tooth_count  = 10;     // number of teeth
tooth_w      = 5.0;    // tooth width along X
tooth_len    = 12;     // tooth length along Y (from spine edge)
tooth_gap    = 1.5;    // gap between neighbouring teeth
cut_extra    = 0.5;    // overshoot for clean boolean cuts

$fn = 64;

// ---------- Derived ----------
spine_x0 = -spine_len / 2;              // spine start in X
tooth_x0 = spine_x0 + (spine_len - tooth_count * tooth_w) / 2; // first tooth X
pitch    = tooth_w + tooth_gap;         // centre-to-centre spacing

// ---------- Model ----------
difference() {
    union() {
        // Spine (back bar)
        translate([spine_x0, -spine_w / 2, 0])
            cube([spine_len, spine_w, plate_t]);

        // Teeth hanging in -Y from the spine
        for (i = [0 : tooth_count - 1])
            translate([tooth_x0 + i * pitch, -spine_w / 2, 0])
                cube([tooth_w, tooth_len, plate_t]);
    }

    // 10 rectangular wells cut from the plate, running along X
    for (i = [0 : tooth_count - 1])
        translate([tooth_x0 + i * pitch - tooth_w / 2,
                   -spine_w / 2 - cut_extra,
                   -cut_extra])
            cube([tooth_w, tooth_len + cut_extra, plate_t + 2 * cut_extra]);
}
```
