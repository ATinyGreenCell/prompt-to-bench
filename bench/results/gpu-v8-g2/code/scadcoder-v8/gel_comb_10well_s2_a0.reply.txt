// 10-well agarose gel comb, printed flat
// Units: mm

// ---- Spine ----
spine_len   = 70;   // spine length along X
spine_wid   = 12;   // spine width along Y
spine_thk   = 1.5;  // spine thickness along Z

// ---- Teeth ----
tooth_count = 10;   // number of teeth
tooth_w     = 5.0;  // tooth width along X
tooth_l     = 12;   // tooth length along Y (hangs from -Y edge)
tooth_gap   = 1.5;  // gap between neighbouring teeth along X

// ---- Misc ----
eps = 0.5;          // cutter overshoot past faces

$fn = 64;

// Offset so the tooth row is centred on the spine in X
row_start = -(tooth_count - 1) * tooth_w / 2 - (tooth_count - 2) * tooth_gap / 2;

difference() {
    union() {
        // Spine (back bar)
        translate([-spine_len / 2, 0, 0])
            cube([spine_len, spine_wid, spine_thk]);

        // Teeth hanging from the -Y edge
        for (i = [0 : tooth_count - 1])
            translate([row_start + i * (tooth_w + tooth_gap), -tooth_l, 0])
                cube([tooth_w, tooth_l, spine_thk]);
    }

    // Cut out the wells (through the full thickness)
    for (i = [0 : tooth_count - 1])
        translate([row_start + i * (tooth_w + tooth_gap) - tooth_w / 2,
                   -tooth_l + eps,
                   -eps])
            cube([tooth_w, tooth_l + 2 * eps, spine_thk + 2 * eps]);
}
