```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on the bed, walls rising in +Z.
// The rack is printed upside down and flipped over after printing.

// ---------------- Parameters ----------------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm), z = 0..4

hole_d       = 30.5;  // tube hole diameter (mm)
hole_nx      = 3;     // holes per row along X
hole_ny      = 2;     // number of rows along Y
hole_pitch   = 40;    // centre-to-centre spacing, both directions (mm)

wall_h       = 70;    // wall height above plate (mm), z = 4..74
wall_t       = 4;     // wall thickness (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 128;   // circle resolution

// ---------------- Derived ----------------
wall_x0 = -plate_x / 2 + wall_t;   // inner wall X start (-63)
wall_x1 = plate_x / 2 - wall_t;    // outer wall X end (63)

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([-plate_x / 2, -plate_y / 2, 0])
            cube([plate_x, plate_y, plate_t]);

        // Two solid walls along the long edges
        for (sx = [-1, 1])
            translate([sx * wall_x0, -plate_y / 2, plate_t])
                cube([wall_x1 - wall_x0 + 2 * wall_t,
                      wall_t,
                      wall_h]);
    }

    // Six through-holes in a 2 x 3 grid
    for (i = [0 : hole_ny - 1])
        for (j = [0 : hole_nx - 1])
            translate([(j - (hole_nx - 1) / 2) * hole_pitch,
                       (i - (hole_ny - 1) / 2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2 * eps);
}
```
