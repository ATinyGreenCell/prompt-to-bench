// Replacement knob for hotplate stirrer with D-shaft
// Print orientation: side facing hotplate down on bed (z = 0)

// ---------------- Parameters ----------------
knob_d          = 30;    // knob outer diameter
knob_h          = 15;    // knob height
shaft_d         = 6;     // D-shaft diameter
bore_depth      = 12;    // blind bore depth from bottom face
bore_flat       = 4.7;   // distance from D-bore flat to opposite side
bore_r          = bore_d / 2;  // bore radius (6.2 mm)
groove_count    = 18;    // number of grip grooves
groove_d        = 2;     // grip groove diameter
pointer_width   = 1.5;   // pointer groove width
pointer_depth   = 1;     // pointer groove depth
eps             = 0.5;   // cutter overshoot past faces
fn_curve        = 96;    // facets for curved shapes

// Derived
bore_d = 6.2;            // bore diameter from spec
pitch  = 360 / groove_count;  // angle between groove centres

// ---------------- Model ----------------
difference() {
    // Main knob body
    cylinder(d = knob_d, h = knob_h, $fn = fn_curve);

    // D-shaped blind bore from bottom face
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = bore_depth + 2 * eps, $fn = fn_curve);

    // Flat on D-bore, cut from bottom face
    translate([-(bore_d - bore_flat) / 2, 0, -eps])
        cube([bore_d - bore_flat, bore_d, bore_depth + 2 * eps]);

    // Vertical grip grooves along full height
    for (i = [0 : groove_count - 1]) {
        rotate([0, 0, i * pitch])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 16);
    }

    // Pointer groove on top face, same side as D-bore flat
    translate([-(bore_d - bore_flat) / 2, 0, knob_h - pointer_depth])
        cube([pointer_width, knob_d, pointer_depth + eps]);
}
