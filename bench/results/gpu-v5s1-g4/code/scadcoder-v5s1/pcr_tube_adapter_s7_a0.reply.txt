// PCR tube adapter sleeve - fits a 0.2 mL tube into a 1.5 mL rack hole
// Print orientation: collar on the bed (z = 0), sleeve body builds upward.

// ---------- Parameters ----------
collar_d        = 13;     // collar outer diameter (mm)
collar_t        = 1.5;    // collar thickness (mm)
sleeve_d        = 10.8;   // sleeve outer diameter (mm)
sleeve_h        = 20;     // sleeve body height above collar (mm)
bore_d          = 6.2;    // through-bore diameter (mm)
eps             = 0.5;    // cutter overshoot past faces (mm)

$fn = 96;

// ---------- Model ----------
total_h = collar_t + sleeve_h;   // 21.5 mm overall height

difference() {
    union() {
        // Collar: z = 0 to 1.5, centred on origin in XY
        cylinder(d = collar_d, h = collar_t);

        // Sleeve body: z = 1.5 to 21.5, centred on origin in XY
        translate([0, 0, collar_t])
            cylinder(d = sleeve_d, h = sleeve_h);
    }

    // Straight through-bore, extends past both end faces
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = total_h + 2 * eps);
}
