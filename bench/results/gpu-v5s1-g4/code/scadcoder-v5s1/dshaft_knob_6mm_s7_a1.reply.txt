// Replacement knob for hotplate stirrer with D-shaft
// Print orientation: side facing hotplate down on bed (z = 0)

// ---------------- Parameters ----------------
knob_d          = 30;    // knob outer diameter
knob_h          = 15;    // knob height
shaft_d         = 6;     // D-shaft diameter
bore_depth      = 12;    // blind bore depth from bottom face
bore_flat       = 4.7;   // distance from D-bore flat to opposite side
bore_r          = 3.1;   // bore radius (6.2 mm / 2)
groove_count    = 18;    // number of vertical grip grooves
groove_d        = 2;     // groove diameter (half-cylinder)
pointer_width   = 1.5;   // pointer groove width
pointer_depth   = 1;     // pointer groove depth
eps             = 0.5;   // overcut for clean differences

$fn = 96;

// ---------------- Model ----------------
difference() {
    // Main cylinder
    translate([0, 0, 0])
        cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, -eps])
        linear_extrude(height = bore_depth + 2 * eps)
            circle(r = bore_r, $fn = 64);   // circle with one side cut flat

    // Vertical grip grooves along full height, centred on outer surface
    for (i = [0 : groove_count - 1]) {
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 16);
    }

    // Pointer groove on top face, from centre to outer edge
    // (placed on the same side as the flat of the D-bore)
    translate([0, 0, knob_h - pointer_depth])
        linear_extrude(height = pointer_depth + eps)
            translate([-knob_d / 2, -pointer_width / 2, 0])
                cube([knob_d, pointer_width, eps]);
}
