// Replacement knob for a hotplate stirrer with a 6 mm D-shaft
// Print orientation: bottom face (facing the hotplate) on the bed at z = 0

// ---------------- Parameters ----------------
knob_d      = 30;    // knob outer diameter (mm)
knob_h      = 15;    // knob height (mm)

bore_depth  = 12;    // D-bore depth from bottom face (mm)
bore_w      = 4.7;   // D-bore width: flat to opposite side of circle (mm)
bore_d      = 6.2;   // diameter of the circle the D-bore is based on (mm)

groove_n    = 18;    // number of vertical grip grooves
groove_d    = 2;     // grip groove diameter (mm)

pointer_w   = 1.5;   // pointer groove width (mm)
pointer_d   = 1;     // pointer groove depth (mm)

eps         = 0.5;   // overcut so cutting shapes pass beyond faces (mm)

$fn = 128;           // resolution for curved surfaces

// ---------------- Derived ----------------
// D-bore geometry: circle of diameter bore_d with one side cut flat.
// The distance from the flat to the circle centre is (bore_d - bore_w)/2.
bore_offset = (bore_d - bore_w) / 2;   // distance from flat to circle centre

// ---------------- Model ----------------
difference() {
    // Main body: cylinder, centred in X/Y, sitting on z = 0
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from the bottom face, running up to bore_depth
    translate([0, 0, -eps])
        cylinder(r = bore_d / 2, h = bore_depth + 2 * eps, $fn = 64);
    // Remove the flat side of the D (extends past the top of the bore)
    translate([bore_offset, 0, -eps])
        cube([bore_w / 2 + 2 * eps, knob_h + 2 * eps, 2 * bore_offset]);

    // 18 evenly spaced vertical grip grooves along the full height
    for (i = [0 : groove_n - 1])
        rotate([0, 0, i * 360 / groove_n])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 16);

    // Pointer groove on the top face, from the centre to the outer edge,
    // on the same side as the flat of the D
    translate([-bore_offset, 0, knob_h - pointer_d])
        linear_extrude(height = pointer_w)
            circle(r = knob_d / 2, $fn = 64);
}
