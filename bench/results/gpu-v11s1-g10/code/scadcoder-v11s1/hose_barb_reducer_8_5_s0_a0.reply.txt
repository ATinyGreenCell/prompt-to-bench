```openscad
// Hose-barb reducer: 8 mm ID -> 5 mm ID
// Print orientation: standing upright along +Z

$fn = 64;

// ---- Dimensions (mm) ----
id_8     = 8;    // inner diameter of 8 mm tubing
id_5     = 5;    // inner diameter of 5 mm tubing
bore     = 3;    // through-bore diameter
length_8 = 20;   // height of 8 mm tubing section
length_c = 5;    // height of collar section
length_5 = 16;   // height of 5 mm tubing section (z = 25 to z = 41)
eps      = 0.5;  // overcut for clean differences

// ---- Derived ----
od_8_top = 9.5;  // outer diameter at top of 8 mm barbs
od_5_bot = 6.5;  // outer diameter at bottom of 5 mm barbs

module hose_barb(id, height, od_top, od_bot) {
    difference() {
        union() {
            // Sawtooth profile, stacked twice
            for (s = [0, 1])
                translate([0, 0, s * height])
                    linear_extrude(height = height)
                        polygon([
                            [-(od_bot + eps) / 2, 0],
                            [-(od_top + eps) / 2, height],
                            [(od_top + eps) / 2, height],
                            [(od_bot + eps) / 2, 0]
                        ]);
        }
        // Through bore
        translate([0, 0, -eps])
            cylinder(d = id, h = height + 2 * eps);
    }
}

difference() {
    union() {
        // 8 mm tubing section (z = 0 to 20)
        hose_barb(id_8, length_8, od_8_top, od_8_bot = 7.5);

        // Collar (z = 20 to 25)
        translate([0, 0, length_8])
            cylinder(d = 12, h = length_c);

        // 5 mm tubing section (z = 25 to 41)
        hose_barb(id_5, length_5, od_5_bot, 5.0);
    }

    // Through bore along full length
    translate([0, 0, -eps])
        cylinder(d = bore, h = length_8 + length_c + length_5 + 2 * eps);
}
```
