```openscad
// Hose-barb reducer: 8 mm ID -> 5 mm ID
// Print orientation: standing upright along +Z

$fn = 64;

// ---- Dimensions (mm) ----
tube_id_8     = 8;    // 8 mm tubing inner diameter
tube_id_5     = 5;    // 5 mm tubing inner diameter
bore          = 3;    // through-bore diameter
collar_d      = 12;   // reducer collar diameter (z = 20..25)

 Barb_h_8     = 10;   // 8 mm tubing barb height
 Barb_od_8_b  = 7.5;  // 8 mm barb outer diameter (bottom)
 Barb_od_8_t  = 9.5;  // 8 mm barb outer diameter (top)

 Barb_h_5     = 8;    // 5 mm tubing barb height
 Barb_od_5_b  = 6.5;  // 5 mm barb outer diameter (bottom)
 Barb_od_5_t  = 5.0;  // 5 mm barb outer diameter (top)

eps           = 0.5;  // cutter overshoot

// ---- Derived ----
z0 = 0;
z8 = z0 + Barb_h_8;              // 10
zC = z8 + 5;                      // 15 (collar start)
z2 = zC + collar_d;               // 27 (collar end)
z5 = z2 + Barb_h_5;               // 35
zT = z5 + Barb_h_8;               // 45 (top of second 8 mm barb)

// ---- Model ----
difference() {
    union() {
        // 8 mm tubing: two identical barbs, stacked
        for (s = [-1, 1])
            translate([0, 0, z0 + s * Barb_h_8 / 2])
                linear_extrude(height = eps)
                    polygon([
                        [0, -Barb_od_8_b / 2],
                        [Barb_od_8_t / 2, Barb_od_8_t / 2],
                        [Barb_od_8_b / 2, Barb_od_8_b / 2]
                    ]);

        // 5 mm tubing: two identical barbs, stacked
        for (s = [-1, 1])
            translate([0, 0, z5 + s * Barb_h_5 / 2])
                linear_extrude(height = eps)
                    polygon([
                        [0, -Barb_od_5_b / 2],
                        [Barb_od_5_t / 2, Barb_od_5_t / 2],
                        [Barb_od_5_b / 2, Barb_od_5_b / 2]
                    ]);

        // Collar: z = 20 to z = 25 (sits between the two tube sizes)
        translate([0, 0, z8])
            cylinder(d = collar_d, h = 5);
    }

    // Through-bore along Z
    translate([0, 0, -eps])
        cylinder(d = bore, h = zT + 2 * eps);
}
```
