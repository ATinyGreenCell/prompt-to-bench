// Straight hose-barb reducer: 8 mm tubing (z = 0..20) -> 5 mm tubing (z = 25..41)
// Print orientation: standing upright along +Z, centred on origin in X/Y

$fn = 96;

// ---- 8 mm tubing section ----
barb_h_8      = 10;   // barb height
od_8_bottom   = 7.5;  // outer diameter at bottom of 8 mm barbs
od_8_top      = 9.5;  // outer diameter at top of 8 mm barbs
bore          = 3;    // through-bore diameter

// ---- 5 mm tubing section ----
barb_h_5      = 8;    // barb height
od_5_bottom   = 6.5;  // outer diameter at bottom of 5 mm barbs
od_5_top      = 5.0;  // outer diameter at top of 5 mm barbs

// ---- Collar ----
collar_d      = 12;   // collar diameter
collar_top_z  = 25;   // collar top height (also start of 5 mm barbs)

// ---- Misc ----
eps           = 0.5;  // overlap for clean booleans / steps
n_barbs       = 2;    // stacked barbs per section

// Barb profile (sawtooth): grow from bottom OD to top OD, then step back to bottom OD
module barb_profile(h, d_bottom, d_top) {
    r_bottom = d_bottom / 2;
    r_top    = d_top / 2;
    // 1.5 rad (~86 deg) of growth, then a straight drop back to r_bottom
    hull() {
        for (a = [0, 90])
            rotate([0, 0, a])
                translate([r_bottom, 0, 0])
                    cylinder(h = h, r = r_bottom);
        for (a = [0, 90])
            rotate([0, 0, a])
                translate([r_top, 0, 0])
                    cylinder(h = h, r = r_top);
    }
}

union() {
    // 8 mm tubing section (z = 0..20)
    for (i = [0 : n_barbs - 1])
        translate([0, 0, i * barb_h_8])
            difference() {
                barb_profile(barb_h_8, od_8_bottom, od_8_top);
                translate([0, 0, -eps])
                    cylinder(h = barb_h_8 + 2 * eps, d = bore);
            }

    // Collar (z = 20..25)
    translate([0, 0, collar_top_z - eps])
        cylinder(h = eps + 5, d = collar_d);

    // 5 mm tubing section (z = 25..41)
    for (i = [0 : n_barbs - 1])
        translate([0, 0, collar_top_z + i * barb_h_5])
            difference() {
                barb_profile(barb_h_5, od_5_bottom, od_5_top);
                translate([0, 0, -eps])
                    cylinder(h = barb_h_5 + 2 * eps, d = bore);
            }
}
