```openscad
// 10-well comb for agarose gel casting, printed lying flat
// Units: mm

// ---- Spine (back bar) ----
spine_len   = 70;    // spine length along X
spine_wid   = 12;    // spine width along Y

// ---- Teeth ----
tooth_count = 10;    // number of wells
tooth_w     = 5.0;   // tooth width along X
tooth_len   = 12;    // tooth length along Y (hangs from -Y edge)
gap         = 1.5;   // gap between neighbouring teeth

// ---- Misc ----
eps         = 0.5;   // overcut for clean differences

$fn = 64;

pitch = tooth_w + gap;   // centre-to-centre spacing of teeth

difference() {
    // Spine plate
    translate([-spine_len/2, 0, 0])
        cube([spine_len, spine_wid, 1.5]);

    // Teeth, row centred on spine in X
    for (i = [0 : tooth_count - 1]) {
        x = (i - (tooth_count - 1)/2) * pitch;
        translate([x - tooth_w/2, -tooth_len, 0])
            cube([tooth_w, tooth_len + eps, 1.5]);
    }
}
```
