// Tissue-grinding pestle for a 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the XY plane, z = 0 at the bottom.

// ---------------- Parameters ----------------
handle_d      = 8;    // handle diameter (mm)
handle_h      = 45;   // handle height (mm), z = 0 to 45
cone_start_d  = 8;    // cone diameter at the handle top (mm)
cone_end_d    = 3;    // cone diameter at the tip base (mm)
cone_top_z    = 62;   // cone top z (mm), cone spans z = 45 to 62
tip_r         = 1.5;  // rounded tip radius (mm), sphere centred at z = 62
groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width in Z (mm)
groove_d      = 1;    // groove depth (mm)
groove_z      = [10, 15, 20, 25]; // groove centre z positions (mm)
eps           = 0.5;  // cutter overshoot past faces (mm)

$fn = 96;

// ---------------- Model ----------------
module pestle() {
    difference() {
        union() {
            // (1) Cylindrical handle
            cylinder(d = handle_d, h = handle_h);

            // (2) Tapering cone: 8 mm at z = 45 to 3 mm at z = 62
            translate([0, 0, handle_h])
                cylinder(d1 = cone_start_d, d2 = cone_end_d, h = cone_top_z - handle_h);

            // (3) Rounded tip: 3 mm diameter sphere centred at z = 62
            translate([0, 0, cone_top_z])
                sphere(r = tip_r);
        }

        // Grip grooves: circumferential, 2 mm wide in Z, 1 mm deep
        for (i = [0 : groove_count - 1]) {
            rotate([0, 0, i * 360 / groove_count])
                translate([0, -1 * groove_d, groove_z[i]])
                    cube([handle_d + 2 * eps, 2 * groove_d, groove_w + eps]);
        }
    }
}

pestle();
