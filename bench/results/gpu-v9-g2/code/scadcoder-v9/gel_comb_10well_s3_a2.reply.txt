```openscad
// 10-well comb for casting agarose gels
// Print orientation: flat on bed, teeth pointing up (+Z)

// ---------- Parameters ----------
plate_len    = 70;    // spine length along X (mm)
plate_wid    = 12;    // spine width along Y (mm)
plate_t      = 1.5;   // plate thickness along Z (mm)

tooth_count  = 10;    // number of wells
tooth_w      = 5.0;   // tooth width along X (mm)
tooth_len    = 12;    // tooth length along Y (mm)
tooth_h      = 12;    // tooth height above plate (Z) (mm)
tooth_pitch  = 1.5;   // gap between neighbouring teeth (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 64;    // curve resolution

// ---------- Derived ----------
row_span    = (tooth_count - 1) * tooth_pitch;  // total gap width
x_start     = -(plate_len / 2 + row_span / 2) + tooth_w / 2;  // first tooth centre X

// ---------- Model ----------
union() {
    // Spine (back bar)
    translate([-plate_len / 2, 0, 0])
        cube([plate_len, plate_wid, plate_t]);

    // Teeth
    for (i = [0 : tooth_count - 1])
        translate([x_start + i * (tooth_w + tooth_pitch),
                   plate_wid,
                   0])
            cube([tooth_w, tooth_len, tooth_h]);
}

// ---------- Cut wells ----------
difference() {
    // Comb body
    union() {
        translate([-plate_len / 2, 0, 0])
            cube([plate_len, plate_wid, plate_t]);
        for (i = [0 : tooth_count - 1])
            translate([x_start + i * (tooth_w + tooth_pitch),
                       plate_wid,
                       0])
                cube([tooth_w, tooth_len, tooth_h]);
    }

    // Well cutters, extend past top face
    for (i = [0 : tooth_count - 1])
        translate([x_start + i * (tooth_w + tooth_pitch) - eps,
                   plate_wid + eps,
                   -eps])
            cube([tooth_w + 2 * eps, tooth_len + eps, tooth_h + 2 * eps]);
}
```
