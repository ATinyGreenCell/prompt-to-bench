```openscad
// Mini horizontal gel-electrophoresis box: open-top rectangular tank with a central
// raised gel-tray platform that divides the bottom into two buffer chambers.
// Print orientation: tank rests on the XY plane (z = 0) and builds upward (+Z).

$fn = 64;

// ---------------- Dimensions (mm) ----------------
outer_x      = 120;   // tank outer length (X)
outer_y      = 70;    // tank outer width (Y)
outer_h      = 40;    // tank outer height (Z)

wall         = 3;     // tank wall thickness
floor        = 3;     // tank floor thickness

platform_x   = 60;    // platform length (X), centred
platform_w   = outer_y; // platform spans full inner width (Y)
platform_top = 13;    // platform top height (Z) = inner floor (3) + 10

hole_d       = 2;     // electrode hole diameter
hole_y       = 0;     // electrode hole Y position
hole_z       = 30;    // electrode hole Z position

eps          = 0.5;   // overshoot for clean boolean cuts

// ---------------- Derived ----------------
inner_x = outer_x - 2 * wall;     // 114
inner_z = outer_h - floor;        // 37 (inner floor is at z = 3)

// ---------------- Model ----------------
union() {
    // Tank body (outer solid)
    translate([-outer_x / 2, -outer_y / 2, 0])
        cube([outer_x, outer_y, outer_h]);

    // Central platform for the gel tray
    translate([-platform_x / 2, -platform_w / 2, floor])
        cube([platform_x, platform_w, platform_top - floor]);

    // Electrode holes through each end wall, along X
    for (sx = [-1, 1])
        translate([sx * (outer_x / 2 - eps), hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = wall + eps);
}
```
