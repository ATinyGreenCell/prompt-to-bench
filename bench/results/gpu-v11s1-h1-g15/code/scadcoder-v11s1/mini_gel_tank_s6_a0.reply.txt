// Mini horizontal gel-electrophoresis box: open-top rectangular tank with a central
// raised gel-tray platform that divides the bottom into two buffer chambers.
// Print orientation: tank standing on its base, z = 0 at the bottom.

$fn = 64;

// ---- Tank dimensions ----
outer_x   = 120;   // outer length along X
outer_y   = 70;    // outer width along Y (full width, used for chamber spacing)
outer_z   = 40;    // outer height along Z
wall      = 3;     // wall thickness (sides and floor)

// ---- Derived inner dimensions ----
inner_x   = outer_x - 2 * wall;   // 114 mm inner length
inner_z   = outer_z - wall;       // 37 mm inner floor height

// ---- Platform (gel tray support) ----
plat_x    = 60;    // platform length along X, centred
plat_top  = 10;    // platform top height above inner floor (z = 13)
plat_y    = outer_y; // platform spans the full inner width

// ---- Electrode holes ----
hole_d    = 2;     // hole diameter
hole_y    = 0;     // hole Y position
hole_z    = 30;    // hole Z position

// ---- Modelling helpers ----
eps       = 0.5;   // overshoot for clean boolean cuts

difference() {
    union() {
        // Tank body: outer block minus the floor thickness (floor sits at z = 0)
        translate([-outer_x / 2, -outer_y / 2, wall])
            cube([outer_x, outer_y, outer_z - wall]);

        // Raised platform in the middle of the tank
        translate([-plat_x / 2, -plat_y / 2, inner_z - plat_top])
            cube([plat_x, plat_y, plat_top]);
    }

    // Horizontal electrode holes through each end wall, along X
    for (sx = [-1, 1])
        translate([sx * (outer_x / 2 - eps), hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = outer_y + 2 * eps);
}
