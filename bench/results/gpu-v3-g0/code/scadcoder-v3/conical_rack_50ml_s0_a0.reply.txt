// Rack for six 50 mL conical (Falcon-type) tubes
// Printed upside down: plate on bed, walls rising from plate edges.
// Flipped over after printing: rack stands on walls, tubes pass through holes.

$fn = 96;

// ---- Plate ----
plate_x = 130;     // plate length (X)
plate_y = 90;      // plate width (Y)
plate_t = 4;       // plate thickness (Z)

// ---- Holes ----
hole_d     = 30.5; // through-hole diameter
hole_nx    = 3;    // holes along X
hole_ny    = 2;    // holes along Y
hole_pitch = 40;   // centre-to-centre spacing (X and Y)

// ---- Walls ----
wall_h = 70;       // wall height above plate (z = 4 to 70)

// ---- Cutting overlap ----
eps = 0.5;         // extension past cut faces

difference() {
    union() {
        // Plate
        translate([-plate_x/2, -plate_y/2, 0])
            cube([plate_x, plate_y, plate_t]);
        // Two edge walls
        for (side = [-1, 1])
            translate([-plate_x/2, side*plate_y/2, 0])
                cube([plate_x, 4, plate_t + wall_h]);
    }

    // Six through-holes, centred on plate
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([(i - (hole_nx - 1)/2) * hole_pitch,
                       (j - (hole_ny - 1)/2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2*eps);
}
