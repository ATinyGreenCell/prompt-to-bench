```openscad
// Bench rack for 1.5 mL microcentrifuge tubes
// Print orientation: block rests on XY plane, holes open upward (+Z)

// ---------- Block ----------
block_l = 106;     // block length (X), mm
block_w = 72;      // block width (Y), mm
block_h = 30;      // block height (Z), mm

// ---------- Holes ----------
cols      = 6;      // holes along X
rows      = 4;      // holes along Y
pitch_x   = 16;     // centre-to-centre spacing in X, mm
pitch_y   = 16;     // centre-to-centre spacing in Y, mm
hole_d    = 11.2;   // hole diameter, mm
hole_depth = 25;    // hole depth from top face, mm
floor_t   = block_h - hole_depth; // solid floor thickness (5 mm)

// ---------- Misc ----------
overcut = 0.5;      // extension of cutters past the top face, mm
$fn = 96;           // facet count for round holes

// Grid offsets so the hole pattern is centred on the block
x0 = -(cols - 1) * pitch_x / 2;
y0 = -(rows - 1) * pitch_y / 2;

difference() {
    // Solid block, centred in X/Y, sitting on z = 0
    translate([-block_l / 2, -block_w / 2, 0])
        cube([block_l, block_w, block_h]);

    // Tube holes: start at floor_t, extend past top face
    for (i = [0 : cols - 1])
        for (j = [0 : rows - 1])
            translate([x0 + i * pitch_x, y0 + j * pitch_y, floor_t])
                cylinder(d = hole_d, h = hole_depth + overcut, $fn = 96);
}
```
