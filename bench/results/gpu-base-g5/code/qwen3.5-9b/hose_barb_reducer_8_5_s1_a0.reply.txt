// Hose-barb Reducer: 8mm ID to 5mm ID
// Orientation: Standing upright on XY plane (Z axis up)
// Centered at origin in X and Y

// --- Dimensions ---
// Main bore diameter (constant through part)
$main_bore_d = 3.0;

// Upper Section (z=0 to z=20): 8mm tubing connection
$upper_start_z = 0;
$upper_end_z = 20;
$upper_barb_height = 10;
$upper_barb_count = 2;
$upper_barb_bottom_od = 7.5;
$upper_barb_top_od = 9.5;
$upper_collar_od = 12;
$upper_collar_height = 5; // 20 to 25

// Lower Section (z=25 to z=41): 5mm tubing connection
$lower_start_z = 25;
$lower_end_z = 41;
$lower_barb_height = 8;
$lower_barb_count = 2;
$lower_barb_bottom_od = 6.5;
$lower_barb_top_od = 5.0;

// --- Geometry Generation ---

// Function to create a single tapered barb with a step
// z_start: base of the barb
// z_end: top of the barb
// bottom_od: outer diameter at z_start
// top_od: outer diameter at z_end
// step_od: outer diameter after the step (should be bottom_od)
// z_step: height of the tapered section
// bore_d: inner diameter of the main bore
function create_barb(z_start, z_end, bottom_od, top_od, step_od, z_step, bore_d) = 
    union(
        // Tapered section
        difference() {
            cylinder(
                h = z_step,
                r1 = bottom_od / 2,
                r2 = top_od / 2,
                $fn = 64
            );
            // Cut out the main bore
            cylinder(
                h = z_step,
                r = bore_d / 2 + 0.01, // Slight overcut to ensure clean hole
                $fn = 64
            );
        },
        // Step section (constant diameter)
        difference() {
            cylinder(
                h = z_end - z_start - z_step,
                r = bottom_od / 2,
                $fn = 64
            );
            // Cut out the main bore
            cylinder(
                h = z_end - z_start - z_step,
                r = bore_d / 2 + 0.01,
                $fn = 64
            );
        }
    );

// --- Main Part Construction ---

difference() {
    union() {
        // 1. Upper Section (0 to 20)
        // Base cylinder for the upper section (from 0 to 20)
        cylinder(
            h = $upper_end_z,
            r = $upper_collar_od / 2,
            $fn = 64
        );
        
        // Add Upper Barbs
        for (i = [0 : $upper_barb_count - 1]) {
            translate([0, 0, $upper_start_z + i * $upper_barb_height]) {
                create_barb(
                    z_start = $upper_start_z + i * $upper_barb_height,
                    z_end = $upper_start_z + (i + 1) * $upper_barb_height,
                    bottom_od = $upper_barb_bottom_od,
                    top_od = $upper_barb_top_od,
                    step_od = $upper_barb_bottom_od,
                    z_step = $upper_barb_height / 2, // Split height for linear taper
                    bore_d = $main_bore_d
                );
            }
        }

        // 2. Collar Section (20 to 25)
        cylinder(
            h = $upper_collar_height,
            r = $upper_collar_od / 2,
            $fn = 64
        );

        // 3. Lower Section (25 to 41)
        // Base cylinder for the lower section (from 25 to 41)
        cylinder(
            h = $lower_end_z - $lower_start_z,
            r = $lower_barb_bottom_od / 2, // Max OD of lower barbs
            $fn = 64
        );

        // Add Lower Barbs
        for (i = [0 : $lower_barb_count - 1]) {
            translate([0, 0, $lower_start_z + i * $lower_barb_height]) {
                create_barb(
                    z_start = $lower_start_z + i * $lower_barb_height,
                    z_end = $lower_start_z + (i + 1) * $lower_barb_height,
                    bottom_od = $lower_barb_bottom_od,
                    top_od = $lower_barb_top_od,
                    step_od = $lower_barb_bottom_od,
                    z_step = $lower_barb_height / 2, // Split height for linear taper
                    bore_d = $main_bore_d
                );
            }
        }
    },
    
    // Cut the main bore through the entire length
    cylinder(
        h = $lower_end_z,
        r = $main_bore_d / 2 + 0.01, // Slight overcut
        $fn = 64
    )
}
