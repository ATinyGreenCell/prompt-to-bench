```openscad
// Tube rack for six 50 mL conical (Falcon-type) tubes - printed upside down and flipped over.
// A flat plate with a 2x3 hole grid and two solid side walls. The walls become the feet when
// the part is flipped, so the rack stands on a flat surface. The two short ends are open.
//

cols   = 3;
rows   = 2;
tube_d = 30.5;
pitch_x = 40;
pitch_y = 40;
plate_w = 130;
plate_h = 90;
plate_t = 4;
wall_h = 70;

$fn = 48;

plate_n = cols * rows;

difference() {
    union() {
        translate([-plate_w / 2, -plate_h / 2, 0]) cube([plate_w, plate_h, plate_t]);
        for (i = [0 : cols - 1])
            translate([-plate_w / 2, -plate_h / 2 + i * pitch_y, 0]) cube([plate_w, pitch_y, wall_h]);
    }
    for (i = [0 : cols - 1], j = [0 : rows - 1])
        translate([0, j * pitch_y, -1])
            cylinder(h = plate_t + 2, d = tube_d);
}
```
