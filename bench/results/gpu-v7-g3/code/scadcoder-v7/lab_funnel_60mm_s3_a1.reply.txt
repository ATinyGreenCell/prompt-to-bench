```openscad
// Funnel printed upright: spout at bottom (z=0), cone on top, open both ends
// Units: mm. Wall thickness: 1.6 mm.

$fn = 96;

// ---- Dimensions ----
spout_od      = 10;     // spout outer diameter
spout_id      = spout_od - 2 * 1.6;  // spout bore (6.8 mm)
spout_len     = 20;     // spout length (z=0 to z=20)
cone_d0       = 10;     // cone outer diameter at z=20 (same as spout top)
cone_d1       = 60;     // cone outer diameter at rim (z=50)
cone_h        = 30;     // cone height (z=20 to z=50)
eps           = 0.5;    // cutter overshoot past faces

// ---- Model ----
union() {
    // Spout (tube)
    translate([0, 0, 0])
        cylinder(h = spout_len, d = spout_od);

    // Cone (wider at top, open at both ends)
    translate([0, 0, spout_len])
        linear_extrude(cone_h)
            circle(d = cone_d1);
}

// Cut the bore through the full length so liquid runs straight through
translate([0, 0, -eps])
    cylinder(h = spout_len + cone_h + 2 * eps, d = spout_id);
```
