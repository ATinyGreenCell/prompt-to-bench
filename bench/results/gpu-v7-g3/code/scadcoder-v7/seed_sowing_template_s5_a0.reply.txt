```openscad
// Petri dish sowing template
// Print orientation: flat on bed, z = 0 up to disc thickness

// ---------- Parameters ----------
disc_d        = 85;    // disc outer diameter
disc_t        = 2;     // disc thickness
hole_d        = 3;     // through-hole diameter
hole_n        = 7;     // holes per row/column
hole_pitch    = 9;     // centre-to-centre spacing
notch_w       = 10;    // orientation notch width (X)
notch_d       = 4;     // orientation notch depth from rim (Y)
eps           = 0.5;   // cutter overshoot past faces
$fn           = 128;   // circle resolution

// ---------- Derived ----------
x0 = -(hole_n - 1) * hole_pitch / 2;  // first hole X
y0 = -(hole_n - 1) * hole_pitch / 2;  // first hole Y
r  = disc_d / 2;                      // disc radius
rim_t = r - notch_d;                  // rim thickness at notch

// ---------- Model ----------
difference() {
    // Main disc
    cylinder(d = disc_d, h = disc_t);

    // Through-hole grid
    for (i = [0 : hole_n - 1])
        for (j = [0 : hole_n - 1])
            translate([x0 + i * hole_pitch, y0 + j * hole_pitch, -eps])
                cylinder(d = hole_d, h = disc_t + 2 * eps);

    // Orientation notch on +Y side, centred on Y axis
    translate([-(notch_w / 2), rim_t, 0])
        cube([notch_w, notch_d, disc_t + eps]);
}
```
