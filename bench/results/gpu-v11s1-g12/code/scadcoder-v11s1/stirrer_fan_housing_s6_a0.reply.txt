```openscad
// DIY magnetic stirrer housing (print orientation: top plate down, box open upward)
// Units: mm

$fn = 64;

// ---- Dimensions ----
plate_t      = 2;     // top plate thickness (z = 0 to 2)
wall_t       = 2.5;   // wall thickness (z = 2 to 40)
box_h        = 40;    // outer height of box top (z = wall top)
outer_x      = 90;    // outer size in X
outer_y      = 90;    // outer size in Y
hole_d       = 4.5;   // M4 screw hole diameter
hole_count   = 4;     // number of screw holes
hole_pitch   = 71.5;  // square pattern side for holes
notch_w      = 12;    // cable notch width
notch_depth  = 8;     // cable notch depth from open edge (z = 32 to 40)
eps          = 0.5;   // cutter overshoot past faces

// ---- Model ----
difference() {
    union() {
        // Top plate (lies on bed)
        translate([-outer_x/2, -outer_y/2, 0])
            cube([outer_x, outer_y, plate_t]);

        // Four walls rising from plate to z = box_h
        for (sx = [-1, 1], sy = [-1, 1]) {
            translate([sx > 0 ? -outer_x/2 + wall_t : outer_x/2 - wall_t,
                       sy > 0 ? -outer_y/2 + wall_t : outer_y/2 - wall_t,
                       plate_t])
                cube([outer_x - 2*wall_t, outer_y - 2*wall_t, box_h - plate_t]);
        }
    }

    // Screw holes through top plate on 71.5 mm square pattern
    for (i = [0 : hole_count - 1]) {
        a = (i % 2 - 0.5) * hole_pitch;
        b = (i / 2 - 0.25) * hole_pitch;
        translate([a, b, -eps])
            cylinder(d = hole_d, h = plate_t + 2*eps);
    }

    // Cable notch in one wall: 12 mm wide, centred along the wall,
    // reaching 8 mm down from the open edge (z = 32 to z = 40)
    translate([-outer_x/2 + wall_t - notch_w/2, -outer_y/2 + wall_t - notch_w/2,
               box_h - notch_depth])
        cube([notch_w, notch_w, notch_depth + eps]);
}
```
