```openscad
// DIY magnetic stirrer housing for 80 mm PC case fan
// Print orientation: top plate on bed (z = 0), walls build upward (+Z)
// Note: this is the upside-down open box used in practice

// ---------- Parameters ----------
outer_size   = 90;    // outer size of top plate and box (X x Y)
plate_t      = 2;     // top plate thickness (z = 0 to 2)
wall_t       = 2.5;   // wall thickness (z = 2 to 40)
box_h        = 40;    // total height of box from bed (top edge at z = 40)
hole_d       = 4.5;   // screw hole diameter (M4)
hole_pitch   = 71.5;  // square pattern pitch for screw holes
hole_count   = 4;     // number of screw holes
notch_w      = 12;    // power cable notch width
notch_depth  = 8;     // notch depth from open edge (z = 32 to 40)
eps          = 0.5;   // cutter overshoot past faces

$fn = 64;

// ---------- Model ----------
difference() {
    union() {
        // Top plate (lies on bed)
        translate([-outer_size/2, -outer_size/2, 0])
            cube([outer_size, outer_size, plate_t]);

        // Four walls rising from plate to open top (z = 40)
        for (sx = [-1, 1], sy = [-1, 1]) {
            translate([sx > 0 ? outer_size/2 - wall_t : -outer_size/2,
                       sy > 0 ? outer_size/2 - wall_t : -outer_size/2,
                       plate_t])
                cube([wall_t, wall_t, box_h - plate_t]);
        }
    }

    // Screw holes through top plate on square pattern
    for (i = [0 : hole_count - 1]) {
        x = (i % 2 - 0.5) * hole_pitch;
        y = (i / 2 - 0.5) * hole_pitch;
        translate([x, y, -eps])
            cylinder(d = hole_d, h = plate_t + 2*eps);
    }

    // Power cable notch in one wall (z = 32 to 40, open edge at z = 40)
    translate([-outer_size/2 - eps, -outer_size/2 - eps, plate_t])
        cube([notch_w + 2*eps, outer_size + 2*eps, notch_depth + eps]);
}
```
