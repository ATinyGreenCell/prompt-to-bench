```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate flat on bed, motor plate vertical

// ---------- Parameters ----------
base_x      = 25;    // base plate half-width (X)
base_y      = 45;    // base plate length (Y)
base_t      = 5;     // base plate thickness (Z)

motor_h     = 50;    // motor plate height from bed (Z)
motor_w     = 5;     // motor plate thickness (Y)

boss_d      = 23;    // motor boss hole diameter
boss_x      = 0;     // boss hole centre X
boss_z      = 28;    // boss hole centre Z

screw_d     = 3.4;   // M3 screw hole diameter
screw_pitch = 31;    // square pattern side length
screw_n     = 4;     // number of screw holes

gusset_t    = 5;     // gusset thickness (X)
gusset_r    = 20;    // gusset leg along Y
gusset_z    = 20;    // gusset leg along Z

eps         = 0.5;   // cutter overshoot past faces

$fn = 96;

// ---------- Model ----------
difference() {
    union() {
        // Base plate
        translate([-base_x, 0, 0])
            cube([base_x * 2, base_y, base_t]);

        // Motor plate (overlaps base plate for a solid union)
        translate([-base_x, base_y - motor_w, 0])
            cube([base_x * 2, motor_w, motor_h]);

        // Gussets, flush with the outer edges of the base
        for (sx = [-1, 1])
            translate([sx > 0 ? base_x - gusset_t : -base_x,
                       base_y - motor_w,
                       0])
                cube([gusset_t, gusset_r, gusset_z]);
    }

    // Base plate mounting holes
    for (h = [base_y / 2 - eps, base_y / 2 + eps])
        translate([-17, h, -eps])
            cylinder(d = 4.5, h = base_t + 2 * eps);

    // Motor boss hole
    translate([boss_x, base_y - motor_w - eps, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_w + 2 * eps);

    // M3 screw holes on a square pattern centred on the boss
    for (i = [0 : screw_n - 1]) {
        a = i * 90;
        translate([boss_x + (i < 2 ? -1 : 1) * screw_pitch / 2,
                   base_y - motor_w - eps,
                   boss_z + (i >= 2 ? -1 : 1) * screw_pitch / 2])
            rotate([0, 90, a])
                cylinder(d = screw_d, h = motor_w + 2 * eps);
    }
}
```
