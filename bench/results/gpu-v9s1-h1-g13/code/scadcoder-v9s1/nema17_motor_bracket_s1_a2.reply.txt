```openscad
// L-shaped mounting bracket for a NEMA 17 stepper motor (e.g. syringe pump)
// Print orientation: base plate on the bed (z = 0), motor plate builds upward.

// ---------------- Parameters (mm) ----------------
base_x      = 25;    // base plate half-width in X
base_y      = 45;    // base plate length in Y
base_z      = 5;     // base plate thickness

motor_h     = 50;    // motor plate height in Z
motor_y     = 5;     // motor plate thickness in Y (overlap into base)

boss_d      = 23;    // motor centring boss hole diameter
boss_x      = 0;     // boss hole X position
boss_z      = 28;    // boss hole Z position

screw_d     = 3.4;   // M3 screw hole diameter
screw_pitch = 31;    // square pattern side length
screw_n     = 4;     // number of screw holes

gusset_t    = 5;     // gusset thickness in X
gusset_y    = 20;    // gusset leg along +Y
gusset_z    = 20;    // gusset leg along +Z

eps         = 0.5;   // cutter overshoot past faces

$fn = 96;

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([-base_x, 0, 0])
            cube([base_x * 2, base_y, base_z]);

        // Motor plate (standing on back edge of base)
        translate([-base_x, base_y - motor_y, 0])
            cube([base_x * 2, motor_y, motor_h]);

        // Gussets: right triangles in YZ, flush with outer edges
        for (sx = [-1, 1])
            translate([sx > 0 ? base_x - gusset_t : -base_x, 0, 0])
                rotate([90, 0, 0])
                    linear_extrude(height = gusset_t)
                        polygon(points = [
                            [gusset_y, gusset_z],
                            [gusset_y, 0],
                            [0, gusset_z]
                        ]);
    }

    // Base plate through-holes (M4)
    for (h = [-1, 1])
        translate([-17, 30, -eps * h])
            cylinder(d = 4.5, h = base_z + 2 * eps * h);

    // Motor boss hole (along Y)
    translate([boss_x, base_y / 2 - eps, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_y + 2 * eps);

    // M3 screw holes on square pattern, centred on boss hole
    for (i = [0 : screw_n - 1])
        for (j = [0 : screw_n - 1])
            if (i + j % 2 == 1)
                translate([(i - (screw_n - 1) / 2) * screw_pitch / 2,
                           (j - (screw_n - 1) / 2) * screw_pitch / 2,
                           base_y / 2 - eps])
                    rotate([0, 90, 0])
                        cylinder(d = screw_d, h = motor_y + 2 * eps);
}
```
