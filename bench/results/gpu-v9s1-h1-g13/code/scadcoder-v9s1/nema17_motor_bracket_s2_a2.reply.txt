```openscad
// L-shaped mounting bracket for a NEMA 17 stepper motor
// Print orientation: base plate on the bed (z = 0), motor plate builds upward (+Z)

// ---------------- Parameters ----------------
base_x       = 25;    // base plate half-width (X)
base_y       = 45;    // base plate length (Y)
base_z       = 5;     // base plate thickness (Z)

motor_x      = 25;    // motor plate half-width (X)
motor_y      = 5;     // motor plate thickness (Y)
motor_h      = 50;    // motor plate height (Z)

boss_d       = 23;    // motor centring boss hole diameter
boss_x       = 0;     // boss hole centre X
boss_z       = 28;    // boss hole centre Z

screw_d      = 3.4;   // M3 screw hole diameter
screw_p      = 31;    // square pattern side length
screw_n      = 4;     // number of screw holes

gusset_t     = 5;     // gusset thickness (X)
gusset_y     = 20;    // gusset leg along +Y
gusset_z     = 20;    // gusset leg along +Z

eps          = 0.5;   // cutter overshoot past faces
$fn          = 96;    // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate: x -25..25, y 0..45, z 0..5
        translate([-base_x, 0, 0])
            cube([2 * base_x, base_y, base_z]);

        // Motor plate: x -25..25, y 0..5, z 0..50 (standing on back edge)
        translate([-motor_x, base_y - motor_y, 0])
            cube([2 * motor_x, motor_y, motor_h]);

        // Gussets: right triangles in YZ, extruded along X, flush with plate edges
        for (sx = [-1, 1])
            translate([sx * (motor_x - gusset_t), 0, 0])
                rotate([90, 0, 0])
                    linear_extrude(height = gusset_t)
                        polygon(points = [
                            [gusset_y, gusset_z],
                            [0,        gusset_z],
                            [0,        0]
                        ]);
    }

    // Base plate through-holes (M4) at (-17, 30) and (17, 30)
    for (hx = [-17, 17])
        translate([hx, 30, -eps])
            cylinder(d = 4.5, h = base_z + 2 * eps);

    // Boss hole along Y, centred at (0, 28)
    translate([boss_x, 0, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_y + 2 * eps);

    // M3 screw holes on a 31 mm square pattern, centred on the boss hole
    for (i = [0 : screw_n - 1])
        for (j = [0 : screw_n - 1])
            translate([(i - (screw_n - 1) / 2) * screw_p,
                       (j - (screw_n - 1) / 2) * screw_p,
                       -eps])
                cylinder(d = screw_d, h = motor_y + 2 * eps);
}
```
