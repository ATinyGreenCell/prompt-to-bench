```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor (syringe pump)
// Print orientation: base plate on bed, motor plate builds upward (+Z)

// ---------------- Parameters ----------------
plate_x       = 25;    // base plate half-width (X)
plate_y       = 45;    // base plate length (Y)
plate_t       = 5;     // base plate thickness (Z)

motor_h       = 50;    // motor plate height (Z)
motor_w       = 5;     // motor plate thickness (Y)

boss_d        = 23;    // motor boss hole diameter
boss_x        = 0;     // boss hole centre X
boss_z        = 28;    // boss hole centre Z

screw_d       = 3.4;   // M3 screw hole diameter
screw_p       = 31;    // square pattern side length
screw_count   = 4;     // number of screw holes

gusset_t      = 5;     // gusset thickness (X)
gusset_y      = 20;    // gusset leg along +Y
gusset_z      = 20;    // gusset leg along +Z

eps           = 0.5;   // cutter overshoot past faces
$fn           = 96;    // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([-plate_x, 0, 0])
            cube([plate_x * 2, plate_y, plate_t]);

        // Motor plate (stands on back edge of base)
        translate([-plate_x, plate_y - motor_w, 0])
            cube([plate_x * 2, motor_w, motor_h]);

        // Left gusset: right triangle in YZ, flush with left edge
        // rotate([90,0,0]) maps polygon X -> world Y, polygon Y -> world Z
        translate([-plate_x - gusset_t, plate_y - motor_w, 0])
            rotate([90, 0, 0])
                linear_extrude(height = gusset_t)
                    polygon(points = [
                        [plate_y - motor_w, plate_t],   // right-angle corner
                        [plate_y - motor_w + gusset_y, plate_t],
                        [plate_y - motor_w, plate_t + gusset_z]
                    ]);

        // Right gusset: right triangle in YZ, flush with right edge
        translate([plate_x + gusset_t, plate_y - motor_w, 0])
            rotate([90, 0, 0])
                linear_extrude(height = gusset_t)
                    polygon(points = [
                        [plate_y - motor_w, plate_t],
                        [plate_y - motor_w + gusset_y, plate_t],
                        [plate_y - motor_w, plate_t + gusset_z]
                    ]);
    }

    // Base plate through-holes (M4)
    for (h = [plate_t - eps, plate_t + eps])
        translate([-17, 30, h])
            cylinder(d = 4.5, h = 2 * eps, $fn = 64);

    // Motor boss hole (along Y)
    translate([boss_x, plate_y - motor_w - eps, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_w + 2 * eps, $fn = 96);

    // M3 screw holes on square pattern centred on boss hole
    for (i = [0 : screw_count - 1]) {
        a = (i < 2) ? 0 : 90;   // two orthogonal rows
        translate([boss_x + (i < 2 ? -1 : 1) * screw_p / 2,
                   plate_y - motor_w - eps,
                   boss_z + (i >= 2 ? -1 : 1) * screw_p / 2])
            rotate([0, 90, a])
                cylinder(d = screw_d, h = motor_w + 2 * eps, $fn = 64);
    }
}
```
