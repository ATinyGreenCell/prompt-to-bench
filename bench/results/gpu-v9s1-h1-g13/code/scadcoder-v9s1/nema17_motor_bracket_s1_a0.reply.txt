```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor (syringe pump)
// Print orientation: base plate on bed, motor plate building up in +Z

// ---------------- Parameters ----------------
plate_x       = 25;    // base plate half-width (X)
plate_y       = 45;    // base plate length (Y)
plate_z       = 5;     // base plate thickness (Z)

motor_h       = 50;    // motor plate height (Z)
motor_w       = 5;     // motor plate thickness (Y)

boss_d        = 23;    // motor boss hole diameter
boss_x        = 0;     // boss hole X position
boss_z        = 28;    // boss hole Z position

screw_d       = 3.4;   // M3 screw hole diameter
screw_pattern = 31;    // square pitch of screw holes

gusset_t      = 5;     // gusset thickness (X)
gusset_leg    = 20;    // gusset leg lengths (Y and Z)

eps           = 0.5;   // cutter overshoot past faces
$fn           = 96;    // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([-plate_x, 0, 0])
            cube([plate_x * 2, plate_y, plate_z]);

        // Motor plate (standing on back edge of base)
        translate([-plate_x, plate_y - motor_w, 0])
            cube([plate_x * 2, motor_w, motor_h]);

        // Left gusset: right triangle in YZ, flushed with left edge
        translate([-plate_x, 0, 0])
            rotate([90, 0, 0])
                linear_extrude(height = gusset_t)
                    polygon(points = [
                        [gusset_leg, gusset_leg],
                        [gusset_leg, 0],
                        [0, gusset_leg]
                    ]);

        // Right gusset: right triangle in YZ, flushed with right edge
        translate([plate_x - gusset_t, 0, 0])
            rotate([90, 0, 0])
                linear_extrude(height = gusset_t)
                    polygon(points = [
                        [gusset_leg, gusset_leg],
                        [gusset_leg, 0],
                        [0, gusset_leg]
                    ]);
    }

    // Base plate through-holes (M4)
    for (h = [-1, 1])
        translate([-17, 30, -eps * h])
            cylinder(d = 4.5, h = plate_z + 2 * eps * h);

    // Motor boss hole (along Y)
    translate([boss_x, plate_y / 2 - eps, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_w + 2 * eps);

    // M3 screw holes on square pattern, centred on boss hole
    for (i = [-1, 1])
        for (j = [-1, 1])
            translate([i * screw_pattern / 2, j * screw_pattern / 2, plate_y / 2 - eps])
                rotate([0, 90, 0])
                    cylinder(d = screw_d, h = motor_w + 2 * eps);
}
```
