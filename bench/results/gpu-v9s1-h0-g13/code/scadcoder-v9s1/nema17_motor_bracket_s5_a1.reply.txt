```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate on bed (z = 0), vertical plate builds upward (+Z)

// ---------------- Parameters ----------------
$fn = 64;

// Base plate
base_x0 = -25;      // base plate X start
base_x1 = 25;       // base plate X end
base_y0 = 0;        // base plate Y start
base_y1 = 45;       // base plate Y end
base_z0 = 0;        // base plate Z start
base_z1 = 5;        // base plate Z end
mount_hole_d = 4.5; // M4 through-hole diameter
mount_hole_x = -17; // M4 hole X position
mount_hole_y = 30;  // M4 hole Y position
mount_hole_n = 2;   // number of M4 holes

// Vertical motor plate
motor_plate_x0 = -25; // motor plate X start
motor_plate_x1 = 25;  // motor plate X end
motor_plate_y0 = 0;   // motor plate Y start
motor_plate_y1 = 5;   // motor plate Y end
motor_plate_z0 = 0;   // motor plate Z start
motor_plate_z1 = 50;  // motor plate Z end

// Boss hole (motor centring boss)
boss_d = 23;         // boss hole diameter
boss_x = 0;          // boss hole X centre
boss_z = 28;         // boss hole Z centre

// Screw holes for boss
screw_d = 3.4;       // M3 screw hole diameter
screw_pattern = 31;  // square pattern side length
screw_n = 4;         // number of screw holes

// Gussets
gusset_t = 5;        // gusset thickness (along X)
gusset_w = 20;       // gusset leg along +Y
gusset_h = 20;       // gusset leg along +Z

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([base_x0, base_y0, base_z0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);

        // Vertical motor plate
        translate([motor_plate_x0, motor_plate_y0, motor_plate_z0])
            cube([motor_plate_x1 - motor_plate_x0,
                  motor_plate_y1 - motor_plate_y0,
                  motor_plate_z1 - motor_plate_z0]);

        // Left gusset (x = -25 to -20)
        translate([-25, gusset_w, gusset_h])
            rotate([90, 0, 0])
                cube([gusset_t, gusset_w, gusset_h]);

        // Right gusset (x = 20 to 25)
        translate([20, gusset_w, gusset_h])
            rotate([90, 0, 0])
                cube([gusset_t, gusset_w, gusset_h]);
    }

    // Base plate through-holes (M4)
    for (i = [0 : mount_hole_n - 1]) {
        translate([mount_hole_x, mount_hole_y, -1])
            cylinder(d = mount_hole_d, h = base_z1 + 2);
    }

    // Boss hole (through motor plate along Y)
    translate([boss_x, -1, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_plate_y1 - motor_plate_y0 + 2);

    // Screw holes on square pattern centred on boss hole
    for (j = [0 : screw_n - 1]) {
        for (k = [0 : screw_n - 1]) {
            translate([(j - k) * screw_pattern / 2,
                       (j + k) * screw_pattern / 2,
                       boss_z - 1])
                cylinder(d = screw_d, h = motor_plate_z1 + 2);
        }
    }
}
```
