```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor
// Print orientation: base plate on bed, motor plate builds upward (+Z)

// ---------------- Parameters ----------------
$fn = 64;

// Base plate
base_x0 = -25;    // base plate X start
base_x1 = 25;     // base plate X end
base_y0 = 0;      // base plate Y start
base_y1 = 45;     // base plate Y end
base_z0 = 0;      // base plate Z start
base_z1 = 5;      // base plate Z end
hole_m4_x = -17;  // M4 through-hole X position
hole_m4_y = 30;   // M4 through-hole Y position
hole_m4_d = 4.5;  // M4 through-hole diameter

// Motor plate
motor_x0 = -25;   // motor plate X start
motor_x1 = 25;    // motor plate X end
motor_y0 = 0;     // motor plate Y start
motor_y1 = 5;     // motor plate Y end
motor_z0 = 0;     // motor plate Z start
motor_z1 = 50;    // motor plate Z end
boss_d = 23;      // motor centring boss hole diameter
boss_x = 0;       // boss hole X centre
boss_z = 28;      // boss hole Z centre
screw_d = 3.4;    // M3 screw hole diameter
screw_p = 31;     // square pitch for screw holes
screw_n = 4;      // number of screw holes

// Gussets
gusset_t = 5;     // gusset thickness (along X)
gusset_y0 = 5;    // gusset leg along Y
gusset_z0 = 5;    // gusset leg along Z

// ---------------- Model ----------------
union() {
    // Base plate with M4 through-holes
    difference() {
        translate([base_x0, base_y0, base_z0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);
        for (sx = [-1, 1])
            translate([sx * hole_m4_x, hole_m4_y, -1])
                cylinder(d = hole_m4_d, h = base_z1 + 2);
    }

    // Motor plate with boss hole and M3 screw holes
    difference() {
        translate([motor_x0, motor_y0, motor_z0])
            cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1 - motor_z0]);
        // Boss hole along Y
        translate([boss_x, -1, boss_z])
            rotate([90, 0, 0])
                cylinder(d = boss_d, h = motor_y1 - motor_y0 + 2);
        // Screw holes on square pattern
        for (i = [0 : screw_n - 1]) {
            a = (i < 2) ? 0 : 90;
            translate([
                (i < 2) ? -screw_p / 2 : screw_p / 2,
                (i < 2) ? -1 : motor_y1 + 1,
                (i % 2 == 0) ? -1 : motor_z1 + 1
            ])
                rotate([0, a, 0])
                    cylinder(d = screw_d, h = motor_y1 - motor_y0 + 2);
        }
    }

    // Left gusset (flush with left edge, x = -25 to -20)
    translate([-25, motor_y0, motor_z0])
        rotate([90, 0, 0])
            linear_extrude(height = gusset_t)
                polygon(points = [
                    [motor_y0, motor_z0],
                    [gusset_y0, motor_z0],
                    [motor_y0, gusset_z0]
                ]);

    // Right gusset (flush with right edge, x = 20 to 25)
    translate([20, motor_y0, motor_z0])
        rotate([90, 0, 0])
            linear_extrude(height = gusset_t)
                polygon(points = [
                    [motor_y0, motor_z0],
                    [gusset_y0, motor_z0],
                    [motor_y0, gusset_z0]
                ]);
}
```
