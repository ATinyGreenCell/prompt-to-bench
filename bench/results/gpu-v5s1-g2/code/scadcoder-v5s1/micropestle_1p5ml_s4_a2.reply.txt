```openscad
// Pestle for grinding tissue in a 1.5 mL microcentrifuge tube
// Print orientation: standing upright, handle at the bottom

// ---------------- Parameters ----------------
handle_d      = 8;     // handle diameter (mm)
handle_h      = 45;    // handle height, z = 0 to 45 (mm)

cone_top_d    = 8;     // cone top diameter at z = 45 (mm)
cone_bottom_d = 3;     // cone bottom diameter at z = 62 (mm)
cone_h        = 17;    // cone height, z = 45 to 62 (mm)

tip_d         = 3;     // tip sphere diameter (mm)
tip_z         = 62;    // tip sphere centre height (mm)

groove_w      = 2;     // groove width in Z (mm)
groove_depth  = 1;     // groove depth from surface (mm)
groove_z      = [10, 15, 20, 25]; // groove centre heights (mm)

overcut       = 0.5;   // extra length of cutters past faces (mm)
$fn           = 96;    // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // (1) Cylindrical handle
        cylinder(d = handle_d, h = handle_h);

        // (2) Tapering cone
        translate([0, 0, handle_h])
            cylinder(d1 = cone_top_d, d2 = cone_bottom_d, h = cone_h);

        // (3) Rounded tip sphere
        translate([0, 0, tip_z - tip_d / 2])
            sphere(d = tip_d);
    }

    // Grip grooves in the handle
    for (z = groove_z)
        translate([0, 0, z - overcut])
            cylinder(d = handle_d + 2 * groove_depth, h = groove_w + 2 * overcut, $fn = 64);
}
```
