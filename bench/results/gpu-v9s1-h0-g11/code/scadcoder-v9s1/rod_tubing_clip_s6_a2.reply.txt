```openscad
// Snap-on tubing clip for a 12.7 mm (1/2 inch) lab stand rod
// Print orientation: flat on bed, extruded upward in +Z

// ---- Dimensions (mm) ----
rod_id     = 12.7;   // stand rod inner diameter
rod_wall   = 3;      // stand rod ring wall thickness
rod_od     = rod_id + 2 * rod_wall;  // stand rod ring outer diameter (18.7)

tubing_id  = 6;      // tubing inner diameter
tubing_wall = 2;     // tubing ring wall thickness
tubing_od  = tubing_id + 2 * tubing_wall;  // tubing ring outer diameter (10)

bridge_w   = 6;      // solid bridge width (X), joins x=8 to x=12
bridge_h   = 6;      // solid bridge height (Y)

opening_r  = 10;     // rod ring opening width (-X side)
opening_t  = 4.5;    // tubing ring opening width (+X side)

eps        = 0.5;    // cutter overshoot past faces
$fn        = 96;     // curve resolution

// ---- 2D profile in XY, extruded to Z ----
profile() {
    union() {
        // Rod ring with -X opening
        difference() {
            cylinder(d = rod_od, h = 1, $fn = 96);
            translate([-rod_od/2 - eps, -opening_r/2, 0])
                cube([opening_r + 2*eps, opening_r, 1]);
        }

        // Tubing ring with +X opening
        translate([15, 0, 0])
            difference() {
                cylinder(d = tubing_od, h = 1, $fn = 96);
                translate([15 + eps, -opening_t/2, 0])
                    cube([opening_t + 2*eps, opening_t, 1]);
            }

        // Solid bridge between the two rings
        translate([8 - eps, -bridge_h/2, 0])
            cube([bridge_w + 2*eps, bridge_h, 1]);
    }
}

translate([0, 0, 0])
    linear_extrude(height = 10)
        profile();
```
