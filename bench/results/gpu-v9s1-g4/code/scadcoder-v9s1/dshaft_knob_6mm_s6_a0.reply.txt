```openscad
// Replacement knob for hotplate stirrer with D-shaft
// Print orientation: flat bottom face on bed (z = 0), builds upward.

// ---------------- Parameters ----------------
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
bore_depth    = 12;    // depth of D-shaped blind bore from bottom face
shaft_d       = 6.2;   // D-shaft diameter (bore width across flat)
bore_flat     = 4.7;   // distance from flat to opposite side of bore circle
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // grip groove diameter (full cylinder)
pointer_w     = 1.5;   // pointer groove width
pointer_d     = 1;     // pointer groove depth
eps           = 0.5;   // overcut for clean boolean differences

$fn = 96;

// ---------------- Model ----------------
difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, -eps])
        cylinder(d = shaft_d, h = bore_depth + 2 * eps, $fn = 64);   // shaft circle
    translate([0, 0, -eps])
        linear_extrude(height = bore_depth + 2 * eps)
            polygon([[0, 0], [shaft_d / 2, shaft_d / 2], [shaft_d / 2, -shaft_d / 2]]); // flat cut

    // Vertical grip grooves around full height
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 64);

    // Pointer groove on top face, same side as bore flat
    translate([0, 0, knob_h - pointer_d])
        linear_extrude(height = pointer_d + eps)
            circle(d = pointer_w, $fn = 64);
}
```
