// Replacement knob for a hotplate stirrer with a 6 mm D-shaft
// Print orientation: flat bottom face on the bed (z = 0), builds upward.

$fn = 96;

// ---- Dimensions (mm) ----
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
bore_depth    = 12;    // D-bore depth from bottom face
bore_w        = 4.7;   // D-bore width (flat to opposite side of circle)
bore_d        = 6.2;   // D-bore circle diameter
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // grip groove diameter (half-cylinder)
pointer_w     = 1.5;   // pointer groove width
pointer_d     = 1;     // pointer groove depth
eps           = 0.5;   // overcut for clean boolean differences

// ---- Derived ----
flat_x = -bore_d/2;    // x position of D-bore flat side
angle_per_groove = 360 / groove_count;

difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face (extended past top face -> blind)
    translate([flat_x, 0, eps])
        cube([bore_d, knob_d/2, bore_depth + eps]);

    // Vertical grip grooves around the full height
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * angle_per_groove])
            translate([knob_d/2, 0, 0])
                cylinder(d = groove_d, h = knob_h, $fn = 64);

    // Pointer groove from centre to outer edge, same side as flat
    translate([flat_x, 0, knob_h - pointer_d])
        cube([pointer_w, knob_d/2, pointer_d + eps]);
}
