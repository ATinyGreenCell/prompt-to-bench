```openscad
// DIY magnetic stirrer housing - print orientation: upside down, open box
// Units: mm

// ---- Dimensions ----
fan_d        = 80;    // fan diameter
outer_x      = 90;    // outer size X
outer_y      = 90;    // outer size Y
top_t        = 2;     // top plate thickness (z = 0 to 2)
wall_t       = 2.5;   // wall thickness
wall_h       = 40;    // wall height (z = 0 to 40)
notch_w      = 12;    // cable notch width
notch_bottom = 8;     // notch depth from open edge (z = 32 to 40)
hole_d       = 4.5;   // M4 screw hole diameter
hole_p       = 71.5;  // screw hole pattern size (square)
eps          = 0.5;   // cutter overshoot

$fn = 96;

// ---- Model ----
difference() {
    union() {
        // Top plate
        translate([-outer_x/2, -outer_y/2, 0])
            cube([outer_x, outer_y, top_t]);
        // Walls (open top)
        for (sx = [-1, 1], sy = [-1, 1])
            translate([sx > 0 ? outer_x/2 - wall_t : -outer_x/2,
                       sy > 0 ? outer_y/2 - wall_t : -outer_y/2,
                       0])
                cube([wall_t, outer_y - outer_y/2 + wall_t, wall_h]);
    }
    // Screw holes
    for (x = [-hole_p/2, hole_p/2], y = [-hole_p/2, hole_p/2])
        translate([x, y, -eps])
            cylinder(d = hole_d, h = top_t + 2*eps);
    // Cable notch in one wall (z = 32 to 40)
    translate([outer_x/2 - wall_t - eps, -notch_w/2, wall_h - notch_bottom])
        cube([eps, notch_w, notch_bottom + eps]);
}
```
