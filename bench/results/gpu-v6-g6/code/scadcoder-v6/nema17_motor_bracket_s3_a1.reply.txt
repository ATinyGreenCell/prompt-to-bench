```openscad
// L-shaped mounting bracket for a NEMA 17 stepper motor (e.g. syringe pump)
// Print orientation: base plate on the bed, vertical motor plate building up in +Z

// ---------------- Parameters ----------------
base_x      = 25;    // base plate half-length in X (mm)
base_y      = 45;    // base plate length in Y (mm)
base_t      = 5;     // base plate thickness in Z (mm)

hole_m4_x   = 17;    // M4 through-hole offset from centre in X (mm)
hole_m4_y   = 30;    // M4 through-hole Y position (mm)

motor_h     = 50;    // vertical motor plate height (mm)
motor_w     = 5;     // vertical motor plate width in Y (mm)

boss_d      = 23;    // motor centring boss hole diameter (mm)
boss_x      = 0;     // boss hole X position (mm)
boss_z      = 28;    // boss hole Z position (mm)

screw_d     = 3.4;   // M3 screw hole diameter (mm)
screw_p     = 31;    // square pitch of the four screw holes (mm)

gusset_t    = 5;     // gusset thickness (mm)
gusset_leg  = 20;    // right-triangle leg length along +Y and +Z (mm)

eps         = 0.5;   // cutter overshoot past faces (mm)
$fn         = 96;    // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([-base_x, 0, 0])
            cube([2 * base_x, base_y, base_t]);

        // Vertical motor plate, standing on the back edge of the base
        translate([-base_x, 0, 0])
            cube([2 * base_x, motor_w, motor_h]);

        // Gussets, flush with the left and right edges
        for (sx = [-1, 1])
            translate([sx > 0 ? base_x - gusset_t : -base_x, 0, 0])
                cube([gusset_t, gusset_leg, gusset_leg]);
    }

    // Base plate M4 through-holes
    for (sx = [-1, 1])
        translate([sx * hole_m4_x, hole_m4_y, -eps])
            cylinder(d = 4.5, h = base_t + motor_w + 2 * eps);

    // Motor centring boss hole, along Y
    translate([boss_x, 0, boss_z])
        rotate([90, 0, 0])
            cylinder(d = boss_d, h = motor_w + 2 * eps);

    // Four M3 screw holes on a square pattern, centred on the boss hole
    for (a = [0, 90, 180, 270])
        rotate([0, 0, a])
            translate([screw_p / 2, screw_p / 2, 0])
                cylinder(d = screw_d, h = motor_h + eps);
}
```
