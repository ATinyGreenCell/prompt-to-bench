```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor (e.g. syringe pump)
// Print orientation: base plate on bed, vertical motor plate builds up in +Z

// ---------------- Parameters ----------------
base_x      = 25;    // base plate half-length in X
base_y      = 45;    // base plate length in Y
base_t      = 5;     // base plate thickness in Z

hole_m4_x   = 17;    // M4 through-hole offset from centre in X (both holes)
hole_m4_y   = 30;    // M4 through-hole position in Y
hole_m4_d   = 4.5;   // M4 through-hole diameter

motor_h     = 50;    // vertical motor plate height in Z
motor_w     = 5;     // vertical motor plate thickness in Y
boss_d      = 23;    // motor centring boss hole diameter
boss_x      = 0;     // boss hole centre X
boss_z      = 28;    // boss hole centre Z
screw_d     = 3.4;   // M3 screw hole diameter
screw_p     = 31;    // square pattern side length for screw holes

gusset_t    = 5;     // gusset thickness along X
gusset_leg  = 20;    // right-triangle leg length along Y and Z
gusset_y    = 5;     // gusset corner position in Y
gusset_z    = 5;     // gusset corner position in Z

eps         = 0.5;   // cutter overshoot past faces

$fn = 96;            // curve resolution

// ---------------- Model ----------------
union() {
    // Base plate with M4 through-holes
    difference() {
        translate([-base_x, 0, 0])
            cube([2 * base_x, base_y, base_t]);
        for (sx = [-1, 1])
            translate([sx * hole_m4_x, hole_m4_y, -eps])
                cylinder(d = hole_m4_d, h = base_t + 2 * eps);
    }

    // Vertical motor plate, standing on the back edge of the base
    difference() {
        translate([-base_x, 0, 0])
            cube([2 * base_x, motor_w, motor_h]);
        // Motor centring boss hole
        translate([boss_x, -motor_w / 2 - eps, boss_z])
            rotate([-90, 0, 0])
                cylinder(d = boss_d, h = motor_w + 2 * eps);
        // M3 screw holes on a square pattern centred on the boss hole
        for (a = [0, 90, 180, 270])
            rotate([0, 0, a])
                translate([screw_p / 2, screw_p / 2, 0])
                    cylinder(d = screw_d, h = motor_h + 1);
    }

    // Triangular gussets flush with the left and right edges
    // Right triangle in the YZ plane: legs along +Y and +Z, right angle at (y = 5, z = 5)
    for (sx = [-1, 1]) {
        x0 = sx > 0 ? 20 : -25;   // gusset start in X
        translate([x0, 0, 0])
            rotate([90, 0, 0])
                linear_extrude(height = gusset_t)
                    polygon(points = [
                        [gusset_y, gusset_z],
                        [gusset_y + gusset_leg, gusset_z],
                        [gusset_y, gusset_z + gusset_leg]
                    ]);
    }
}
```
