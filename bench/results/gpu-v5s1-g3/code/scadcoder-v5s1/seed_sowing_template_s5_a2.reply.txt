```openscad
// Petri dish sowing template - flat disc with a 7x7 grid of through-holes and a
// rectangular orientation notch into the rim on the +Y side. Units: mm.
//
// NOTE: This is a solid model. To get the intended through-holes, print it
// upside down (disc face down on the bed) so the holes open upward, or drill
// them after printing.

$fn = 128;

// ---- Disc ----
disc_d        = 85;    // disc outer diameter
disc_t        = 2;     // disc thickness (Z)

// ---- Hole grid ----
hole_count    = 7;     // holes per side (7 x 7)
hole_d        = 3;     // through-hole diameter
hole_pitch    = 9;     // centre-to-centre spacing

// ---- Notch ----
notch_w       = 10;    // notch width (X)
notch_depth   = 4;     // notch depth from rim (Y)

// ---- Cutting overlap ----
eps           = 0.5;   // extension of cutters past faces

difference() {
    // Main disc, centred in X/Y, resting on z = 0
    cylinder(d = disc_d, h = disc_t);

    // Hole grid, centred on the disc
    for (i = [-hole_count / 2 : hole_count / 2])
        for (j = [-hole_count / 2 : hole_count / 2])
            translate([i * hole_pitch, j * hole_pitch, -eps])
                cylinder(d = hole_d, h = disc_t + 2 * eps);

    // Rectangular orientation notch into the rim on the +Y side, centred on Y
    translate([(disc_d / 2 - notch_w / 2), notch_depth, -eps])
        cube([notch_w, notch_w - notch_depth, disc_t + 2 * eps]);
}
```
