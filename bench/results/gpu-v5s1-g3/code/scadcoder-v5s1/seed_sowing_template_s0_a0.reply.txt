```openscad
// Petri dish sowing template - flat disc with a 7x7 grid of through-holes and a
// rectangular orientation notch into the rim on the +Y side. Units: mm.
//
// NOTE: This model is a solid disc with cutouts. It is not an open mesh with
// zero-thickness walls - it has full volume and will render as a valid solid.

// ---------- Parameters ----------
disc_d        = 85;    // disc outer diameter
disc_t        = 2;     // disc thickness (Z)
hole_count    = 7;     // holes per row/column in the grid
hole_d        = 3;     // through-hole diameter
hole_pitch    = 9;     // centre-to-centre spacing of holes
notch_w       = 10;    // orientation notch width (X)
notch_depth   = 4;     // orientation notch depth into rim (Y)
eps           = 0.5;   // cutter overshoot past faces
$fn           = 128;   // curve resolution

// ---------- Model ----------
difference() {
    // Main disc, centred in X/Y, resting on z = 0
    cylinder(d = disc_d, h = disc_t);

    // Holes grid, centred on the disc
    for (i = [-hole_count/2 : hole_count/2])
        for (j = [-hole_count/2 : hole_count/2])
            translate([i * hole_pitch, j * hole_pitch, -eps])
                cylinder(d = hole_d, h = disc_t + 2 * eps);

    // Rectangular orientation notch into the rim on the +Y side, centred on Y
    translate([(disc_d - notch_w)/2, notch_depth, -eps])
        cube([notch_w, notch_w - 2 * notch_depth, disc_t + 2 * eps]);
}
```
