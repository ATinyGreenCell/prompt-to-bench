cone_bot = 3;

$fn = 64;

handle_h = 45;
cone_h   = 62 - 45;
tip_h    = 62;

difference() {
    union() {
        cylinder(h = handle_h, d = handle_d);
        translate([0, 0, handle_h]) cylinder(h = cone_h, d1 = cone_top, d2 = cone_bot);
        translate([0, 0, handle_h + cone_h]) sphere(d = tip_d);
    }
    for (z = [10, 15, 20, 25])
        translate([0, 0, z]) cylinder(h = groove_w, d = handle_d + groove_d);
}
