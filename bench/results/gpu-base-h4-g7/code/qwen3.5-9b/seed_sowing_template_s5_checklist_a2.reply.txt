HOLE_COUNT = 7;
HOLE_DIA = 3;
HOLE_SPACING = 9;
NOTCH_WIDTH = 10;
NOTCH_DEPTH = 4;

$fn = 64;

// Calculate grid offsets to centre the grid on the origin
// For 7 holes, indices are 0..6. Centre index is 3.

grid_offset = (HOLE_COUNT - 1) / 2 * HOLE_SPACING;

difference() {
