```openscad
// Petri dish sowing template
// Print orientation: flat on bed, z = 0 up to disc thickness

// ---------- Parameters ----------
disc_d        = 85;    // disc outer diameter
disc_h        = 2;     // disc thickness
hole_d        = 3;     // through-hole diameter
hole_spacing  = 9;     // centre-to-centre hole spacing (X and Y)
hole_count    = 7;     // holes per row/column (7 x 7 grid)
notch_w       = 10;    // orientation notch width (X)
notch_depth   = 4;     // orientation notch depth from rim (Y)
eps           = 0.5;   // cutter overshoot past faces

$fn = 128;

// ---------- Model ----------
difference() {
    // Main disc
    cylinder(d = disc_d, h = disc_h);

    // Through-hole grid, centred on disc
    for (i = [0 : hole_count - 1])
        for (j = [0 : hole_count - 1])
            translate([(i - (hole_count - 1) / 2) * hole_spacing,
                       (j - (hole_count - 1) / 2) * hole_spacing,
                       -eps])
                cylinder(d = hole_d, h = disc_h + 2 * eps);

    // Rectangular orientation notch on +Y side, centred on Y axis
    translate([0, notch_depth / 2, 0])
        cube([notch_w, notch_depth, disc_h + eps]);
}
```
