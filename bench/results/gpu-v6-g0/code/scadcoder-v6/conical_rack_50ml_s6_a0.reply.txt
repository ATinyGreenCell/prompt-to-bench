```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, two end walls rising from the plate.
// After printing, flip the part so the walls stand on the bed and the
// tubes pass through the holes.
//
// Units: mm

// ---------------- Parameters ----------------
plate_x      = 130;   // plate length along X
plate_y      = 90;    // plate width along Y
plate_t      = 4;     // plate thickness (z = 0 to 4)

hole_d       = 30.5;  // tube hole diameter
hole_nx      = 3;     // holes per row along X
hole_ny      = 2;     // number of rows along Y
hole_pitch   = 40;    // centre-to-centre spacing in X and Y

wall_t       = 4;     // wall thickness (same as plate thickness)
wall_top_z   = 70;    // walls rise from plate to this height
wall_h       = wall_top_z - plate_t;  // wall height above plate (66)

eps          = 0.5;   // cutter overshoot past faces
$fn          = 96;    // circle resolution

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([-plate_x/2, -plate_y/2, 0])
            cube([plate_x, plate_y, plate_t]);

        // Two solid walls along the long edges, flush with the plate
        for (sx = [-1, 1])
            translate([sx > 0 ? plate_x/2 - wall_t : -plate_x/2,
                       -plate_y/2, 0])
                cube([wall_t, plate_y, wall_h]);
    }

    // Tube holes through the full height (walls included)
    for (i = 0; i < hole_nx; i++)
        for (j = 0; j < hole_ny; j++)
            translate([(i - (hole_nx - 1)/2) * hole_pitch,
                       (j - (hole_ny - 1)/2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + wall_h + 2*eps);
}
```
