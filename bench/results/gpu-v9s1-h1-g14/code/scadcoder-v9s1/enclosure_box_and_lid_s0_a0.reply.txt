```openscad
// Two-part enclosure for small electronics board
// Print orientation: both parts on bed, lid on +X side of box

$fn = 64;

// ---- Box ----
box_x = 70;      // outer length (X)
box_y = 50;      // outer width (Y)
box_z = 30;      // outer height (Z)
wall = 2;        // wall thickness
floor = 2;       // floor thickness

// ---- Lid ----
lip_w = 65.6;    // lip outer length (X)
lip_d = 45.6;    // lip outer width (Y)
lip_h = 4;       // lip height (Z)
lip_t = 1.5;     // lip wall thickness

// ---- Misc ----
gap = 10;        // gap between parts
eps = 0.5;       // cutter overshoot

// Derived
inner_x = box_x - 2 * wall;
inner_y = box_y - 2 * wall;
lip_f  = lip_w - 0.4;   // lip outer size with 0.2 clearance each side
lip_b  = lip_d - 0.4;

module box() {
    difference() {
        translate([0, 0, 0])
            cube([box_x, box_y, box_z]);
        translate([wall, wall, floor])
            cube([inner_x, inner_y, box_z - floor]);   // open top
    }
}

module lid() {
    translate([0, 0, 0]) {
        // base plate
        translate([-box_x / 2, -box_y / 2, 0])
            cube([box_x, box_y, 2]);
        // lip on top, centred
        translate([-lip_f / 2, -lip_b / 2, 2])
            cube([lip_f, lip_b, lip_h]);
    }
}

box();
translate([box_x + gap, 0, 0])
    lid();
```
