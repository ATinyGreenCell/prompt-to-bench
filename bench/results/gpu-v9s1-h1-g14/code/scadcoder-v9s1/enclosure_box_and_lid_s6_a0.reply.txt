```openscad
// Two-part enclosure for small electronics board
// Print orientation: both parts on bed, lid on +X side of box

// ---- Box ----
box_x = 70;      // outer length (X)
box_y = 50;      // outer width (Y)
box_z = 30;      // outer height (Z)
wall = 2;        // wall thickness
floor = 2;       // floor thickness

// ---- Lid ----
lip_w = 65.6;    // lip outer length (X)
lip_d = 45.6;    // lip outer width (Y)
lip_h = 4;       // lip height (above plate)
plate_t = 2;     // plate thickness
eps = 0.2;       // clearance between lip and box opening

// Derived
lip_r = wall + eps;  // lip wall thickness (wall + clearance)

$fn = 64;

// Box (open top)
module box() {
    translate([-box_x/2, -box_y/2, 0])
        cube([box_x, box_y, box_z - floor]);
}

// Lid (plate + lip, printed upside down)
module lid() {
    translate([box_x/2 + 10, -box_y/2, 0]) {
        // Plate
        cube([lip_w, lip_d, plate_t]);
        // Lip: wall on bottom face, extended top face
        translate([0, 0, plate_t - lip_h])
            cube([lip_w, lip_d, lip_h]);
        // Side walls of lip
        translate([0, 0, plate_t])
            cube([lip_w, lip_d, lip_r]);
    }
}

union() {
    box();
    translate([0, 0, 10])
        lid();
}
```
