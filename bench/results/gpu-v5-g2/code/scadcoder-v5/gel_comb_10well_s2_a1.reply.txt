```openscad
// 10-well comb for casting agarose gels, printed lying flat
// Units: mm

// ---- Parameters ----
plate_t      = 1.5;   // plate thickness (Z)
spine_l      = 70;    // spine length (X)
spine_w      = 12;    // spine width (Y)
teeth_n      = 10;    // number of teeth
teeth_w      = 5.0;   // tooth width (X)
teeth_l      = 12;    // tooth length (Y)
teeth_gap    = 1.5;   // gap between teeth (X)
overcut      = 0.5;   // extra length on cutters past faces (Z)

// Derived
pitch        = teeth_w + teeth_gap;   // centre-to-centre tooth spacing
x0           = -(teeth_n - 1) * pitch / 2;  // first tooth X (centred row)

// ---- Model ----
difference() {
    union() {
        // Spine (back bar)
        translate([0, -spine_w / 2, 0])
            cube([spine_l, spine_w, plate_t]);

        // Teeth hanging in -Y from the spine's +Y edge
        for (i = [0 : teeth_n - 1])
            translate([x0 + i * pitch, spine_w / 2 - teeth_l, 0])
                cube([teeth_w, teeth_l, plate_t]);
    }

    // Cut out the 10 wells from the +Y face of the plate
    for (i = [0 : teeth_n - 1])
        translate([x0 + i * pitch, spine_w / 2 - teeth_l - overcut, -overcut])
            cube([teeth_w, teeth_l, plate_t + 2 * overcut]);
}
```
