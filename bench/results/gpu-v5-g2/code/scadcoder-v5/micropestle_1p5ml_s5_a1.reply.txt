// Tissue-pestle for 1.5 mL microcentrifuge tube
// Print orientation: standing upright, handle on bed

// ---- Dimensions (mm) ----
handle_d      = 8;     // handle diameter
handle_h      = 45;    // handle height (z = 0 to 45)
cone_top_d    = 8;     // cone diameter at top of handle (z = 45)
cone_tip_d    = 3;     // cone diameter at z = 62
tip_r         = 1.5;   // tip sphere radius (3 mm diameter)
groove_w      = 2;     // groove width (Z)
groove_depth  = 1;     // groove depth (radial)
groove_z      = [10, 15, 20, 25]; // groove centre heights
eps           = 0.5;   // cutter overshoot past faces

$fn = 96;

// ---- Model ----
difference() {
    union() {
        // Handle
        cylinder(d = handle_d, h = handle_h);
        // Cone (tapers 8 mm -> 3 mm over 17 mm)
        translate([0, 0, handle_h])
            cylinder(d1 = cone_top_d, d2 = cone_tip_d, h = 17);
        // Rounded tip: sphere centred at z = 62
        translate([0, 0, handle_h + 17])
            sphere(r = tip_r);
    }
    // Grip grooves, cut fully through (X and Y)
    for (z = groove_z)
        translate([0, 0, z])
            linear_extrude(height = groove_w)
                circle(d = groove_depth + 2 * eps);
}
