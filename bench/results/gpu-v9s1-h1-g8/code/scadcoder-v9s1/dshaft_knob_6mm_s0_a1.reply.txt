```openscad
// Replacement knob for a hotplate stirrer with a 6 mm D-shaft
// Print orientation: bottom face (with D-bore) on the bed at z = 0

// ---------- Dimensions (mm) ----------
knob_d      = 30;    // knob outer diameter
knob_h      = 15;    // knob height
bore_depth  = 12;    // D-bore depth from bottom face
shaft_d     = 6;     // D-shaft diameter (fit clearance included)
groove_n    = 18;    // number of vertical grip grooves
groove_d    = 2;     // grip groove diameter
pointer_w   = 1.5;   // pointer groove width
pointer_d   = 1;     // pointer groove depth
eps         = 0.5;   // overcut for clean boolean differences

$fn = 96;            // curve resolution

// ---------- Derived ----------
bore_h = sqrt(knob_d**2 - shaft_d**2);   // chord length of D-bore (4.7 mm side to side)

// ---------- Model ----------
difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, eps])
        cylinder(d = bore_h, h = bore_depth + eps);

    // Vertical grip grooves around the outside, full height
    for (i = [0 : groove_n - 1])
        rotate([0, 0, i * 360 / groove_n])
            translate([0, knob_d / 2, 0])
                cylinder(d = groove_d, h = knob_h + 2 * eps);

    // Pointer groove on top face, same side as the flat of the D
    translate([shaft_d / 2, 0, knob_h - pointer_d])
        cube([pointer_w, knob_d / 2 + eps, pointer_d + eps]);
}
```
