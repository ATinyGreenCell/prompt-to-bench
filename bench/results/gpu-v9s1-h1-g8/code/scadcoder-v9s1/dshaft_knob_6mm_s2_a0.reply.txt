```openscad
// Replacement knob for hotplate stirrer (D-shaft 6 mm)
// Print orientation: flat bottom face on bed (z = 0)

$fn = 96;

// --- Dimensions (mm) ---
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
bore_depth    = 12;    // D-bore depth from bottom face
bore_w        = 4.7;   // D-bore width (flat to opposite side)
bore_d        = 6.2;   // D-bore circle diameter
groove_count  = 18;    // number of grip grooves
groove_d      = 2;     // grip groove diameter
pointer_w     = 1.5;   // pointer groove width
pointer_d     = 1;     // pointer groove depth

// --- Derived ---
eps = 0.5;             // overcut for clean differences
flat_x = -bore_d / 2;  // flat side of D (x = -3.1)

difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, eps])
        linear_extrude(height = bore_depth + eps)
            circle(d = bore_d, $fn = 64);
    translate([flat_x, 0, eps])
        linear_extrude(height = bore_depth + eps)
            polygon([[-bore_d / 2, -bore_w / 2],
                     [-bore_d / 2, bore_w / 2]]);

    // Vertical grip grooves along full height
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, 0])
                cylinder(d = groove_d, h = knob_h, $fn = 64);

    // Pointer groove from centre to outer edge, same side as flat
    translate([0, 0, knob_h - pointer_d])
        linear_extrude(height = pointer_d)
            circle(d = pointer_w, $fn = 64);
}
```
