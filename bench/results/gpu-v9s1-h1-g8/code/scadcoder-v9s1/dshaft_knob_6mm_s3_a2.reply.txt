```openscad
// Replacement knob for a hotplate stirrer with a 6 mm D-shaft
// Print orientation: bottom face (with D-bore) on the bed at z = 0

$fn = 96;

// ---- Dimensions (mm) ----
knob_d      = 30;    // knob outer diameter
knob_h      = 15;    // knob height
bore_d      = 6.2;   // D-bore circle diameter
bore_depth  = 12;    // D-bore depth from bottom face
bore_flat   = 4.7;   // distance from D-bore flat to opposite side
groove_n    = 18;    // number of vertical grip grooves
groove_d    = 2;     // grip groove diameter (half-cylinder)
pointer_w   = 1.5;   // pointer groove width
pointer_d   = 1;     // pointer groove depth
eps         = 0.5;   // overcut for clean boolean differences

// ---- Derived ----
bore_r = bore_d / 2;
start_angle = atan2(-bore_flat / 2, bore_r);  // flat on +X side

difference() {
    // Main body: cylinder, centred in XY, resting on z = 0
    translate([0, 0, 0])
        cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, -eps])
        rotate([0, 0, start_angle])
            cylinder(r = bore_r, h = bore_depth + 2 * eps, $fn = 64);

    // 18 vertical grip grooves along the full height
    for (i = [0 : groove_n - 1])
        translate([knob_d / 2, 0, -eps])
            rotate([0, 90, i * 360 / groove_n])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 64);

    // Pointer groove on top face, from centre to outer edge
    translate([0, 0, knob_h - pointer_d])
        rotate([90, 0, 0])
            linear_extrude(height = knob_d / 2)
                polygon([[-pointer_w / 2, -1], [pointer_w / 2, -1], [pointer_w / 2, 1]]);
}
```
