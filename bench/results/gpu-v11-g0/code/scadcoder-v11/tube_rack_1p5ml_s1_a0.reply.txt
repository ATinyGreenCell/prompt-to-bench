// Bench rack for 1.5 mL microcentrifuge tubes
// Print orientation: flat on bed, holes open at the top

// ---- Block dimensions ----
block_len   = 106;   // length along X (mm)
block_wid   = 72;    // width along Y (mm)
block_ht    = 30;    // height along Z (mm)

// ---- Hole parameters ----
hole_d      = 11.2;  // hole diameter (mm)
hole_depth  = 25;    // hole depth from top face (mm)
hole_count  = 24;    // total number of holes
hole_rows   = 4;     // holes along Y
hole_cols   = 6;     // holes along X
hole_pitch  = 16;    // centre-to-centre spacing, both directions (mm)

// ---- Cutting overlap ----
eps         = 0.5;   // extension past cut faces (mm)

$fn = 64;

difference() {
    // Main block
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_ht]);

    // Hole grid, centred on block
    for (i = [0 : hole_rows - 1])
        for (j = [0 : hole_cols - 1])
            translate([(j - (hole_cols - 1)/2) * hole_pitch,
                       (i - (hole_rows - 1)/2) * hole_pitch,
                       block_ht - hole_depth])
                cylinder(d = hole_d, h = hole_depth + eps);
}
