// 96-well PCR tube rack (0.2 mL, microplate footprint)
// Print orientation: base on XY plane, builds up +Z

// ---- Block ----
block_len   = 127.76;   // X length (mm)
block_wid   = 85.48;    // Y width (mm)
block_h     = 20;       // Z height (mm)

// ---- Holes ----
hole_d      = 6.2;      // through-hole diameter (mm)
hole_nx     = 12;       // columns along X
hole_ny     = 8;        // rows along Y
hole_pitch  = 9.0;      // centre-to-centre spacing (mm)
hole_x0     = 14.38;    // first hole X offset from left edge (mm)
hole_y0     = 11.24;    // first hole Y offset from back edge (mm)

// ---- Chamfer ----
chamfer     = 5;        // 45-degree chamfer leg length at A1 corner (mm)

// ---- Misc ----
eps         = 0.5;      // cutter overshoot past faces (mm)
$fn         = 96;       // curve resolution

difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_h]);

    // 96 vertical through-holes
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([hole_x0 + i * hole_pitch - hole_d/2 - eps,
                       hole_y0 + j * hole_pitch - hole_d/2 - eps,
                       -eps])
                cylinder(d = hole_d, h = block_h + 2 * eps);

    // 45-degree chamfer at the A1 corner (vertical edge, top and bottom faces)
    translate([hole_x0 - chamfer, hole_y0, 0])
        cube([chamfer, chamfer, block_h]);
}
