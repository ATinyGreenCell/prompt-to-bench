// 96-well PCR tube rack (standard microplate footprint)
// Units: mm. Print orientation: base on XY plane, building up +Z.

// ---------------- Parameters ----------------
block_len    = 127.76;   // block length along X
block_wid    = 85.48;    // block width along Y
block_h      = 20;       // block height along Z

hole_d       = 6.2;      // through-hole diameter
hole_pitch   = 9.0;      // centre-to-centre spacing (X and Y)
hole_rows    = 8;        // number of rows along Y
hole_cols    = 12;       // number of columns along X

first_x      = 14.38;    // X of first hole (A1) from left edge
first_y      = 11.24;    // Y of first hole (A1) from back edge

chamfer      = 5;        // 45-degree chamfer leg length at A1 corner
eps          = 0.5;      // cutter overshoot past faces

$fn = 64;

// ---------------- Model ----------------
difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_h]);

    // 96 vertical through-holes
    for (r = [0 : hole_rows - 1])
        for (c = [0 : hole_cols - 1])
            translate([first_x + c * hole_pitch - hole_d/2,
                       first_y + r * hole_pitch - hole_d/2,
                       -eps])
                cylinder(d = hole_d, h = block_h + 2*eps);

    // 45-degree chamfer at the A1 corner (vertical edge, not the top face)
    translate([first_x - hole_d/2 - 1, first_y, 0])
        linear_extrude(height = block_h)
            polygon(points = [
                [0, 0],
                [hole_d/2 + 1, 0],
                [hole_d/2 + 1 - chamfer, chamfer],
                [0, chamfer]
            ]);
}
