// Microscope slide drying rack
// 10 slides stand upright on their long edge, side by side

// ---- Block ----
block_l = 90;    // X length of base block (mm)
block_w = 45;    // Y width of base block (mm)
block_h = 20;    // Z height of base block (mm)

// ---- Slots ----
slot_n     = 10;   // number of slots
slot_len   = 77;   // slot length along X (mm)
slot_w     = 1.6;  // slot width along Y (mm)
slot_depth = 10;   // slot depth from top face (mm)
slot_pitch = 3.5;  // centre-to-centre spacing along Y (mm)

// ---- Misc ----
eps = 0.5;        // overcut past faces to avoid coplanar surfaces (mm)

difference() {
    // Base block, centred in X/Y, resting on z = 0
    translate([-block_l/2, -block_w/2, 0])
        cube([block_l, block_w, block_h]);

    // Slots running along X, open at the top
    for (i = [0 : slot_n - 1]) {
        y = (i - (slot_n - 1)/2) * slot_pitch;
        translate([-slot_len/2, y, block_h - slot_depth])
            cube([slot_len, slot_w, slot_depth + eps]);
    }
}
