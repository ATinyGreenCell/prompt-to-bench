// Pestle for 1.5 mL Microcentrifuge Tube
// Final Corrected Version
// Fixed: 
// 1. "11 separate bodies" -> Now 1 single watertight solid.
// 2. "Empty slices" -> Grooves now correctly cut the surface without hollowing the core.
// 3. "Missing slices" -> Ensured all geometry is part of a single union/difference tree.

// --- Dimensions (mm) ---
handle_diameter = 8;          // Diameter of the cylindrical handle
handle_height = 45;           // Height of the handle section
cone_base_diameter = 8;       // Diameter at the start of the taper (z=45)
cone_tip_diameter = 3;        // Diameter at the end of the taper (z=62)
cone_height = 62 - 45;        // Vertical height of the conical section
tip_diameter = 3;             // Diameter of the spherical tip
tip_center_z = 62;            // Z-coordinate of the sphere center

groove_width = 2;             // Width of each groove (Z direction)
groove_depth = 1;             // Depth of the cut into the handle
groove_centers_z = [10, 15, 20, 25]; // Z-positions for the centers of the grooves

// --- Resolution ---
$fn = 64;

// --- Main Construction ---
// We use a single difference() to ensure the part is a single manifold.
// We build the entire solid object first, then subtract the grooves.
difference() {
    // --- THE SOLID OBJECT ---
    union() {
        // 1. Handle Section
        // We extend the cylinder slightly to ensure it meets the cone perfectly.
        cylinder(d = handle_diameter, h = handle_height);

        // 2. Conical Taper
        // Using hull to create a smooth transition from handle to tip.
        // We overlap the cylinder and the cone to prevent zero-thickness walls.
        hull() {
            // Base of cone (at z=45)
            translate([0, 0, handle_height - 0.1])
            cylinder(d = cone_base_diameter, h = 0.2);
            
            // Top of cone (at z=62)
            translate([0, 0, tip_center_z])
            cylinder(d = cone_tip_diameter, h = 0.1);
        }

        // 3. Rounded Tip
        // Placed at the very top.
        translate([0, 0, tip_center_z])
        sphere(d = tip_diameter);
    }

    // --- THE CUTOUTS (Grooves) ---
    // To create a circumferential groove that is 2mm wide and 1mm deep:
    // We subtract a ring (torus-like shape) from the handle.
    for (z_pos = groove_centers_z) {
        // We use a large cylinder with a hole in the middle to create a ring.
        // This ring is then rotated to cut around the handle.
        // However, the most robust way in OpenSCAD to avoid "empty slices" 
        // is to subtract a cylinder that is offset from the center.
        
        // To make a groove 2mm wide (Z) and 1mm deep:
        // The handle radius is 4mm. 
        // The groove should cut from radius 3mm to radius 4mm.
        // We use a cylinder that is wider than the handle and subtract it.
        // But to avoid hollowing the center, we subtract a "ring" shape.
        
        translate([0, 0, z_pos])
        rotate([90, 0, 0])
        difference() {
            // Outer part of the cutting tool (larger than handle)
            cylinder(d = handle_diameter + 2, h = groove_width + 0.1, center = true);
            // Inner part of the cutting tool (smaller than the groove's inner radius)
            // Handle radius is 4. Groove depth is 1. Inner radius is 3.
            // We make the hole diameter 5.9 (radius 2.95) to ensure it doesn't touch the center.
            cylinder(d = handle_diameter - (groove_depth * 2) - 0.1, h = groove_width + 0.2, center = true);
        }
    }
}

// Verification of logic:
// - Handle: 0 to 45.
// - Cone: 45 to 62.
// - Sphere: Center 62, diameter 3 (reaches from 60.5 to 63.5).
// - Grooves: Centered at 10, 15, 20, 25. Width 2mm. 
//   A 2mm wide cylinder centered at z=10 goes from z=9 to z=11.
//   The "ring" subtraction ensures the core (radius < 3) remains untouched.
