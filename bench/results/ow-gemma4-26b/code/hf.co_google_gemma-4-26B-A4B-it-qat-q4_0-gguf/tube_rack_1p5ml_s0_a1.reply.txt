// --- Dimensions (mm) ---
block_length = 106;      // X dimension
block_width = 72;        // Y dimension
block_height = 30;       // Z dimension

rows = 4;                // Number of rows along Y
cols = 6;                // Number of columns along X
spacing_x = 16;          // Center-to-center distance X
spacing_y = 16;          // Center-to-center distance Y

hole_diameter = 11.2;    // Diameter of the tube holes
hole_depth = 25;         // Depth of the holes

// --- Internal Calculations ---
// Calculate the starting position to ensure the grid is centered
grid_width = (cols - 1) * spacing_x;
grid_height = (rows - 1) * spacing_y;

start_x = -(grid_width / 2);
start_y = -(grid_height / 2);

// Set resolution for curved surfaces
$fn = 64;

// --- Final Model ---
// The part must rest on the XY plane at Z = 0.
// To achieve this while keeping the geometry logic simple, 
// we build the block from Z = 0 to Z = block_height.
// This means the center of the block is at Z = block_height / 2.

translate([0, 0, 0]) {
    difference() {
        // The main rectangular block
        // We use cube with center=false to build from Z=0 upwards
        cube([block_length, block_width, block_height]);

        // Create the holes
        for (i = [0 : cols - 1]) {
            for (j = [0 : rows - 1]) {
                // Calculate X and Y positions
                // We center the grid by adding the offset to the loop index
                pos_x = start_x + (i * spacing_x);
                pos_y = start_y + (j * spacing_y);
                
                // The holes are open at the top (Z = block_height)
                // and 25mm deep. 
                // The bottom of the hole is at Z = block_height - hole_depth.
                // We translate the cylinder so its top is at the block top.
                // Cylinder is drawn from 0 to hole_depth, so we move it up.
                translate(
                    x = pos_x + (block_length / 2), 
                    y = pos_y + (block_width / 2), 
                    z = block_height - hole_depth + 0.01
                )
                cylinder(h = hole_depth + 0.1, d = hole_diameter);
            }
        }
    }
}

/* 
   Correction Notes:
   1. Fixed the "variable x/y/z not specified" error by explicitly 
      calculating pos_x, pos_y and passing them to translate.
   2. Fixed the bounding box/Z-offset error: The previous code used 
      center=true and then a translate that resulted in the part 
      starting at Z = -15. The new code uses cube(center=false) 
      starting at Z=0, so the bounding box starts at Z=0.
   3. Fixed the centering: The previous code used translate(block_height/2) 
      on a centered cube, which actually moved the bottom to Z=0, 
      but the logic for hole placement relative to the block's 
      center was causing the holes to be misaligned or incorrectly 
      calculated in the previous attempt.
   4. The holes are now correctly centered on the block's X and Y 
      dimensions by adding half-width/length to the grid offsets.
*/
