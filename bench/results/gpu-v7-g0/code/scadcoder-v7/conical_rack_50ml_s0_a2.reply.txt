```openscad
// Falcon tube rack for six 50 mL conical tubes
// Print orientation: flat plate on bed (z = 0), walls build upward (+Z)
// After printing, flip the part so it stands on the walls.

// ---------------- Parameters ----------------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm)

hole_d       = 30.5;  // through-hole diameter (mm)
hole_nx      = 3;     // holes along X (per row)
hole_ny      = 2;     // holes along Y (rows)
hole_pitch   = 40;    // centre-to-centre spacing in X and Y (mm)

wall_h       = 70;    // wall height above plate (mm)
wall_t       = 4;     // wall thickness along Y (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 96;    // circle resolution

// ---------------- Derived ----------------
wall_top_z   = plate_t + wall_h;   // top of walls (z = 74)

// ---------------- Model ----------------
union() {
    // Base plate
    translate([-plate_x/2, -plate_y/2, 0])
        cube([plate_x, plate_y, plate_t]);

    // Two solid walls along the long edges, flush with the plate
    for (s = [-1, 1])
        translate([-plate_x/2 + s * (plate_x/2 - wall_t),
                   -plate_y/2, 0])
            cube([wall_t, plate_y, wall_top_z]);

    // Through-holes for tubes, centred on the plate
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([
                -plate_x/2 + hole_pitch * i - hole_d/2,
                -plate_y/2 + hole_pitch * j - hole_d/2,
                -eps
            ])
                cylinder(d = hole_d, h = plate_t + 2*eps);
}
```
