```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, holes vertical.
// After printing: flip the part so the two wall strips stand on the bed.

$fn = 64;

// ---- Plate ----
plate_x     = 130;   // plate length (X)
plate_y     = 90;    // plate width (Y)
plate_t     = 4;     // plate thickness (Z)

// ---- Holes ----
hole_d      = 30.5;  // hole diameter
hole_nx     = 3;     // holes along X
hole_ny     = 2;     // holes along Y
hole_pitch  = 40;    // centre-to-centre spacing (both directions)

// ---- Walls ----
wall_h      = 66;    // wall height above plate (Z)

// ---- Misc ----
eps         = 0.5;   // cutter overshoot past faces

difference() {
    union() {
        // Base plate
        translate([-plate_x/2, -plate_y/2, 0])
            cube([plate_x, plate_y, plate_t]);

        // Two solid walls along the long edges
        for (s = [-1, 1])
            translate([s > 0 ? plate_x/2 - 1 : -plate_x/2,
                       s * (plate_y/2 - 1), 0])
                cube([plate_x, plate_y - 2, plate_t + wall_h]);
    }

    // Through-holes in the plate
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([(i - (hole_nx - 1)/2) * hole_pitch,
                       (j - (hole_ny - 1)/2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2 * eps);
}
```
