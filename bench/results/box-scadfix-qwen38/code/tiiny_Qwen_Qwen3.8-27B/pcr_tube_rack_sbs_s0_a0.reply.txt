```openscad
// 96-well PCR tube rack (0.2 mL) — ANSI/SLAS microplate footprint
// Print orientation: rests on XY plane (z = 0), builds upward (+Z), centred on origin in X and Y.

// ---- Block dimensions (measured with calipers) ----
L = 127.76;   // length along X
W = 85.48;    // width  along Y
H = 20.0;     // height along Z

// ---- Hole grid ----
n_rows  = 8;  // rows along Y
n_cols  = 12; // columns along X
pitch   = 9.0;    // centre-to-centre spacing (both X and Y)
hole_d  = 6.2;    // vertical through-hole diameter
hole_r  = hole_d / 2;

// A1 (first hole) is 14.38 mm from the left edge and 11.24 mm from the back edge.
// The grid is centred on the block, so the first hole sits at:
x1 = -L/2 + 14.38;   // X coordinate of A1
y1 = -W/2 + 11.24;   // Y coordinate of A1 (back edge = -W/2)

// ---- 45° chamfer on the vertical edge at the A1 corner ----
// The A1 corner is the (left, back) vertical edge of the block.
// A 5 mm leg chamfer removes a 5×5 mm square from that corner in plan,
// producing a 45° bevel on the vertical edge.
chamfer_leg = 5.0;

$fn = 64;

// ---- Build the rack ----
difference() {
    // Solid block
    translate([-L/2, -W/2, 0])
        cube([L, W, H]);

    // 96 vertical through-holes (extend 1 mm past top and bottom faces)
    for (i = [0 : n_rows - 1]) {
        for (j = [0 : n_cols - 1]) {
            hx = x1 + j * pitch;
            hy = y1 + i * pitch;
            translate([hx, hy, -1])
                cylinder(h = H + 2, r = hole_r, $fn = $fn);
        }
    }

    // 45° chamfer on the vertical edge at the A1 corner (left-back corner)
    // Cut a triangular prism that removes a 5 mm leg from the corner.
    // The corner is at (-L/2, -W/2). The chamfer removes the region
    // where x < -L/2 + chamfer_leg AND y < -W/2 + chamfer_leg,
    // bounded by the line x - (-L/2) + y - (-W/2) = chamfer_leg.
    // We model this as a polygon extruded in Z, extending 1 mm past top and bottom.
    translate([-L/2, -W/2, -1])
        linear_extrude(h = H + 2, center = false)
            polygon(
                points = [
                    [0, 0],
                    [chamfer_leg, 0],
                    [0, chamfer_leg]
                ],
                paths = [[0, 1, 2, 0]]
            );
}
```
