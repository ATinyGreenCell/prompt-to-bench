```openscad
// 96-well PCR tube rack (0.2 mL) – ANSI/SLAS microplate footprint
// Print orientation: rests on XY plane (z = 0), builds upward (+Z), centred on origin in X and Y.

// ---- Block dimensions (measured) ----
L = 127.76;   // length along X (mm)
W = 85.48;    // width  along Y (mm)
H = 20.0;     // height along Z (mm)

// ---- Hole grid ----
nrows   = 8;  // rows along Y
ncols   = 12; // columns along X
pitch   = 9.0;    // centre-to-centre spacing (mm)
hole_d  = 6.2;    // hole diameter (mm)
hole_r  = hole_d / 2;

// A1 (first hole) offset from the left (X-) and back (Y-) edges
off_x   = 14.38;  // from left edge
off_y   = 11.24;  // from back edge

// ---- Chamfer (orientation mark) ----
chamfer_leg = 5.0; // 45° chamfer leg length on the vertical edge at the A1 corner

// ---- Derived centred coordinates ----
// Block spans x in [-L/2, L/2], y in [-W/2, W/2], z in [0, H]
x0 = -L/2;        // left edge
y0 = -W/2;        // back edge

// Hole centre positions (centred on the block)
hole_x = [for (c = 0; c < ncols; c++) x0 + off_x + c * pitch];
hole_y = [for (r = 0; r < nrows; r++) y0 + off_y + r * pitch];

// Chamfer corner = A1 corner = left edge / back edge = (x0, y0)
cx = x0;
cy = y0;

// ---- Build ----
$fn = 64;

difference() {
    // Solid block
    translate([0, 0, 0])
        cube([L, W, H], center = [true, true, false]);

    // 96 vertical through-holes (extend 1 mm past top and bottom faces)
    for (hx = hole_x)
        for (hy = hole_y)
            translate([hx, hy, -1])
                cylinder(h = H + 2, r = hole_r, $fn = 64);

    // 45° chamfer off the vertical edge at the A1 corner (orientation mark)
    // Wedge: right-triangle cross-section in XY, extruded full height,
    // hypotenuse runs from (cx+leg, cy) to (cx, cy+leg) -> 45° chamfer.
    translate([cx, cy, -1])
        linear_extrude(height = H + 2, center = false)
            polygon(points = [[0, 0], [chamfer_leg, 0], [0, chamfer_leg]]);
}
```
