// Drying rack for 10 microscope slides (75 x 25 x 1 mm)
// Slides stand upright on their long edge, held in slots cut into the top face.

// ---- Base block dimensions (measured with calipers) ----
base_L = 90;   // length along X
base_W = 45;   // width  along Y
base_H = 20;   // height along Z

// ---- Slot dimensions ----
slot_L   = 77;   // slot length along X
slot_W   = 1.6;  // slot width  along Y
slot_D   = 10;   // slot depth  along Z (open at top)
slot_Pitch = 3.5; // centre-to-centre spacing in Y
n_slots  = 10;   // number of slots

// ---- Derived slot positions (centred on origin in X and Y) ----
// For 10 slots with 3.5 mm pitch, the Y-centres are:
// y_i = (i - (n_slots-1)/2) * slot_Pitch,  i = 0..9
// This gives a symmetric group centred at Y = 0.

$fn = 64;

// ---- Build the rack ----
difference() {
    // Solid base block, resting on z=0, centred on origin in X and Y
    translate([-base_L/2, -base_W/2, 0])
        cube([base_L, base_W, base_H]);

    // Cut 10 parallel slots into the top face
    for (i = [0 : n_slots-1]) {
        y_c = (i - (n_slots - 1) / 2) * slot_Pitch;

        // Each slot: 77 mm long (X), 1.6 mm wide (Y), 10 mm deep (Z), open at top.
        // The cutter extends 1 mm above the top face (z = base_H) to avoid coplanar faces.
        translate([-slot_L/2, y_c - slot_W/2, base_H - slot_D])
            cube([slot_L, slot_W, slot_D + 1]);  // +1 mm past the top face
    }
}
