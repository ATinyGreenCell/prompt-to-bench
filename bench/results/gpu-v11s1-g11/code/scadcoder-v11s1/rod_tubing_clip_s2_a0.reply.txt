```openscad
// Snap-on clip for tubing next to a 12.7 mm lab stand rod
// Print orientation: flat on bed, extruded upward in +Z

$fn = 96;

// ---- Dimensions (mm) ----
ring_od        = 18.7;   // rod ring outer diameter
ring_id        = 12.7;   // rod ring inner diameter (rod bore)
ring_wall      = 3;      // rod ring wall thickness
ring_open      = 10;     // rod ring opening width (along -X)

tube_od        = 8;      // tubing ring outer diameter
tube_id        = 6;      // tubing ring inner diameter
tube_wall      = 2;      // tubing ring wall thickness
tube_open      = 4.5;    // tubing opening width (along +X)

bridge_x0      = 8;      // bridge start (X)
bridge_x1      = 12;     // bridge end (X)
bridge_w       = 6;      // bridge width (Y)

eps            = 0.5;    // cutter overshoot past faces
standoff       = 0.5;    // ring centre offset from rod surface

// ---- Derived ----
ring_r = ring_od / 2;
tube_r = tube_od / 2;

difference() {
    union() {
        // Rod ring
        difference() {
            cylinder(h = 10, r = ring_r);
            // Opening cut from -X side
            translate([0, -ring_open / 2, -eps])
                cube([ring_r + eps, ring_open, 10 + 2 * eps]);
        }

        // Tubing ring
        translate([15, 0, 0])
            difference() {
                cylinder(h = 10, r = tube_r);
                // Opening cut from +X side
                translate([0, -tube_open / 2, -eps])
                    cube([tube_r + eps, tube_open, 10 + 2 * eps]);
            }

        // Solid bridge between rings
        translate([bridge_x0, -bridge_w / 2, 0])
            cube([bridge_x1 - bridge_x0, bridge_w, 10]);
    }

    // Rod stand bore (12.7 mm rod + standoff clearance)
    translate([0, 0, -eps])
        cylinder(h = 10, r = ring_id / 2 + standoff);
}
```
