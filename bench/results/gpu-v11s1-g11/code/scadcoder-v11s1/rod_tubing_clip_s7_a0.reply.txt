```openscad
// Tubing clip for 12.7 mm lab stand rod
// Print orientation: flat on bed, extruded up +Z

$fn = 96;

// ---- Ring 1: rod ring ----
ring1_id   = 12.7;   // inner diameter (rod clearance)
ring1_wall = 3;      // wall thickness
ring1_od   = ring1_id + 2 * ring1_wall; // outer diameter = 18.7
ring1_x    = 0;      // centre X
ring1_y    = 0;      // centre Y

// ---- Ring 2: tubing ring ----
ring2_id   = 6;      // inner diameter
ring2_wall = 2;      // wall thickness
ring2_od   = ring2_id + 2 * ring2_wall; // outer diameter = 10
ring2_x    = 15;     // centre X
ring2_y    = 0;      // centre Y

// ---- Bridge ----
bridge_x0  = 8;      // start X
bridge_x1  = 12;     // end X
bridge_w   = 6;      // width in Y

// ---- Cutouts ----
cut1_w     = 10;     // rod ring opening width (-X side)
cut2_w     = 4.5;    // tubing ring opening width (+X side)
eps        = 0.5;    // cutter overshoot

// ---- Model ----
h = 10;              // extrusion height
difference() {
    union() {
        // Rod ring
        cylinder(h = h, d = ring1_od, $fn = 128);
        // Tubing ring
        translate([ring2_x, ring2_y, 0])
            cylinder(h = h, d = ring2_od, $fn = 64);
        // Bridge
        translate([bridge_x0, -bridge_w / 2, 0])
            cube([bridge_x1 - bridge_x0, bridge_w, h]);
    }
    // Rod ring opening: x < 0, -5 < y < 5
    translate([-ring1_od - eps, -cut1_w / 2, -eps])
        cube([ring1_od + 2 * eps, cut1_w, h + 2 * eps]);
    // Tubing ring opening: x > 15, -2.25 < y < 2.25
    translate([ring2_x, -cut2_w / 2, -eps])
        cube([ring2_od + 2 * eps, cut2_w, h + 2 * eps]);
}
```
