```openscad
// Lab stand rod clip - printed flat, extruded along Z
// Units: mm

// ---- Parameters ----
rod_id      = 12.7;   // stand rod inner diameter
rod_wall    = 3;      // stand rod ring wall thickness
rod_open    = 10;     // width of opening cut from -X side of rod ring
rod_dia     = rod_id + 2 * rod_wall; // outer diameter of rod ring (18.7)

tub_id      = 6;      // tubing inner diameter
tub_wall    = 2;      // tubing ring wall thickness
tub_x       = 15;     // tubing ring centre X
tub_open    = 4.5;    // width of opening cut from +X side of tubing ring

bridge_x1   = 8;      // bridge start X
bridge_x2   = 12;     // bridge end X
bridge_w    = 6;      // bridge width along Y

eps         = 0.5;    // cutter overshoot past faces
$fn         = 96;     // curve resolution

// ---- Profile (2D in XY) ----
profile = union() {
    // Rod ring with -X opening
    difference() {
        translate([0, 0]) circle(d = rod_dia);
        translate([0, -eps])
            linear_extrude(height = rod_open + 2 * eps)
                polygon([[-rod_dia / 2, -eps], [-rod_dia / 2, eps]]);
    }

    // Tubing ring with +X opening
    translate([tub_x, 0])
        difference() {
            circle(d = tub_id + 2 * tub_wall);
            translate([tub_x, -tub_open / 2])
                linear_extrude(height = tub_open + eps)
                    polygon([[0, -tub_open / 2], [0, tub_open / 2]]);
        }

    // Solid bridge between the two rings
    translate([bridge_x1, -bridge_w / 2])
        cube([bridge_x2 - bridge_x1, bridge_w, 1]);
}

// ---- Extrude to final part ----
translate([0, 0, 0])
    linear_extrude(height = 10)
        profile;
```
