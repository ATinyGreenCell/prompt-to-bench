```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, walls pointing up.
// After printing, flip over so the rack stands on its two walls.

// ---------- Parameters ----------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm), z = 0..4

hole_d       = 30.5;  // tube hole diameter (mm)
hole_nx      = 3;     // holes along X
hole_ny      = 2;     // holes along Y
hole_pitch   = 40;    // centre-to-centre spacing, X and Y (mm)
hole_eps     = 0.5;   // cutter overshoot past faces (mm)

wall_len     = plate_x; // wall length along X (mm)
wall_t       = 4;     // wall thickness along Y (mm)
wall_h       = 66;    // wall height above plate (mm), z = 4..70

$fn = 96;             // curve resolution

// ---------- Model ----------
union() {
    // Base plate
    translate([-plate_x/2, -plate_y/2, 0])
        cube([plate_x, plate_y, plate_t]);

    // Through-holes, centred on plate
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([(i - (hole_nx - 1)/2) * hole_pitch,
                       (j - (hole_ny - 1)/2) * hole_pitch,
                       -hole_eps])
                cylinder(d = hole_d, h = plate_t + 2*hole_eps);

    // Two solid walls along the long edges (flush with plate)
    for (s = [-1, 1])
        translate([-plate_x/2, s * (plate_y/2 - wall_t/2), 0])
            cube([wall_len, wall_t, wall_h + plate_t]);
}
```
