```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate down on the bed, walls rising in +Z.
// After printing, flip the part so it stands on the two walls.

// ---------- Parameters ----------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm), z = 0..4

hole_d       = 30.5;  // tube hole diameter (mm)
hole_nx      = 3;     // number of holes along X
hole_ny      = 2;     // number of holes along Y
hole_pitch   = 40;    // centre-to-centre spacing in X and Y (mm)

wall_h       = 70;    // wall height above plate top (mm), z = 4..74
wall_t       = 4;     // wall thickness along Y (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 96;    // circle resolution

// ---------- Model ----------
difference() {
    union() {
        // Base plate
        translate([-plate_x/2, -plate_y/2, 0])
            cube([plate_x, plate_y, plate_t]);

        // Two solid walls along the long edges (flush with plate's long edge)
        for (sx = [-1, 1])
            translate([sx > 0 ? plate_x/2 - wall_t : -plate_x/2, -wall_t/2, plate_t - 0.01])
                cube([wall_t, wall_t, wall_h + 0.01]);
    }

    // Six through-holes in a 2 x 3 grid (along Y x along X), centred on the plate
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([(i - (hole_nx - 1)/2) * hole_pitch,
                       (j - (hole_ny - 1)/2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2*eps);
}
```
