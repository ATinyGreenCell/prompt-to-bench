// Helper to create a hollow cylinder (ring) based on radius and wall thickness
function make_ring(r_inner_mm, r_outer_mm, center_x, center_y, width_mm = 0) {
    // Calculate vertices for a square-based polygon approximation of a circle
    const num_vertices = 8; 
    const side_length = 2 * r_outer_mm / num_vertices;
    
    var v = [];
    for (i = 0; i < num_vertices; i++) {
        let theta_deg = (i + 1) * 360.0 / num_vertices;
        // Convert to radians and calculate coordinates relative to center
        let r = side_length / 2; 
        let x_rel = r * cos(theta_deg * PI / 180);
        let y_rel = -r * sin(theta_deg * PI / 180); // Negative Y for clockwise
        
        v.push([center_x + x_rel, center_y + y_rel]);
    }
    
    // Close the loop by adding a vertex at the start of the last segment to ensure connectivity
    // Actually, standard polygon requires closing. Let's add one more point back to close it properly if needed, 
    // but usually 'polygon' closes automatically if we list vertices correctly or use a specific method.
    // OpenSCAD polygon: [x1,y1]...[xn,yn]. It connects xn to x1.
    
    return v;
}
