```openscad
// Dimensions in millimeters
box_width = 70;
box_height = 50;
box_depth = 30;
wall_thickness = 2;
floor_thickness = 2;
lid_thickness = 2;
lip_width = 65.6;
lip_height = 45.6;
lip_wall_thickness = 1.5;
lip_clearance = 0.2;

// Box
box = difference() {
    cube([box_width, box_height, box_depth]);
    translate([wall_thickness, wall_thickness, 0]) cube([box_width - 2*wall_thickness, box_height - 2*wall_thickness, box_depth - floor_thickness]);
};

// Lid
lid = difference() {
    cube([lip_width + 2*(lip_wall_thickness + lip_clearance), lip_height + 2*(lip_wall_thickness + lip_clearance), lid_thickness]);
    translate([lip_wall_thickness + lip_clearance, lip_wall_thickness + lip_clearance, 0]) cube([lip_width - 2*(lip_wall_thickness + lip_clearance), lip_height - 2*(lip_wall_thickness + lip_clearance), lid_thickness]);
};

// Positioning
union() {
    translate([-box_width/2, -box_height/2, 0]) box;
    translate([box_width/2 + 10, -box_height/2, 0]) lid;
};
```
