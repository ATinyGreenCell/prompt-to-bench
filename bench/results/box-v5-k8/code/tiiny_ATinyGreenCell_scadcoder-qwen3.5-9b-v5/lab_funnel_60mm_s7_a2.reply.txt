// Lab funnel - printed upright, wide opening at the top
// Units: mm. Straight spout (tube) from z = 0 to z = 20,
// then a cone widens to the rim at z = 50. Wall 1.6 mm thick.

$fn = 64;

// ---- Dimensions ----
spout_d        = 10;   // spout outer diameter
spout_bore     = 6.8;  // spout bore (tube) diameter
spout_h        = 20;   // spout height (z = 0 to z = 20)
cone_top_d     = 10;   // cone outer diameter at z = 20 (top of spout)
cone_rim_d     = 60;   // cone outer diameter at the rim (z = 50)
wall           = 1.6;  // wall thickness everywhere
eps            = 0.5;  // cutter overshoot past faces

// ---- Derived ----
cone_rim_h      = 50;                        // rim height (z = 50)
cone_slope      = (cone_rim_d - cone_top_d) / (cone_rim_h - spout_h);  // 1 mm per 1.667 mm Z
cone_bore_d     = spout_bore + 2 * wall;     // bore diameter through the cone (8.4 mm)

// ---- Model ----
difference() {
    union() {
        // Spout: tube 10 mm OD, 6.8 mm bore, z = 0 to z = 20
        cylinder(d = spout_d, h = spout_h);
        
        // Cone: OD 10 mm at z = 20, 60 mm at z = 50
        cylinder(d1 = cone_top_d, h = cone_rim_h - spout_h, $fn = 128);
    }
    
    // Through-hole: 8.4 mm diameter from z = 0 to z = 50
    translate([0, 0, -eps])
        cylinder(d = cone_bore_d, h = cone_rim_h + 2 * eps);
}
