```openscad
// NEMA 17 stepper motor L-shaped mounting bracket
// Print orientation: base plate on the bed, vertical plate builds upward (+Z)

// ---------------- Parameters ----------------
plate_x      = 25;     // base plate length along X (mm)
plate_y      = 45;     // base plate width along Y (mm)
plate_t      = 5;      // base plate thickness along Z (mm)

hole_m4      = 4.5;    // M4 through-hole diameter (mm)
hole_m4_x    = 17;     // M4 hole offset from centre in X (mm)
hole_m4_y    = 30;     // M4 hole Y position (mm)

plate_z      = 50;     // vertical motor plate height (mm)
boss_d       = 23;     // motor centring boss hole diameter (mm)
boss_z       = 28;     // motor boss hole centre height (mm)

screw_d      = 3.4;    // M3 screw hole diameter (mm)
screw_pitch  = 31;     // M3 screw hole square pattern size (mm)

gusset_t     = 5;      // gusset thickness along X (mm)
gusset_y     = 20;     // gusset leg along +Y (mm)
gusset_z     = 20;     // gusset leg along +Z (mm)

eps          = 0.5;    // cutter overshoot past faces (mm)

$fn = 96;              // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // Base plate
        translate([-plate_x/2, 0, 0])
            cube([plate_x, plate_y, plate_t]);

        // Vertical motor plate
        translate([-plate_x/2, 0, 0])
            cube([plate_x, plate_y, plate_z]);
    }

    // M4 through-holes in base plate
    for (sx = [-1, 1])
        translate([sx * hole_m4_x, hole_m4_y, -eps])
            cylinder(d = hole_m4, h = plate_t + 2 * eps);

    // Motor centring boss hole through vertical plate
    translate([0, -eps, boss_z])
        cylinder(d = boss_d, h = plate_z + 2 * eps);

    // M3 screw holes on square pattern, centred on boss hole
    for (i = [0 : 3]) {
        a = screw_pitch / sqrt(2);
        rotate([0, 90, 0])
            translate([0, 0, boss_z])
                translate([-a * (i & 1 ? -1 : 1), -a * (i % 4 < 2 ? -1 : 1), 0])
                    cylinder(d = screw_d, h = plate_z + 2 * eps);
    }

    // Gussets, flush with left and right edges
    for (dir = [-1, 1]) {
        x0 = dir == -1 ? -plate_x/2 : plate_x/2 - gusset_t;
        translate([x0, 0, 0])
            linear_extrude(height = gusset_t)
                translate([0, plate_y - gusset_y, 0])
                    cube([gusset_y, gusset_y, gusset_z]);
    }
}
```
