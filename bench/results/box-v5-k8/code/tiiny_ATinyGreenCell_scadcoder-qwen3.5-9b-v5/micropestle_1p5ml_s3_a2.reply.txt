// Pestle for grinding tissue inside a 1.5 mL microcentrifuge tube
// Print orientation: standing upright, handle at the bottom

// ---------------- Parameters ----------------
handle_d      = 8;     // handle diameter
handle_h      = 45;    // handle height (z = 0 to 45)

cone_top_d    = 8;     // cone diameter at top of handle (z = 45)
cone_tip_d    = 3;     // cone diameter at tip (z = 62)
cone_h        = cone_tip_d - cone_top_d; // cone height (19 mm)

tip_d         = 3;     // rounded tip diameter
tip_z         = 62;    // tip centre height (top of cone)

groove_w      = 2;     // groove width in Z
groove_depth  = 1;     // groove depth into handle
groove_z      = [10, 15, 20, 25]; // groove centre heights

eps           = 0.5;   // cutter overshoot past faces
$fn           = 96;    // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // Handle: cylinder from z = 0 to 45
        cylinder(d = handle_d, h = handle_h);

        // Cone: tapers from 8 mm at z = 45 to 3 mm at z = 62
        translate([0, 0, handle_h])
            cylinder(d1 = cone_top_d, d2 = cone_tip_d, h = cone_h);

        // Rounded tip: 3 mm diameter sphere centred at z = 62
        translate([0, 0, tip_z])
            sphere(d = tip_d);
    }

    // Circumferential grooves into the handle
    for (z = groove_z)
        translate([0, 0, z])
            rotate([0, 90, 0])
                cylinder(d = groove_depth, h = groove_w, $fn = 96);
}
