```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, walls rising to z = 70
// After printing, flip the part over so the walls stand on the bed

// ---------- Parameters ----------
plate_x      = 130;   // plate length along X
plate_y      = 90;    // plate width along Y
plate_t      = 4;     // plate thickness (z = 0 to 4)

hole_d       = 30.5;  // through-hole diameter
hole_nx      = 3;     // holes along X per row
hole_ny      = 2;     // holes along Y per row
hole_pitch   = 40;    // centre-to-centre spacing in X and Y

wall_t       = 4;     // wall thickness (Y)
wall_h       = 70;    // wall top height (z)

eps          = 0.5;   // cutter overshoot past faces
$fn          = 96;    // curve resolution

// ---------- Derived ----------
wall_h_top = wall_h - plate_t;  // wall height above plate (66 mm)

// ---------- Model ----------
union() {
    // Flat base plate
    translate([-plate_x/2, -plate_y/2, 0])
        cube([plate_x, plate_y, plate_t]);

    // Two solid walls along the long edges
    for (sx = [-1, 1])
        translate([sx > 0 ? plate_x/2 - wall_t : -plate_x/2,
                   -plate_y/2,
                   0])
            cube([plate_x - 2*wall_t + (sx > 0 ? 2 : 0),
                  wall_t,
                  wall_h_top]);

    // Through-holes, centred on the plate
    for (i = [0, 2])
        for (j = [0, 1])
            translate([-plate_x/2 + i*hole_pitch,
                       -plate_y/2 + j*hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2*eps);
}
```
