// Microscope slide drying rack
// Print orientation: base on XY plane, slots open upward (+Z)

// ---- Block dimensions ----
block_len   = 90;    // block length along X
block_wid   = 45;    // block width along Y
block_ht    = 20;    // block height along Z

// ---- Slot dimensions ----
slot_count  = 10;    // number of slide slots
slot_len    = 77;    // slot length along X
slot_w      = 1.6;   // slot width along Y
slot_depth  = 10;    // slot depth into block (from top face)
slot_pitch  = 3.5;   // centre-to-centre spacing in Y

// ---- Cutting overlap ----
eps = 0.5;           // extension past cut faces to avoid coplanar faces

$fn = 64;

// X position of first slot so the group is centred on the block
slot_x0 = -(slot_count - 1) * slot_len / 2;

difference() {
    // Solid base block, centred in X and Y, resting on z = 0
    translate([-block_len / 2, -block_wid / 2, 0])
        cube([block_len, block_wid, block_ht]);

    // Parallel slots open at the top, running along X
    for (i = [0 : slot_count - 1])
        translate([slot_x0 + i * slot_len, -block_wid / 2, block_ht - slot_depth])
            cube([slot_len, slot_w, slot_depth + eps]);
}
