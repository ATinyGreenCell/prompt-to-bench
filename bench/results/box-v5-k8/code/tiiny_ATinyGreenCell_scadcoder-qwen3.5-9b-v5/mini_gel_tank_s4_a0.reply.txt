```openscad
// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: open top facing up (+Z), resting on XY plane at z = 0

// ---------- Parameters ----------
outer_x   = 120;   // outer length along X (mm)
outer_y   = 70;    // outer width along Y (mm)
outer_z   = 40;    // outer height along Z (mm)
wall      = 3;     // wall thickness (mm)
floor     = 3;     // floor thickness (mm)

platform_x = 60;   // platform length along X (mm), centred
platform_y = outer_y - 2 * wall; // platform spans full inner width (mm)
platform_top = outer_z - wall - floor + 10; // platform top height, 10 mm above inner floor (mm)

hole_d = 2;        // electrode hole diameter (mm)
hole_y = 0;        // electrode hole Y position (mm)
hole_z = 30;       // electrode hole Z position (mm)

eps = 0.5;         // cutter overshoot past faces (mm)

$fn = 96;

// ---------- Model ----------
difference() {
    // Main tank body
    translate([-outer_x / 2, -outer_y / 2, 0])
        cube([outer_x, outer_y, outer_z]);

    // Raised platform for gel tray
    translate([-platform_x / 2, -platform_y / 2, platform_z])
        cube([platform_x, platform_y, platform_top - platform_z]);

    // Horizontal electrode holes through end walls along X
    for (s = [-1, 1])
        translate([s > 0 ? outer_x / 2 - eps : -outer_x / 2 + eps, hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = outer_x + 2 * eps);
}
```
