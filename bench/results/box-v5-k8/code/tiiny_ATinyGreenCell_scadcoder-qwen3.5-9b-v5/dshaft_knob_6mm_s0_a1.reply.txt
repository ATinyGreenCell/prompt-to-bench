```openscad
// Replacement knob for hotplate stirrer (D-shaft 6 mm)
// Print orientation: side facing hotplate down on bed (z = 0)

// ---------------- Parameters ----------------
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
bore_depth    = 12;    // D-shaft blind bore depth from bottom face
shaft_d       = 6.2;   // D-shaft diameter (circle diameter)
flat_to_curve = 4.7;   // distance from flat side of D to opposite circle side
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // groove diameter (half-cylinder)
pointer_w     = 1.5;   // pointer groove width
pointer_d     = 1;     // pointer groove depth
eps           = 0.5;   // cutter overshoot past faces

$fn = 96;

// ---------------- Model ----------------
difference() {
    // Main knob body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, -eps])
        cylinder(d = shaft_d, h = bore_depth + 2 * eps, $fn = 64);

    // Flat side of D bore (cut into the cylinder along +X)
    translate([knob_d / 2 - flat_to_curve, 0, -eps])
        cube([flat_to_curve, knob_d, bore_depth + 2 * eps]);

    // Vertical grip grooves around full height, centred on outer surface
    for (i = [0 : groove_count - 1]) {
        a = (i - groove_count / 2) * 360 / groove_count;
        translate([knob_d / 2, 0, 0])
            rotate([0, 0, a])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 16);
    }

    // Pointer groove on top face, same side as D flat (along +X)
    translate([-pointer_d, 0, knob_h - pointer_d])
        cube([pointer_w, knob_d + 2 * pointer_d, pointer_d + eps]);
}
```
