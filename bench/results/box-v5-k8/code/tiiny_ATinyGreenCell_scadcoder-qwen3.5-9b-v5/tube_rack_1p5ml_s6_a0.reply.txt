// Bench rack for 1.5 mL microcentrifuge tubes
// Print orientation: base on XY plane, holes open upward (+Z)

// ---------------- Parameters ----------------
block_len   = 106;   // block length along X (mm)
block_wid   = 72;    // block width along Y (mm)
block_ht    = 30;    // block height along Z (mm)

hole_count  = 24;    // total number of tube holes
hole_rows   = 4;     // number of rows along Y
hole_per_row = 6;    // number of holes per row along X

hole_pitch  = 16;    // centre-to-centre spacing in X and Y (mm)
hole_d      = 11.2;  // hole diameter (mm)
hole_depth  = 25;    // hole depth from top surface (mm)
floor_th    = block_ht - hole_depth; // remaining solid floor = 5 mm

eps         = 0.5;   // cutter overshoot past faces (mm)
$fn         = 96;    // circle resolution

// ---------------- Model ----------------
difference() {
    // Main block, centred in X/Y, sitting on z = 0
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_ht]);

    // Hole grid centred on the block
    for (r = 0; r < hole_rows; ++r)
        for (c = 0; c < hole_per_row; ++c) {
            x = (c - (hole_per_row - 1)/2) * hole_pitch;
            y = (r - (hole_rows - 1)/2) * hole_pitch;
            // Cylinder along Z, open at top, closed at bottom
            translate([x, y, block_ht - hole_depth])
                cylinder(d = hole_d, h = hole_depth + eps, $fn = 96);
        }
}
