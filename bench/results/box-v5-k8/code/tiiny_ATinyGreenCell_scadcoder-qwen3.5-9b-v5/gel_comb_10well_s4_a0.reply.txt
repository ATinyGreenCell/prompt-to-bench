```openscad
// 10-well comb for agarose gel casting, printed lying flat
// Units: mm

// ---- Parameters ----
plate_t      = 1.5;    // plate thickness (Z)
spine_l      = 70;     // spine length (X)
spine_w      = 12;     // spine width (Y)
teeth_n      = 10;     // number of teeth
teeth_w      = 5.0;    // tooth width (X)
teeth_l      = 12;     // tooth length (Y)
teeth_gap    = 1.5;    // gap between teeth (X)
overcut      = 0.5;    // extra length on cutting shapes past faces (Z)

// ---- Derived ----
teeth_f = teeth_n * teeth_w + (teeth_n + 1) * teeth_gap;  // total row length (X)
y0      = -spine_w / 2 + teeth_l / 2;                     // aligns teeth with -Y edge

// ---- Model ----
union() {
    // Spine (back bar)
    translate([0, 0, 0])
        cube([spine_l, spine_w, plate_t]);

    // Teeth, centred on spine in X
    for (i = [0 : teeth_n - 1]) {
        translate([-teeth_f / 2 + i * (teeth_w + teeth_gap), y0, 0])
            cube([teeth_w, teeth_l, plate_t + overcut]);
    }
}
```
