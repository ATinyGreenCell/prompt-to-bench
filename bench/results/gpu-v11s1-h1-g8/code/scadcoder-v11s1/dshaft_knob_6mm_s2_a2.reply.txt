// Hotplate stirrer knob - D-shaft 6 mm
// Print orientation: bottom face (with D-bore) on the bed, z = 0

$fn = 96;

// ---- Dimensions (mm) ----
knob_d        = 30;    // knob diameter
knob_h        = 15;    // knob height
bore_d        = 6.2;   // D-bore circle diameter
bore_depth    = 12;    // D-bore depth from bottom face
bore_flat     = bore_d - 4.7;  // flat width (circle diameter - chord length)
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // groove diameter
pointer_w     = 1.5;   // pointer groove width
pointer_d     = 1;     // pointer groove depth
eps           = 0.5;   // overshoot for clean cuts

difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, -eps])
        difference() {
            cylinder(d = bore_d, h = bore_depth + eps);
            // flat cut on one side of the D
            translate([-bore_flat / 2, -bore_d / 2, 0])
                cube([bore_flat, bore_d, bore_depth + eps]);
        }

    // Vertical grip grooves, evenly spaced
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps);

    // Pointer groove on top face, from centre to outer edge
    translate([0, 0, knob_h - pointer_d])
        cylinder(d = pointer_w, h = pointer_d + eps);
}
