// Replacement knob for hotplate stirrer (D-shaft 6 mm)
// Print orientation: bottom face (with D-bore) on bed, z = 0

$fn = 96;

// --- Knob body ---
knob_d      = 30;    // outer diameter
knob_h      = 15;    // height

// --- D-blind bore (from bottom face) ---
bore_d      = 6.2;   // circle diameter
bore_depth  = 12;    // depth from bottom face
bore_flat   = bore_d - 4.7;  // flat length so flat-to-opposite side = 4.7 mm

// --- Grip grooves (outside, full height) ---
groove_n    = 18;    // number of evenly spaced grooves
groove_d    = 2;     // half-cylinder diameter

// --- Pointer groove (top face) ---
pointer_w   = 1.5;   // width
pointer_d   = 1;     // depth
pointer_t   = knob_h * 0.8;   // length from centre to near outer edge

eps         = 0.5;   // overcut for clean differences

difference() {
    // Main body, centred in XY, sitting on z = 0
    cylinder(d = knob_d, h = knob_h);

    // D-blind bore: circle with one side cut flat, centred on -X
    translate([-(bore_d / 2 + eps), -(bore_d / 2 + eps), eps])
        rotate([-90, 0, 0])
            cylinder(d = bore_d, h = bore_depth + eps);

    // Grip grooves along the full height, centred on the surface
    for (i = [0 : groove_n - 1])
        rotate([0, 0, i * 360 / groove_n])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps);

    // Pointer groove on top face, same side as the D-flat
    translate([0, -pointer_w / 2, knob_h - pointer_d])
        cube([pointer_t, pointer_w, pointer_d + eps]);
}
