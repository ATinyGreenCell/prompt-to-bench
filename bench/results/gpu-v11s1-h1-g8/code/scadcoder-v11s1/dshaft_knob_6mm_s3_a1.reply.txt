// Hotplate stirrer knob for D-shaft
// Print orientation: flat bottom on bed (z = 0), builds upward

// ---------- Parameters ----------
shaft_d      = 6;     // shaft diameter (reference for bore size)
knob_d       = 30;    // knob outer diameter
knob_h       = 15;    // knob height
bore_depth   = 12;    // blind bore depth from bottom face
bore_flat    = 4.7;   // D-bore flat width (distance from flat to opposite side = 4.7 mm)
groove_count = 18;    // number of vertical grip grooves
groove_d     = 2;     // groove diameter (half-cylinder)
pointer_w    = 1.5;   // pointer groove width
pointer_d    = 1;     // pointer groove depth
eps          = 0.5;   // overcut for clean boolean differences
$fn          = 96;    // curve resolution

// ---------- Derived ----------
bore_r = bore_d / 2;               // bore radius (using D-shaft diameter as reference)
bore_x = -(bore_d - bore_flat) / 2; // offset of bore centre toward flat side

// ---------- Model ----------
difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore
    translate([bore_x, 0, -eps])
        linear_extrude(height = bore_depth + eps)
            circle(r = bore_r, $fn = 64);

    // Vertical grip grooves
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2 - groove_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps);

    // Pointer groove from centre to outer edge
    translate([knob_d / 2 - eps, -pointer_w / 2, knob_h - pointer_d])
        cube([eps + pointer_d, pointer_w, pointer_d + eps]);
}
