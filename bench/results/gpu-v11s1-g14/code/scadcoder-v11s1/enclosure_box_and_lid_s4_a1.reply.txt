```openscad
// Two-part enclosure for small electronics board
// Print orientation: box standing on the bed, lid on top (overlapping slightly)

$fn = 64;

// ---- Box ----
box_x = 70;    // outer length (X)
box_y = 50;    // outer width (Y)
box_z = 30;    // outer height (Z)
wall  = 2;     // wall thickness
floor = 2;     // floor thickness (separate from wall)

// ---- Lid ----
lip_x   = 65.6;      // outer length of lip (X)
lip_y   = 45.6;      // outer width of lip (Y)
lip_w   = 1.5;       // lip thickness (Z)
lip_h   = 4;         // lip height above plate (Z)
plate_t = 2;         // plate thickness (Z)
eps     = 0.5;       // overlap/overhang allowance

// ---- Layout ----
gap = 10;       // gap between box and lid (X direction)

module box() {
    difference() {
        union() {
            // Walls
            for (sx = [-1, 1], sy = [-1, 1])
                translate([sx > 0 ? wall : 0, sy > 0 ? box_y - wall : 0, 0])
                    cube([box_x - 2 * wall, box_y - 2 * wall, box_z]);
            // Floor
            translate([0, 0, floor])
                cube([box_x, box_y, box_z - floor]);
        }
        // Open top cut (extends above top face)
        translate([wall, wall, box_z - 1])
            cube([box_x - 2 * wall, box_y - 2 * wall, box_z + 1]);
    }
}

module lid() {
    union() {
        // Base plate
        translate([0, 0, plate_t])
            cube([lip_x, lip_y, plate_t]);
        // Lip on top of plate
        translate([0, 0, plate_t])
            cube([lip_x, lip_y, lip_h + eps]);
    }
}

difference() {
    union() {
        box();
        translate([box_x + gap, 0, 0])
            lid();
    }
    // Keep the lid's open-top cut clear of any lid material
    translate([box_x + gap - 1, -1, box_z - 1])
        cube([lip_x + 2, lip_y + 2, box_z + 1]);
}
```
