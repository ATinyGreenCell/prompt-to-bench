```openscad
// Dimensions
base_length = 90;  // Length of the base block (X)
base_width = 45;   // Width of the base block (Y)
base_height = 20;  // Height of the base block (Z)
slot_length = 77;  // Length of each slot (X)
slot_width = 1.6;  // Width of each slot (Y)
slot_depth = 10;   // Depth of each slot (Z)
slot_spacing = 3.5; // Center-to-center spacing of slots (Y)
num_slots = 10;    // Number of slots

// Set the number of fragments for curved shapes
$fn = 64;

// Base block
base = cube(size = [base_length, base_width, base_height]);

// Slot dimensions and positioning
slot_start_y = (base_width - (num_slots * slot_width + (num_slots - 1) * slot_spacing)) / 2;

// Create slots using a for loop
slots = union() {
    for (i = [0:num_slots-1]) {
        translate([0, slot_start_y + i * (slot_width + slot_spacing), base_height - slot_depth])
            cube(size = [slot_length, slot_width, slot_depth + 0.01], center = false);
    }
};

// Final part: base with slots cut out
drying_rack = difference() {
    base;
    slots;
};

// Center the drying rack on the origin
translate([-base_length/2, -base_width/2, 0])
    drying_rack;
```
