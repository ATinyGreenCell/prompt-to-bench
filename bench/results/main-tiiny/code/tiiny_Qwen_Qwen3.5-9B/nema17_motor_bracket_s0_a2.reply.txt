```openscad
$fn = 128;

// --- Dimensions ---
// Base Plate
base_x_start = -25;
base_x_end = 25;
base_y_start = 0;
base_y_end = 45;
base_z_thick = 5;

// Base Mounting Holes (M4)
hole_m4_diameter = 4.5;
hole_m4_x1 = -17;
hole_m4_y1 = 30;
hole_m4_x2 = 17;
hole_m4_y2 = 30;

// Vertical Motor Plate
motor_plate_x_start = -25;
motor_plate_x_end = 25;
motor_plate_y_start = 0;
motor_plate_y_end = 5;
motor_plate_z_height = 50;

// Motor Plate Through Hole (Centring Boss)
boss_hole_diameter = 23;
boss_hole_x = 0;
boss_hole_z = 28;

// Motor Plate Screw Holes (M3)
screw_hole_diameter = 3.4;
screw_pattern_size = 31; // 31mm square pattern
screw_hole_count = 4;

// Gussets
gusset_thickness = 5;
gusset_leg_y = 20;
gusset_leg_z = 20;
gusset_right_angle_y = 5;
gusset_right_angle_z = 5;

// --- Construction ---

module base_plate() {
    // Create the main L-shape block
    // Base: x[-25, 25], y[0, 45], z[0, 5]
    // Vertical Plate: x[-25, 25], y[0, 5], z[0, 50]
    // Combined bounding box: x[-25, 25], y[0, 45], z[0, 50]
    // Width = 50, Height = 50, Depth = 45
    difference() {
        cube([base_x_end - base_x_start, base_y_end - base_y_start, motor_plate_z_height], center = false);
        
        // Cut Base Mounting Holes (M4)
        // Position: (x = -17, y = 30) and (x = 17, y = 30)
        // Extend slightly past faces for clean cut
        for (i = [0, 1]) {
            translate([
                i == 0 ? hole_m4_x1 : hole_m4_x2,
                hole_m4_y1,
                0
            ]) {
                cylinder(
                    h = motor_plate_z_height + 1,
                    r = hole_m4_diameter / 2 + 0.1,
                    center = true
                );
            }
        }
    }
}

module motor_plate_holes() {
    // Cut the boss hole and screw holes from the vertical plate
    // Boss Hole: x=0, z=28, diameter 23mm (along Y)
    // Screw Holes: 4 holes, diameter 3.4mm, 31mm square pattern centered on boss hole
    
    // Boss Hole
    translate([boss_hole_x, 0, boss_hole_z]) {
        cylinder(
            h = motor_plate_y_end - base_y_start + 1, // Cut through full height of vertical plate
            r = boss_hole_diameter / 2 + 0.1,
            center = true
        );
    }
    
    // Screw Holes
    // Pattern center is at (0, 0, 28)
    // Square pattern size 31mm means distance from center to hole is 15.5mm
    // Coordinates: x = +/- 15.5, z = +/- 15.5
    for (i = [-1, 1]) {
        for (j = [-1, 1]) {
            translate([
                i * (screw_pattern_size / 2),
                0,
                j * (screw_pattern_size / 2) + boss_hole_z
            ]) {
                cylinder(
                    h = motor_plate_y_end - base_y_start + 1,
                    r = screw_hole_diameter / 2 + 0.1,
                    center = true
                );
            }
        }
    }
}

module gusset() {
    // Right triangle in YZ plane
    // Legs: 20mm along +Y, 20mm along +Z
    // Right angle corner at (y=5, z=5)
    // Thickness: 5mm along X (flush with edges)
    // Left Gusset (x = -25 to -20)
    // Right Gusset (x = 20 to 25)
    
    // Define the polygon vertices in the YZ plane relative to the corner (5, 5)
    // Points: (5, 5) -> (25, 5) -> (5, 25)
    // This creates a right triangle with legs 20mm along +Y and +Z.
    
    // Left Gusset
    translate([-25, 0, 0]) {
        linear_extrude(height = gusset_thickness, center = false) {
            polygon([
                [gusset_right_angle_y, gusset_right_angle_z], // Corner (5, 5)
                [gusset_right_angle_y + gusset_leg_y, gusset_right_angle_z], // (25, 5)
                [gusset_right_angle_y, gusset_right_angle_z + gusset_leg_z]  // (5, 25)
            ]);
        }
    }
    
    // Right Gusset
    translate([20, 0, 0]) {
        linear_extrude(height = gusset_thickness, center = false) {
            polygon([
                [gusset_right_angle_y, gusset_right_angle_z], // Corner (5, 5)
                [gusset_right_angle_y + gusset_leg_y, gusset_right_angle_z], // (25, 5)
                [gusset_right_angle_y, gusset_right_angle_z + gusset_leg_z]  // (5, 25)
            ]);
        }
    }
}

// --- Final Assembly ---
difference() {
    union() {
        base_plate();
        motor_plate_holes();
        gusset();
    }
}
```
