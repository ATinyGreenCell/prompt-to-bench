```openscad
// Parameters (mm)
plate_x = 130;          // Plate length (X direction)
plate_y = 90;           // Plate width  (Y direction)
plate_thick = 4;        // Plate thickness (Z direction)
hole_d = 30.5;          // Diameter of tube holes
hole_spacing = 40;      // Centre‑to‑centre spacing of holes (both X and Y)
wall_thick = 4;         // Thickness of the side walls (Y direction)
wall_height = 70;       // Total height of the side walls (Z direction)
$fn = 64;                // Resolution for circles and cylinders

// Helper: plate base (centered on origin, Z from 0 to plate_thick)
module plate() {
    translate([-plate_x/2, -plate_y/2, 0])
        cube([plate_x, plate_y, plate_thick]);
}

// Helper: side walls (flush with long edges of the plate)
module side_walls() {
    // Wall runs along X, thickness in Y, rises from plate_thick to wall_height
    wall_z = wall_height - plate_thick; // height above the plate
    // Positive Y wall (inner face at Y = +plate_y/2)
    translate([-plate_x/2,  plate_y/2, plate_thick])
        cube([plate_x, wall_thick, wall_z]);
    // Negative Y wall (inner face at Y = -plate_y/2)
    translate([-plate_x/2, -plate_y/2 - wall_thick, plate_thick])
        cube([plate_x, wall_thick, wall_z]);
}

// Helper: holes (through the plate only)
module plate_holes() {
    // Cylinder height slightly larger than plate to avoid coplanar faces
    cyl_h = plate_thick + 0.2; // extra 0.1 mm on each side
    cyl_z = -0.1; // start just below the plate bottom
    for (ix = [-1, 0, 1])          // three columns
        for (iy = [-1, 1])          // two rows
            translate([ix*hole_spacing, iy*hole_spacing, cyl_z])
                cylinder(d = hole_d, h = cyl_h, center = false);
}

// Assemble the rack
union() {
    plate();
    side_walls();
}
difference() {
    // The object we just built (union of plate and walls)
    union() {
        plate();
        side_walls();
    }
    // Subtract only the plate holes
    plate_holes();
}
```
