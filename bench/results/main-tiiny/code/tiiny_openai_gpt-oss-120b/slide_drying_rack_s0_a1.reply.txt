```openscad
// ---------------
// Parameters (mm)
// ---------------
base_len   = 90;    // X length of the base block
base_wid   = 45;    // Y width of the base block
base_h     = 20;    // Z height of the base block

slot_len   = 77;    // Length of each slot along X
slot_wid   = 1.6;   // Width of each slot along Y
slot_depth = 10;    // Depth of each slot down from the top face

slot_count = 10;    // Number of slots
slot_spacing = 3.5; // Centre‑to‑centre spacing of slots in Y

extra = 0.2;        // Clearance added to each cut face (mm)

// ---------------
// Model
// ---------------
module base_block() {
    // Bottom at Z=0, centred in X and Y
    translate([0, 0, base_h/2])
        cube([base_len, base_wid, base_h], center = true);
}

// Cut shape for a single slot (centered at its own origin)
module slot_cut() {
    // Cube that extends a little beyond the slot dimensions
    cube([slot_len + extra,
          slot_wid + extra,
          slot_depth + extra],
         center = true);
}

// Assemble the whole part
difference() {
    // Solid base
    base_block();

    // Slots cut out of the top face
    for (i = 0; i < slot_count; i = i + 1) {
        // Compute Y centre of the i‑th slot
        // Total span of the slot group = (slot_count-1)*slot_spacing
        total_span = (slot_count - 1) * slot_spacing;
        y_center = -total_span/2 + i * slot_spacing;

        // Position the cut shape:
        // - X centre = 0 (block centred)
        // - Y centre = y_center
        // - Z centre = base_h - slot_depth/2
        //   (so the cut extends just above the top surface)
        translate([0, y_center, base_h - slot_depth/2])
            slot_cut();
    }
}
```
