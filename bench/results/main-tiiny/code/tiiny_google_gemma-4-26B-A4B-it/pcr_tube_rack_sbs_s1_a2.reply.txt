// --- Dimensions (mm) ---
block_x = 127.76;          // Total length (X)
block_y = 85.48;           // Total width (Y)
block_z = 20.00;           // Total height (Z)

hole_dia = 6.2;            // Diameter of PCR tube holes
hole_spacing = 9.0;        // Center-to-center spacing

rows = 8;                  // Number of rows (Y direction)
cols = 12;                 // Number of columns (X direction)

// Offset from edges to the center of the first hole (A1)
offset_x = 14.38;          // Distance from left edge to first hole center
offset_y = 11.24;          // Distance from back edge to first hole center

// Chamfer parameters
chamfer_leg = 5.0;         // Length of the chamfer legs

// --- Rendering Settings ---
$fn = 64;

// --- Main Construction ---

// The user requested:
// 1. Rest on XY plane at z = 0 and build upward (+Z).
// 2. Centered on the origin in X and Y.
// 3. 96 vertical through-holes (6.2mm dia) in 8 rows by 12 columns.
// 4. Spacing 9.0mm.
// 5. A1 center is 14.38 from left edge and 11.24 from back edge.
// 6. 45-degree chamfer with 5mm legs on the vertical edge at the A1 corner.

module pcr_rack() {
    difference() {
        // Main Block
        // To be centered on X and Y and start at Z=0:
        translate([-block_x/2, -block_y/2, 0])
            cube([block_x, block_y, block_z]);

        // 96 Holes
        // A1 is the first hole. 
        // Left edge is at -block_x/2. 
        // Back edge is at block_y/2.
        // Hole X = (-block_x/2) + offset_x + (col_index * spacing)
        // Hole Y = (block_y/2) - offset_y - (row_index * spacing)
        for (i = [0 : cols - 1]) {
            for (j = [0 : rows - 1]) {
                translate([
                    (-block_x / 2) + offset_x + (i * hole_spacing), 
                    (block_y / 2) - offset_y - (j * hole_spacing), 
                    -0.01 // Slight overlap to ensure clean cut through bottom
                ])
                cylinder(h = block_z + 0.02, d = hole_dia);
            }
        }

        // 45-degree chamfer at the A1 corner
        // The A1 corner is the vertical edge where X = -block_x/2 and Y = block_y/2.
        // We use a wedge to cut this edge.
        // The wedge is a prism with a triangular base.
        // We translate it to the corner and extend it through the Z height.
        translate([-block_x/2 - 1, block_y/2 - chamfer_leg, -0.01])
        linear_extrude(height = block_z + 0.02)
        polygon(points=[[0, 0], [chamfer_leg, 0], [0, chamfer_leg]]);
    }
}

// Execute the model
pcr_rack();
