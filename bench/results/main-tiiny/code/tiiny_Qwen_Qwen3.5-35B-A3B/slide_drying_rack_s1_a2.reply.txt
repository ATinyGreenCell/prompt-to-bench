```openscad
// Drying rack for 10 microscope slides
// Units: millimetres

// Base block dimensions
base_length = 90;      // X dimension
base_width = 45;       // Y dimension
base_height = 20;      // Z dimension

// Slide dimensions (for reference)
slide_length = 75;
slide_width = 25;
slide_thickness = 1;

// Slot dimensions
slot_length = 77;      // X dimension
slot_width = 1.6;      // Y dimension
slot_depth = 10;     // Z dimension (depth from top)

// Slot placement
num_slots = 10;
slot_spacing = 3.5;    // Centre-to-centre spacing in Y

// Calculate total width of slot group
total_slot_width = (num_slots - 1) * slot_spacing;

// Calculate offset to centre the slots on the base
slot_offset_y = (base_width - total_slot_width) / 2;

// Calculate X offset to centre the slots on the base
slot_offset_x = (base_length - slot_length) / 2;

// Base block
base = cube([base_length, base_width, base_height], center = true);

// Create the slots
for (i = [0 : num_slots - 1]) {
    // Calculate Y position for each slot
    slot_y = slot_offset_y + i * slot_spacing;
    
    // Create a slot cutter
    // The base is centered at Z=0, so top face is at Z=10.
    // We want to cut from Z=10 down to Z=0 (depth 10).
    // To ensure no coplanar faces, we extend the cutter slightly past the bottom.
    // Cutter top: Z = 10 (aligned with base top)
    // Cutter height: 10 + 0.01 (extends 0.01mm past the bottom face)
    // Cutter center Z: 10 - (10 + 0.01)/2 = 10 - 5.005 = 4.995
    // Alternatively, simpler: translate to top face, then use cube with height.
    // Let's use translate to the top face and a cube that goes down.
    // Top face Z = 10.
    // Cutter Z position: top = 10, bottom = -0.01 (to go past 0).
    // Center Z = (10 + -0.01) / 2 = 4.995.
    // Or simply: translate([slot_offset_x, slot_y, 10 - slot_depth - 0.01])
    // Wait, if we use center=false, we need to position the bottom of the cube at 0 - 0.01.
    // Let's position the cube so its top is at 10 and it extends down to -0.01.
    // Cube size: [slot_length, slot_width, slot_depth + 0.02]
    // Position: x = slot_offset_x, y = slot_y, z = 10 - (slot_depth + 0.02)
    // This places the bottom of the cube at 10 - (10 + 0.02) = -0.02.
    // This is safe.
    
    slot_cutter = translate([slot_offset_x, slot_y, 10 - (slot_depth + 0.02)])
                  * cube([slot_length, slot_width, slot_depth + 0.02], center = false);
    
    base = difference() {
        base;
        slot_cutter;
    }
}

// Render the part
base;
```
