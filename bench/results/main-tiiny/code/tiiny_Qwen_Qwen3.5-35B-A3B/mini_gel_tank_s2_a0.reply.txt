```openscad
// Buffer tank for mini horizontal gel-electrophoresis box
// Dimensions in millimetres (mm)

// Outer dimensions
outer_x = 120;
outer_y = 70;
outer_z = 40;

// Wall and floor thickness
wall_thickness = 3;
floor_thickness = 3;

// Platform dimensions
platform_length_x = 60;
platform_height_offset = 10; // Height above inner floor
platform_width_y = outer_y - 2 * wall_thickness; // Spans full inner width

// Electrode hole dimensions
hole_diameter = 2;
hole_z = 30;
hole_y = 0;

// Calculate inner dimensions
inner_x = outer_x - 2 * wall_thickness;
inner_y = outer_y - 2 * wall_thickness;
inner_z = outer_z - floor_thickness;

// Calculate platform top Z position
platform_top_z = floor_thickness + platform_height_offset;

// Calculate hole position relative to bottom
// Hole is at z=30, floor is at z=0, so hole is 30mm from bottom
// Wall thickness is 3mm, so inner wall is at 3mm
// Hole is drilled through the end walls (X ends)

module tank_body() {
    // Main tank shell (outer box)
    difference() {
        // Outer solid block
        cube([outer_x, outer_y, outer_z], center = true);
        
        // Subtract inner cavity
        translate([0, 0, floor_thickness])
            cube([inner_x, inner_y, inner_z], center = true);
    }
}

module platform() {
    // Platform is a solid block
    // Centered in X, spans full inner Y, height from floor + offset
    translate([0, 0, floor_thickness + platform_height_offset])
        cube([platform_length_x, platform_width_y, inner_z - platform_height_offset], center = true);
}

module electrode_holes() {
    // Two holes, one at each X end (left and right)
    // Hole center at y=0, z=30
    // Hole diameter 2mm
    
    // Left hole (at x = -outer_x/2)
    translate([-outer_x/2, 0, hole_z])
        cylinder(d = hole_diameter, h = outer_y + 10, center = true);
        
    // Right hole (at x = +outer_x/2)
    translate([outer_x/2, 0, hole_z])
        cylinder(d = hole_diameter, h = outer_y + 10, center = true);
}

// Main assembly
union() {
    // Tank body
    tank_body();
    
    // Raised platform
    platform();
    
    // Electrode holes (cut out from the tank)
    // We need to subtract the holes from the tank body
    // The holes go through the end walls, so we subtract them from the main body
    // Since the platform is inside, we need to ensure the holes don't cut the platform
    // The platform is at z=13 to z=23 (floor=3, height=10, so top=13, bottom=3)
    // The holes are at z=30, so they are above the platform, no conflict
    
    // Subtract the holes from the tank body
    difference() {
        tank_body();
        electrode_holes();
    }
}

// Final assembly: Tank body with platform and holes
// We need to combine the tank body (with holes) and the platform
// The platform is inside the tank, so we union them

// Create the final part
union() {
    // Tank body with holes
    difference() {
        // Outer tank
        cube([outer_x, outer_y, outer_z], center = true);
        // Inner cavity
        translate([0, 0, floor_thickness])
            cube([inner_x, inner_y, inner_z], center = true);
        // Electrode holes
        translate([-outer_x/2, 0, hole_z])
            cylinder(d = hole_diameter, h = outer_y + 10, center = true);
        translate([outer_x/2, 0, hole_z])
            cylinder(d = hole_diameter, h = outer_y + 10, center = true);
    }
    
    // Platform
    translate([0, 0, floor_thickness + platform_height_offset])
        cube([platform_length_x, platform_width_y, inner_z - platform_height_offset], center = true);
}
```
