```openscad
// Pestle for 1.5 mL microcentrifuge tube, printed upright
// All dimensions in millimetres

$fn = 64;

// Handle
handle_d = 8;          // handle diameter
handle_r = handle_d / 2;
handle_z0 = 0;         // bottom of handle
handle_z1 = 45;        // top of handle / start of cone

// Cone (taper)
cone_d_bottom = 8;     // cone diameter at z = 45
cone_d_top = 3;        // cone diameter at z = 62
cone_z0 = 45;
cone_z1 = 62;

// Rounded tip
tip_d = 3;             // sphere diameter
tip_r = tip_d / 2;
tip_cz = 62;           // sphere centre height

// Grip grooves (cut into handle)
groove_w = 2;          // groove width in Z
groove_depth = 1;      // groove depth (radial)
groove_zs = [10, 15, 20, 25];  // groove centre heights

// ---- Build the solid body (handle + cone + tip) ----
body = union() {
    // Cylindrical handle
    translate([0, 0, handle_z0])
        cylinder(h = handle_z1 - handle_z0, r = handle_r, $fn = $fn);

    // Tapering cone (8 mm dia at z=45 -> 3 mm dia at z=62)
    translate([0, 0, cone_z0])
        cylinder(h = cone_z1 - cone_z0,
                 r1 = cone_d_bottom / 2,
                 r2 = cone_d_top / 2,
                 $fn = $fn);

    // Rounded tip: sphere centred at z = 62
    translate([0, 0, tip_cz])
        sphere(r = tip_r, $fn = $fn);
};

// ---- Cut four circumferential grip grooves into the handle ----
// Each groove is a torus (ring) whose tube radius equals the groove depth,
// centred on the handle axis at the given height. The torus extends past the
// handle surface by exactly the depth, so the cut is clean and not coplanar.
grooves = for (gz = groove_zs)
    translate([0, 0, gz])
        rotate_extrude($fn = $fn)
            circle(r = groove_depth, $fn = $fn);

// Final part: body minus grip grooves
difference() {
    body;
    grooves;
}
```
