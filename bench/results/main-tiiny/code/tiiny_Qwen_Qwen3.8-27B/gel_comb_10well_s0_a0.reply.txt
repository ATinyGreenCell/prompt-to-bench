// 10-well agarose gel comb, printed flat on the bed (Z = thickness)

// --- Dimensions (mm) ---
plate_thick   = 1.5;   // Z thickness of the whole comb
spine_len     = 70.0;  // X length of the back bar
spine_width   = 12.0;  // Y width of the back bar
tooth_w       = 5.0;   // X width of each tooth
tooth_len     = 12.0;  // Y length of each tooth (hangs in -Y from spine)
gap           = 1.5;   // X gap between neighbouring teeth
n_tooth       = 10;    // number of teeth

// --- Derived layout ---
row_width     = n_tooth * tooth_w + (n_tooth - 1) * gap;  // total X span of the tooth row
row_x0        = -row_width / 2.0;                          // left edge of the row (centred on X)
tooth_x0      = row_x0;                                    // X of the left edge of tooth 0
tooth_y0      = -tooth_len;                                // Y of the tip (bottom) of each tooth

$fn = 64;

// --- Model (rests on z = 0, centred on origin in X and Y) ---
union() {
    // Back bar (spine): centred on origin in X and Y
    translate([-spine_len/2, -spine_width/2, 0])
        cube([spine_len, spine_width, plate_thick]);

    // Teeth: hang from the -Y edge of the spine
    for (i = [0 : n_tooth - 1]) {
        tx = tooth_x0 + i * (tooth_w + gap);   // X of left edge of tooth i
        translate([tx, tooth_y0, 0])
            cube([tooth_w, tooth_len, plate_thick]);
    }
}
