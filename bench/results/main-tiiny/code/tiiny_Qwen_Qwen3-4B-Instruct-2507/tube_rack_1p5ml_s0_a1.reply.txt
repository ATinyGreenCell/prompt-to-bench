// Dimensions of the bench rack
$fn = 64;

// Block dimensions (in mm)
length_X = 106;
width_Y = 72;
height_Z = 30;

// Hole grid: 24 holes in 4 rows (Y direction), 6 holes per row (X direction)
// Centre-to-centre distance in both directions
hole_center_to_center = 16.0;
hole_diameter = 11.2;
hole_depth = 25.0;
floor_thickness = 5.0;  // solid floor at the bottom

// Calculate hole positions
num_holes_per_row = 6;
num_rows = 4;

// X positions: 6 holes per row, centered on origin
hole_x_positions = [];
for (i = [0 : num_holes_per_row - 1]) {
    hole_x_positions.push(-5 * hole_center_to_center / 2 + i * hole_center_to_center);
}

// Y positions: 4 rows, centered on origin
hole_y_positions = [];
for (i = [0 : num_rows - 1]) {
    hole_y_positions.push(-3 * hole_center_to_center / 2 + i * hole_center_to_center);
}

// Create the base solid block
block = cube([length_X, width_Y, height_Z], center = true);

// Create the final part using difference
// Start with the full block, then subtract each hole

// Define the hole as a cylinder from z = floor_thickness to z = floor_thickness + hole_depth
// To avoid coplanarity, extend the cylinder slightly beyond the top and bottom
// We use difference to remove the holes

// The hole is 11.2 mm in diameter, 25 mm deep, and starts at z = 5 mm (floor)
// So the cylinder goes from z = 5 to z = 30, and we extend it to z = 30.1 to avoid coplanarity

// Build the final part as a difference: base block minus all holes
difference() {
    // Full base block
    cube([length_X, width_Y, height_Z], center = true);
    
    // Subtract each hole
    for (i = [0 : num_rows - 1]) {
        for (j = [0 : num_holes_per_row - 1]) {
            x = hole_x_positions[j];
            y = hole_y_positions[i];
            
            // Translate to hole position
            translate([x, y, 0]) {
                // Cylinder with diameter 11.2 mm, depth 25 mm, from z=5 to z=30.1
                // This ensures the hole is fully cut and avoids coplanarity
                cylinder(
                    r = hole_diameter / 2,
                    h = hole_depth + 0.1,
                    center = false,
                    f = 64
                );
            }
        }
    }
}
