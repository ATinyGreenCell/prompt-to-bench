// Designed to fit 0.2 mL tube into a 1.5 mL rack hole
// Print orientation: collar on z=0 to z=1.5, body from z=1.5 to z=21.5

$fn = 64;

sleeve_outer_diameter = 10.8;
sleeve_height = 20.0;  // Corrected: from z=1.5 to z=21.5 is 20.0 mm height, not 21.5
bore_diameter = 6.2;
bore_radius = bore_diameter / 2;

// Corrected: The sleeve body must rise from z=1.5 to z=21.5, so total height is 20.0 mm
// The full part must have:
// - A collar from z=0 to z=1.5

// - A central bore from z=0 to z=23.0 (full height of collar + body)

difference() {
    // Base: collar (13 mm diameter, 1.5 mm thick)
    translate([0, 0, 0])
        cylinder(r = collar_diameter / 2, h = collar_thickness, $fn = 64);

    translate([0, 0, collar_thickness])

}
