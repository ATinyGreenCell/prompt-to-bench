// Build the tank body: outer dimensions 120 x 70 x 40, with 3 mm walls and 3 mm floor
// The inner cavity is from z = 3 to z = 40 (since floor is 3 mm thick), and x,y offset by wall_thickness

module tank() {
    difference() {
        // Outer box: from z = 0 to z = 40
        cube([outer_x, outer_y, outer_z], center = false);
        // Inner cavity: from z = 3 to z = 40, and x,y offset by wall_thickness
        translate([wall_thickness, wall_thickness, 3]) {
            cube([outer_x - 2*wall_thickness, outer_y - 2*wall_thickness, outer_z - 3], center = false);
        }
    }
}
