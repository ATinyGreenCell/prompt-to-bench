// Bench rack for 1.5 mL microcentrifuge tubes
// Corrected to match specifications exactly

// Overall block dimensions
length_x = 106;    // Length along X axis
width_y = 72;       // Width along Y axis
height_z = 30;      // Height along Z axis

// Hole grid parameters
hole_diameter = 11.2;           // Diameter of each hole
hole_depth = 25;                 // Depth of each hole (from top)
floor_thickness = 5;             // Solid floor thickness at bottom
hole_spacing_x = 16;           // Centre-to-centre spacing along X
hole_spacing_y = 16;            // Centre-to-centre spacing along Y
num_holes_x = 6;                // Number of holes along X (length)
num_holes_y = 4;                  // Number of holes along Y (width)

// Ensure the block sits on the XY plane (z = 0) and builds upward
// The original code used `center = true` which places the block symmetrically around z = 0
// But the specification says: "rests on the XY plane at z = 0 and builds upward (+Z)"
// So we must shift the block so that its bottom is at z = 0

// Adjusted: use cube with bottom at z = 0
// Also, the hole grid must be centred on the block in X and Y

// Calculate grid centre offset to centre the hole array on the block
grid_centre_x = length_x / 2;
grid_centre_y = width_y / 2;

// Calculate hole grid start offsets to centre the array
start_x = grid_centre_x - ((num_holes_x - 1) * hole_spacing_x) / 2;
start_y = grid_centre_y - ((num_holes_y - 1) * hole_spacing_y) / 2;

// Set resolution for smooth holes
$fn = 64;

// Main block: solid rectangular prism, sitting on XY plane (z = 0)
// Use translate to shift the block so that its bottom is at z = 0
// The original cube was centered at (0,0,0), so we shift it up by height_z/2
// But we want the bottom at z = 0, so we translate by (0, 0, height_z/2)

// However, we can build the block from z = 0 to z = height_z directly
// So we define the block as:
//   cube([length_x, width_y, height_z], center = false) placed at origin

// Now create the holes using difference
difference() {
    // Main block: from z = 0 to z = height_z
    cube([length_x, width_y, height_z], center = false);

    // Subtract cylinders for each hole
    for (i = [0 : num_holes_x - 1]) {
        for (j = [0 : num_holes_y - 1]) {
            // Calculate hole centre
            hole_x = start_x + i * hole_spacing_x;
            hole_y = start_y + j * hole_spacing_y;

            // Create cylinder hole: from z = 0 to z = hole_depth (25 mm)
            // The hole is open at the top (z = 25) and closed at the bottom (z = 0) with a 5 mm floor
            // So the hole should be 25 mm deep, meaning it goes from z = 0 to z = 25
            // But to ensure clean cut, extend slightly below the floor (z = 0) and above (z = 25)
            // We'll use a cylinder from z = -0.1 to z = 25.1 to ensure full removal

            cylinder(
                h = height_z + 0.1,  // Slightly over to ensure full cut
                d = hole_diameter,
                center = false,
                $fn = 64
            );
            translate([hole_x, hole_y, -0.1])
                cylinder(
                    h = height_z + 0.1,
                    d = hole_diameter,
                    center = false,
                    $fn = 64
                );
        }
    }
}
