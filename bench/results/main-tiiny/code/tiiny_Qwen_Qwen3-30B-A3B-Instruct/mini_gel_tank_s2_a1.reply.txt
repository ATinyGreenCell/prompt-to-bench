inner_x = outer_x - 2 * wall_thickness;
inner_y = outer_y - 2 * wall_thickness;
inner_z = outer_z - floor_thickness;

    difference() {

    }

    }
}

// Create the two end-wall holes (horizontal, along X, at y=0, z=30)

    // Cylinder along X-axis, centered at origin, length = wall_thickness + 1 to ensure cut
    cylinder(r = hole_radius, h = wall_thickness + 1, $fn = 64, center = true);
}

// Place a hole at a given X position (on the end wall), at y=0, z=30
module hole_at_end(x_pos) {
    // Translate to the end wall (x_pos), at y=0, z=30
    // Then rotate to align cylinder along X-axis (so it's horizontal)
    translate([x_pos, 0, 30]) {
        rotate([90, 0, 0]) {
            end_hole();
        }
    }
}

module tank() {
    difference() {

        cube([outer_x, outer_y, outer_z], center = false);

    // Rotate to align cylinder along X-axis (so it's horizontal)

// Main assembly: build the tank, add the platform, then subtract the holes
// We do this in one union to ensure solid is watertight

    tank();

}

// Now subtract the two end holes (through the end walls)
difference() {
    // Start with the unioned body (tank + platform)
    union() {
        tank();
        platform();
    }

    // Subtract the left hole (at x = outer_x/2 - wall_thickness)
    hole_at_end(outer_x/2 - wall_thickness);

    // Subtract the right hole (at x = -outer_x/2 + wall_thickness)
    hole_at_end(-outer_x/2 + wall_thickness);
}
