```openscad
// Dimensions for the PCR tube adapter sleeve
$fn = 64; // Ensure smooth curves

// User-specified dimensions
collar_diameter = 13;           // Diameter of the base collar (z = 0 to 1.5 mm)
collar_height = 1.5;           // Height of the collar
sleeve_outer_diameter = 10.8; // Outer diameter of the sleeve body (z = 1.5 to 21.5 mm)
sleeve_height = 20;        // Height of the sleeve body (from z = 1.5 to 21.5 mm)
bore_diameter = 6.2;        // Diameter of the central bore through the full height

// Center the part on the origin in X and Y
// The collar is from z = 0 to z = 1.5
// The sleeve body extends from z = 1.5 to z = 21.5 (total height 21.5 mm)

// Create the collar (base) - placed at z = 0
collar = cylinder(h = collar_height, d = collar_diameter, center = false);

// Create the sleeve body (cylindrical section from z = 1.5 to 21.5)
sleeve_body = cylinder(h = sleeve_height, d = sleeve_outer_diameter, center = false);

// Create the bore (a hole through the full height, centered)
// Extend slightly beyond the collar and sleeve to ensure clean subtraction
bore = cylinder(h = collar_height + sleeve_height + 0.1, d = bore_diameter, center = false);

// Assemble the full adapter: collar + sleeve body, with bore subtracted
difference() {
    union() {
        translate([0, 0, 0]) collar;
        translate([0, 0, collar_height]) sleeve_body;
    }
    translate([0, 0, -0.05]) bore;
}
```
