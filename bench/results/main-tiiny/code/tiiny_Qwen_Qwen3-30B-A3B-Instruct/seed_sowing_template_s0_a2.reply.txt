// Dimensions and parameters
$fn = 64; // Ensure smooth curves

// Petri dish inner diameter: 90 mm
// Template disc: 85 mm diameter, 2 mm thick
disc_diameter = 85;
disc_thickness = 2;

// Grid: 7x7 holes, 3 mm diameter, 9 mm center-to-center spacing
hole_diameter = 3;
grid_spacing = 9;
grid_rows = 7;
grid_cols = 7;

// Notch: 10 mm wide (X), 4 mm deep (Y), on +Y side, centered on Y axis
notch_width = 10;
notch_depth = 4;

// Calculate grid center offset (to center grid on disc)
grid_center_offset = (grid_spacing * (grid_rows - 1)) / 2;

// Create the base disc, centered at origin and resting on z = 0
module base_disc() {
    // Cube centered in X and Y, but shifted so that bottom is at z = 0
    // So we place the center at (0, 0, -disc_thickness / 2)
    translate([0, 0, -disc_thickness / 2])
    cube([disc_diameter, disc_diameter, disc_thickness], center = true);
}

// Create the grid of through-holes
module hole_grid() {
    for (i = [0 : grid_rows - 1]) {
        for (j = [0 : grid_cols - 1]) {
            // Calculate hole center position (offset from center)
            x = (j - (grid_cols - 1) / 2) * grid_spacing;
            y = (i - (grid_rows - 1) / 2) * grid_spacing;
            // Hole extends through the full thickness, centered in Z
            translate([x, y, -disc_thickness / 2])
            cylinder(h = disc_thickness + 0.02, r = hole_diameter / 2, center = true);
        }
    }
}

// Create the orientation notch on the +Y rim
module orientation_notch() {
    // Notch is centered on Y axis, extends 10 mm in X, 4 mm inward from rim
    // Rim is at Y = disc_diameter / 2 = 42.5 mm
    // Notch goes from Y = 42.5 - notch_depth = 38.5 mm to Y = 42.5 mm
    // X spans from -notch_width/2 to +notch_width/2 = -5 to +5
    // Cut from top (z = 0) down to z = -disc_thickness (but we extend slightly below to ensure full removal)
    // The notch is a rectangular hole through the disc, so we cut from z = -disc_thickness to z = 0
    // We use a cube that extends slightly beyond the disc in Y and Z to ensure full removal
    translate([0, disc_diameter / 2 - notch_depth, -disc_thickness - 0.01])
    cube([notch_width, notch_depth + 1, disc_thickness + 0.02], center = false);
}

// Assemble the final part: disc with holes and notch
difference() {
    base_disc();
    hole_grid();
    orientation_notch();
}
