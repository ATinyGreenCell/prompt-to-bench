// Pestle for 1.5 mL microcentrifuge tube
// Dimensions measured in mm

// Handle dimensions
handle_diameter = 8;
handle_radius = handle_diameter / 2;
handle_height = 45;

// Cone dimensions
cone_bottom_diameter = 8;
cone_bottom_radius = cone_bottom_diameter / 2;
cone_top_diameter = 3;
cone_top_radius = cone_top_diameter / 2;
cone_bottom_z = 45;
cone_top_z = 62;
cone_height = cone_top_z - cone_bottom_z;

// Tip dimensions
tip_diameter = 3;
tip_radius = tip_diameter / 2;
tip_center_z = 62;

// Groove dimensions
groove_width = 2;
groove_depth = 1;
groove_centers = [10, 15, 20, 25];

$fn = 64;

// Main body: handle + cone + tip
union() {
    // Handle
    translate([0, 0, 0])
        cylinder(h = handle_height, r = handle_radius, center = false);

    // Cone
    translate([0, 0, cone_bottom_z])
        cylinder(h = cone_height, r1 = cone_bottom_radius, r2 = cone_top_radius, center = false);

    // Tip (sphere)
    translate([0, 0, tip_center_z])
        sphere(r = tip_radius);
}

// Cut grooves into the handle
difference() {
    // Empty - the union above is the base, but we need to subtract from it
    // We'll restructure: do the union first, then subtract grooves
}

// Restructured: build solid, then subtract grooves
difference() {
    union() {
        // Handle
        translate([0, 0, 0])
            cylinder(h = handle_height, r = handle_radius, center = false);

        // Cone
        translate([0, 0, cone_bottom_z])
            cylinder(h = cone_height, r1 = cone_bottom_radius, r2 = cone_top_radius, center = false);

        // Tip (sphere)
        translate([0, 0, tip_center_z])
            sphere(r = tip_radius);
    }

    // Grooves
    for (z_center = groove_centers) {
        translate([0, 0, z_center])
            cylinder(
                h = groove_width + 0.02,
                r = handle_radius - groove_depth,
                center = true
            );
    }
}
