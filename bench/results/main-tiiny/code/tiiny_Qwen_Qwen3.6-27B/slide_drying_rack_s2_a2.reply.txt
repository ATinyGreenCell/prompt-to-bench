```openscad
// Drying Rack for 10 Microscope Slides
// Dimensions measured with calipers

// Base block dimensions
base_length = 90;   // X
base_width  = 45;   // Y
base_height = 20;   // Z

// Slot dimensions
slot_length = 77;   // X
slot_width  = 1.6;  // Y
slot_depth  = 10;   // Z

// Slot arrangement
num_slots = 10;
slot_spacing = 3.5; // centre-to-centre in Y

// Rendering quality
$fn = 64;

// Calculate total width of the slot group
total_slot_group_width = (num_slots - 1) * slot_spacing;

// Start Y position so the group is centred on the block
start_y = -total_slot_group_width / 2;

// Build the solid base
difference() {
    // Main block, centred on origin in X and Y, sitting on z=0
    translate([-base_length/2, -base_width/2, 0])
        cube([base_length, base_width, base_height]);

    // Cut out the 10 slots
    for (i = [0 : num_slots - 1]) {
        y_pos = start_y + i * slot_spacing;
        
        // Slot extends slightly past the block in X to ensure clean cut
        translate([
            -slot_length/2, 
            y_pos, 
            base_height - slot_depth - 0.01 // extend 0.01mm past the bottom of the cut
        ])
        cube([
            slot_length + 0.02, // extend 0.01mm past each side in X
            slot_width, 
            slot_depth + 0.02   // extend 0.01mm past the top and bottom
        ]);
    }
}
```
