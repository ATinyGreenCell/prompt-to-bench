bore_diameter = 3;
collar_diameter = 12;

// Helper function to create a single sawtooth barb
module sawtooth_barb(bottom_diameter, top_diameter, height, z_offset) {
    half_height = height / 2;
    
    // Create the sawtooth profile using hull for the step transition
    hull() {
        // Bottom circle
        translate([0, 0, z_offset])
            circle(d = bottom_diameter, $fn = 64);
        
        // Top of first ramp (at half height)
        translate([0, 0, z_offset + half_height])
            circle(d = top_diameter, $fn = 64);
    }
    
    hull() {
        // Top of first ramp (at half height)
        translate([0, 0, z_offset + half_height])
            circle(d = top_diameter, $fn = 64);
        
        // Bottom circle at top of barb (step back)
        translate([0, 0, z_offset + height - 0.01])
            circle(d = bottom_diameter, $fn = 64);
    }
}

// Main body with bore
module main_body_with_bore() {
