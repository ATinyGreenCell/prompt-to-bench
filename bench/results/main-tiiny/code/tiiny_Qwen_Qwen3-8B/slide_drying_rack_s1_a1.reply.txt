```openscad
// Dimensions for the drying rack
base_length = 90;       // Length of the base in X direction
base_width = 45;        // Width of the base in Y direction
base_height = 20;       // Height of the base in Z direction

slot_length = 77;       // Length of each slot in X direction
slot_width = 1.6;       // Width of each slot in Y direction
slot_depth = 10;        // Depth of each slot in Z direction
slot_spacing = 3.5;     // Centre-to-centre spacing between slots in Y direction

// Calculate the number of slots and their positions
num_slots = floor(base_width / slot_spacing) - 1;
slot_start_y = (base_width - (num_slots * slot_spacing)) / 2;

// Create the base block
base = cube([base_length, base_width, base_height], center=true);

// Create the slots using a for loop
slots = union() for (i = [0 : num_slots]) {
    // Position each slot in Y direction
    translate([0, slot_start_y + i * slot_spacing, 0]) {
        // Create a slot as a cube with negative Z extrusion
        cube([slot_length, slot_width, -slot_depth], center=true);
    }
}

// Combine the base and slots
drying_rack = difference() {
    base
    // Add a small protrusion to the top of the slots to ensure they are cut properly
    // and not coplanar with the base
    for (i = [0 : num_slots]) {
        translate([0, slot_start_y + i * slot_spacing, base_height - 0.01]) {
            cube([slot_length, slot_width, 0.02], center=true);
        }
    }
}

// Final output
drying_rack;
```
