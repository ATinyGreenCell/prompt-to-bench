```openscad
// Define all dimensions
base_plate_x_width = 50; // x from -25 to 25
base_plate_y_height = 45; // y from 0 to 45
base_plate_z_height = 5; // z from 0 to 5

motor_plate_x_width = 50; // x from -25 to 25
motor_plate_y_height = 5; // y from 0 to 5
motor_plate_z_height = 50; // z from 0 to 50

centring_boss_diameter = 23; // diameter of the centring boss hole
centring_boss_x_center = 0;
centring_boss_y_center = 0;
centring_boss_z_center = 28;

screw_hole_diameter = 3.4; // diameter of M3 screw holes
screw_pattern_side = 31; // square pattern side length for screw holes
screw_pattern_offset = 15.5; // offset from center to align with motor plate

gusset_thickness = 5; // thickness of triangular gussets
gusset_leg_y = 20; // length along +Y
gusset_leg_z = 20; // length along +Z
gusset_corner_y = 5; // y position of the right-angle corner
gusset_corner_z = 5; // z position of the right-angle corner

$fn = 64;

// Base plate
base_plate = cube([base_plate_x_width, base_plate_y_height, base_plate_z_height]);

// Motor plate
motor_plate = cube([motor_plate_x_width, motor_plate_y_height, motor_plate_z_height]);

// Centring boss hole
centring_boss = cylinder(r = centring_boss_diameter / 2, h = motor_plate_y_height, $fn = 64);

// Screw holes
screw_holes = [
  [screw_pattern_offset, 0, 0],
  [screw_pattern_offset, screw_pattern_side, 0],
  [0, screw_pattern_side, 0],
  [0, 0, 0]
];

// Create screw hole cylinders
screw_hole_cylinders = [
  cylinder(r = screw_hole_diameter / 2, h = motor_plate_y_height, $fn = 64),
  cylinder(r = screw_hole_diameter / 2, h = motor_plate_y_height, $fn = 64),
  cylinder(r = screw_hole_diameter / 2, h = motor_plate_y_height, $fn = 64),
  cylinder(r = screw_hole_diameter / 2, h = motor_plate_y_height, $fn = 64)
];

// Position screw holes
screw_hole_positions = [
  translate([centring_boss_x_center, centring_boss_y_center, 0]) * screw_hole_cylinders[0],
  translate([centring_boss_x_center, centring_boss_y_center - screw_pattern_side, 0]) * screw_hole_cylinders[1],
  translate([centring_boss_x_center - screw_pattern_offset, centring_boss_y_center, 0]) * screw_hole_cylinders[2],
  translate([centring_boss_x_center + screw_pattern_offset, centring_boss_y_center, 0]) * screw_hole_cylinders[3]
];

// Left gusset
left_gusset = 
  translate([-25, 5, 5]) * 
  rotate([90, 0, 0]) * 
  polygon(points=[[0,0], [0, gusset_leg_y], [gusset_leg_z, gusset_leg_y], [gusset_leg_z, 0]], 
          paths=[[0,1,2,3]]);

// Right gusset
right_gusset = 
  translate([20, 5, 5]) * 
  rotate([90, 0, 0]) * 
  polygon(points=[[0,0], [0, gusset_leg_y], [gusset_leg_z, gusset_leg_y], [gusset_leg_z, 0]], 
          paths=[[0,1,2,3]]);

// Combine all parts
union() {
  base_plate
  - translate([0, 0, 0]) * cylinder(r = 4.5 / 2, h = base_plate_z_height + 0.5, $fn = 64) // Left through-hole
  - translate([0, 0, 0]) * cylinder(r = 4.5 / 2, h = base_plate_z_height + 0.5, $fn = 64) // Right through-hole
  - motor_plate
  - centring_boss
  - screw_hole_positions[0]
  - screw_hole_positions[1]
  - screw_hole_positions[2]
  - screw_hole_positions[3]
  - left_gusset
  - right_gusset
}
```
