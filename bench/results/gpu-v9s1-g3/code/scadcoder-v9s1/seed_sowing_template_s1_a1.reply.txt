```openscad
// Petri dish sowing template
// Print orientation: flat on bed, z = 0 up to disc thickness

// ---------- Parameters ----------
disc_d        = 85;    // disc outer diameter
disc_h        = 2;     // disc thickness
hole_d        = 3;     // through-hole diameter
hole_nx       = 7;     // holes along X
hole_ny       = 7;     // holes along Y
hole_pitch    = 9;     // centre-to-centre spacing
notch_w       = 10;    // orientation notch width (X)
notch_d       = 4;     // orientation notch depth from rim (Y)
eps           = 0.5;   // cutter overshoot past faces
$fn           = 128;   // curve resolution

// ---------- Derived ----------
grid_x0 = -(hole_nx - 1) * hole_pitch / 2;  // first hole X centre
grid_y0 = -(hole_ny - 1) * hole_pitch / 2;  // first hole Y centre
notch_y0 = disc_h - notch_d + eps;          // notch floor (z), inside material

// ---------- Model ----------
difference() {
    // Main disc
    cylinder(d = disc_d, h = disc_h);

    // Hole grid
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([grid_x0 + i * hole_pitch,
                       grid_y0 + j * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = disc_h + 2 * eps);

    // Rectangular orientation notch on +Y side, centred on Y axis
    translate([-(notch_w / 2), disc_h - eps, notch_y0])
        cube([notch_w, eps, notch_d]);
}
```
