// Dimensions for agarose comb (flat print orientation)
$fn = 64;

// Spine dimensions
spine_length_x = 70;       // Long edge length (X)
spine_width_y = 12;        // Width along Y

// Tooth dimensions
tooth_count = 10;
tooth_width_x = 5.0;       // Width in X
tooth_length_y = 12;       // Length in Y
tooth_gap_y = 1.5;         // Gap between teeth in Y

// Thickness and print orientation
thickness_z = 1.5;        // Height (Z) of comb plate

module agarose_comb() {
    // Calculate X position of teeth (centered on spine)
    tooth_x_start = spine_width_y / 2 - tooth_count * (tooth_width_x + tooth_gap_y)/2;

    // Spine
    spine = cube([spine_length_x, spine_width_y, thickness_z]);

    // Teeth (cut from top surface)
    teeth_cut = [];
    for (i in [0:tooth_count-1]) {
        x_pos = tooth_x_start - i * (tooth_width_x + tooth_gap_y) + tooth_width_x/2;
        teeth_cut.push(
            translate([x_pos, spine_width_y/2], linear_extrude(height=-thickness_z),
                polygon(points=[[0, 0], [tooth_length_y, 0], [tooth_length_y, tooth_width_x], [0, tooth_width_x]])
            )
        );
    }

    // Create teeth by subtracting cutouts from spine
    comb = difference(spine, union(teeth_cut));
}

// Print the model
module() {
    agarose_comb();
}
