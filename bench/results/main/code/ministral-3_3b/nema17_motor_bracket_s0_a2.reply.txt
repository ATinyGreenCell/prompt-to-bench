```openscad
// Dimensions for L-shaped mounting bracket
$fn = 128;

// Base plate dimensions
base_width = 45;
base_height = 5;
hole_x_positions = [-17, 17];
hole_y_pos = 30;
hole_diameter = 4.5;

// Motor plate dimensions and boss
motor_plate_height = 50;
motor_plate_thickness = base_height + 0.2; // Extra thickness for clearance
boss_hole_diameter = 23;
boss_center_z = 28;
screw_spacing = 31;
motor_screws_offset = [screw_spacing/2 - boss_hole_diameter/2, screw_spacing/2 + boss_hole_diameter/2];
motor_screw_diameter = 3.4;

// Gusset dimensions
gusset_length = 20;
gusset_height = 20;
gusset_thickness = 5;

// Create base plate with holes
base_plate = difference(
    cube([base_width + 1, base_height + 1, base_height]),
    [
        for (x in hole_x_positions)
            translate([x, hole_y_pos, 0]) cylinder(h=base_height+1, $fn=$fn, diameter=hole_diameter)
    ]
);

// Create motor plate with boss and screws
motor_plate = difference(
    cube([base_width + motor_plate_thickness, base_height + 1, motor_plate_height]),
    [
        translate([0, 0, boss_center_z]) cylinder(h=25, $fn=$fn, diameter=boss_hole_diameter),
        for (offset in screw_spacing/2:motor_screws_offset)
            translate([offset, screw_spacing/2 + motor_screw_diameter/2, boss_center_z])
                cylinder(h=25, $fn=$fn, diameter=motor_screw_diameter)
    ]
);

// Position motor plate vertically behind base
motor_plate_positioned = rotate(
    [0, 90, 0],
    translate([-base_width/2 + (motor_plate_thickness - 1)/2, -base_height/2, 0]) motor_plate
);

// Create gussets
gusset_left = linear_extrude(height=gusset_thickness) polygon([
    [0, 0],
    [-gusset_length, 0],
    [-gusset_length, gusset_height]
]);
gusset_right = linear_extrude(height=gusset_thickness) polygon([
    [0, 0],
    [gusset_length, 0],
    [gusset_length, gusset_height]
]);

// Assemble all parts
bracket = union(
    base_plate,
    translate([-base_width/2 + motor_plate_thickness/2 - 1, -base_height/2, 0])
        rotate([0, 90, 0], motor_plate_positioned),
    difference(
        cube([
            base_width + motor_plate_thickness + gusset_length,
            base_height + gusset_thickness,
            base_height
        ]),
        [
            translate([-25 - gusset_length/2, 0, 0]) hull([
                [gusset_length, 0],
                [0, gusset_height]
            ]),
            translate([25 - gusset_length/2, 0, 0]) hull([
                [-gusset_length, 0],
                [0, gusset_height]
            ])
        ]
    ),
    [
        translate([-25 - gusset_thickness/2, 5 + gusset_thickness/2, 0]) gusset_left,
        translate([25 - gusset_thickness/2, 5 + gusset_thickness/2, 0]) gusset_right
    ]
);
```
