```openscad
// Dimensions in millimetres
$fn = 64;
width_plate = 130;      // Long edge (X)
height_plate = 90;     // Short edge (Y)
thickness_plate = 4;   // Plate thickness (Z)

hole_diameter = 30.5;  // Tube hole diameter
tube_hole_spacing_x = 40; // Centre-to-centre spacing along X
tube_hole_spacing_y = 40; // Centre-to-centre spacing along Y

wall_height = 70;      // Wall height (Z)

module create_tube_holes() {
    difference() {
        for (x in [1:3]) {
            for (y in [1:2]) {
                translate([
                    ((x-1)*tube_hole_spacing_x - hole_diameter/2),
                    ((y-1)*tube_hole_spacing_y - hole_diameter/2),
                    0
                ])
                    cylinder(diameter=hole_diameter+1.5, height=thickness_plate+1);
            }
        }
    }
}

module create_walls() {
    translate([-width_plate/2 - 0.3, -height_plate/2 + thickness_plate/2, 0])
        rotate([90, 0, 0]) cube([width_plate+0.6, thickness_plate+1, wall_height]);

    translate([-width_plate/2 - 0.3, height_plate/2 + thickness_plate/2, 0])
        rotate([90, 0, 0]) cube([width_plate+0.6, thickness_plate+1, wall_height]);
}

module plate_with_holes_and_walls() {
    union() {
        cube([width_plate, height_plate, thickness_plate]);

        create_tube_holes();
        create_walls();
    }
}

// Main rack construction
module final_rack() {
    // Base plate (printed upside down)
    translate([0, 0, -thickness_plate])
        rotate([0, 180, 0])
            difference() {
                cube([width_plate + wall_height*2, height_plate + thickness_plate*2, wall_height]);
                create_walls();
            };

    // Top part (mirrored)
    mirror([1, 1, 0]) {
        translate([0, 0, thickness_plate])
            plate_with_holes_and_walls();
    }
}

// Create the final solid object
final_rack();
```
