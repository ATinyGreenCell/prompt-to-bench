// Dimensions in millimetres
$fn = 128;
d_inner_bore = 3;                     // Inner bore diameter
d_outer_barb_8mm_top = 9.5;          // Max outer barb diameter for 8mm tubing (top)
d_outer_barb_8mm_bottom = 7.5;       // Min outer barb diameter for 8mm tubing (bottom)
height_barb_8mm = 10;                 // Height of each 8mm barb section
length_collar = 5;                   // Length of cylindrical collar

d_outer_barb_5mm_top = 6.5;          // Max outer barb diameter for 5mm tubing (top)
d_outer_barb_5mm_bottom = 5.0;       // Min outer barb diameter for 5mm tubing (bottom)
height_barb_5mm = 8;                  // Height of each 5mm barb section
total_height = 41;                   // Total height

// Main inner bore cylinder
inner_bore = cylinder(d=d_inner_bore, h=total_height);

// Barb for 8mm tubing (stacked twice)
for (i in [0,1]) {
    translate([0, 0, i * height_barb_8mm])
        rotate(45, [0, 0, 1])
            difference(
                cylinder(d=d_inner_bore + 0.5, h=height_barb_8mm),
                [
                    // Sawtooth outer profile
                    translate([0, 0, height_barb_8mm - (d_outer_barb_8mm_top - d_outer_barb_8mm_bottom)/2])
                        rotate(45, [0, 0, 1])
                            polygon(points=[
                                [0, 0],
                                [(d_outer_barb_8mm_top + d_inner_bore)/2, 0],
                                [(d_outer_barb_8mm_top + d_inner_bore)/2, (d_outer_barb_8mm_top - d_inner_bore)/2],
                                [-(d_outer_barb_8mm_bottom + d_inner_bore)/2, -(d_outer_barb_8mm_bottom - d_inner_bore)/2],
                                [-d_inner_bore/2, 0]
                            ], center=true),
                    // Step cut
                    translate([0, 0, height_barb_8mm - (d_outer_barb_8mm_top - d_outer_barb_8mm_bottom)/2])
                        rotate(45, [0, 0, 1])
                            polygon(points=[
                                [0, 0],
                                [(d_inner_bore + d_inner_bore)/2, 0],
                                [(d_inner_bore + d_inner_bore)/2, (d_outer_barb_8mm_bottom - d_inner_bore)/2],
                                [-(d_inner_bore + d_inner_bore)/2, -(d_outer_barb_8mm_bottom - d_inner_bore)/2],
                                [-d_inner_bore/2, 0]
                            ], center=true)
                ]
            );
}

// Cylindrical collar (z=20 to z=25)
collar = translate([0, 0, height_barb_8mm * 2])
    cylinder(d=d_outer_barb_8mm_top + 1, h=length_collar);

// Barb for 5mm tubing (stacked twice)
for (i in [0,1]) {
    translate([0, 0, i * height_barb_5mm + length_collar])
        rotate(45, [0, 0, 1])
            difference(
                cylinder(d=d_inner_bore + 0.5, h=height_barb_5mm),
                [
                    // Linear taper cut
                    translate([0, 0, height_barb_5mm - (d_outer_barb_5mm_top - d_outer_barb_5mm_bottom)/2])
                        rotate(45, [0, 0, 1])
                            polygon(points=[
                                [0, 0],
                                [(d_outer_barb_5mm_top + d_inner_bore)/4, 0],
                                [(d_outer_barb_5mm_top - d_inner_bore)/2, (d_outer_barb_5mm_top - d_inner_bore)/2],
                                [-(d_outer_barb_5mm_bottom - d_inner_bore)/2, -(d_outer_barb_5mm_bottom - d_inner_bore)/2],
                                [-(d_outer_barb_5mm_top + d_inner_bore)/4, 0]
                            ], center=true),
                    // Step cut
                    translate([0, 0, height_barb_5mm - (d_outer_barb_5mm_top - d_outer_barb_5mm_bottom)/2])
                        rotate(45, [0, 0, 1])
                            polygon(points=[
                                [0, 0],
                                [(d_inner_bore + d_inner_bore)/4, 0],
                                [(d_inner_bore - d_inner_bore)/2, (d_outer_barb_5mm_bottom - d_inner_bore)/2],
                                [-(d_inner_bore - d_inner_bore)/2, -(d_outer_barb_5mm_bottom - d_inner_bore)/2],
                                [-(d_inner_bore + d_inner_bore)/4, 0]
                            ], center=true)
                ]
            );
}

// Final assembly
union(
    inner_bore,
    difference(
        union(
            translate([0, 0, height_barb_8mm * 2])
                cylinder(d=d_outer_barb_8mm_top + 1, h=length_collar),
            translate([0, 0, height_barb_5mm * 2 + length_collar])
                cylinder(d=d_outer_barb_5mm_top + 1, h=length_collar)
        ),
        inner_bore
    )
);
