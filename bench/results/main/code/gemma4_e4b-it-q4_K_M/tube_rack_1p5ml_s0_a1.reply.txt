```openscad
// Bench Rack for 1.5 mL Microcentrifuge Tubes

// --- Dimensions ---
rack_length = 106;      // Total length along X (mm)
rack_width = 72;        // Total width along Y (mm)
rack_height = 30;       // Total height of the block (mm)
hole_diameter = 11.2;   // Diameter of each tube hole (mm)
hole_depth = 25;        // Depth of the holes (mm)
floor_thickness = 5;    // Thickness of the solid floor (mm)

// --- Hole Grid Parameters ---
num_rows_y = 4;         // Number of rows along Y
holes_per_row_x = 6;   // Number of holes along X
spacing_xy = 16;        // Center-to-center spacing in both X and Y (mm)

// --- Calculations for Centering ---
// Total span covered by the grid centers: (N-1) * Spacing
span_x = (holes_per_row_x - 1) * spacing_xy;
span_y = (num_rows_y - 1) * spacing_xy;

// Offset from center to the first hole center
offset_x = span_x / 2;
offset_y = span_y / 2;

// --- Constants ---
$fn = 64; // High resolution for curves

// --- Main Module: The Rack Body ---
module bench_rack() {
    difference() {
        // 1. Solid Outer Block (The main body)
        // We define the block from Z=0 to Z=rack_height
        cube([rack_length, rack_width, rack_height]);

        // 2. Cutting the Holes
        for (i = [0 : holes_per_row_x - 1]) { // Loop along X (columns)
            for (j = [0 : num_rows_y - 1]) { // Loop along Y (rows)

                // Calculate center position for this hole
                // Center X: Rack center + offset from rack center to first hole + i * spacing
                hole_center_x = -(rack_length / 2) + offset_x + i * spacing_xy;
                // Center Y: Rack center + offset from rack center to first row + j * spacing
                hole_center_y = -(rack_width / 2) + offset_y + j * spacing_xy;

                // The cutting cylinder must extend past the top and bottom faces.
                // It starts at Z=0 (the base of the rack).
                // We cut down to -1mm relative to the floor, ensuring it goes through the 5mm floor.
                // Since the block is defined from Z=0 to Z=30, we need to cut deep enough.
                // The hole needs to go through the 5mm floor (Z=0 to Z=-5) and then 25mm into the body (down to Z=-30).
                // We center the cylinder at Z = -hole_depth/2 - floor_thickness/2, and extend it by hole_depth + 1.
                cylinder(h = hole_depth + 1, r = hole_diameter / 2, center = [hole_center_x, hole_center_y, -(floor_thickness/2) - (hole_depth/2)]);

            }
        }

        // 3. Cutting the Solid Floor (Creating the open top structure)
        // The original design used a difference cut to create the floor thickness.
        // If we define the main block from Z=0 to Z=30, and the holes go down to Z=-30,
        // then the bottom 5mm must be solid (Z=0 to Z=-5).
        // To achieve this:
        // A) The main body is [106 x 72 x 30] starting at Z=0.
        // B) We cut holes down to Z = -hole_depth.
        // C) We must ensure the bottom 5mm (Z=-5 to Z=0) remains solid.

        // The previous implementation was flawed because it used a large floor cut that removed the entire base structure if not carefully managed relative to the main block's origin.
        // Let's redefine: The rack is a solid block of H=30, but we remove material from Z=5 up to Z=30 (the tube area).

        // New approach: Start with the full 106x72x30 block. Cut holes through it.
        // To leave a 5mm floor, we only cut down to Z = -floor_thickness.
        // The hole depth is 25mm. If the rack starts at Z=0, the bottom of the tube cavity should be at Z=-5.
        // Total required cut depth: 5 (floor) + 25 (tube space) = 30mm.

        // Let's adjust the cylinder center and height to ensure the floor remains solid from Z=0 down to Z=-5.
        // The tube cavity must be 25mm deep, starting at Z=-5. So it ends at Z=-30.
        // We cut cylinders that are 25mm long, centered such that their top is at Z=-5.

        for (i = [0 : holes_per_row_x - 1]) { // Loop along X (columns)
            for (j = [0 : num_rows_y - 1]) { // Loop along Y (rows)

                hole_center_x = -(rack_length / 2) + offset_x + i * spacing_xy;
                hole_center_y = -(rack_width / 2) + offset_y + j * spacing_xy;

                // Cut cylinder: Height = hole_depth (25mm). Top surface at Z=-floor_thickness (-5mm).
                // Center Z must be -5 - (25/2) = -17.5
                cylinder(h = hole_depth, r = hole_diameter / 2, center = [hole_center_x, hole_center_y, -(floor_thickness + hole_depth)/2]);

            }
        }
    }
}

bench_rack();
```
