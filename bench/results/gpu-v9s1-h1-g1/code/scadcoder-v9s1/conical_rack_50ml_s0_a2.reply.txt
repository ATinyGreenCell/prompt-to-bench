```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed (z = 0 to 4), walls rise to z = 70
// After printing, flip over so the rack stands on the two walls

// ---------- Parameters ----------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm)

hole_d       = 30.5;  // tube hole diameter (mm)
hole_nx      = 3;     // holes per row along X
hole_py      = 40;    // hole pitch along Y (mm)
hole_x0      = 0;     // first hole X offset (mm)
hole_y0      = 0;     // first hole Y offset (mm)

wall_len     = 130;   // wall length along X (mm)
wall_t       = 4;     // wall thickness along Y (mm)
wall_h       = 70;    // wall height above plate (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 96;    // curve resolution

// ---------- Derived ----------
hole_ny = 2;          // number of rows along Y
wall_r  = plate_y / 2 - wall_t / 2;   // inner Y of plate / walls

// ---------- Model ----------
difference() {
    union() {
        // Plate
        translate([-plate_x / 2, -plate_y / 2, 0])
            cube([plate_x, plate_y, plate_t]);

        // Two solid walls along the long edges
        for (s = [-1, 1])
            translate([0, s * wall_r, 0])
                cube([wall_len, wall_t, plate_t + wall_h]);
    }

    // Tube through-holes in 2 rows (along Y), 3 per row (along X)
    for (i = [0 : hole_ny - 1])
        for (j = [0 : hole_nx - 1])
            translate([(j - (hole_nx - 1) / 2 + hole_x0) * (plate_x / (hole_nx + 1)),
                       (i - (hole_ny - 1) / 2 + hole_y0) * hole_py,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2 * eps);
}
```
