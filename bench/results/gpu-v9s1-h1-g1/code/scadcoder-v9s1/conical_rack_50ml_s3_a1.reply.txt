```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, walls rising to z = 70
// After printing, flip over so the rack stands on the walls

// ---- Dimensions (mm) ----
plate_x      = 130;   // plate length (X)
plate_y      = 90;    // plate width (Y)
plate_t      = 4;     // plate thickness (Z)
hole_d       = 30.5;  // tube hole diameter
hole_nx      = 3;     // holes along X
hole_ny      = 2;     // holes along Y
hole_pitch_x = 40;    // centre-to-centre spacing along X
hole_pitch_y = 40;    // centre-to-centre spacing along Y
wall_h       = 70;    // wall height above plate (top at z = 74)
eps          = 0.5;   // cutter overshoot past faces

$fn = 96;

difference() {
    union() {
        // Plate
        translate([-plate_x/2, -plate_y/2, 0])
            cube([plate_x, plate_y, plate_t]);
        // Two solid walls along the long edges
        for (sx = [-1, 1])
            translate([sx > 0 ? plate_x/2 - 1 : -plate_x/2, -plate_y/2, 0])
                cube([1 + (sx > 0 ? 128 : 128), plate_y, wall_h + plate_t]);
    }
    // Tube through-holes
    for (i = [0 : hole_ny - 1])
        for (j = [0 : hole_nx - 1])
            translate([(j - (hole_nx - 1)/2) * hole_pitch_x,
                       (i - (hole_ny - 1)/2) * hole_pitch_y,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2*eps);
}
```
