```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, walls rise to z = 70
// After printing, flip over so the rack stands on the walls

// ---------- Parameters ----------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm), z = 0..4

hole_d       = 30.5;  // tube hole diameter (mm)
hole_nx      = 3;     // holes along X
hole_ny      = 2;     // holes along Y
hole_pitch   = 40;    // centre-to-centre spacing, X and Y (mm)

wall_len     = 130;   // wall length along X (mm)
wall_t       = 4;     // wall thickness along Y (mm)
wall_h       = 70;    // wall top height (mm), z = 0..70

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 128;   // circle resolution

// ---------- Derived ----------
x0 = -(hole_nx - 1) * hole_pitch / 2;   // first hole X centre
y0 = -(hole_ny - 1) * hole_pitch / 2;   // first hole Y centre

// ---------- Model ----------
union() {
    // Plate
    translate([-plate_x / 2, -plate_y / 2, 0])
        cube([plate_x, plate_y, plate_t]);

    // Walls along the long edges, flush with the plate
    for (s = [-1, 1])
        translate([s > 0 ? plate_x / 2 - wall_len / 2 : -plate_x / 2,
                   s * wall_t, 0])
            cube([wall_len, wall_t, wall_h]);

    // Through-holes for the tubes
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([x0 + i * hole_pitch, y0 + j * hole_pitch, -eps])
                cylinder(d = hole_d, h = plate_t + 2 * eps);
}
```
