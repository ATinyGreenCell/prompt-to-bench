```openscad
// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, walls rising to z = 70
// After printing, flip over so the rack stands on the walls

// ---------- Parameters ----------
plate_x      = 130;   // plate length (X)
plate_y      = 90;    // plate width (Y)
plate_t      = 4;     // plate thickness (Z)

hole_d       = 30.5;  // tube hole diameter
hole_nx      = 3;     // holes along X
hole_ny      = 2;     // holes along Y
hole_pitch   = 40;    // centre-to-centre spacing (X and Y)

wall_len     = plate_x; // wall length along X
wall_t       = 4;     // wall thickness along Y

eps          = 0.5;   // cutter overshoot past faces

$fn = 96;             // hole smoothness

// ---------- Model ----------
difference() {
    union() {
        // Base plate
        translate([-plate_x/2, -plate_y/2, 0])
            cube([plate_x, plate_y, plate_t]);

        // Two long-edge walls, flush with plate, open at top
        for (s = [-1, 1])
            translate([-wall_len/2, s * (plate_y/2 - wall_t/2), 0])
                cube([wall_len, wall_t, 70]);
    }

    // Tube holes through plate and walls
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([(i - (hole_nx - 1)/2) * hole_pitch,
                       (j - (hole_ny - 1)/2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = 74, $fn = 96);
}
```
