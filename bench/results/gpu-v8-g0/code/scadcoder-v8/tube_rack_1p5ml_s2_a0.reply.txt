```openscad
// Bench rack for 1.5 mL microcentrifuge tubes
// Print orientation: flat on bed, holes open upward (+Z)

// ---------- Parameters ----------
block_l      = 106;   // block length along X (mm)
block_w      = 72;    // block width along Y (mm)
block_h      = 30;    // block height along Z (mm)

hole_d       = 11.2;  // tube hole diameter (mm)
hole_depth   = 25;    // tube hole depth from top face (mm)
floor_t      = block_h - hole_depth; // solid floor thickness = 5 mm

hole_count_x = 6;     // holes per row along X
hole_count_y = 4;     // number of rows along Y
pitch        = 16;    // centre-to-centre spacing both directions (mm)

overcut      = 0.5;   // extra length on cutters past faces (mm)

$fn = 96;             // circle resolution

// ---------- Derived ----------
grid_x0 = -(hole_count_x - 1) * pitch / 2;  // first hole X (centred grid)
grid_y0 = -(hole_count_y - 1) * pitch / 2;  // first hole Y (centred grid)

// ---------- Model ----------
difference() {
    // Main block
    translate([-block_l / 2, -block_w / 2, 0])
        cube([block_l, block_w, block_h]);

    // Vertical tube holes, open at top, closed at bottom
    for (i = [0 : hole_count_x - 1])
        for (j = [0 : hole_count_y - 1])
            translate([grid_x0 + i * pitch,
                       grid_y0 + j * pitch,
                       floor_t])
                cylinder(d = hole_d, h = hole_depth + overcut, $fn = 96);
}
```
