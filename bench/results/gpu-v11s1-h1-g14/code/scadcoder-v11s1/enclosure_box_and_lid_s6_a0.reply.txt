```openscad
// Two-part enclosure for small electronics board
// Print orientation: box on bed (z = 0), lid on +X side, both build upward

// ---------- Parameters ----------
box_x      = 70;    // box outer length (X)
box_y      = 50;    // box outer width (Y)
box_z      = 30;    // box outer height (Z)
box_wall   = 2;     // box wall thickness
box_floor  = 2;     // box floor thickness

lip_x      = 65.6;  // lip outer length (X)
lip_y      = 45.6;  // lip outer width (Y)
lip_wall   = 1.5;   // lip wall thickness
lip_h      = 4;     // lip height (from plate top)

gap        = 10;    // gap between box and lid
plate_t    = 2;     // lid plate thickness

eps        = 0.5;   // cutter overshoot

$fn = 64;

// ---------- Model ----------
union() {
    // Box
    difference() {
        cube([box_x, box_y, box_z]);
        translate([box_wall, box_wall, box_floor])
            cube([box_x - 2*box_wall, box_y - 2*box_wall, box_z - box_floor]); // hollow body
    }

    // Lid: plate + lip, printed upside down
    translate([box_x + gap, -box_y/2, 0]) {
        union() {
            // Plate
            cube([lip_x, lip_y, plate_t]);
            // Lip
            translate([0, 0, plate_t])
                cube([lip_x, lip_y, lip_h]);
        }
        // Cut lip pocket (extends past top face)
        translate([0, 0, plate_t])
            cube([lip_x - 0.4, lip_y - 0.4, lip_h + eps]);
    }
}
```
