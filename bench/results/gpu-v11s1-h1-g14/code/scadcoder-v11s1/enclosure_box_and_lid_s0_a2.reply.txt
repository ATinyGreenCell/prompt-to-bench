```openscad
// Two-part enclosure for small electronics board (e.g. Arduino-based logger)
// Print orientation: box on the bed (z = 0), lid on top (+Z)

$fn = 64;

// ---- Box ----
box_x = 70;      // outer length (X)
box_y = 50;      // outer width (Y)
box_z = 30;      // outer height (Z)
wall  = 2;       // wall thickness
floor = 2;       // floor thickness

// ---- Lid ----
lip_x   = 65.6;  // lip outer length (X)
lip_y   = 45.6;  // lip outer width (Y)
lip_t   = 1.5;   // lip wall thickness
lip_h   = 4;     // lip height
clear   = 0.2;   // clearance on each side of the opening

// ---- Derived ----
opening_x = box_x - 2 * wall - 2 * clear;   // 66.0
opening_y = box_y - 2 * wall - 2 * clear;   // 46.0
overlap   = lip_h + 0.5;                    // lip extends into box for a clean union

module box() {
    difference() {
        cube([box_x, box_y, box_z]);
        translate([wall, wall, floor])
            cube([opening_x + 2 * clear, opening_y + 2 * clear, box_z - floor]);
    }
}

module lid() {
    union() {
        // base plate
        translate([0, 0, box_z])
            cube([box_x, box_y, plate_t]);

        // lip
        translate([0, 0, box_z + plate_t])
            cube([lip_x, lip_y, lip_t]);
    }
}

module plate() {
    translate([0, 0, box_z])
        cube([box_x, box_y, plate_t]);
}

union() {
    box();

    // Lid printed upside down: plate on the bed, lip pointing down into the box
    translate([0, 0, box_z])
        rotate([180, 0, 0])
            union() {
                plate();

                // lip, extended 0.2 mm into the box so it fuses with the opening
                translate([-lip_x / 2, -lip_y / 2, -overlap])
                    cube([lip_x, lip_y, lip_t + overlap]);
            }
}
```
