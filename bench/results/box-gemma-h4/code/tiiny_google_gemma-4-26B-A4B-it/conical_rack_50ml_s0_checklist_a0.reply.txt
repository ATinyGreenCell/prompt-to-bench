// 50mL Conical Tube Rack
// Designed for printing upside down (walls at Z=0 to Z=66) 
// and flipped to stand on the walls.

// --- Dimensions ---
plate_x      = 130.0; // Length of the plate (X)
plate_y      = 90.0;  // Width of the plate (Y)
plate_z      = 4.0;   // Thickness of the plate (Z)

hole_d       = 30.5;  // Diameter of the tube holes
rows         = 2;     // Number of rows (along Y)
cols         = 3;     // Number of columns (along X)
spacing      = 40.0;  // Center-to-center distance between holes

wall_x       = 130.0; // Length of the side walls (X)
wall_y       = 4.0;   // Thickness of the side walls (Y)
wall_h       = 70.0;  // Total height of the walls from the base (Z)

$fn = 64;

// --- Calculations ---
// Calculate offsets to center the grid of holes on the plate
x_offset = (cols - 1) * spacing / 2;
y_offset = (rows - 1) * spacing / 2;

// The user specifies the part is printed "upside down".
// In OpenSCAD, we model the part in its final functional orientation 
// and then rotate/translate it to the print orientation.
// Functional orientation: Plate at Z=0, Walls rising to Z=70.
// Print orientation: Walls at Z=0, Plate at Z=66.

module functional_rack() {
    difference() {
        union() {
            // The Base Plate
            translate([-plate_x/2, -plate_y/2, 0])
                cube([plate_x, plate_y, plate_z]);
            
            // The Two Side Walls
            // Wall 1 (along one long edge)
            translate([-plate_x/2, -plate_y/2, plate_z])
                cube([plate_x, wall_y, wall_h - plate_z]);
            
            // Wall 2 (along the other long edge)
            translate([-plate_x/2, plate_y/2 - wall_y, plate_z])
                cube([plate_x, wall_y, wall_h - plate_z]);
        }

        // The Six Holes
        for (i = [0 : cols - 1]) {
            for (j = [0 : rows - 1]) {
                // Calculate hole position relative to center
                hx = (i * spacing) - x_offset;
                hy = (j * spacing) - y_offset;
                
                // Subtracting the hole. 
                // We extend the hole in Z to ensure a clean cut through the plate.
                translate([hx, hy, -1])
                    cylinder(h = plate_z + 2, d = hole_d);
            }
        }
    }
}

// --- Print Orientation Logic ---
// The user wants to print it "upside down".
// This means the walls are on the bed (Z=0) and the plate is at the top.
// To achieve this, we rotate the functional model 180 degrees around the X axis.

rotate([180, 0, 0]) functional_rack();
