// Pestle for grinding tissue inside a 1.5 mL microcentrifuge tube
// Printed standing upright: handle at the bottom, tip at the top

// ---------------- Parameters ----------------
handle_d      = 8;    // handle diameter (mm)
handle_h      = 45;   // handle height, z = 0 to 45 (mm)

cone_top_d    = 8;    // cone diameter at the handle joint, z = 45 (mm)
cone_tip_d    = 3;    // cone diameter at its top, z = 62 (mm)
cone_top_z    = 62;   // height of the cone top (mm)

tip_d         = 3;    // rounded tip diameter (mm)
tip_z         = 62;   // centre height of the tip sphere (mm)

groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width (mm), measured along Z
groove_depth  = 1;    // groove depth (mm)
groove_z      = [10, 15, 20, 25]; // groove centre heights (mm)

eps           = 0.5;  // overshoot for clean boolean cuts (mm)

$fn = 96;             // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // (1) Cylindrical handle
        cylinder(d = handle_d, h = handle_h);

        // (2) Tapering cone from handle diameter down to tip diameter
        translate([0, 0, handle_h])
            linear_extrude(height = cone_top_z - handle_h)
                polygon(points = [
                    [cone_top_d / 2, 0],
                    [cone_tip_d / 2, cone_top_z - tip_z]
                ]);

        // (3) Rounded tip: sphere centred at z = tip_z
        translate([0, 0, tip_z])
            sphere(d = tip_d);
    }

    // Grip grooves cut into the handle, extended past the top face
    for (g = [0 : groove_count - 1])
        translate([0, 0, groove_z[g] - groove_w / 2])
            cylinder(d = groove_depth, h = groove_w + 2 * eps);
}
