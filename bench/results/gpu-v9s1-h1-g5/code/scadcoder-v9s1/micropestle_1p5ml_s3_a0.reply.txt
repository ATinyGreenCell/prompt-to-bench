// Pestle for grinding tissue inside a 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the bed (z = 0)

// ---------------- Parameters ----------------
handle_d      = 8;    // handle diameter (mm)
handle_h      = 45;   // handle height (mm), z = 0 .. 45
cone_top_d    = 8;    // cone base diameter at z = handle_h (mm)
cone_tip_d    = 3;    // cone tip diameter (mm)
cone_top_z    = 62;   // cone tip height (mm)
tip_d         = 3;    // rounded tip diameter (mm)
tip_center_z  = 62;   // sphere tip centre height (mm)
groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width (mm) along Z
groove_depth  = 1;    // groove depth (mm) from surface
groove_z      = [10, 15, 20, 25]; // groove centre heights (mm)
eps           = 0.5;  // overshoot for clean boolean cuts (mm)

$fn = 96;             // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // (1) Cylindrical handle
        cylinder(d = handle_d, h = handle_h);

        // (2) Tapering cone: 8 mm at z = 45 to 3 mm at z = 62
        translate([0, 0, handle_h])
            cylinder(d1 = cone_top_d, d2 = cone_tip_d, h = cone_top_z - handle_h);

        // (3) Rounded tip: 3 mm sphere centred at z = 62
        translate([0, 0, tip_center_z])
            sphere(d = tip_d);
    }

    // Grip grooves, each extending past the top face
    for (z = groove_z)
        translate([0, 0, z - groove_w / 2])
            cylinder(d = handle_d + 2 * eps, h = groove_w + eps);
}
