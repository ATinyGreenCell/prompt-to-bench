// Pestle for grinding tissue inside a 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the XY plane, building up +Z

// ---------------- Parameters ----------------
handle_d      = 8;    // handle diameter (mm), z = 0 to 45
handle_h      = 45;   // handle height (mm)
cone_d_top    = 8;    // cone diameter at top (mm), z = 45
cone_d_bot    = 3;    // cone diameter at bottom (mm), z = 62
cone_h        = 17;   // cone height (mm), z = 45 to 62
tip_d         = 3;    // tip sphere diameter (mm), centre z = 62
tip_z         = 62;   // tip centre height (mm)
groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width (mm), in Z
groove_d      = 1;    // groove depth (mm)
groove_z      = [10, 15, 20, 25]; // groove centre heights (mm)
eps           = 0.5;  // cutter overshoot past faces (mm)

$fn = 96;             // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // (1) Cylindrical handle
        cylinder(d = handle_d, h = handle_h);

        // (2) Tapered cone: 8 mm at z = 45 down to 3 mm at z = 62
        translate([0, 0, handle_h])
            cylinder(d1 = cone_d_top, d2 = cone_d_bot, h = cone_h);

        // (3) Rounded tip: 3 mm sphere centred at z = 62
        translate([0, 0, tip_z])
            sphere(d = tip_d);
    }

    // Grip grooves, cut through the handle (z = 0 to 45)
    for (z = groove_z)
        translate([0, 0, z - groove_w / 2])
            cylinder(d = handle_d + 2 * eps, h = groove_w, $fn = 96);
}
