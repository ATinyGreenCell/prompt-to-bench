// Pestle for 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the tube bottom (z = 0)

// ---------------- Parameters ----------------
tube_h          = 30;   // tube height (z = 0 to 30), clearance for tube cap
handle_d        = 8;    // handle diameter (mm), z = 0 to 45
cone_d_top      = 8;    // cone diameter at z = 45 (mm)
cone_d_tip      = 3;    // cone diameter at z = 62 (mm)
tip_d           = 3;    // sphere diameter (mm), centred at z = 62
groove_count    = 4;    // circumferential grooves for grip
groove_w        = 2;    // groove width in Z (mm)
groove_depth    = 1;    // groove depth (mm)
groove_z        = [10, 15, 20, 25]; // groove centre heights (mm)
overlap         = 0.01; // small overlap so handle and cone fuse cleanly (mm)
$fn             = 96;   // curve resolution

// ---------------- Derived ----------------
handle_h = 45;
cone_h   = 62 - 45;   // cone height from z = 45 to z = 62

// ---------------- Model ----------------
difference() {
    union() {
        // 1) Cylindrical handle: z = 0 to 45
        cylinder(d = handle_d, h = handle_h);

        // 2) Cone: z = 45 to 62, tapers from 8 mm down to 3 mm
        translate([0, 0, handle_h - overlap])
            cylinder(d1 = cone_d_top, d2 = cone_d_tip, h = cone_h);

        // 3) Rounded tip: 3 mm sphere centred at z = 62
        translate([0, 0, 62 - tip_d / 2])
            sphere(d = tip_d);
    }

    // 4) Grip grooves: 4 circumferential cuts, each 2 mm wide in Z and 1 mm deep
    for (z = groove_z)
        translate([0, 0, z - groove_w / 2])
            cylinder(d = handle_d + 2 * groove_depth, h = groove_w, $fn = 64);
}
