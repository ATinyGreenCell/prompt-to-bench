```openscad
// 10-well agarose gel comb - printed lying flat
// Units: mm

// ---------- Parameters ----------
plate_t      = 1.5;    // plate thickness (Z)
spine_l      = 70;     // spine length (X)
spine_w      = 12;     // spine width (Y)
tooth_n      = 10;     // number of teeth
tooth_w      = 5.0;    // tooth width (X)
tooth_h      = 12;     // tooth length (Y)
gap          = 1.5;    // gap between teeth (X)
eps          = 0.5;    // cutter overshoot past faces

$fn = 64;

// ---------- Derived ----------
pitch   = tooth_w + gap;
x0      = -(tooth_n - 1) * gap / 2;   // first tooth left edge (row centred)
y_tooth = -tooth_h / 2;               // tooth extends into -Y

// ---------- Model ----------
difference() {
    // Plate with spine and teeth
    union() {
        translate([-spine_l / 2, -spine_w / 2, 0])
            cube([spine_l, spine_w, plate_t]);
        for (i = [0 : tooth_n - 1])
            translate([x0 + i * pitch - tooth_w / 2, y_tooth, 0])
                cube([tooth_w, tooth_h, plate_t]);
    }

    // Cut out the wells through the full thickness
    for (i = [0 : tooth_n - 1])
        translate([x0 + i * pitch - tooth_w / 2, y_tooth - eps, -eps])
            cube([tooth_w, tooth_h + 2 * eps, plate_t + 2 * eps]);
}
```
