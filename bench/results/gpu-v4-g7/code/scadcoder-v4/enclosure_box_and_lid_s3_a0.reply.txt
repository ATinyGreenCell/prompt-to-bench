// Two-part enclosure for a small electronics board (e.g. Arduino-based temperature logger)
// Print orientation: both parts rest on the bed (z = 0), side by side, with a 10 mm gap.

// ---------------- Parameters ----------------
box_x      = 70;    // box outer length (X)
box_y      = 50;    // box outer width (Y)
box_z      = 30;    // box height (Z)
box_wall   = 2;     // box wall thickness
box_floor  = 2;     // box floor thickness

lid_t      = 2;     // lid plate thickness
lip_t      = 1.5;   // lip (overhang) thickness
lip_h      = 4;     // lip height above plate
lip_x      = 65.6;  // lip outer length (X)
lip_y      = 45.6;  // lip outer width (Y)
lip_over_x = 4;     // lip overhang past box end (X)
clearance  = 0.2;   // clearance between lip and box opening

$fn = 64;

// ---------------- Derived ----------------
lip_x2 = lip_x - 2 * clearance;   // lip size inside box (X)
lip_y2 = lip_y - 2 * clearance;   // lip size inside box (Y)
overlap = lip_over_x - clearance; // lip extends past box end by this amount

// ---------------- Model ----------------
union() {
    // Part 1: Box (open top)
    difference() {
        cube([box_x, box_y, box_z + box_floor]);
        translate([box_wall, box_wall, box_floor])
            cube([box_x - 2 * box_wall, box_y - 2 * box_wall, box_z + 1]);
    }

    // Part 2: Lid (plate with a centred lip on top, printed upside down)
    union() {
        // Plate
        translate([0, 0, 0])
            cube([box_x, box_y, lid_t]);

        // Lip
        translate([0, 0, lid_t])
            cube([lip_x, lip_y, lip_h]);

        // Cut out the part of the lip that sticks out past the box end
        translate([box_x - overlap, 0, lid_t])
            cube([overlap, lip_y, lip_h + 1]);
    }
}
