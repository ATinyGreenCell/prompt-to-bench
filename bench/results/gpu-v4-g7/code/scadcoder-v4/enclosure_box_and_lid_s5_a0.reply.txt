```openscad
// Two-part enclosure for small electronics board
// Print orientation: box standing on bed, lid printed upside down on +X side

// ---- Box parameters ----
box_x = 70;      // outer length (X)
box_y = 50;      // outer width (Y)
box_z = 30;      // outer height (Z)
box_w = 2;       // wall thickness
box_f = 2;       // floor thickness

// ---- Lid parameters ----
lip_t = 1.5;     // lip wall thickness
lip_h = 4;       // lip height
lip_x = 65.6;    // outer lip length (X)
lip_y = 45.6;    // outer lip width (Y)
clear = 0.2;     // clearance between lip and box opening

// ---- Misc ----
eps = 0.5;       // cutter overshoot
$fn = 64;

// Derived
inner_x = box_x - 2 * box_w;
inner_y = box_y - 2 * box_w;
plate_x = lip_x + 2 * clear;
plate_y = lip_y + 2 * clear;

union() {
    // Box
    difference() {
        cube([box_x, box_y, box_z]);
        translate([box_w, box_w, box_f])
            cube([inner_x, inner_y, box_z - box_f]);
    }

    // Lid (upside down: plate on bed, lip pointing up into box)
    union() {
        translate([plate_x / 2, plate_y / 2, 0])
            cube([plate_x, plate_y, 2]);  // plate
        translate([box_x / 2 - lip_x / 2, box_y / 2 - lip_y / 2, 2])
            cube([lip_x, lip_y, lip_h]);  // lip
    }
}
```
