```openscad
// Mini horizontal gel-electrophoresis buffer tank
// Print orientation: tank rests on XY plane at z = 0, builds upward (+Z)

// ---------------- Parameters ----------------
outer_x   = 120;   // outer length along X (mm)
outer_y   = 70;    // outer width along Y (mm)
height    = 40;    // overall height along Z (mm)
wall      = 3;     // wall thickness (mm)
floor     = 3;     // floor thickness (mm)

platform_x = 60;   // platform length along X (mm), centred
platform_top_z = 13; // platform top height above inner floor (mm) -> z = floor + 10

hole_d     = 2;    // electrode hole diameter (mm)
hole_y     = 0;    // electrode hole Y position (mm)
hole_z     = 30;   // electrode hole Z position (mm)

eps        = 0.5;  // cutter overshoot past faces (mm)
$fn        = 96;   // curve resolution

// ---------------- Derived ----------------
inner_x = outer_x - 2 * wall;        // inner length (X)
inner_y = outer_y - 2 * wall;        // inner width (Y)
inner_floor_z = floor;              // inner floor height (Z)
platform_x0 = (inner_x - platform_x) / 2;  // platform start X (centred)

// ---------------- Model ----------------
difference() {
    // Tank body (open top)
    translate([-outer_x / 2, -outer_y / 2, 0])
        cube([outer_x, outer_y, height]);

    // Central raised platform for gel tray
    translate([platform_x0, -inner_y / 2, inner_floor_z])
        cube([platform_x, inner_y, platform_top_z]);

    // Horizontal electrode holes through each end wall, along X
    for (sx = [-1, 1])
        translate([sx > 0 ? outer_x / 2 - eps : -outer_x / 2 - eps, hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = outer_x + 2 * eps);
}
```
