// Snap-on tubing clip for a 12.7 mm (1/2 inch) lab stand rod
// Print orientation: flat on the bed, extruded along +Z

// ---------------- Parameters ----------------
rod_d        = 12.7;   // stand rod diameter
ring_w       = 3;      // rod ring wall thickness
ring_od      = rod_d + 2 * ring_w;  // rod ring outer diameter (18.7 mm)
ring_open    = 10;     // rod ring opening width (cut out of -X side)

tube_d       = 6;      // tubing inner diameter
tube_w       = 2;      // tubing wall thickness
tube_od      = tube_d + 2 * tube_w; // tubing outer diameter (10 mm)
tube_x       = 15;     // tubing ring centre X
tube_open    = 4.5;    // tubing ring opening width (cut out of +X side)

bridge_w     = 6;      // solid bridge width (Y)
bridge_x0    = 8;      // bridge start X
bridge_x1    = 12;     // bridge end X

eps          = 0.5;    // cutter overshoot past faces
$fn          = 96;     // curve resolution

// ---------------- Profile (in XY) ----------------
module profile() {
    union() {
        // Rod ring with -X opening
        difference() {
            translate([0, 0])
                cylinder(d = ring_od, h = ring_w);
            translate([-ring_open / 2, -ring_open / 2])
                cube([ring_open, ring_open, ring_w + eps]);
        }

        // Tubing ring with +X opening
        difference() {
            translate([tube_x, 0])
                cylinder(d = tube_od, h = tube_w);
            translate([tube_x, -tube_open / 2])
                cube([tube_open, tube_open, tube_w + eps]);
        }

        // Solid bridge between the two rings
        translate([bridge_x0, -bridge_w / 2])
            cube([bridge_x1 - bridge_x0, bridge_w, ring_w + tube_w + 1]);
    }
}

// ---------------- Part ----------------
linear_extrude(height = 10)
    profile();
