```openscad
// Hose-barb reducer: 8 mm tubing -> 5 mm tubing
// Print orientation: standing upright along +Z

$fn = 64;

// ---- Dimensions (mm) ----
reducer_id      = 8;    // input tubing inner diameter
output_id       = 5;    // output tubing inner diameter
bore_d          = 3;    // through-bore diameter
barb_h_8        = 10;   // height of each 8 mm barb
barb_od_8_top   = 9.5;  // outer diameter of 8 mm barb at top
barb_od_8_bot   = 7.5;  // outer diameter of 8 mm barb at bottom
barb_h_5        = 8;    // height of each 5 mm barb
barb_od_5_top   = 5.0;  // outer diameter of 5 mm barb at top
barb_od_5_bot   = 6.5;  // outer diameter of 5 mm barb at bottom
collar_d        = 12;   // collar diameter
collar_h        = 5;    // collar height (z = 20 to 25)
eps             = 0.5;  // overshoot for clean cuts

// ---- Derived ----
total_h = barb_h_8 * 2 + collar_h + barb_h_5 * 2;  // 41 mm

// ---- Model ----
difference() {
    union() {
        // 8 mm barbs: z = 0..20
        for (z0 = [0, 10])
            translate([0, 0, z0])
                linear_extrude(height = barb_h_8)
                    polygon([
                        [0, barb_od_8_bot],
                        [0, barb_od_8_top],
                        [6.25, barb_od_8_top],
                        [6.25, barb_od_8_bot]
                    ]);

        // Collar: z = 20..25
        translate([0, 0, 20])
            cylinder(d = collar_d, h = collar_h);

        // 5 mm barbs: z = 25..41
        for (z0 = [25, 33])
            translate([0, 0, z0])
                linear_extrude(height = barb_h_5)
                    polygon([
                        [0, barb_od_5_bot],
                        [0, barb_od_5_top],
                        [3.25, barb_od_5_top],
                        [3.25, barb_od_5_bot]
                    ]);
    }

    // Through-bore along Z
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = total_h + 2 * eps);
}
```
