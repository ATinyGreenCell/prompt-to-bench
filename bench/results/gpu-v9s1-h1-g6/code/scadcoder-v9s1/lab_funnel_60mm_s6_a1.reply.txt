```openscad
// Lab funnel - printed upright, open at top and bottom
// Units: mm

$fn = 64;

// --- Dimensions ---
spout_od      = 10;   // spout outer diameter (z = 0 to 20)
spout_h       = 20;   // spout height
cone_od_top   = 60;   // cone rim outer diameter (z = 50)
cone_od_bot   = 10;   // cone base outer diameter at spout junction (z = 20)
cone_h        = 30;   // cone height (z = 20 to 50)
wall          = 1.6;  // wall thickness everywhere

// --- Derived ---
spout_id      = spout_od - 2 * wall;   // bore of spout
cone_h_total  = spout_h + cone_h;      // z of rim from spout start

// --- Model ---
difference() {
    union() {
        // Spout (tube)
        cylinder(d = spout_od, h = spout_h);
        // Cone (widens from spout junction to rim)
        translate([0, 0, spout_h])
            cylinder(d1 = cone_od_bot, d2 = cone_od_top, h = cone_h);
    }
    // Through-bore: spout bore up to cone, then cone's internal cone up to rim
    translate([0, 0, -1])
        linear_extrude(height = cone_h_total + 2)
            circle(d = spout_id);
    translate([0, 0, spout_h - 1])
        cylinder(d1 = spout_id, d2 = cone_od_top - 2 * wall, h = cone_h + 2);
}
```
