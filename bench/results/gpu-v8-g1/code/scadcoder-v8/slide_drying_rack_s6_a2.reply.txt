```openscad
// Microscope slide drying rack
// Units: mm

// ---- Block ----
block_len   = 90;    // X length
block_wid   = 45;    // Y width
block_ht    = 20;    // Z height

// ---- Slots ----
slot_count  = 10;    // number of slots
slot_len    = 77;    // slot length along X
slot_wid    = 1.6;   // slot width along Y
slot_depth  = 10;    // slot depth from top face
slot_pitch  = 3.5;   // centre-to-centre spacing along Y

// ---- Misc ----
eps         = 0.5;   // overshoot for clean cuts
$fn         = 64;

difference() {
    // Base block, centred in X/Y, resting on z = 0
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_ht]);

    // Slots, centred as a group in X and Y
    for (i = [0 : slot_count - 1]) {
        y = (i - (slot_count - 1)/2) * slot_pitch;
        translate([-slot_len/2, y, block_ht - slot_depth])
            cube([slot_len, slot_wid, slot_depth + eps]);
    }
}
```
