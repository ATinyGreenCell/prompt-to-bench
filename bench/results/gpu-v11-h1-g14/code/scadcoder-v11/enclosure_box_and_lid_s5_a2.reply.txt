// Two-part enclosure for a small electronics board (e.g. Arduino-based temperature logger)
// Print orientation: both parts flat on the bed, side by side along X

// ---------------- Parameters ----------------
box_x        = 70;    // box outer size in X (mm)
box_y        = 50;    // box outer size in Y (mm)
box_z        = 30;    // box outer size in Z (mm)
box_wall     = 2;     // box wall thickness (mm)
box_floor    = 2;     // box floor thickness (mm)

lip_x        = 65.6;  // lip outer size in X (mm)
lip_y        = 45.6;  // lip outer size in Y (mm)
lip_t        = 1.5;   // lip wall thickness (mm)
lip_h        = 4;     // lip height above plate (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)
gap          = 10;    // gap between box and lid (mm)

$fn = 64;

// ---------------- Derived ----------------
box_inner_x = box_x - 2 * box_wall;   // 66
box_inner_y = box_y - 2 * box_wall;   // 46
box_h       = box_z - box_floor;      // 28 (open top)

plate_t     = 2;                   // lid plate thickness

// ---------------- Model ----------------
union() {
    // Part 1: Box, centred in X/Y, sitting on z = 0
    translate([-box_x / 2, -box_y / 2, 0])
        cube([box_x, box_y, box_h]);

    // Part 2: Lid, printed upside down
    translate([box_x / 2 - gap, -box_y / 2, 0])
        union() {
            // Base plate
            cube([lip_x, lip_y, plate_t]);

            // Rectangular lip on top, centred, with 0.2 mm clearance inside the box
            translate([-lip_x / 2, -lip_y / 2, plate_t])
                cube([lip_x / 2 + 0.2, lip_y / 2 + 0.2, lip_t]);
        }
}
