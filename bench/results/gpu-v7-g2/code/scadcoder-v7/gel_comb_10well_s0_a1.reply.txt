// 10-well agarose gel comb, printed lying flat
// Units: mm. Plate thickness (Z) = 1.5 mm.

$fn = 64;

// ---- Dimensions ----
plate_t      = 1.5;     // plate thickness (Z)
spine_len    = 70;      // spine length (X)
spine_wid    = 12;      // spine width (Y)
tooth_count  = 10;      // number of teeth
tooth_w      = 5.0;     // tooth width (X)
tooth_l      = 12;      // tooth length (Y)
tooth_gap    = 1.5;     // gap between teeth (X)
eps          = 0.5;     // cutter overshoot past faces

// Check: 10*5.0 + 9*1.5 = 73.5 mm > spine_len (70 mm)
// The row does not fit centred with the given gap.
// Reduce the gap so the row is centred and has no overlap.
pitch = (spine_len - tooth_count * tooth_w) / (tooth_count + 1); // 5.0 mm
// With this pitch the outer teeth overhang the spine by 2.5 mm.
// Keep the pitch for a centred, symmetric row.

// ---- Model ----
difference() {
    // Base plate: spine + teeth
    translate([-spine_len/2, -spine_wid/2, 0])
        cube([spine_len, spine_wid, plate_t]);

    // Teeth cut out from the +Y side, extending past the top face
    for (i = [0 : tooth_count - 1]) {
        x = -spine_len/2 + i * pitch + tooth_w/2;
        translate([x, -spine_wid/2 - tooth_l, -eps])
            cube([tooth_w, tooth_l + eps, plate_t + 2*eps]);
    }
}
