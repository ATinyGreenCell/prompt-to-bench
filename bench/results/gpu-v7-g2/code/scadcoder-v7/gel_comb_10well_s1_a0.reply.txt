```openscad
// 10-well agarose gel comb, printed lying flat
// Units: mm

$fn = 64;

// ---- Spine (back bar) ----
spine_len   = 70;   // spine length along X
spine_wid   = 12;   // spine width along Y
plate_thk   = 1.5;  // plate thickness along Z

// ---- Teeth ----
tooth_count = 10;   // number of wells
tooth_w     = 5.0;  // tooth width along X
tooth_l     = 12;   // tooth length along Y (overlaps spine)
gap         = 1.5;  // centre-to-centre spacing of teeth

// ---- Derived ----
pitch = tooth_w + gap;  // centre-to-centre tooth spacing
x0    = -(tooth_count - 1) * pitch / 2;  // first tooth X so row is centred
eps   = 0.5;  // cutter overshoot past faces

difference() {
    // Base plate: spine + teeth
    union() {
        translate([-spine_len / 2, -spine_wid / 2, 0])
            cube([spine_len, spine_wid, plate_thk]);
        for (i = [0 : tooth_count - 1])
            translate([x0 + i * pitch - tooth_w / 2,
                       -spine_wid / 2 - tooth_l, 0])
                cube([tooth_w, tooth_l, plate_thk]);
    }

    // Wells: cut through the teeth (and into the spine)
    for (i = [0 : tooth_count - 1])
        translate([x0 + i * pitch - tooth_w / 2,
                   -spine_wid / 2 - tooth_l - eps, 0])
            cube([tooth_w, tooth_l + 2 * eps, plate_thk + 2 * eps]);
}
```
