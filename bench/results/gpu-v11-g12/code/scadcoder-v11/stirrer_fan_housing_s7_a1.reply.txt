```openscad
// DIY magnetic stirrer housing for an 80 mm PC case fan
// Print orientation: top plate on the bed (z = 0), walls build upward (+Z)
// Note: this is upside down in use; the open top becomes the stirring surface

// ---------- Dimensions (mm) ----------
plate_t      = 2;     // top plate thickness (z = 0 to 2)
wall_t       = 2.5;   // wall thickness
box_h        = 40;    // outer wall height (z = 0 to 40)
outer_x      = 90;    // outer size in X
outer_y      = 90;    // outer size in Y
hole_d       = 4.5;   // M4 screw hole diameter
hole_pitch   = 71.5;  // square pattern centre-to-centre distance for holes
notch_w      = 12;    // cable notch width
notch_depth  = 8;     // cable notch depth from open edge (z = 32 to 40)
eps          = 0.5;   // cutter overshoot past faces

$fn = 64;

// ---------- Model ----------
difference() {
    union() {
        // Top plate
        translate([-outer_x/2, -outer_y/2, 0])
            cube([outer_x, outer_y, plate_t]);

        // Four walls rising from the top plate to z = box_h
        for (dx = [-1, 1], dy = [-1, 1])
            translate([dx > 0 ? outer_x/2 - wall_t : -outer_x/2,
                       dy > 0 ? outer_y/2 - wall_t : -outer_y/2,
                       0])
                cube([wall_t, outer_y - 2*wall_t, box_h]);
    }

    // Screw holes through the top plate on a 71.5 mm square pattern
    for (a = [0, 90, 180, 270])
        rotate([0, 0, a])
            translate([hole_pitch/2, 0, -eps])
                cylinder(d = hole_d, h = plate_t + 2*eps);

    // Cable notch cut into one wall, centred along the wall
    translate([-outer_x/2, -outer_y/2/2 + notch_w/2, box_h - notch_depth])
        cube([outer_x, notch_w, notch_depth + eps]);
}
```
