difference() {
    union() {
        // Base Plate
        translate([-base_x / 2, 0, 0]) cube([base_x, base_y, base_z]);
        
        // Motor Plate (standing on the back edge of the base, y=0)
        // It extends from y=0 to y=5.
        translate([-motor_x / 2, 0, 0]) cube([motor_x, motor_y, motor_z]);
    }
    
    // Cut Base Plate Holes (M4)
    // Hole 1
    translate([base_hole_x1, base_hole_y1, 0]) cylinder(h = base_z + 1, d = m4_diameter);
    // Hole 2
    translate([base_hole_x2, base_hole_y2, 0]) cylinder(h = base_z + 1, d = m4_diameter);
    
    // Cut Motor Plate Boss Hole (along Y axis)
    // Centre at x=0, z=28. Extends through the plate (y=0 to y=5).
    translate([boss_hole_x, 0, boss_hole_z]) cylinder(h = motor_y + 1, d = boss_hole_diameter);
    
    // Cut Motor Plate Screw Holes (M3)
    // 4 holes on a 31mm square pattern centred on the boss hole (x=0, z=28).
    // Pattern size 31mm means distance from center to edge is 15.5mm.
    // Holes are at +/- 15.5mm in X and Z.
    hole_offset = 15.5;
    for (i = [-1, 1]) {
        for (j = [-1, 1]) {
            translate([i * hole_offset, 0, boss_hole_z + j * hole_offset]) cylinder(h = motor_y + 1, d = m3_diameter);
        }
    }
}
