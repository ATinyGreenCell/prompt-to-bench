```openscad
// Lab Funnel Design
// Units: millimetres
// Orientation: Upright, wide opening at top, rests on XY plane at z=0

// Dimensions
$fn = 128; // High resolution for smooth curves

// Spout (Tube)
spout_outer_diameter = 10;
spout_inner_diameter = 6.8;
spout_length = 20;
spout_wall_thickness = 1.6;

// Cone (Funnel)
cone_bottom_diameter = 10; // Matches spout outer diameter
cone_top_diameter = 60;
cone_height = 30; // From z=20 to z=50

// Wall Thickness (applies to cone as well)
wall_thickness = 1.6;

// Derived Cone Dimensions
cone_bottom_inner_diameter = cone_bottom_diameter - 2 * wall_thickness;
cone_top_inner_diameter = cone_top_diameter - 2 * wall_thickness;

// Build the part
// We construct the solid by creating the outer shell and subtracting the inner void.
// The part sits on the XY plane at z=0.

difference() {
    // --- OUTER SHELL ---
    union() {
        // 1. Spout Outer Cylinder (z=0 to z=20)
        cylinder(
            h = spout_length,
            d = spout_outer_diameter,
            center = false
        );

        // 2. Cone Outer Frustum (z=20 to z=50)
        // Translate to start at z=20
        translate([0, 0, spout_length]) {
            // Linear extrusion of a triangle profile rotated 90 deg around Y to form cone
            // Profile: Base at z=0 (relative) is cone_bottom_diameter, Top at z=cone_height is cone_top_diameter
            // We use linear_extrude with path = circle to create a cone, but simpler is rotate_extrude with a line segment.
            // Let's use rotate_extrude for the cone frustum.
            // Profile: A line from (cone_bottom_diameter/2, 0) to (cone_top_diameter/2, cone_height)
            rotate_extrude($fn = $fn) {
                // Draw the profile line
                // Start at bottom radius, go up to top radius
                // Note: rotate_extrude rotates around Z axis. We need to define the profile in XZ plane.
                // However, rotate_extrude expects the profile to be in the XZ plane (Y is the axis of rotation).
                // Wait, standard rotate_extrude rotates around Z.
                // To make a cone, we rotate a line segment around the Z axis.
                // The line segment goes from (radius_bottom, 0) to (radius_top, height).
                
                // Let's define the profile points
                // Point 1: (cone_bottom_diameter/2, 0)
                // Point 2: (cone_top_diameter/2, cone_height)
                // We need to draw this line.
                
                // Using polygon for the profile
                polygon([
                    [cone_bottom_diameter/2, 0],
                    [cone_top_diameter/2, cone_height]
                ]);
            }
        }
    }

    // --- INNER VOID ---
    // We subtract the inner volume to create the bore.
    // The void consists of the inner cylinder and the inner cone frustum.
    
    union() {
        // 1. Inner Cylinder Void (z=0 to z=20)
        cylinder(
            h = spout_length,
            d = spout_inner_diameter,
            center = false
        );

        // 2. Inner Cone Void (z=20 to z=50)
        translate([0, 0, spout_length]) {
            rotate_extrude($fn = $fn) {
                // Profile for inner cone
                // Start at bottom inner radius, go up to top inner radius
                polygon([
                    [cone_bottom_inner_diameter/2, 0],
                    [cone_top_inner_diameter/2, cone_height]
                ]);
            }
        }
    }
}
```
