// Pestle for 1.5 mL microcentrifuge tube
// Print orientation: standing upright, bottom on bed (z = 0)

$fn = 96;

// ---- Dimensions (mm) ----
handle_d      = 8;    // handle diameter
handle_h      = 45;   // handle height (z = 0 to 45)
cone_top_d    = 8;    // cone diameter at z = 45
cone_bot_d    = 3;    // cone diameter at z = 62
cone_h        = 17;   // cone height (z = 45 to 62)
tip_r         = 3;    // rounded tip radius (z = 62)
groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width in Z
groove_d      = 1;    // groove depth
overlap       = 0.5;  // overlap between adjacent grooves

eps           = 0.5;  // overshoot for clean boolean cuts

// ---- Derived ----
groove_start  = [10, 15, 20, 25];  // groove centre heights

// ---- Model ----
module pestle() {
    union() {
        // Handle with grooves
        translate([0, 0, 0])
            difference() {
                cylinder(d = handle_d, h = handle_h);
                for (g = groove_start)
                    translate([0, 0, g - groove_w / 2])
                        cylinder(d = handle_d + 2 * eps, h = groove_w + eps);
            }

        // Cone (tapered from top of handle to tip)
        translate([0, 0, handle_h - overlap])
            linear_extrude(height = cone_h + overlap)
                circle(d = cone_top_d);

        // Rounded tip
        translate([0, 0, 62 - tip_r])
            sphere(r = tip_r);
    }
}

pestle();
