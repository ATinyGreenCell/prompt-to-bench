// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: tank resting on XY plane, open top facing up (+Z)

// ---------------- Parameters ----------------
outer_x      = 120;   // outer length in X (mm)
outer_y      = 70;    // outer width in Y (mm)
outer_z      = 40;    // outer height in Z (mm)

wall         = 3;     // wall thickness (mm)

floor        = 3;     // floor thickness under the tank (mm)

platform_x   = 60;    // platform length along X (mm), centred
platform_y   = outer_y; // platform spans full inner width (mm)

platform_top_z = 13;  // height of platform top above outer floor (mm)

hole_d       = 2;     // electrode hole diameter (mm)

hole_y       = 0;     // electrode hole Y position (mm)

hole_z       = 30;    // electrode hole Z position (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)

$fn = 96;             // curve resolution

// ---------------- Derived ----------------
inner_x = outer_x - 2 * wall;   // 114 mm
inner_y = outer_y - 2 * wall;   // 64 mm
inner_z = outer_z - floor - 2 * wall; // 34 mm (space between floor and open top)

platform_bot_z = platform_top_z - wall; // 10 mm (matches user statement: top at z = 13, wall 3)

// ---------------- Model ----------------
union() {
    // Tank body: floor + side walls, open top
    difference() {
        cube([outer_x, outer_y, outer_z]);

        // Cut out the open-top inner volume (through the top face)
        translate([wall, wall, floor])
            cube([inner_x, inner_y, inner_z + eps]);
    }

    // Gel-tray platform, centred in X, spanning full inner width
    translate([-inner_x / 2, 0, platform_bot_z])
        cube([platform_x, platform_y, wall]);
}

// ---------------- Electrode holes ----------------
// Horizontal 2 mm holes along X, through each end wall, at y = 0, z = 30
for (sx = [-1, 1]) {
    translate([sx > 0 ? inner_x / 2 + eps : -inner_x / 2 - eps,
               hole_y,
               hole_z])
        rotate([0, 90, 0])
            cylinder(d = hole_d, h = outer_x + 2 * eps);
}
