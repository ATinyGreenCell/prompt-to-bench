```openscad
// Tube rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: flat plate on bed, walls rising to z = 70
// Flipped over after printing so the rack stands on the two walls

// ---------- Parameters ----------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm), z = 0 to 4

hole_d       = 30.5;  // tube hole diameter (mm)
hole_count   = 6;     // number of tube holes
hole_pitch   = 40;    // centre-to-centre spacing of holes, both directions (mm)

wall_len     = plate_x;// wall length along X (mm)
wall_t       = 4;     // wall thickness along Y (mm)
wall_top_z   = 70;    // top of walls (mm)

eps          = 0.5;   // overshoot for clean boolean cuts (mm)

$fn = 96;             // curve resolution

// ---------- Model ----------
union() {
    // Base plate
    translate([-plate_x/2, -plate_y/2, 0])
        cube([plate_x, plate_y, plate_t]);

    // Two solid walls along the long edges
    for (s = [-1, 1])
        translate([s > 0 ? plate_x/2 - 1 : -plate_x/2,
                   -wall_t/2, 0])
            cube([wall_len, wall_t, wall_top_z]);

    // Tube holes, 2 rows along Y, 3 along X
    for (i = [0 : hole_count - 1])
        for (j = [0 : 1])
            translate([(i - (hole_count - 1)/2) * hole_pitch,
                       (j - 0.5) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2*eps);
}
```
