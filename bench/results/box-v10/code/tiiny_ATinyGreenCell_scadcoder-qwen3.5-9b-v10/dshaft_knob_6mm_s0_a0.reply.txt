```openscad
// Replacement knob for hotplate stirrer (D-shaft 6 mm)
// Print orientation: bottom face (with D-bore) on bed, z = 0

// ---------- Parameters ----------
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
shaft_d       = 6.2;   // D-bore diameter
shaft_flat    = 4.7;   // distance from flat side to opposite side of D
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // groove diameter (half-cylinder)
pointer_w     = 1.5;   // pointer groove width
pointer_d     = knob_d - 2 * groove_d; // pointer groove outer diameter
eps           = 0.5;   // overcut for clean differences

$fn = 96;

// ---------- Model ----------
difference() {
    // Main knob body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, 0])
        linear_extrude(height = shaft_flat + eps)
            circle(d = shaft_d, $fn = 64);   // 6.2 mm circle -> D shape

    // Vertical grip grooves along full height
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps);

    // Pointer groove on top face (same side as flat of D)
    translate([0, 0, knob_h - pointer_w])
        rotate([90, 0, 0])
            linear_extrude(height = pointer_w)
                circle(d = pointer_d / 2, $fn = 64);
}
```
