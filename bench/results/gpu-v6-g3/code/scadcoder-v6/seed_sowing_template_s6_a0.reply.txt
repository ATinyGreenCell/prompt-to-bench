```openscad
// Petri dish sowing template
// Print orientation: flat on bed, z = 0 up to disc_t

$fn = 128;

// --- Disc ---
disc_d     = 85;    // disc diameter
disc_t     = 2;     // disc thickness

// --- Hole grid ---
hole_d     = 3;     // through-hole diameter
hole_pitch = 9;     // centre-to-centre spacing
n_x        = 7;     // holes in X
n_y        = 7;     // holes in Y

// --- Notch ---
notch_w    = 10;    // notch width (X)
notch_d    = 4;     // notch depth from rim (Y)

// --- Misc ---
eps        = 0.5;   // cutter overshoot

difference() {
    // Disc body
    cylinder(d = disc_d, h = disc_t);

    // Hole grid, centred
    for (i = [0 : n_x - 1])
        for (j = [0 : n_y - 1])
            translate([j * hole_pitch - (n_x - 1) * hole_pitch / 2,
                       i * hole_pitch - (n_y - 1) * hole_pitch / 2,
                       -eps])
                cylinder(d = hole_d, h = disc_t + 2 * eps);

    // Orientation notch on +Y side, centred on Y axis
    translate([-notch_w / 2, 0, -eps])
        cube([notch_w, notch_d + eps, disc_t + 2 * eps]);
}
```
