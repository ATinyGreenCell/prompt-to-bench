// Tissue pestle for 1.5 mL microcentrifuge tube
// Print orientation: handle down on the bed (z = 0), tip up (+Z)

$fn = 64;

// ---- Dimensions (mm) ----
handle_d      = 8;    // handle diameter
handle_h      = 45;   // handle height (z = 0 to 45)
cone_d_bottom = 8;    // cone diameter at z = 45
cone_d_top    = 3;    // cone diameter at z = 62
tip_d         = 3;    // tip sphere diameter
tip_z         = 62;   // tip sphere centre height
groove_n      = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width in Z
groove_depth  = 1;    // groove depth into handle
groove_z      = [10, 15, 20, 25]; // groove centre heights

eps = 0.5;            // overlap for clean boolean cuts

difference() {
    union() {
        // Handle
        cylinder(d = handle_d, h = handle_h);
        // Cone (tapers from handle_d_bottom to cone_d_top over cone_h)
        translate([0, 0, handle_h - eps])
            cylinder(d1 = cone_d_bottom, d2 = cone_d_top, h = tip_z - handle_h + eps);
        // Tip sphere
        translate([0, 0, tip_z])
            sphere(d = tip_d);
    }
    // Grooves
    for (z = groove_z)
        translate([0, 0, z])
            cylinder(d = handle_d, h = groove_w, $fn = 64);
}
