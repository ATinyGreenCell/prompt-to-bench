```openscad
// Sowing template for a 90 mm Petri dish
// Print orientation: flat on bed, z = 0 up to disc thickness

$fn = 128;

// ---- Dimensions (mm) ----
disc_d      = 85;    // disc diameter
disc_h      = 2;     // disc thickness
hole_d      = 3;     // through-hole diameter
hole_n      = 7;     // holes per row/column
hole_pitch  = 9;     // centre-to-centre spacing
notch_w     = 10;    // orientation notch width (X)
notch_depth = 4;     // orientation notch depth from rim (Y)
eps         = 0.5;   // cutter overshoot past faces

// ---- Derived ----
hole_x0 = -(hole_n - 1) * hole_pitch / 2;   // first hole X (centred grid)

difference() {
    // Main disc
    cylinder(d = disc_d, h = disc_h);

    // 7 x 7 through-hole grid
    for (i = [0 : hole_n - 1])
        for (j = [0 : hole_n - 1])
            translate([hole_x0 + i * hole_pitch,
                       -(hole_n - 1) * hole_pitch / 2 + j * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = disc_h + 2 * eps);

    // Rectangular orientation notch on +Y side, centred on Y axis
    translate([-notch_w / 2, disc_d / 2 - notch_depth, -eps])
        cube([notch_w, notch_depth + eps, disc_h + 2 * eps]);
}
```
