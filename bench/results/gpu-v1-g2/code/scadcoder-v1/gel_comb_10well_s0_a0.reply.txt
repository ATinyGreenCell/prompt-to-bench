```openscad
// Agarose gel comb - a flat plate with a back bar and a row of rectangular teeth, for casting
// horizontal gels. NOVEL general part.
//

cols   = 10;
teeth_d = 5.0;
teeth_l = 12;
pitch  = 1.5;
bar_w   = 12;
t       = 1.5;

$fn = 48;

p2  = teeth_d + pitch;                 // centre-to-centre pitch
over = min(t/2, pitch/2);              // butt the teeth into the bar so no gap at the joint
L   = 70;                              // spine length (X)

difference() {
    union() {
        cube([L, bar_w, t]);                 // spine (back bar)
        for (i = [0 : cols - 1])
            translate([-(teeth_d / 2 - over) + i * p2, 0, 0]) cube([teeth_d, teeth_l, t]);   // teeth
    }
    for (i = [0 : cols - 1])
        translate([-(teeth_d / 2 - over) + i * p2, 0, -1]) cube([teeth_d + 0.01, teeth_l, t + 2]);   // pocket (open top) in each tooth
}
```
