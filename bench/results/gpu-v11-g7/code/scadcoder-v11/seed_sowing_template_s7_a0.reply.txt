```openscad
// Petri dish sowing template
// Print orientation: flat on bed, z = 0 upward

// ---- Dimensions (mm) ----
disc_d      = 85;    // disc diameter
disc_t      = 2;     // disc thickness
hole_d      = 3;     // through-hole diameter
hole_n_x    = 7;     // holes per row (X)
hole_n_y    = 7;     // holes per column (Y)
hole_pitch  = 9;     // centre-to-centre spacing
notch_w     = 10;    // orientation notch width (X)
notch_d     = 4;     // orientation notch depth (Y, from rim)
eps         = 0.5;   // cut overshoot past faces

$fn = 128;

difference() {
    // Main disc
    cylinder(d = disc_d, h = disc_t);

    // Hole grid, centred on disc
    for (i = [0 : hole_n_x - 1])
        for (j = [0 : hole_n_y - 1])
            translate([(i - (hole_n_x - 1) / 2) * hole_pitch,
                       (j - (hole_n_y - 1) / 2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = disc_t + 2 * eps);

    // Rectangular orientation notch on +Y side, centred on Y axis
    translate([0, disc_t / 2 - notch_d, 0])
        cube([notch_w, notch_d, disc_t]);
}
```
