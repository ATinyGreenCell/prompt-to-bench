// Adapter sleeve: 0.2 mL PCR tube for a 1.5 mL rack hole
// Print orientation: collar down on the bed (z = 0), sleeve up (+Z)

$fn = 64;

// ---- Dimensions (mm) ----
collar_d      = 13;    // collar diameter
collar_h      = 1.5;   // collar thickness (z = 0 to 1.5)
sleeve_od     = 10.8;  // sleeve outer diameter
sleeve_top_z  = 21.5;  // top of sleeve (sleeve runs z = 1.5 to 21.5)
bore_d        = 6.2;   // through-bore diameter
eps           = 0.5;   // cutter overshoot past faces

total_h = sleeve_top_z; // full height for the through-bore

// ---- Model ----
difference() {
    union() {
        // Collar
        cylinder(d = collar_d, h = collar_h);
        // Sleeve body (slightly overlaps collar to fuse solids)
        translate([0, 0, collar_h - 0.01])
            cylinder(d = sleeve_od, h = sleeve_top_z - collar_h + 0.01);
    }
    // Straight through-bore
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = total_h + 2 * eps);
}
