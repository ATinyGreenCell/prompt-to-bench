```openscad
// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: tank rests on XY plane, open top facing +Z

// ---------- Parameters ----------
outer_x      = 120;   // outer length (X)
outer_y      = 70;    // outer width (Y)
outer_h      = 40;    // outer height (Z)
wall         = 3;     // wall thickness
floor        = 3;     // floor thickness
platform_x   = 60;    // platform length in X (centred)
platform_top = 10;    // platform top height above inner floor
hole_d       = 2;     // electrode hole diameter
eps          = 0.5;   // cutter overshoot past faces

$fn = 64;

// ---------- Derived ----------
inner_h   = outer_h - floor - wall;              // 34 mm
pitch     = outer_x / 2 - platform_x / 2;        // chamber length
y1        = -outer_y / 2 + wall + 1;             // left end wall Y
y2        = outer_y / 2 - wall - 1;              // right end wall Y

// ---------- Model ----------
difference() {
    // Tank body
    translate([-outer_x / 2, -outer_y / 2, 0])
        cube([outer_x, outer_y, outer_h]);

    // Raised central platform (solid, no cut through)
    translate([-platform_x / 2, -outer_y / 2, outer_h - floor - wall + platform_top])
        cube([platform_x, outer_y, floor + wall + eps]);

    // Horizontal electrode holes through end walls along X
    for (s = [-1, 1])
        translate([s * (outer_x / 2 - hole_d / 2), y1, outer_h - floor - wall + eps])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = outer_x + 2 * eps, $fn = 64);
}
```
