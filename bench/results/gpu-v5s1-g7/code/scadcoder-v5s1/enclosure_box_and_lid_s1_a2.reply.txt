```openscad
// Two-part enclosure for a small electronics board (e.g. Arduino-based temperature logger)
// Prints together in one job: box on the left, upside-down lid on the right, with a 10 mm gap.
// Units: mm. Wall thicknesses are 2 mm (box) and 1.5 mm (lid), with a 0.2 mm clearance
// between lid and box so they do not fuse.

// ---------------- Parameters ----------------
box_x      = 70;    // box outer length (X)
box_y      = 50;    // box outer width (Y)
box_z      = 30;    // box height (Z)
wall       = 2;     // box wall thickness
floor      = 2;     // box floor thickness

lip_t      = 1.5;   // lid lip wall thickness
lip_h      = 4;     // lid lip height (from plate top)
lip_x      = 65.6;  // lid lip outer length (X)
lip_y      = 45.6;  // lid lip outer width (Y)

gap        = 10;    // horizontal gap between box and lid
eps        = 0.5;   // overshoot for clean boolean cuts

$fn = 64;

// Derived: lid plate is the same size as the lip, centred on it
plate_x = lip_x;
plate_y = lip_y;

// ---------------- Model ----------------
union() {
    // Part 1: open-top box, centred in X/Y, resting on z = 0
    difference() {
        cube([box_x, box_y, box_z + floor]);
        translate([wall, wall, floor])
            cube([box_x - 2 * wall, box_y - 2 * wall, box_z + 1]);   // open top cavity
    }

    // Part 2: lid printed upside down: plate with a rectangular lip on top of it
    translate([box_x + gap, -plate_y / 2, 0])
        union() {
            // Base plate, 2 mm thick
            cube([plate_x, plate_y, 2]);
            // Lip: a wall 1.5 mm thick and 4 mm tall, centred on the plate
            translate([0, 0, 2])
                cube([lip_x, lip_y, lip_h]);
        }
        // Small cutouts under the lip arms so the two parts do not fuse at the corners
        translate([box_x + gap - eps, -plate_y / 2, 2])
            cube([eps, plate_y + 4, lip_h + 1]);
        translate([box_x + gap + plate_x - eps, -plate_y / 2, 2])
            cube([eps, plate_y + 4, lip_h + 1]);
}
```
