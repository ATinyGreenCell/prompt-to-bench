// Two-part enclosure for a small electronics board (e.g. Arduino-based temperature logger)
// Prints together in one job: box on the left, upside-down lid on the right, with a gap.

// ---------------- Parameters ----------------
box_x      = 70;    // box outer length (X)
box_y      = 50;    // box outer width (Y)
box_z      = 30;    // box height (Z)
wall       = 2;     // box wall thickness
floor      = 2;     // box floor thickness
eps        = 0.5;   // cutter overshoot past faces

lip_t      = 1.5;   // lid lip wall thickness
lip_h      = 4;     // lid lip height
lip_x      = 65.6;  // lid lip outer length (X)
lip_y      = 45.6;  // lid lip outer width (Y)

gap        = 10;    // horizontal gap between the two parts
clearance  = 0.2;   // clearance between lip and box opening

$fn = 64;

// Derived
box_in_x = box_x - 2 * wall;
box_in_y = box_y - 2 * wall;
lip_x1   = lip_x - 2 * clearance;   // lip inner size (X)
lip_y1   = lip_y - 2 * clearance;   // lip inner size (Y)

// ---------------- Model ----------------
union() {
    // Part 1: box, open top
    translate([-box_x / 2, -box_y / 2, 0])
        cube([box_x, box_y, box_z - floor]);

    // Part 2: lid, printed upside down with a rectangular lip
    translate([box_x / 2 + gap, -lip_y / 2, 0])
        cube([lip_x, lip_y, lip_t]);                                     // lip base plate
    translate([lip_x1 / 2, lip_y1 / 2, lip_h])
        cube([lip_x - lip_x1, lip_y - lip_y1, lip_t]);                   // lip walls (open top)
}
