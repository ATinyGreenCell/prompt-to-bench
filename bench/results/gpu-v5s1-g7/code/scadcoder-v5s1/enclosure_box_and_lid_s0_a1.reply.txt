```openscad
// Two-part enclosure for a small electronics board (e.g. Arduino-based temperature logger)
// Prints together in one job: box on the left, upside-down lid on the right, with a 10 mm gap.
// Units: mm. Wall thicknesses are 2 mm (box) and 1.5 mm (lid), with a 0.2 mm clearance
// between lid and box so they do not fuse.

// ---------------- Parameters ----------------
box_x      = 70;    // box outer length (X)
box_y      = 50;    // box outer width (Y)
box_z      = 30;    // box outer height (Z)
wall       = 2;     // box wall thickness
floor      = 2;     // box floor thickness

lip_t      = 1.5;   // lid lip wall thickness
lip_h      = 4;     // lid lip height (from plate top)
lip_x      = 65.6;  // lid lip outer length (X)
lip_y      = 45.6;  // lid lip outer width (Y)

gap        = 10;    // horizontal gap between box and lid
eps        = 0.5;   // overshoot for clean boolean cuts

$fn = 64;

// Derived: lid plate thickness = lip height - wall thickness = 2.5 mm
plate_t = lip_h - wall;   // 2.5 mm overlap of the lip into the box

// ---------------- Model ----------------
union() {
    // Part 1: Box, open top, centred in X/Y, resting on z = 0
    difference() {
        cube([box_x, box_y, box_z + floor]);
        translate([wall, wall, floor])
            cube([box_x - 2 * wall, box_y - 2 * wall, box_z + 1]);   // open-top cavity
    }

    // Part 2: Lid, printed upside down. Plate lies on the bed; the lip (a wall with the
    // same outer dimensions as the box opening) overhangs the plate by plate_t and rises
    // lip_h above the plate top. Positioned on the +X side of the box with the specified gap.
    translate([box_x + gap, 0, 0])
        difference() {
            union() {
                // Base plate
                translate([0, 0, 0])
                    cube([lip_x, lip_y, plate_t]);
                // Lip: wall 1.5 mm thick, 4 mm tall, centred on the plate
                translate([(lip_x - 65.6) / 2, (lip_y - 45.6) / 2, plate_t])
                    cube([65.6, 45.6, lip_h]);
            }
            // Small cutout under the lip centre so it does not fuse with the box floor
            translate([(lip_x - 65.6) / 2 - eps, (lip_y - 45.6) / 2 - eps, plate_t])
                cube([65.6 + 2 * eps, 45.6 + 2 * eps, eps]);
        }
}
```
