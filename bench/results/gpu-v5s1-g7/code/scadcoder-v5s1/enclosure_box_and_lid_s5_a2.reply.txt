// Two-part enclosure for a small electronics board (e.g. Arduino-based temperature logger)
// Prints together in one job: box on the left, upside-down lid on the right, with a gap.
// Units: mm. No external libraries.

// ---------------- Parameters ----------------
box_x      = 70;    // box outer length (X)
box_y      = 50;    // box outer width (Y)
box_z      = 30;    // box height (Z)
wall       = 2;     // box wall thickness
floor      = 2;     // box floor thickness
eps        = 0.5;   // cutter overshoot past box faces

lip_t      = 1.5;   // lid lip wall thickness
lip_h      = 4;     // lid lip height (from plate top)
lip_x      = 65.6;  // lid lip outer length (X)
lip_y      = 45.6;  // lid lip outer width (Y)
clearance  = 0.2;   // side clearance between lip and box opening
plate_t    = 2;     // lid plate thickness (lies on bed)
gap        = 10;    // gap between box and lid on the bed

$fn = 64;

// Derived
box_in_x   = box_x - 2 * wall;   // box inner length
box_in_y   = box_y - 2 * wall;   // box inner width
lip_x2     = lip_x + 2 * clearance;
lip_y2     = lip_y + 2 * clearance;

// ---------------- Model ----------------
union() {
    // Part 1: Box, open top, centred in X/Y, resting on z = 0
    difference() {
        translate([-box_x / 2, -box_y / 2, 0])
            cube([box_x, box_y, box_z]);
        translate([-box_in_x / 2, -box_in_y / 2, floor])
            cube([box_in_x, box_in_y, box_z - floor + eps]);
    }

    // Part 2: Lid, printed upside down. Plate on the bed, lip above it.
    // Placed on the +X side of the box with the given gap.
    union() {
        // Base plate
        translate([gap - (lip_x2 - lip_x) / 2, -lip_y / 2, 0])
            cube([lip_x2, lip_y, plate_t]);

        // Rectangular lip (wall) on top of the plate
        translate([gap - (lip_x2 - lip_x) / 2, -lip_y2 / 2, plate_t])
            cube([lip_x2, lip_y2, lip_h]);
    }
}
