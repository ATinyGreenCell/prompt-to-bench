```openscad
// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: tank rests on XY plane, open top facing up (+Z)

// ---------------- Parameters ----------------
tank_x      = 120;   // outer length along X (mm)
tank_y      = 70;    // outer width along Y (mm)
tank_h      = 40;    // outer height along Z (mm)

wall        = 3;     // wall thickness (mm)
floor       = 3;     // floor thickness (mm)

platform_x  = 60;    // platform length along X (mm), centred
platform_z  = 13;    // platform top height above inner floor (mm)

hole_d      = 2;     // electrode hole diameter (mm)
hole_y      = 0;     // electrode hole Y position (mm)
hole_z      = 30;    // electrode hole Z position (mm)

eps         = 0.5;   // cutter overshoot past faces (mm)
$fn         = 96;    // curve resolution

// ---------------- Derived ----------------
inner_x = tank_x - 2 * wall;          // inner length
inner_y = tank_y - 2 * wall;          // inner width
inner_h = tank_h - floor;             // inner height (open top)
floor_z = floor;                      // floor bottom height (z = 3)
platform_t = platform_z - floor;      // platform thickness above floor (10)

// ---------------- Model ----------------
difference() {
    // Main tank body
    translate([-tank_x/2, -tank_y/2, floor_z])
        cube([tank_x, tank_y, tank_h - floor]);

    // Central raised platform (spans full inner width Y)
    translate([-platform_x/2, -inner_y/2, floor_z + floor])
        cube([platform_x, inner_y, platform_t]);

    // Horizontal electrode holes through each end wall, along X
    for (sx = [-1, 1])
        translate([sx > 0 ? inner_x/2 - eps : -inner_x/2 - eps, hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = tank_x + 2*eps);
}
```
