// Replacement knob for hotplate stirrer (D-shaft 6 mm)
// Print orientation: bottom face (with D-bore opening) on the bed at z = 0

$fn = 96;

// ---- Dimensions (mm) ----
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
bore_d        = 6.2;   // D-bore circle diameter
bore_depth    = 12;    // D-bore depth from bottom face
bore_flat     = 4.7;   // D-bore width (flat side to opposite side of circle)
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // groove diameter (half-cylinder)
pointer_w     = 1.5;   // pointer groove width
pointer_depth = 1;     // pointer groove depth
eps           = 0.5;   // overcut for clean differences

// Derived
outer_r = knob_d / 2;

difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, -eps])
        difference() {
            cylinder(d = bore_d, h = bore_depth + eps);
            // flat side of D, centred on x = 0
            translate([-(bore_flat / 2), -(bore_d / 2), 0])
                cube([bore_flat, bore_d, bore_depth + eps]);
        }

    // Vertical grip grooves along the full height
    for (i = [0 : groove_count - 1]) {
        rotate([0, 0, i * 360 / groove_count])
            translate([outer_r, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps);
    }

    // Pointer groove on top face, from centre to outer edge
    translate([-(pointer_w / 2), 0, knob_h - pointer_depth])
        cube([outer_r + eps, pointer_w, pointer_depth + eps]);
}
