// Hotplate stirrer knob - D-shaft adapter with grip grooves
// Print orientation: flat bottom on bed (z = 0), D-bore opens downward

$fn = 96;

// ---- Dimensions (mm) ----
knob_d       = 30;    // knob outer diameter
knob_h       = 15;    // knob height
shaft_d      = 6;     // D-shaft diameter
bore_d       = 6.2;   // D-bore diameter (6.2 circle, 4.7 flat-to-opposite)
bore_depth   = 12;    // blind bore depth from bottom face
groove_count = 18;    // vertical grip grooves evenly spaced
groove_d     = 2;     // groove half-cylinder diameter
pointer_w    = 1.5;   // pointer groove width
pointer_d    = 1;     // pointer groove depth
eps          = 0.5;   // overcut for clean differences

difference() {
    // Main body
    cylinder(d = knob_d, h = knob_h);

    // D-blind bore: circle with one side cut flat
    translate([0, 0, -eps])
        hull() {
            cylinder(d = bore_d, h = bore_depth + eps);
            translate([-(bore_depth + eps), -bore_d / 2, -eps])
                cube([bore_depth + eps, bore_d, bore_depth + eps]);
        }

    // Vertical grip grooves along full height
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, 0])
                cylinder(d = groove_d, h = knob_h, $fn = 64);

    // Pointer groove from centre to outer edge, on flat side
    translate([-(bore_depth + eps), -pointer_w / 2, knob_h - pointer_d])
        cube([bore_depth + eps, pointer_w, pointer_d + eps]);
}
