```openscad
// Mini horizontal gel-electrophoresis buffer tank
// Print orientation: base on XY plane, open top at z = 40

// ---- Tank dimensions ----
outer_x = 120;   // outer length along X
outer_y = 70;    // outer width along Y
height  = 40;    // overall height along Z
wall    = 3;     // wall thickness
floor   = 3;     // floor thickness

// ---- Platform dimensions ----
platform_len   = 60;   // platform length along X
platform_top_z = 13;   // platform top height (10 mm above inner floor)

// ---- Electrode holes ----
hole_d = 2;     // hole diameter
hole_y = 0;     // hole Y position
hole_z = 30;    // hole Z position

// ---- Modelling helpers ----
eps = 0.5;       // cutter overshoot past faces
$fn = 64;        // curve resolution

// ---- Derived ----
inner_h = height - floor;              // inner floor to open top
platform_h = platform_top_z - floor;   // platform height above floor

difference() {
    // Main tank body
    translate([-outer_x/2, -outer_y/2, 0])
        cube([outer_x, outer_y, height]);

    // Raised platform in the middle (spans full inner width)
    translate([-outer_x/2 + (outer_x - platform_len)/2,
               -outer_y/2,
               floor])
        cube([platform_len, outer_y, platform_h]);

    // Horizontal electrode holes through each end wall along X
    for (sx = [-1, 1])
        translate([sx > 0 ? outer_x/2 - 1 : -outer_x/2 - 1,
                   hole_y,
                   hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = outer_y + 2*eps, center = true);
}
```
