```openscad
// PCR tube adapter sleeve: fits a 0.2 mL tube into a 1.5 mL rack hole
// Print orientation: collar on the bed (z = 0), sleeve body builds upward

// ---------- Parameters ----------
collar_d        = 13;     // collar outer diameter
collar_t        = 1.5;    // collar thickness (z = 0 to 1.5)
sleeve_d        = 10.8;   // sleeve body outer diameter
sleeve_h        = 21.5;   // sleeve body height above collar (z = 1.5 to 23.0)
bore_d          = 6.2;    // through-bore diameter
eps             = 0.5;    // cutter overshoot past faces
$fn             = 96;     // curve resolution

// ---------- Model ----------
union() {
    // Collar lying on the bed
    cylinder(d = collar_d, h = collar_t);

    // Sleeve body rising from the collar top
    translate([0, 0, collar_t])
        cylinder(d = sleeve_d, h = sleeve_h);
}

// Full-height bore, open at both ends
translate([0, 0, -eps])
    cylinder(d = bore_d, h = sleeve_h + collar_t + 2 * eps);
```
