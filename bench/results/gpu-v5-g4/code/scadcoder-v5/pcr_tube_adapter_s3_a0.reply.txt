```openscad
// PCR tube adapter sleeve - print upside down
// Units: mm

$fn = 64;

// --- Dimensions ---
collar_d      = 13;    // collar outer diameter
collar_t      = 1.5;   // collar thickness (z = 0 to 1.5)
sleeve_d      = 10.8;  // sleeve body outer diameter
sleeve_h      = 20;    // sleeve body height (z = 1.5 to 21.5)
bore_d        = 6.2;   // through-bore diameter
eps           = 0.5;   // cutter overshoot past faces

// Derived
sleeve_top_z  = collar_t + sleeve_h;  // 22.0 mm

difference() {
    union() {
        // Collar lying on the bed
        cylinder(d = collar_d, h = collar_t);
        // Sleeve body rising from the collar
        translate([0, 0, collar_t])
            cylinder(d = sleeve_d, h = sleeve_h);
    }
    // Straight bore, open at both ends
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = sleeve_top_z + 2 * eps);
}
```
