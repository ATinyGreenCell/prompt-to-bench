```openscad
// PCR tube adapter sleeve: prints upside down, collar on the bed (z = 0 to 1.5),
// sleeve body rises from the collar to z = 21.5. A 6.2 mm bore runs through the
// full height, open at both ends. Units: mm.
//

$fn = 64;

// ---- Dimensions ----
collar_d        = 13;     // collar outer diameter
collar_t        = 1.5;    // collar thickness (z = 0 to 1.5)
sleeve_d        = 10.8;   // sleeve body outer diameter
sleeve_top_z    = 21.5;   // top of sleeve body (absolute Z)
bore_d          = 6.2;    // through-bore diameter
eps             = 0.5;    // cutter overshoot past faces

// ---- Model ----
difference() {
    union() {
        // Collar lying on the bed, centred on the origin
        cylinder(d = collar_d, h = collar_t);

        // Sleeve body rising from the top of the collar
        translate([0, 0, collar_t])
            cylinder(d = sleeve_d, h = sleeve_top_z - collar_t);
    }

    // Straight bore through the full height, open at both ends
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = sleeve_top_z + 2 * eps);
}
```
