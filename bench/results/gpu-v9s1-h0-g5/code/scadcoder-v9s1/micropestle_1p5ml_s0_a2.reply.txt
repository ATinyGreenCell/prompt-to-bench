// Pestle for grinding tissue inside a 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the bed (z = 0)

// ---------- Parameters ----------
handle_d      = 8;    // handle diameter (mm)
handle_h      = 45;   // handle height (mm), z = 0 to 45
cone_top_d    = 8;    // cone diameter at handle top (mm)
cone_bottom_d = 3;    // cone diameter at tip (mm)
cone_h        = 17;   // cone height (mm), z = 45 to 62
tip_r         = 1.5;  // tip sphere radius (mm), centred at z = 62
groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width (mm), in Z
groove_depth  = 1;    // groove depth (mm)
groove_z      = [10, 15, 20, 25]; // groove centre heights (mm)
eps           = 0.5;  // overshoot for clean cuts (mm)

$fn = 96;

// ---------- Model ----------
difference() {
    union() {
        // Handle
        cylinder(d = handle_d, h = handle_h);
        // Cone (overlaps slightly into handle for a solid union)
        translate([0, 0, handle_h - 0.01])
            cylinder(d1 = cone_top_d, d2 = cone_bottom_d, h = cone_h + 0.01);
        // Tip sphere
        translate([0, 0, tip_r])
            sphere(r = tip_r);
    }
    // Grooves, extended past faces
    for (z = groove_z)
        translate([0, 0, z - groove_w / 2])
            rotate([0, 90, 0])
                cylinder(d = groove_w, h = groove_depth + eps);
}
