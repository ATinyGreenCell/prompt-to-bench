```openscad
// Pestle for grinding tissue inside a 1.5 mL microcentrifuge tube
// Print orientation: standing upright (handle on bed, tip on top)

$fn = 96;

// ---- Dimensions (mm) ----
handle_d      = 8;    // handle diameter
handle_h      = 45;   // handle height (z = 0 to 45)
cone_top_d    = 8;    // cone diameter at z = 45
cone_bottom_d = 3;    // cone diameter at z = 62
tip_d         = 3;    // tip sphere diameter
tip_center_z  = 62;   // tip sphere center height

groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width in Z
groove_depth  = 1;    // groove depth
groove_z      = [10, 15, 20, 25]; // groove center heights

eps           = 0.5;  // overshoot for clean cuts

// ---- Model ----
difference() {
    union() {
        // Handle
        cylinder(d = handle_d, h = handle_h);
        // Cone (z = 45 to 62)
        translate([0, 0, handle_h])
            cylinder(d1 = cone_top_d, d2 = cone_bottom_d, h = tip_center_z - handle_h);
        // Tip sphere
        translate([0, 0, tip_center_z])
            sphere(d = tip_d);
    }
    // Grooves
    for (z = groove_z)
        translate([0, 0, z - groove_w / 2])
            cylinder(d = groove_depth * 2, h = groove_w, $fn = 64);
}
```
