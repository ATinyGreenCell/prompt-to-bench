```openscad
// Small lab funnel, printed upright (wide opening at top)
// All dimensions in mm

// ---- Parameters ----
spout_od      = 10;    // spout outer diameter
spout_h       = 20;    // spout height (z = 0 to 20)
cone_od_bottom = 10;    // cone outer diameter at z = 20
cone_od_top   = 60;    // cone outer diameter at rim (z = 50)
rim_h         = 30;    // cone height (z = 20 to 50)
wall          = 1.6;   // wall thickness everywhere
eps           = 0.5;   // cutter overshoot past faces

$fn = 96;

// ---- Derived ----
spout_id = spout_od - 2 * wall;   // 6.8 mm bore

// ---- Model ----
difference() {
    union() {
        // Spout
        cylinder(d = spout_od, h = spout_h);
        // Cone (slightly overlapping spout for a clean union)
        translate([0, 0, spout_h - 0.01])
            cylinder(d1 = cone_od_bottom, d2 = cone_od_top, h = rim_h + 0.01);
    }
    // Hollow out the full height so liquid runs straight through
    translate([0, 0, -eps])
        cylinder(d = spout_id, h = spout_h + rim_h + 2 * eps);
}
```
