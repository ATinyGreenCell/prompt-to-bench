```openscad
// Small lab funnel, printed upright: spout at the bottom, wide rim at the top
// All dimensions in millimetres

// ---- Parameters ----
spout_od      = 10;    // outer diameter of straight spout
spout_h       = 20;    // spout height (z = 0 to 20)
cone_od_top   = 60;    // outer diameter of cone rim (z = 50)
cone_od_bot   = 10;    // cone outer diameter at z = 20
wall          = 1.6;   // wall thickness everywhere
eps           = 0.5;   // overshoot for clean boolean cuts

$fn = 96;

// ---- Derived ----
bore = spout_od - 2 * wall;   // 6.8 mm bore in the spout

// ---- Model ----
difference() {
    union() {
        // Straight spout from z = 0 to z = 20
        cylinder(h = spout_h, d = spout_od);

        // Cone from z = 20 to z = 50
        translate([0, 0, spout_h])
            cylinder(h = cone_od_top - cone_od_bot,
                     d1 = cone_od_bot,
                     d2 = cone_od_top);
    }

    // Through-bore: starts below z = 0, runs through the spout and cone
    translate([0, 0, -eps])
        cylinder(h = cone_od_top + 2 * eps, d = bore);
}
```
