```openscad
// Lab funnel - printed upright, wide opening at the top
// All dimensions in millimetres

// ---- Parameters ----
spout_od      = 10;    // outer diameter of straight spout
spout_length  = 20;    // spout runs from z = 0 to z = 20
cone_bottom_d = 10;    // cone outer diameter at z = 20
cone_top_d    = 60;    // rim outer diameter at z = 50
rim_z         = 50;    // height of the funnel rim
wall          = 1.6;   // wall thickness everywhere

$fn = 128;             // smoothness for curved surfaces

// ---- Derived ----
spout_id = spout_od - 2 * wall;   // spout bore = 6.8 mm
cone_h   = rim_z - spout_length;  // cone height = 30 mm

// ---- Model ----
difference() {
    union() {
        // Straight spout from z = 0 to z = 20
        cylinder(d = spout_od, h = spout_length);

        // Cone from z = 20 to z = 50 (open bottom where it meets the spout)
        translate([0, 0, spout_length])
            cylinder(d1 = cone_bottom_d, d2 = cone_top_d, h = cone_h);
    }

    // Bore through the spout
    translate([0, 0, -1])
        cylinder(d = spout_id, h = spout_length + 2);

    // Hollow out the cone so the bore continues up past the rim
    translate([0, 0, spout_length - 1])
        cylinder(d1 = spout_id, d2 = cone_top_d + 2 * wall, h = cone_h + 1);
}
```
