```openscad
// 96-well PCR tube rack (0.2 mL, standard microplate footprint)
// Units: mm. Print orientation: base on XY plane, building up +Z.

// ---------------- Parameters ----------------
block_len    = 127.76;   // block length along X
block_wid    = 85.48;    // block width along Y
block_ht     = 20;       // block height along Z

hole_d       = 6.2;      // through-hole diameter
hole_pitch   = 9.0;      // centre-to-centre spacing (X and Y)
hole_nx      = 12;       // number of columns along X
hole_ny      = 8;        // number of rows along Y

a1_x         = 14.38;    // X of first hole (A1) from left edge
a1_y         = 11.24;    // Y of first hole (A1) from back edge

chamfer_leg  = 5;        // chamfer leg length off A1 vertical edge
chamfer_ang  = 45;       // chamfer angle (degrees)

eps          = 0.5;      // cutter overshoot past faces
$fn          = 96;       // circle resolution

// ---------------- Derived ----------------
x0 = a1_x - (hole_nx - 1) * hole_pitch / 2;   // grid origin X
y0 = a1_y - (hole_ny - 1) * hole_pitch / 2;   // grid origin Y

// ---------------- Model ----------------
difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_len / 2, -block_wid / 2, 0])
        cube([block_len, block_wid, block_ht]);

    // 96 vertical through-holes
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([x0 + i * hole_pitch, y0 + j * hole_pitch, -eps])
                cylinder(d = hole_d, h = block_ht + 2 * eps);

    // Orientation mark: 45-degree chamfer with 5 mm legs off the A1 corner
    // (vertical edge at minimum X, minimum Y)
    translate([x0 + hole_pitch / 2 - 1e-6, y0 + hole_pitch / 2 - 1e-6, 0])
        linear_extrude(height = block_ht)
            polygon([[-chamfer_leg, -chamfer_leg],
                     [-chamfer_leg,  block_len / 2 - 1],
                     [ block_len / 2 - 1,  block_len / 2 - 1],
                     [ block_len / 2 - 1, -chamfer_leg]]);
}
```
