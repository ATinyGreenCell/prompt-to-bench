// 96-well PCR tube rack (0.2 mL, standard microplate footprint)
// Print orientation: base on XY plane, builds upward in +Z

// ---------- Block ----------
block_l = 127.76;   // block length along X (mm)
block_w = 85.48;    // block width along Y (mm)
block_h = 20;       // block height along Z (mm)

// ---------- Holes ----------
hole_d     = 6.2;   // through-hole diameter (mm)
hole_nx    = 12;    // number of columns along X
hole_ny    = 8;     // number of rows along Y
hole_pitch = 9.0;   // centre-to-centre spacing (mm)
hole_x0    = 14.38; // X of first hole (A1) from left edge (mm)
hole_y0    = 11.24; // Y of first hole (A1) from back edge (mm)

// ---------- Chamfer ----------
chamf_leg  = 5;     // chamfer leg length (mm)
chamf_angle = 45;   // chamfer angle (deg)

// ---------- Misc ----------
eps = 0.5;          // cutter overshoot past faces (mm)
$fn = 96;           // circle resolution

// Derived
z_chamf = (chamf_leg / tan(chamf_angle)) + eps; // height of chamfer cut
x_last  = hole_x0 + (hole_nx - 1) * hole_pitch;
y_last  = hole_y0 + (hole_ny - 1) * hole_pitch;

difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_l / 2, -block_w / 2, 0])
        cube([block_l, block_w, block_h]);

    // Through-holes
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([hole_x0 + i * hole_pitch - hole_d / 2,
                       hole_y0 + j * hole_pitch - hole_d / 2,
                       -eps])
                cylinder(d = hole_d, h = block_h + 2 * eps);

    // Orientation chamfer at A1 corner (vertical edge, 45 deg, 5 mm legs)
    translate([hole_x0 - hole_d / 2 - 1, hole_y0 - hole_d / 2 - 1, 0])
        linear_extrude(height = block_h + 1)
            polygon([[-hole_d / 2 - 1, -hole_d / 2 - 1],
                     [-hole_d / 2 - 1, block_w / 2 - hole_d / 2 - 1],
                     [x_last - hole_d / 2 - 1, block_w / 2 - hole_d / 2 - 1],
                     [x_last - hole_d / 2 - 1, -hole_d / 2 - 1]]);
}
