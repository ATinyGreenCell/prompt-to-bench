```openscad
// Pestle for 1.5 mL microcentrifuge tube
// Print orientation: standing upright on XY plane

// ---- Dimensions (mm) ----
handle_d      = 8;     // handle diameter (z = 0 to 45)
cone_top_d    = 8;     // cone diameter at z = 45
cone_bottom_d = 3;     // cone diameter at z = 62
tip_d         = 3;     // sphere tip diameter, centre at z = 62
groove_w      = 2;     // groove width in Z
groove_depth  = 1;     // groove depth into handle
groove_z      = [10, 15, 20, 25]; // groove centre heights

// ---- Modelling parameters ----
eps = 0.5;             // cutter overshoot past faces
$fn = 96;              // curve resolution

// ---- Model ----
difference() {
    union() {
        // Handle
        cylinder(d = handle_d, h = 45);
        // Cone (z = 45 to 62)
        translate([0, 0, 45])
            cylinder(d1 = cone_top_d, d2 = cone_bottom_d, h = 17);
        // Rounded tip (sphere centred at z = 62)
        translate([0, 0, 62 - tip_d / 2])
            sphere(d = tip_d);
    }
    // Grip grooves
    for (z = groove_z)
        translate([0, 0, z - groove_w / 2 - eps])
            cylinder(d = handle_d + 2 * eps, h = groove_w + 2 * eps, $fn = 64);
}
```
