```openscad
// Pestle for grinding tissue inside a 1.5 mL microcentrifuge tube
// Printed standing upright on the XY plane (z = 0), centred on the origin in X and Y.

// ---------------- Parameters ----------------
handle_d      = 8;    // handle diameter (mm)
handle_h      = 45;   // handle height (mm), z = 0 to z = 45

cone_top_d    = 8;    // cone diameter at z = 45 (mm)
cone_bottom_d = 3;    // cone diameter at z = 62 (mm)
cone_h        = 62 - 45; // cone height (mm)

tip_d         = 3;    // tip sphere diameter (mm)
tip_z         = 62;   // tip centre height (mm)

groove_w      = 2;    // groove width (mm), in Z
groove_depth  = 1;    // groove depth (mm)
groove_z      = [10, 15, 20, 25]; // groove centre heights (mm)

eps           = 0.5;  // overshoot for clean boolean cuts (mm)
$fn           = 96;   // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // (1) Cylindrical handle
        cylinder(d = handle_d, h = handle_h);

        // (2) Conical section
        translate([0, 0, handle_h])
            cylinder(d1 = cone_top_d, d2 = cone_bottom_d, h = cone_h);

        // (3) Rounded tip: sphere centred at z = 62
        translate([0, 0, tip_z])
            sphere(d = tip_d);
    }

    // Grip grooves cut into the handle
    for (gz = groove_z)
        translate([0, 0, gz - eps])
            cylinder(d = handle_d + 2 * groove_depth, h = groove_w + 2 * eps);
}
```
