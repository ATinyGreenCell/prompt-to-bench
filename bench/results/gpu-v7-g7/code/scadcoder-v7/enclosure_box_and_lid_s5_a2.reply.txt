// Two-part enclosure for a small electronics board (e.g. Arduino-based temperature logger)
// Print orientation: both parts lie on the bed (z = 0), side by side, with a 10 mm gap.
// Units: mm

// ---------------- Parameters ----------------
box_x      = 70;    // box outer length (X)
box_y      = 50;    // box outer width (Y)
box_z      = 30;    // box outer height (Z)
wall       = 2;     // box wall thickness
floor      = 2;     // box floor thickness
lip_t      = 1.5;   // lid lip wall thickness
lip_h      = 4;     // lid lip height (from plate top)
lip_outer_x = 65.6; // outer size of lip in X
lip_outer_y = 45.6; // outer size of lip in Y
clearance  = 0.2;   // clearance between lip outer size and box opening
gap        = 10;    // gap between box and lid
eps        = 0.5;   // cutter overshoot past faces

$fn = 64;

// Derived
box_inner_x = box_x - 2 * wall;
box_inner_y = box_y - 2 * wall;
box_inner_z = box_z - floor - wall;   // open-top height
lip_overlap = clearance + wall;       // lip extends into box by clearance + wall (0.2+2=2.2)

// ---------------- Modules ----------------
module box() {
    difference() {
        translate([0, 0, 0])
            cube([box_x, box_y, box_z]);
        // open top: empty space above the top wall
        translate([wall, wall, box_z - wall])
            cube([box_inner_x, box_inner_y, wall + eps]);
    }
}

module lid() {
    // Plate lying on the bed
    translate([0, 0, 0])
        cube([box_x, box_y, 2]);
    // Rectangular lip centred on the plate, printed upside down
    translate([(box_x - lip_outer_x) / 2, (box_y - lip_outer_y) / 2, 2])
        cube([lip_outer_x, lip_outer_y, lip_t + lip_h]);
}

// ---------------- Assembly ----------------
union() {
    box();
    translate([box_x + gap, 0, 0])
        lid();
}
