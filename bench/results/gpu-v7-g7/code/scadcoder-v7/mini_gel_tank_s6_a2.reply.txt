```openscad
// Buffer tank for mini horizontal gel-electrophoresis box
// Print orientation: open top facing +Z, resting on XY plane at z = 0

// ---------------- Parameters ----------------
outer_x   = 120;   // outer length (X)
outer_y   = 70;    // outer width (Y)
outer_z   = 40;    // outer height (Z)
wall      = 3;     // wall thickness
floor     = 3;     // floor thickness

inner_x   = outer_x - 2*wall;   // inner length
inner_y   = outer_y - 2*wall;   // inner width
inner_z   = outer_z - wall - floor;  // inner height (from inner floor up)

platform_x = 60;     // platform length in X (centred)
platform_top_z = 13; // platform top height above inner floor (z = floor + 10)

hole_d    = 2;       // electrode hole diameter
hole_y    = 0;       // hole Y position
hole_z    = 30;      // hole Z position (above inner floor)

eps       = 0.5;     // cutter overshoot past faces

$fn = 96;

// ---------------- Model ----------------
difference() {
    // Tank body: open-top box with solid floor
    translate([-outer_x/2, -outer_y/2, 0])
        cube([outer_x, outer_y, outer_z]);

    // Cut out the open top (leaves the 3 mm wall at z = outer_z - wall)
    translate([-outer_x/2 - eps, -outer_y/2 - eps, outer_z - wall])
        cube([outer_x + 2*eps, outer_y + 2*eps, wall + eps]);

    // Gel-tray platform, raised from the floor, spanning the full inner width
    translate([-outer_x/2, -inner_y/2, floor])
        cube([platform_x, inner_y, platform_top_z - floor + eps]);

    // Horizontal electrode holes through each end wall, along X
    for (sx = [-1, 1])
        translate([sx > 0 ? outer_x/2 - hole_d/2 : -outer_x/2 - eps, hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = outer_y + 2*eps, center = true);
}
```
