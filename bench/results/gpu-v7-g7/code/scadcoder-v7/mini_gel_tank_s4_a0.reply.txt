// Buffer tank for mini horizontal gel-electrophoresis box
// Print orientation: open top facing +Z, resting on XY plane at z = 0

// ---------------- Parameters ----------------
outer_x   = 120;   // outer length in X
outer_y   = 70;    // outer width in Y
outer_z   = 40;    // outer height in Z
wall      = 3;     // wall thickness
floor     = 3;     // floor thickness

inner_h   = outer_z - wall - floor;   // inner height of tank (34)
platform_t = 10;                      // platform thickness (top at z = 13)
platform_x = 60;                      // platform length in X, centred
hole_d     = 2;                       // electrode hole diameter
hole_y     = 0;                       // electrode hole Y position
hole_z     = 30;                      // electrode hole Z position

eps       = 0.5;                      // cutter overshoot past faces
$fn       = 64;                       // curve resolution

// ---------------- Derived ----------------
inner_w = outer_y - 2 * wall;         // inner width (64)
inner_l = outer_x - 2 * wall;         // inner length (114)
chamber_h = inner_h - platform_t;     // chamber height above platform (24)

// ---------------- Model ----------------
difference() {
    // Main tank body (open top)
    translate([-outer_x/2, -outer_y/2, 0])
        cube([outer_x, outer_y, outer_z - wall]);

    // Gel-tray platform, raised above inner floor
    translate([-outer_x/2 + wall, -inner_w/2, floor + chamber_h])
        cube([platform_x, inner_w, platform_t]);

    // Horizontal electrode holes through each end wall along X
    for (s = [-1, 1])
        translate([s * (inner_l/2 - hole_d/2 - eps), hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = inner_l + 2*eps + wall);
}
