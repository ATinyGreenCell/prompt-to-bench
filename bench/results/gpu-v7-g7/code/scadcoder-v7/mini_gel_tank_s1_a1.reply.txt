```openscad
// Buffer tank for mini horizontal gel-electrophoresis box
// Print orientation: base on XY plane, open top facing +Z

// ---------- Parameters ----------
outer_x      = 120;   // outer length (X)
outer_y      = 70;    // outer width (Y)
outer_z      = 40;    // outer height (Z)
wall         = 3;     // wall thickness
floor        = 3;     // floor thickness
platform_x   = 60;    // platform length in X (centred)
platform_top = 13;    // platform top height above tank base (z)
hole_d       = 2;     // electrode hole diameter
hole_y       = 0;     // electrode hole Y position
hole_z       = 30;    // electrode hole Z position

eps          = 0.5;   // cutter overshoot past faces
$fn          = 64;    // curve resolution

// ---------- Derived ----------
inner_x = outer_x - 2 * wall;   // 114
inner_y = outer_y - 2 * wall;   // 64
inner_z = outer_z - wall - floor; // 34 (inner floor at z=0, top at z=34)

// ---------- Model ----------
difference() {
    // Tank body (open top)
    translate([wall, wall, floor])
        cube([inner_x, inner_y, inner_z]);

    // Raised platform for gel tray
    translate([(outer_x - platform_x) / 2, wall, floor])
        cube([platform_x, inner_y, platform_top - floor]);

    // Horizontal electrode holes through each end wall (along X)
    for (sx = [-1, 1])
        translate([sx > 0 ? outer_x - eps : -eps, hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = outer_z + 2 * eps);
}
```
