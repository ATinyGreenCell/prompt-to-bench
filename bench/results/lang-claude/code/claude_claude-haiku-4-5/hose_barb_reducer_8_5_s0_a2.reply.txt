barb_5mm_base_r = barb_5mm_base_od / 2;
barb_5mm_valley_r = barb_5mm_valley_od / 2;

// Collar between barbed sections

collar_r = collar_od / 2;
collar_z_start = 20;
collar_z_end = 25;

// Central hole through entire connector
hole_od = 3;                    // hole outer diameter
hole_r = hole_od / 2;

// Total height

$fn = 64;

// 2D profile for rotate_extrude (XZ plane: X=radius, Z=height)
// Rotated 360° around Z-axis to create 3D solid
profile = [
    [0, 0],  // Center at z=0
    
    // First 8mm barb (z=0 to z=10): triangular peak
    [barb_8mm_base_r, 0],
    [barb_8mm_peak_r, barb_8mm_height / 2],
    [barb_8mm_base_r, barb_8mm_height],

    [barb_5mm_base_r, total_height],
