gusset_right_x_min = 20;

$fn = 64;

// Module for triangular prism gusset
module triangular_prism(x_min, x_max, y_corner, z_corner, y_length, z_length) {
  hull() {
    translate([x_min, y_corner, z_corner]) sphere(r=0.01);
    translate([x_min, y_corner + y_length, z_corner]) sphere(r=0.01);
    translate([x_min, y_corner, z_corner + z_length]) sphere(r=0.01);
    translate([x_max, y_corner, z_corner]) sphere(r=0.01);
    translate([x_max, y_corner + y_length, z_corner]) sphere(r=0.01);
    translate([x_max, y_corner, z_corner + z_length]) sphere(r=0.01);
  }
}

// Main assembly
difference() {
  union() {
    // Base platform: 50x45x5 mm
    translate([-25, 0, 0])
      cube([50, 45, 5]);

    translate([-25, 0, 0])
      cube([50, 5, 50]);

    triangular_prism(gusset_left_x_min, gusset_left_x_max,

    // Right triangular gusset
    triangular_prism(gusset_right_x_min, gusset_right_x_max,

  translate([motor_shaft_x, -0.5, motor_shaft_z])
    rotate([90, 0, 0])
    cylinder(d = motor_shaft_d, h = 6);
  
  // Cut M3 mounting holes in square pattern

        cylinder(d = motor_m3_hole_d, h = 6);
    }
  }
