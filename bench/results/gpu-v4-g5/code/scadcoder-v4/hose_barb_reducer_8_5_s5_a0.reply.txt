// Hose-barb reducer: joins 8 mm inner-diameter silicone tubing to 5 mm inner-diameter tubing
// Print orientation: standing upright along +Z, centred on origin in X/Y
// Units: mm

$fn = 96;

// ---- Parameters ----
reducer_id      = 8;   // inner diameter of larger (8 mm) tubing
reducer2_id     = 5;   // inner diameter of smaller (5 mm) tubing
bore            = 3;   // through-bore diameter
sawtooth        = 0.01;// small overshoot for clean boolean cuts

// Barb dimensions for 8 mm tubing (z = 0 to 20)
barb_h1         = 10;  // barb height
barb_d1_bot     = 7.5; // outer diameter at bottom of barbs
barb_d1_top     = 9.5; // outer diameter at top of barbs

// Collar dimensions (z = 20 to 25)
collar_d        = 12;  // collar diameter

// Barb dimensions for 5 mm tubing (z = 25 to 41)
barb_h2         = 8;   // barb height
barb_d2_bot     = 6.5; // outer diameter at bottom of barbs
barb_d2_top     = 5.0; // outer diameter at top of barbs

// ---- Model ----
difference() {
    union() {
        // Lower section: two stacked barbs for 8 mm tubing
        for (i = [0, 1])
            translate([0, 0, i * barb_h1])
                linear_extrude(height = barb_h1)
                    circle(d = barb_d1_bot + (barb_d1_top - barb_d1_bot) * (z / barb_h1));

        // Collar
        translate([0, 0, 20])
            cylinder(d = collar_d, h = 5);

        // Upper section: two stacked barbs for 5 mm tubing
        for (i = [0, 1])
            translate([0, 0, 25 + i * barb_h2])
                linear_extrude(height = barb_h2)
                    circle(d = barb_d2_bot + (barb_d2_top - barb_d2_bot) * (z / barb_h2));
    }

    // Cutting shapes, extended past faces by sawtooth
    translate([0, 0, -sawtooth])
        linear_extrude(height = 42 + 2 * sawtooth)
            circle(d = reducer_id);

    translate([0, 0, 25 - sawtooth])
        linear_extrude(height = 42 + 2 * sawtooth)
            circle(d = reducer2_id);

    translate([0, 0, -sawtooth])
        linear_extrude(height = 42 + 2 * sawtooth)
            circle(d = bore);
}
