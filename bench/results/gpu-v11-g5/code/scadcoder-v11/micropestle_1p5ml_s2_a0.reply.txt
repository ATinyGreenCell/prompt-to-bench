// Tissue grinding pestle for 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the bed (z = 0)

$fn = 96;

// ---- Dimensions (mm) ----
handle_d      = 8;    // handle diameter (z = 0 to 45)
handle_h      = 45;   // handle height
cone_top_d    = 8;    // cone diameter at z = 45 (base)
cone_tip_d    = 3;    // cone diameter at z = 62 (tip)
cone_base_z   = 45;   // cone base height
cone_tip_z    = 62;   // cone tip height
tip_d         = 3;    // rounded tip sphere diameter
tip_z         = 62;   // tip centre height
groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width (Z)
groove_depth  = 1;    // groove depth into handle
groove_z      = [10, 15, 20, 25]; // groove centre heights

eps = 0.5;            // cutter overshoot

// ---- Model ----
difference() {
    union() {
        // Handle
        cylinder(d = handle_d, h = handle_h);
        // Cone (tapered, overlapping base into handle for a clean union)
        translate([0, 0, cone_base_z - 0.01])
            cylinder(d1 = cone_top_d, d2 = cone_tip_d, h = cone_tip_z - cone_base_z + 0.01);
        // Rounded tip
        translate([0, 0, tip_z])
            sphere(d = tip_d);
    }
    // Grip grooves
    for (z = groove_z)
        translate([0, 0, z - groove_w / 2])
            cylinder(d = handle_d + 2 * eps, h = groove_w + eps);
}
