```openscad
// Mini horizontal gel-electrophoresis buffer tank
// Print orientation: base on XY plane, open top, builds +Z

$fn = 64;

// ---- Tank ----
tank_x = 120;      // outer length (X)
tank_y = 70;       // outer width (Y)
tank_z = 40;       // outer height (Z)
wall   = 3;        // wall thickness
floor  = 3;        // floor thickness

// ---- Platform ----
plat_len   = 60;   // platform length in X
plat_top_z = 13;   // platform top height (z = floor + 10)

// ---- Holes ----
hole_d = 2;        // electrode hole diameter
hole_y = 0;        // hole Y position
hole_z = 30;       // hole Z position

// ---- Derived ----
inner_f = tank_y - 2*wall;              // inner width (Y)
inner_h = tank_z - floor;               // inner floor height to top
eps     = 0.5;                          // cutter overshoot

difference() {
    // Tank body with solid end walls (no top face)
    union() {
        cube([tank_x, tank_y, floor]);                                          // floor
        cube([tank_x, wall, tank_z - floor]);                                  // back wall at y = 0
        cube([tank_x, wall, tank_z - floor]);                                  // front wall at y = tank_y - wall
    }

    // Raised platform, full inner width, centred in X
    translate([-tank_x/2 + plat_len/2, 0, floor])
        cube([plat_len, inner_f, plat_top_z - floor]);

    // Electrode holes through each end wall along X
    for (sx = [-1, 1])
        translate([sx > 0 ? tank_x - wall : 0, hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = wall + eps, center = true);
}
```
