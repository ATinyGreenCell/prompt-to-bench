```openscad
// Hose-barb reducer: 8 mm tubing (barbs) -> 5 mm tubing (barbs)
// Print orientation: standing upright along Z

$fn = 96;

// ---- 8 mm tubing barbs ----
t8_od     = 9.5;    // outer diameter at top of barb
t8_base   = 7.5;    // outer diameter at bottom (step back to this)
t8_h      = 10;     // barb height
t8_count  = 2;      // number of stacked barbs

// ---- Collar ----
collar_d  = 12;     // collar diameter
collar_z0 = 20;     // collar start height
collar_z1 = 25;     // collar end height

// ---- 5 mm tubing barbs ----
t5_od     = 5.0;    // outer diameter at top of barb
t5_base   = 6.5;    // outer diameter at bottom (step back to this)
t5_h      = 8;      // barb height
t5_count  = 2;      // number of stacked barbs

// ---- Bore ----
bore      = 3;      // through-bore diameter
eps       = 0.5;    // cutter overshoot

total_h   = t8_h + (collar_z1 - collar_z0) + t5_h;  // 41 mm

difference() {
    union() {
        // 8 mm barbs
        for (i = [0 : t8_count - 1])
            translate([0, 0, i * t8_h])
                hull() {
                    cylinder(d = t8_base, h = 0.01);   // bottom step
                    translate([0, 0, t8_h - 0.01])
                        cylinder(d = t8_od, h = 0.01);  // top tip
                }

        // Collar
        translate([0, 0, collar_z0])
            cylinder(d = collar_d, h = collar_z1 - collar_z0);

        // 5 mm barbs
        for (i = [0 : t5_count - 1])
            translate([0, 0, collar_z1 + i * t5_h])
                hull() {
                    cylinder(d = t5_base, h = 0.01);   // bottom step
                    translate([0, 0, t5_h - 0.01])
                        cylinder(d = t5_od, h = 0.01);  // top tip
                }
    }

    // Through bore
    translate([0, 0, -eps])
        cylinder(d = bore, h = total_h + 2 * eps);
}
```
