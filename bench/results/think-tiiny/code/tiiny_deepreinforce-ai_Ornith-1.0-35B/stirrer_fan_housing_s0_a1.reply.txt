```openscad
$fn = 64;

// --- Dimensions ---
plate_thickness = 2;
wall_thickness = 2.5;
outer_size = 90;
wall_height = 40;
inner_size = outer_size - 2 * wall_thickness;
screw_pattern = 71.5;
screw_hole_dia = 4.5;
notch_width = 12;
notch_height = 8;
notch_z_start = wall_height - notch_height;

// --- Base Plate ---
base = translate([0, 0, 0]) cube([outer_size, outer_size, plate_thickness]);

// --- Walls ---
walls = difference(
    translate([0, 0, plate_thickness]) cube([outer_size, outer_size, wall_height - plate_thickness]),
    translate([0, 0, plate_thickness - 0.01]) cube([inner_size, inner_size, wall_height - plate_thickness + 0.02])
);

housing = union(base, walls);

// --- Screw Holes ---
module screw_holes() {
    for (x = [-screw_pattern/2, screw_pattern/2]) {
        for (y = [-screw_pattern/2, screw_pattern/2]) {
            translate([x, y, 0])
                cylinder(d=screw_hole_dia, h=plate_thickness + 0.01);
        }
    }
}

// --- Cable Notch (front wall, +Y) ---
notch_cut = translate([
    0,
    outer_size/2 - wall_thickness/2,
    notch_z_start
])
cube([notch_width, wall_thickness + 0.01, notch_height]);

// --- Final Assembly ---
result = difference(housing, screw_holes(), notch_cut);

// Output
result;
```
