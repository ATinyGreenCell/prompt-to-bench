// Sowing template for 90 mm Petri dish
// Disc: 85 mm diameter, 2 mm thick
// Grid: 7x7 holes, 3 mm diameter, 9 mm spacing
// Notch: 10 mm wide (X), 4 mm deep (Y) on +Y side

$fn = 64;

// Disc dimensions
DISC_DIAM = 85;       // Outer diameter of the template disc
DISC_THICK = 2;       // Thickness of the template disc

// Grid hole dimensions
HOLE_DIAM = 3;        // Diameter of each grid hole
GRID_SPACING = 9;     // Centre-to-centre spacing of grid holes

// Orientation notch dimensions
NOTCH_W = 10;         // Width of the notch along X axis
NOTCH_D = 4;          // Depth of the notch along Y axis (measured from rim)

difference() {
    // Base disc
    cylinder(d = DISC_DIAM, h = DISC_THICK);

    // 7x7 grid of through-holes
    for (x = [-3:1:3]) {
        for (y = [-3:1:3]) {
            translate([x * GRID_SPACING, y * GRID_SPACING, 0]) {
                cylinder(d = HOLE_DIAM, h = DISC_THICK + 0.01);
            }
        }
    }

    // Rectangular orientation notch on +Y side
    // Centred on Y axis, extending 4 mm in from the rim (Y = 42.5)
    translate([0, DISC_DIAM / 2 - NOTCH_D / 2, 0]) {
        cube([NOTCH_W, NOTCH_D, DISC_THICK + 0.01], center = true);
    }
}
