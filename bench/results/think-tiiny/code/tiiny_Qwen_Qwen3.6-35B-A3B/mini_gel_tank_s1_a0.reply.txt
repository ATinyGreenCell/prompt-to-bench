```openscad
$fn = 64;

// Outer dimensions
outer_x = 120;
outer_y = 70;
outer_z = 40;

// Wall and floor thickness
wall_thick = 3;
floor_thick = 3;

// Platform dimensions
platform_x = 60;
platform_top_z = 13;

// Hole dimensions
hole_diam = 2;
hole_z = 30;
hole_y = 0;

// Derived dimensions
inner_x = outer_x - 2 * wall_thick;
inner_y = outer_y - 2 * wall_thick;
inner_z = outer_z - floor_thick;
platform_y = inner_y;
platform_z_height = platform_top_z - floor_thick;

// Tank shell (floor + 4 walls)
module tank_shell() {
    union() {
        // Floor
        translate([0, 0, floor_thick / 2]) cube([outer_x, outer_y, floor_thick], center = false);
        // Back wall (Y negative)
        translate([-inner_x / 2, -outer_y / 2 + wall_thick / 2, floor_thick + inner_z / 2]) cube([inner_x, wall_thick, inner_z], center = false);
        // Front wall (Y positive)
        translate([-inner_x / 2, outer_y / 2 - wall_thick / 2, floor_thick + inner_z / 2]) cube([inner_x, wall_thick, inner_z], center = false);
        // Left wall (X negative)
        translate([-outer_x / 2 + wall_thick / 2, -inner_y / 2, floor_thick + inner_z / 2]) cube([wall_thick, inner_y, inner_z], center = false);
        // Right wall (X positive)
        translate([outer_x / 2 - wall_thick / 2, -inner_y / 2, floor_thick + inner_z / 2]) cube([wall_thick, inner_y, inner_z], center = false);
    }
}

// Cut out the inner cavity
difference() {
    tank_shell();
    translate([-inner_x / 2, -inner_y / 2, floor_thick]) cube([inner_x, inner_y, inner_z], center = false);
}

// Add the raised platform
translate([-platform_x / 2, -inner_y / 2, floor_thick]) cube([platform_x, platform_y, platform_z_height], center = false);

// Drill electrode holes through the end walls
for (x_pos = [-outer_x / 2, outer_x / 2]) {
    translate([x_pos, hole_y, hole_z]) cylinder(r = hole_diam / 2, h = 4, center = false);
}
```
