$fn = 64;

block_length = 90;
block_width = 45;
block_height = 20;
slot_length = 77;
slot_width = 1.6;
slot_depth = 10;
slot_spacing = 3.5;
n = 10;

total_span = (slot_spacing * (n - 1)) + slot_width;
start_y = (block_width - total_span) / 2;

difference() {
    cube([block_length, block_width, block_height]);
    for (i = [0:n-1]) {
        translate([0, start_y + i * slot_spacing, 0])
        cube([slot_length, slot_width, slot_depth]);
    }
}
