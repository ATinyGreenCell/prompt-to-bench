// Block dimensions
BLOCK_LENGTH = 127.76;    // X dimension (length)
BLOCK_WIDTH = 85.48;      // Y dimension (width)
BLOCK_HEIGHT = 20;       // Z dimension (height)

// Hole specifications
HOLE_DIAMETER = 6.2;     // Hole diameter
HOLE_SPACING = 9.0;    // Center-to-center spacing
NUM_ROWS = 8;            // Rows along Y axis
NUM_COLS = 12;           // Columns along X axis

// First hole (A1) position from edges
HOLE_A1_X_OFFSET = 14.38;  // From left edge
HOLE_A1_Y_OFFSET = 11.24;  // From back edge

// Chamfer specifications
CHAMFER_SIZE = 5;          // Leg length
CHAMFER_DEPTH = 20.01;    // Slightly past block height

// Calculate hole grid positions
// Block center is at origin (0, 0)
// Left edge at X = -BLOCK_LENGTH/2
// Back edge at Y = -BLOCK_WIDTH/2

FIRST_HOLE_X = -BLOCK_LENGTH/2 + HOLE_A1_X_OFFSET;
FIRST_HOLE_Y = -BLOCK_WIDTH/2 + HOLE_A1_Y_OFFSET;

// Create the base block
module base_block() {
    cube([BLOCK_LENGTH, BLOCK_WIDTH, BLOCK_HEIGHT], center = true);
}

// Create the hole grid
module hole_grid() {
    for (col = [0:NUM_COLS-1]) {
        for (row = [0:NUM_ROWS-1]) {
            hole_x = FIRST_HOLE_X + col * HOLE_SPACING;
            hole_y = FIRST_HOLE_Y + row * HOLE_SPACING;
            translate([hole_x, hole_y, 0]) {
                cylinder(d = HOLE_DIAMETER, h = BLOCK_HEIGHT + 0.01, center = true);
            }
        }
    }
}

// Create the chamfer at A1 corner (45-degree, 5mm legs)
module chamfer() {
    // A1 corner is at FIRST_HOLE_X, FIRST_HOLE_Y
    // Create a wedge that cuts 5mm from both edges at 45 degrees
    translate([FIRST_HOLE_X, FIRST_HOLE_Y, 0]) {
        // Create a triangular prism wedge
        // The wedge cuts diagonally through the corner
        difference() {
            // Base wedge shape
            cube([CHAMFER_SIZE * 2, CHAMFER_SIZE * 2, CHAMFER_DEPTH], center = true);
            // Cut away the corner with a 45-degree plane
            // Using a rotated cube to create the diagonal cut
            rotate([0, 0, 45]) {
                cube([CHAMFER_SIZE * 2, CHAMFER_SIZE * 2, CHAMFER_DEPTH], center = true);
            }
        }
    }
}

// Main assembly - create block with holes and chamfer
difference() {
    base_block();
    // Subtract holes
    hole_grid();
    // Subtract chamfer
    chamfer();
}
