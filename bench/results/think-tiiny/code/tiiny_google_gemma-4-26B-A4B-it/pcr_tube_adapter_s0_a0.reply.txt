```openscad
// Dimensions
collar_diameter = 13.0;        // Diameter of the base collar
collar_thickness = 1.5;        // Thickness of the base collar
sleeve_outer_diameter = 10.8;  // Outer diameter of the main sleeve
total_height = 21.5;           // Total height from the bed to the top
bore_diameter = 6.2;           // Diameter of the central hole
$fn = 64;                      // Resolution for curved surfaces

// Part Construction
difference() {
    // The solid part of the adapter
    union() {
        // Base collar (z = 0 to 1.5)
        cylinder(d=collar_diameter, h=collar_thickness);
        
        // Sleeve body (rises from collar to z = 21.5)
        // The height of this cylinder is total_height minus the collar thickness
        translate([0, 0, collar_thickness])
            cylinder(d=sleeve_outer_diameter, h=total_height - collar_thickness);
    }

    // The central bore (runs through the full height)
    // We extend the cutting cylinder slightly beyond the object boundaries 
    // to ensure a clean through-hole.
    translate([0, 0, -1])
        cylinder(d=bore_diameter, h=total_height + 2);
}
```
