```openscad
// 10-well comb for casting agarose gels
// Print orientation: flat on bed, teeth pointing down (-Z)

// ---------- Parameters ----------
plate_t      = 1.5;   // plate thickness (Z)
spine_len    = 70;    // spine length (X)
spine_w      = 12;    // spine width (Y)
tooth_n      = 10;    // number of teeth
tooth_w      = 5.0;   // tooth width (X)
tooth_len    = 12;    // tooth length (Y)
tooth_pitch  = 1.5;   // gap between neighbouring teeth (X)
eps          = 0.5;   // overcut for clean differences

$fn = 64;

// ---------- Derived ----------
tooth_span = (tooth_n - 1) * tooth_pitch;   // 16.5 mm
x0         = -(spine_len / 2 + tooth_span / 2); // first tooth left edge

// ---------- Model ----------
difference() {
    union() {
        // Spine
        translate([-spine_len / 2, -spine_w / 2, 0])
            cube([spine_len, spine_w, plate_t]);

        // Teeth
        for (i = [0 : tooth_n - 1])
            translate([x0 + i * tooth_pitch - tooth_w / 2,
                       -tooth_len / 2, 0])
                cube([tooth_w, tooth_len, plate_t]);
    }

    // Tooth cutouts (extend past -Y face)
    for (i = [0 : tooth_n - 1])
        translate([x0 + i * tooth_pitch - tooth_w / 2,
                   -tooth_len / 2 - eps, 0])
            cube([tooth_w, tooth_len + 2 * eps, plate_t]);
}
```
