```openscad
// 10-well comb for agarose gel casting
// Printed lying flat, teeth pointing down (-Y), spine along X

// ---------- Parameters ----------
tooth_count  = 10;    // number of wells
tooth_width  = 5.0;   // tooth width along X (mm)
tooth_length = 12;    // tooth length along Y (mm)
teeth_gap    = 1.5;   // gap between neighbouring teeth along X (mm)
spine_width  = 12;    // spine width along Y (mm)
spine_length = 70;    // spine length along X (mm)
plate_thk    = 1.5;   // plate thickness along Z (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)

// ---------- Derived ----------
pitch      = tooth_width + teeth_gap;            // centre-to-centre spacing of teeth
teeth_span = (tooth_count - 1) * pitch;          // distance between first and last tooth centres
x0         = -teeth_span / 2;                    // first tooth centre (row centred on origin)

// ---------- Model ----------
union() {
    // Teeth: extend past -Y edge of spine so the cut is clean
    for (i = [0 : tooth_count - 1])
        translate([x0 + i * pitch - tooth_width / 2,
                   -tooth_length - eps,
                   0])
            cube([tooth_width, tooth_length + eps, plate_thk]);

    // Spine (back bar)
    translate([-spine_length / 2, -spine_width / 2, 0])
        cube([spine_length, spine_width, plate_thk]);
}
```
