// Pestle for grinding tissue inside a 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the bed (z = 0), building up +Z

// ---------------- Parameters ----------------
handle_d      = 8;    // handle diameter (mm)
handle_h      = 45;   // handle height (mm), z = 0..45
cone_d_top    = 8;    // cone diameter at its bottom, z = 45 (mm)
cone_d_tip    = 3;    // cone diameter at its top, z = 62 (mm)
cone_top_z    = 45;   // cone bottom height (mm)
cone_tip_z    = 62;   // cone top height (mm)
tip_d         = 3;    // rounded tip diameter (mm)
tip_z         = 62;   // tip centre height (mm)
groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width (mm), in Z
groove_depth  = 1;    // groove depth (mm)
groove_z      = [10, 15, 20, 25]; // groove centre heights (mm)
eps           = 0.5;  // cutter overshoot past faces (mm)
$fn           = 96;   // curve resolution

// ---------------- Model ----------------
difference() {
    union() {
        // Handle
        cylinder(d = handle_d, h = handle_h);

        // Cone: tapers from 8 mm at z = 45 to 3 mm at z = 62
        translate([0, 0, cone_top_z])
            cylinder(d1 = cone_d_top, d2 = cone_d_tip, h = cone_tip_z - cone_top_z);

        // Rounded tip: 3 mm diameter sphere centred at z = 62
        translate([0, 0, tip_z])
            sphere(d = tip_d);
    }

    // Grip grooves, cut through the handle
    for (g = groove_count) {
        rotate([0, 0, groove_z[g - 1] > 0 ? 0 : 90])
            translate([0, -groove_w / 2, groove_z[g - 1] - eps])
                cube([handle_d + 2 * eps, groove_w, groove_w + 2 * eps]);
    }
}
