```openscad
// 10-well comb for agarose gel casting, printed lying flat
// Units: mm

// ---- Spine ----
spine_len   = 70;    // spine length along X
spine_wid   = 12;    // spine width along Y
spine_thk   = 1.5;   // spine thickness along Z

// ---- Teeth ----
tooth_count = 10;    // number of wells/teeth
tooth_w     = 5.0;   // tooth width along X
tooth_l     = 12;    // tooth length along Y (hangs in -Y)
tooth_gap   = 1.5;   // gap between neighbouring teeth along X

// ---- Misc ----
eps         = 0.5;   // cutter overshoot past faces

$fn = 64;

// Derived
pitch   = tooth_w + tooth_gap;          // centre-to-centre tooth spacing
x_start = -(tooth_count - 1) * pitch / 2; // first tooth centre, row centred

difference() {
    // Spine (back bar)
    translate([-spine_len / 2, 0, 0])
        cube([spine_len, spine_wid, spine_thk]);

    // Teeth cut out from the top edge (at y = spine_wid)
    for (i = [0 : tooth_count - 1])
        translate([x_start + i * pitch - tooth_w / 2,
                   spine_wid - eps,
                   spine_thk - eps])
            cube([tooth_w, tooth_l + eps, tooth_l + eps]);
}
```
