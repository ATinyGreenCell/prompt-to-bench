```openscad
// 10-well comb for agarose gel casting, printed lying flat
// Units: mm

// ---- Spine ----
spine_len   = 70;   // spine length (X)
spine_wid   = 12;   // spine width (Y)

// ---- Teeth ----
n_teeth     = 10;   // number of teeth
tooth_w     = 5.0;  // tooth width (X)
tooth_len   = 12;   // tooth length (Y), hanging in -Y
gap         = 1.5;  // gap between neighbouring teeth (X)

// ---- Misc ----
eps         = 0.5;  // cutter overshoot past faces
$fn         = 64;

// Derived
pitch       = tooth_w + gap;          // centre-to-centre spacing
row_span    = (n_teeth - 1) * gap;    // total gap width
x_start     = -row_span / 2;          // first tooth centre (row centred)

difference() {
    // Spine (back bar)
    translate([-spine_len / 2, 0, 0])
        cube([spine_len, spine_wid, 1.5]);

    // Teeth cut out from the top face
    for (i = [0 : n_teeth - 1])
        translate([x_start + i * pitch - tooth_w / 2,
                   -tooth_len,
                   spine_wid - eps])
            cube([tooth_w, tooth_len + eps, 1.5]);
}
```
