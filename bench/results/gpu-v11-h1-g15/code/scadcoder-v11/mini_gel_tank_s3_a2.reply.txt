```openscad
// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: tank bottom on XY plane, opening upward (+Z)

// ---------------- Parameters ----------------
outer_x      = 120;   // outer tank length (X)
outer_y      = 70;    // outer tank width (Y)
tank_h       = 40;    // overall tank height (Z)
wall         = 3;     // wall thickness
floor        = 3;     // floor thickness (solid bottom)

platform_len = 60;    // gel tray platform length (X)
platform_h   = 10;    // platform height above inner floor (z = 13)
platform_y   = outer_y - 2*wall;  // platform spans full inner width

hole_d       = 2;     // electrode wire hole diameter
hole_z       = 30;    // hole height (through end walls)
hole_y       = 0;     // hole Y position

eps          = 0.5;   // cutter overshoot past faces
$fn          = 96;    // curve resolution

// ---------------- Model ----------------
difference() {
    // Tank body
    translate([-outer_x/2, -outer_y/2, 0])
        cube([outer_x, outer_y, tank_h]);

    // Open top (hollow out above floor)
    translate([-outer_x/2, -outer_y/2, floor])
        cube([outer_x, outer_y, tank_h - floor - eps]);

    // Gel tray platform
    translate([-platform_len/2, -platform_y/2, floor])
        cube([platform_len, platform_y, platform_h + eps]);

    // Electrode holes through end walls
    for (sx = [-1, 1])
        translate([sx * (outer_x/2 - eps), hole_y, hole_z])
            rotate([90, 0, 0])
                cylinder(d = hole_d, h = outer_y + 2*eps);
}
```
