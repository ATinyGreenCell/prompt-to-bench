```openscad
// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: tank bottom on the bed (z = 0), open top.

$fn = 64;

// ---- Tank dimensions ----
outer_x    = 120;   // outer length along X
outer_y    = 70;    // outer width along Y
tank_h     = 40;    // overall height (Z)
wall       = 3;     // wall thickness
floor      = 3;     // floor thickness under the lowest chamber

// ---- Platform (gel tray support) ----
plat_x     = 60;    // platform length along X
plat_y     = outer_y; // platform spans the full inner width
plat_top_z = 10;    // height of platform top above the inner floor

// ---- Electrode holes ----
hole_d     = 2;     // hole diameter
hole_y     = 0;     // hole Y position
hole_z     = 30;    // hole Z position
eps        = 0.5;   // cutter overshoot past end walls

// Derived values
inner_h    = tank_h - floor;                    // height above floor
plat_z0    = floor + (plat_top_z - wall);       // platform bottom height
region_x   = outer_x / 2 - wall;                // half-width of end chambers

difference() {
    // Main tank body
    translate([-outer_x / 2, -outer_y / 2, 0])
        cube([outer_x, outer_y, tank_h]);

    // Raised platform, full width, centred in X
    translate([-plat_x / 2, -plat_y / 2, plat_z0])
        cube([plat_x, plat_y, plat_top_z - plat_z0]);

    // Horizontal electrode holes through each end wall
    for (sx = [-1, 1])
        translate([sx * (region_x - eps), hole_y, hole_z])
            rotate([90, 0, 0])
                cylinder(d = hole_d, h = outer_y + 2 * eps);
}
```
