// Buffer tank for a mini horizontal gel-electrophoresis box. Open-top rectangular tank with a
// solid raised platform in the middle that divides the bottom into two buffer chambers. Each
// end wall gets a horizontal 2 mm hole along X at y = 0, z = 30 for the electrode wires.
//

Lx = 120;
Wy = 70;
H  = 40;
w  = 3;
fl = 3;

$fn = 64;

ilx = Lx - 2 * w;
iy  = Wy - 2 * w;
ih  = H - fl;

difference() {
    cube([Lx, Wy, H]);
    translate([w, w, fl]) cube([ilx, iy, ih]);     // inner cavity
    translate([w, w, 0]) cube([ilx, iy, fl]);      // floor
    cl = min(w, fl - 1);                           // clearance under the platform
    translate([0, w, fl - cl]) cube([Lx, Wy - 2 * w, cl]);  // left chamber floor
    translate([Lx - w, w, fl - cl]) cube([w, Wy - 2 * w, cl]);  // right chamber floor
    platform_t = fl - cl + 10;                     // platform top above inner floor
    translate([0, 0, fl - cl]) cube([Lx, Wy, cl]); // fill space below platform to keep it floating
    translate([(Lx - 60) / 2, 0, fl - cl]) cube([60, Wy, fl - cl]);  // platform X span, full width, at clearance height
    for (x = [-Lx / 2, Lx / 2])
        translate([x, -1, -1]) rotate([-90, 0, 0]) cylinder(h = Lx, d = 2);  // horizontal X holes in the end walls
}
