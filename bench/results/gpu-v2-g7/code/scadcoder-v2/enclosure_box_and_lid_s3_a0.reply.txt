```openscad
// Two-part enclosure for a small electronics board - prints in one go with a part between them.
// Part 1: open-top box (wall .2 mm, floor .2 mm). Part 2: lid plate with a rectangular lip that
// slides into the box opening (.2 mm clearance). The lid is printed upside down (lip up) so the
// wall fuses at the seam when the two parts are pushed together. NOVEL general part.
//

box_x = 70;
box_y = 50;
box_z = 30;
wall  = 2;
floor = 2;

lip_t = 1.5;
lip_h = 4;
lip_x = 65.6;
lip_y = 45.6;

$fn = 64;

box_inner_x = box_x - 2 * wall;
box_inner_y = box_y - 2 * wall;
box_inner_z = box_z - floor;

difference() {
    cube([box_x, box_y, box_z]);
    translate([wall, wall, floor]) cube([box_inner_x, box_inner_y, box_inner_z]);   // hollow, open top
}

plate_t = 2;
lip_w   = wall + lip_h;

difference() {
    union() {
        cube([lip_x, lip_y, plate_t]);                                    // plate
        cube([lip_x, lip_y, lip_h]);                                     // lip
    }
    translate([wall, wall, -1]) cube([lip_x - 2 * wall, lip_y - 2 * wall, plate_t + 2]);  // pocket the lip falls into
}

box   = cube([box_x, box_y, box_z]);
lip   = cube([lip_x, lip_y, lip_h]);
plate = cube([lip_x, lip_y, plate_t]);

g = 10;
union() {
    translate([-box_x / 2, -box_y / 2, 0]) box;
    translate([box_x / 2 + g, -box_y / 2, 0]) plate;
    translate([box_x / 2 + g, -box_y / 2, lip_h]) lip;
}
```
