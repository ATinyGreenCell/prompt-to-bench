```openscad
// Mini horizontal gel-electrophoresis box - buffer tank
// Print orientation: tank resting on XY plane, building up +Z

$fn = 64;

// ---- Tank dimensions ----
outer_x = 120;   // outer length in X
outer_y = 70;    // outer width in Y
outer_z = 40;    // outer height in Z
wall    = 3;     // wall thickness
floor   = 3;     // floor thickness

// ---- Platform dimensions ----
plat_x     = 60;   // platform length in X (centred)
plat_y     = outer_y; // platform spans full inner width
plat_top_z = 13;   // platform top height (10 mm above inner floor)

// ---- Electrode holes ----
hole_d     = 2;    // hole diameter
hole_y     = 0;    // hole Y position
hole_z     = 30;   // hole Z position

// ---- Derived ----
inner_x = outer_x - 2*wall;
inner_z = outer_z - wall;
eps     = 0.5;     // cutter overshoot

difference() {
    // Tank body
    difference() {
        cube([outer_x, outer_y, outer_z]);
        translate([wall, wall, floor])
            cube([inner_x, inner_y = outer_y - 2*wall, inner_z]); // hollow, open top
    }

    // Raised platform
    translate([-plat_x/2, 0, 0])
        cube([plat_x, plat_y, plat_top_z]);

    // Electrode holes through end walls
    for (sx = [-1, 1])
        translate([sx > 0 ? outer_x - wall : wall, hole_y, hole_z])
            rotate([90, 0, 0])
                cylinder(d = hole_d, h = wall + 2*eps, center = true);
}
```
