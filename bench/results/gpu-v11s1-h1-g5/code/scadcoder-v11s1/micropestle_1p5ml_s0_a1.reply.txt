```openscad
// Tissue pestle for 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the bed (z = 0)

$fn = 64;

// ---- Handle ----
handle_d      = 8;    // handle diameter
handle_h      = 45;   // handle height (z = 0 to 45)

// ---- Cone ----
cone_top_d    = 8;    // cone diameter at z = 45
cone_bottom_d = 3;    // cone diameter at z = 62
cone_top_z    = 45;   // cone start height
cone_h        = 17;   // cone height (62 - 45)

// ---- Tip ----
tip_d         = 3;    // tip sphere diameter
tip_cz        = 62;   // tip sphere centre height

// ---- Grooves ----
groove_count  = 4;    // number of circumferential grooves
groove_w      = 2;    // groove width in Z
groove_depth  = 1;    // groove depth from surface
groove_z      = [10, 15, 20, 25]; // groove centre heights

eps = 0.5;            // cutter overshoot

module handle() {
    translate([0, 0, 0])
        cylinder(d = handle_d, h = handle_h);
}

module cone() {
    translate([0, 0, cone_top_z])
        cylinder(d1 = cone_top_d, d2 = cone_bottom_d, h = cone_h);
}

module tip() {
    translate([0, 0, tip_cz])
        sphere(d = tip_d);
}

module grooves() {
    for (z = groove_z)
        translate([0, 0, z - eps])
            linear_extrude(height = groove_w + 2 * eps)
                circle(d = groove_depth * 2);
}

difference() {
    union() {
        handle();
        cone();
        tip();
    }
    for (z = groove_z)
        translate([0, 0, z - eps])
            linear_extrude(height = groove_w + 2 * eps)
                circle(d = groove_depth * 2);
}
```
