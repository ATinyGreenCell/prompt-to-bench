```openscad
// Dimensions
box_width = 70;
box_height = 50;
box_depth = 30;
box_wall = 2;
box_floor = 2;

lid_width = 70;
lid_height = 50;
lid_depth = 2;
lid_wall = 1.5;
lid_lip_height = 4;
lid_lip_width = 65.6;
lid_lip_clearance = 0.2;

// Box
box = difference() {
    cube([box_width, box_height, box_depth]);
    translate([box_wall, box_wall, 0]) cube([box_width - 2*box_wall, box_height - 2*box_wall, box_depth - box_floor]);
    translate([0, 0, box_depth - box_floor]) cube([box_width, box_height, box_floor]);
};

// Lid
lid = difference() {
    cube([lid_width, lid_height, lid_depth]);
    translate([lid_wall, lid_wall, 0]) cube([lid_width - 2*lid_wall, lid_height - 2*lid_wall, lid_depth]);
    translate([(lid_width - lid_lip_width)/2 - lid_lip_clearance, (lid_height - lid_lip_height)/2 - lid_lip_clearance, lid_depth]) cube([lid_lip_width + 2*lid_lip_clearance, lid_lip_height + 2*lid_lip_clearance, lid_wall]);
};

// Assembly
union() {
    translate([0, 0, 0]) box;
    translate([box_width + 10, 0, 0]) lid;
};
```
