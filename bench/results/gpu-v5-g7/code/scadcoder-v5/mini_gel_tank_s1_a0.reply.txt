```openscad
// Mini horizontal gel-electrophoresis buffer tank
// Print orientation: open top facing up (+Z), resting on XY plane at z = 0

// ---------- Parameters ----------
outer_x      = 120;   // outer length in X (mm)
outer_y      = 70;    // outer width in Y (mm)
outer_h      = 40;    // outer height in Z (mm)
wall         = 3;     // wall thickness (mm)
floor        = 3;     // floor thickness (mm)

platform_x   = 60;    // platform length in X (mm), centred
platform_y   = outer_y - 2 * wall;  // platform spans full inner width (mm)
platform_top = outer_h - wall + 10;  // platform top height, 10 mm above inner floor (mm)

hole_d       = 2;     // electrode hole diameter (mm)
hole_y       = 0;     // electrode hole Y position (mm)
hole_z       = 30;    // electrode hole Z position (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 96;    // curve resolution

// ---------- Derived ----------
inner_h = outer_h - floor - wall;   // inner floor-to-top height (mm)

// ---------- Model ----------
difference() {
    // Main tank body
    translate([-outer_x/2, -outer_y/2, 0])
        cube([outer_x, outer_y, outer_h]);

    // Open top (no lid)
    translate([-outer_x/2 - eps, -outer_y/2 - eps, outer_h - wall])
        cube([outer_x + 2*eps, outer_y + 2*eps, wall + eps]);

    // Raised platform in the middle, centred in X
    translate([-platform_x/2, -platform_y/2, floor])
        cube([platform_x, platform_y, platform_top - floor]);

    // Horizontal electrode holes through each end wall, along X
    for (s = [-1, 1])
        translate([s > 0 ? outer_x/2 - wall : -outer_x/2 - eps, hole_y, hole_z])
            rotate([0, 90, 0])
                cylinder(d = hole_d, h = outer_x + 2*eps);
}
```
