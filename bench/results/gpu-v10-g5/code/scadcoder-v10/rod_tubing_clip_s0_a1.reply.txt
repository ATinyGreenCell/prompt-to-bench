```openscad
// Snap-on tubing clip for 12.7 mm lab stand rod
// Print orientation: flat on bed, builds upward in +Z

// ---------- Dimensions (mm) ----------
stand_id     = 12.7;   // stand rod inner diameter
stand_wall   = 3;      // stand ring wall thickness
stand_od     = stand_id + 2 * stand_wall;  // 18.7 outer diameter of stand ring

tube_id      = 6;      // tubing inner diameter
tube_wall    = 2;      // tubing wall thickness
tube_od      = tube_id + 2 * tube_wall;    // 10 outer diameter of tubing ring

bridge_x0    = 8;      // bridge start (X)
bridge_x1    = 12;     // bridge end (X)
bridge_w     = 6;      // bridge width (Y)

stand_cut    = 10;     // stand ring opening width (-X side)
tube_cut     = 4.5;    // tubing ring opening width (+X side)

eps          = 0.5;    // cutter overshoot past faces
$fn          = 96;     // curve resolution

// ---------- Model ----------
difference() {
    union() {
        // Stand rod ring
        cylinder(d = stand_od, h = 10);

        // Tubing ring
        translate([15, 0, 0])
            cylinder(d = tube_od, h = 10);

        // Solid bridge
        translate([bridge_x0, -bridge_w / 2, 0])
            cube([bridge_x1 - bridge_x0, bridge_w, 10]);
    }

    // Stand ring opening
    translate([-stand_od / 2 - eps, -stand_cut / 2, -eps])
        cube([stand_od / 2 + eps, stand_cut, 10 + 2 * eps]);

    // Tubing ring opening
    translate([tube_od / 2 - tube_cut, -tube_cut / 2, -eps])
        cube([tube_cut + eps, tube_cut, 10 + 2 * eps]);
}
```
