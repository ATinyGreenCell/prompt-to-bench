// Straight hose-barb reducer: 8 mm tubing (z = 0..20) -> collar (z = 20..25) -> 5 mm tubing (z = 25..41)
// Print orientation: standing upright along +Z, centred on the origin in X/Y.

// ---------------- Parameters ----------------
reducer_in     = 8;    // 8 mm tubing inner diameter (mm)
reducer_in_h   = 20;   // height of 8 mm tubing barbs (mm)
reducer_in_od  = 7.5;  // outer diameter of 8 mm barbs at bottom (mm)
reducer_in_od2 = 9.5;  // outer diameter of 8 mm barbs at top (mm)

reducer_coll   = 12;   // collar diameter (mm)
reducer_coll_h = 5;    // collar height (mm)

reducer_out    = 5;    // 5 mm tubing inner diameter (mm)
reducer_out_h  = 8;    // height of 5 mm tubing barbs (mm)
reducer_out_od = 6.5;  // outer diameter of 5 mm barbs at bottom (mm)
reducer_out_od2 = 5.0; // outer diameter of 5 mm barbs at top (mm)

eps = 0.5;             // overshoot for clean boolean cuts (mm)

$fn = 96;              // curve resolution

// ---------------- Derived ----------------
total_h = reducer_in_h + reducer_coll_h + reducer_out_h;   // 33 mm
pitch   = 1.5;                                             // barb-to-barb spacing (mm)

// ---------------- Model ----------------
difference() {
    union() {
        // 8 mm tubing barbs (z = 0..20)
        for (i = [0 : 1])
            translate([0, -pitch * i, 0])
                linear_extrude(height = reducer_in_h)
                    polygon([
                        [0, 0],
                        [reducer_in_od / 2, reducer_in_h],
                        [-reducer_in_od / 2, reducer_in_h]
                    ]);

        // Collar (z = 20..25)
        translate([0, 0, reducer_in_h])
            cylinder(d = reducer_coll, h = reducer_coll_h);

        // 5 mm tubing barbs (z = 25..33)
        for (i = [0 : 1])
            translate([0, -pitch * i, reducer_in_h + reducer_coll_h])
                linear_extrude(height = reducer_out_h)
                    polygon([
                        [0, 0],
                        [reducer_out_od / 2, reducer_out_h],
                        [-reducer_out_od / 2, reducer_out_h]
                    ]);
    }

    // Through bore (full length)
    translate([0, 0, -eps])
        linear_extrude(height = total_h + 2 * eps, center = true)
            circle(d = reducer_in < reducer_out ? reducer_in : reducer_out);
}
