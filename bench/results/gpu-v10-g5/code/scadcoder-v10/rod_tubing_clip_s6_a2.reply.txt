```openscad
// Snap-on tubing clip for 12.7 mm stand rod
// Print orientation: flat on bed, extrusion along +Z

$fn = 96;

// ---- Dimensions (mm) ----
stand_id     = 12.7;  // stand rod inner diameter
stand_wall   = 3;     // stand ring wall thickness
stand_od     = stand_id + 2 * stand_wall;  // 18.7 outer diameter
tube_id      = 6;     // tubing inner diameter
tube_wall    = 2;     // tubing wall thickness
tube_od      = tube_id + 2 * tube_wall;    // 10 outer diameter
bridge_w     = 6;     // bridge width (Y)
bridge_x0    = 8;     // bridge start (X)
bridge_x1    = 12;    // bridge end (X)
opening_stand = 10;   // stand ring opening width (-X side)
opening_tube  = 4.5;  // tubing ring opening width (+X side)
eps          = 0.5;   // cutter overshoot past faces
height       = 10;    // extrusion height (Z)

// ---- Model ----
difference() {
    union() {
        // Stand rod ring
        cylinder(d = stand_od, h = height);
        // Tubing ring
        translate([15, 0, 0])
            cylinder(d = tube_od, h = height);
        // Solid bridge
        translate([bridge_x0, -bridge_w / 2, 0])
            cube([bridge_x1 - bridge_x0, bridge_w, height]);
    }
    // Stand ring opening (x < 0, -5 < y < 5)
    translate([-stand_od / 2 - eps, -opening_stand / 2, 0])
        cube([opening_stand + 2 * eps, opening_stand, height]);
    // Tubing ring opening (x > 15, -2.25 < y < 2.25)
    translate([15 + tube_od / 2 - eps, -opening_tube / 2, 0])
        cube([opening_tube + 2 * eps, opening_tube, height]);
}
```
