// Tubing clip for 12.7 mm stand rod
// Print orientation: flat on bed, extrusion along +Z

// ---- Stand rod ring ----
stand_d = 12.7;      // stand rod diameter
stand_w = 3;         // stand ring wall
stand_od = stand_d + 2 * stand_w;  // 18.7 mm outer diameter
stand_x = 0;         // ring centre X
stand_y = 0;         // ring centre Y

// ---- Tubing ring ----
tube_d = 6;          // tubing inner diameter
tube_w = 2;          // tubing wall
tube_od = tube_d + 2 * tube_w;    // 10 mm outer diameter
tube_x = 15;         // ring centre X
tube_y = 0;          // ring centre Y

// ---- Bridge ----
bridge_w = 6;        // bridge width in Y
bridge_x0 = 8;       // bridge start X
bridge_x1 = 12;      // bridge end X

// ---- Misc ----
eps = 0.5;           // cutter overshoot
$fn = 96;

difference() {
    union() {
        // Stand ring
        cylinder(d = stand_od, h = 10, center = true);
        // Tubing ring
        translate([tube_x, tube_y, 0])
            cylinder(d = tube_od, h = 10, center = true);
        // Bridge
        translate([bridge_x0, -bridge_w / 2, 0])
            cube([bridge_x1 - bridge_x0, bridge_w, 10]);
    }
    // Stand rod opening (through the ring)
    translate([stand_x - 1, stand_y, -eps])
        cube([stand_d + 2, 4, 10 + 2 * eps]);
    // Tubing opening (through the ring)
    translate([tube_x - 1, tube_y, -eps])
        cube([tube_d + 2, 4, 10 + 2 * eps]);
}
