// 96-well PCR tube rack (0.2 mL) - microplate footprint
// Print orientation: base on XY plane, builds up +Z

// ---------- Parameters ----------
block_len    = 127.76;   // block length along X (mm)
block_wid    = 85.48;    // block width along Y (mm)
block_h      = 20;       // block height along Z (mm)

hole_d       = 6.2;      // through-hole diameter (mm)
hole_rows    = 8;        // number of rows along Y
hole_cols    = 12;       // number of columns along X
hole_pitch   = 9.0;      // centre-to-centre spacing (mm)
hole_x0      = 14.38;    // first hole offset from left edge (mm)
hole_y0      = 11.24;    // first hole offset from back edge (mm)

chamfer_leg  = 5;        // 45-degree chamfer leg length (mm)
chamfer_r    = 5 * 1.414; // chamfer hypotenuse (mm)

eps          = 0.5;      // cutter overshoot past faces (mm)
$fn          = 64;       // curve resolution

// ---------- Model ----------
difference() {
    // Main block, centred in X/Y, sitting on z = 0
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_h]);

    // 96 through-holes
    for (i = [0 : hole_cols - 1])
        for (j = [0 : hole_rows - 1])
            translate([(i - (hole_cols - 1)/2) * hole_pitch + hole_x0,
                       (j - (hole_rows - 1)/2) * hole_pitch + hole_y0,
                       -eps])
                cylinder(d = hole_d, h = block_h + 2*eps);

    // 45-degree chamfer at A1 corner (left, back, top)
    translate([-block_len/2, -block_wid/2, block_h - eps])
        rotate([90, 0, 0])
            cylinder(r = chamfer_r, h = chamfer_leg + eps);
}
