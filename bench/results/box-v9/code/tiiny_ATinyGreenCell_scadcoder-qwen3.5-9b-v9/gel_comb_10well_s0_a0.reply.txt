```openscad
// 10-well comb for agarose gel casting, printed flat
// Units: mm

// ---- Spine ----
spine_len   = 70;    // spine length (X)
spine_wid   = 12;    // spine width (Y)
plate_thk   = 1.5;   // plate thickness (Z)

// ---- Teeth ----
tooth_count = 10;    // number of teeth
tooth_w     = 5.0;   // tooth width (X)
tooth_len   = 12;    // tooth length (Y), from top edge down
gap         = 1.5;   // gap between neighbouring teeth (X)

// ---- Derived ----
pitch   = tooth_w + gap;                 // centre-to-centre spacing
x_start = -(tooth_count - 1) * pitch / 2; // first tooth centre X
overrun = 0.5;                           // extra tooth length past the top edge

$fn = 64;

difference() {
    // Base plate with spine
    translate([-spine_len / 2, -spine_wid / 2, 0])
        cube([spine_len, spine_wid, plate_thk]);

    // Teeth hanging in -Y, centred on the spine
    for (i = [0 : tooth_count - 1]) {
        translate([x_start + i * pitch,
                   spine_wid / 2 - tooth_len,
                   plate_thk])
            cube([tooth_w, tooth_len + overrun, overrun]);
    }
}
```
