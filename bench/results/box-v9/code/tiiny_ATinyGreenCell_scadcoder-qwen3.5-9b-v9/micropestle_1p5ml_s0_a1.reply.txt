// Tissue-grinding pestle for a 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the XY plane, z = 0 at the bottom.

// ---------------- Parameters ----------------
handle_d      = 8;     // handle diameter (mm)
handle_h      = 45;    // handle height (mm), z = 0 to 45

cone_top_d    = 8;     // cone diameter at the top (mm), z = 45
cone_bottom_d = 3;     // cone diameter at the bottom (mm), z = 62
cone_h        = 17;    // cone height (mm), z = 45 to 62

tip_d         = 3;     // rounded tip diameter (mm), sphere centre at z = 62

groove_count  = 4;     // number of circumferential grip grooves
groove_w      = 2;     // groove width along Z (mm)
groove_depth  = 1;     // groove depth into handle (mm)
groove_z      = [10, 15, 20, 25]; // groove centre heights (mm)

eps           = 0.5;   // cutter overshoot past faces (mm)
$fn           = 96;   // facet count for curved shapes

// ---------------- Model ----------------
difference() {
    union() {
        // (1) Cylindrical handle, z = 0 to 45
        cylinder(d = handle_d, h = handle_h);

        // (2) Tapering cone, z = 45 to 62
        translate([0, 0, handle_h])
            cylinder(d1 = cone_top_d, d2 = cone_bottom_d, h = cone_h);

        // (3) Rounded tip: 3 mm sphere centred at z = 62
        translate([0, 0, cone_bottom_d])
            sphere(d = tip_d);
    }

    // Grip grooves: circumferential cuts into the handle
    for (g = [0 : groove_count - 1]) {
        rotate([0, 0, g * 360 / groove_count])
            translate([0, -groove_w / 2, groove_z[g]])
                cube([handle_d / 2 + eps, groove_w, groove_depth]);
    }
}
