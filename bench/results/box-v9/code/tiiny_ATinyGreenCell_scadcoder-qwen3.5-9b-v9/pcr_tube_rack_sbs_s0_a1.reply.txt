// 96-well PCR tube rack (0.2 mL, ANSI/SLAS footprint)
// Print orientation: base on XY plane, builds upward in +Z

// ---------- Parameters ----------
plate_length   = 127.76;   // block length along X (mm)
plate_width    = 85.48;    // block width along Y (mm)
plate_height   = 20;        // block height along Z (mm)

hole_count     = 96;        // number of tube holes
hole_d         = 6.2;        // hole diameter (mm)
hole_pitch     = 9.0;        // centre-to-centre spacing (mm)
hole_cols      = 12;         // holes per row (along X)
hole_rows      = 8;          // number of rows (along Y)

corner_chamfer = 5;          // 45-degree chamfer leg length at A1 (mm)
cut_extra      = 0.5;        // overshoot of cutters past faces (mm)

first_x        = 14.38;      // distance of A1 from left edge (mm)
first_y        = 11.24;      // distance of A1 from back edge (mm)

$fn = 64;                   // circle resolution

// ---------- Model ----------
difference() {
    // Main block
    translate([-plate_length/2, -plate_width/2, 0])
        cube([plate_length, plate_width, plate_height]);

    // Tube holes, centred on the block
    for (i = [0 : hole_count - 1]) {
        row = floor(i / hole_cols);
        col = i % hole_cols;
        translate([-plate_length/2 + first_x + col * hole_pitch,
                   -plate_width/2 + first_y + row * hole_pitch,
                   -cut_extra])
            cylinder(d = hole_d, h = plate_height + 2 * cut_extra);
    }

    // Orientation mark: 45-degree chamfer at the A1 corner
    translate([plate_length/2 - first_x, -first_y, 0])
        cube([corner_chamfer, corner_chamfer, plate_height]);
}
