// Microscope slide drying rack
// Print orientation: base block on XY plane, slots open upward (+Z)

// ---------- Parameters ----------
slide_len   = 75;    // slide length along X (mm)
slide_wid   = 25;    // slide width along Y (mm)
slide_thk   = 1;     // slide thickness along Z (mm)

base_len    = 90;    // block length along X (mm)
base_wid    = 45;    // block width along Y (mm)
base_ht     = 20;    // block height along Z (mm)

slot_count  = 10;    // number of slide slots
slot_len    = 77;    // slot length along X (mm)
slot_wid    = 1.6;   // slot width along Y (mm)
slot_depth  = 10;    // slot depth from top face (mm)
slot_pitch  = 3.5;   // centre-to-centre slot spacing in Y (mm)

eps         = 0.5;   // cutter overshoot past top face (mm)

$fn = 64;

// ---------- Model ----------
difference() {
    // Solid base block, centred in X/Y, resting on z = 0
    translate([-base_len/2, -base_wid/2, 0])
        cube([base_len, base_wid, base_ht]);

    // Parallel slots running along X, centred on the block
    for (i = [0 : slot_count - 1]) {
        y = (i - (slot_count - 1)/2) * slot_pitch;
        translate([-slot_len/2, y - slot_wid/2, base_ht - slot_depth])
            cube([slot_len, slot_wid, slot_depth + eps]);
    }
}
