```openscad
// Lab funnel - printed upright, open top and bottom
// Units: mm

// ---- Parameters ----
spout_od      = 10;    // spout outer diameter
spout_len     = 20;    // spout length (z = 0 to 20)
cone_od_top   = 60;    // cone rim outer diameter (z = 50)
cone_od_bot   = 10;    // cone base outer diameter (z = 20)
wall          = 1.6;   // wall thickness everywhere
eps           = 0.5;   // cutter overshoot past faces

$fn = 96;

// ---- Model ----
difference() {
    union() {
        // Spout (tube)
        translate([0, 0, 0])
            cylinder(d = spout_od, h = spout_len);

        // Cone (z = 20 to 50)
        translate([0, 0, cone_od_bot / 2])
            cylinder(d1 = spout_od, d2 = cone_od_top, h = spout_len);
    }

    // Through-bore: spout bore (6.8 mm) extended past both ends
    translate([0, 0, -eps])
        cylinder(d = spout_od - 2 * wall, h = spout_len + 2 * eps);
}
```
