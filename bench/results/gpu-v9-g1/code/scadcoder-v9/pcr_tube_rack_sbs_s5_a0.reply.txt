// 96-well PCR tube rack (0.2 mL) with ANSI/SLAS footprint
// Print orientation: base on XY plane, holes open upward (+Z)

// ---------------- Parameters ----------------
block_len    = 127.76;   // block length along X (mm)
block_wid    = 85.48;    // block width along Y (mm)
block_h      = 20;       // block height along Z (mm)

hole_d       = 6.2;      // through-hole diameter (mm)
hole_pitch   = 9.0;      // centre-to-centre spacing in X and Y (mm)
hole_count_x = 12;       // number of columns along X
hole_count_y = 8;        // number of rows along Y

corner_x     = 14.38;    // X of first hole (A1), from left edge (mm)
corner_y     = 11.24;    // Y of first hole (A1), from back edge (mm)

eps          = 0.5;      // overshoot for clean boolean cuts (mm)
chamfer      = 5;        // 45-degree chamfer leg length on A1 corner (mm)

$fn = 64;                // circle resolution

// ---------------- Model ----------------
difference() {
    // Main block, centred in X/Y, sitting on z = 0
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_h]);

    // 96 vertical through-holes
    for (i = [0 : hole_count_x - 1])
        for (j = [0 : hole_count_y - 1])
            translate([corner_x + i * hole_pitch - block_len/2,
                       corner_y + j * hole_pitch - block_wid/2,
                       -eps])
                cylinder(d = hole_d, h = block_h + 2*eps);

    // 45-degree chamfer with 5 mm legs on the A1 corner (vertical edge)
    // Extended 0.5 mm above the corner so the cut passes through the edge
    translate([corner_x - block_len/2 - eps, corner_y - block_wid/2 - eps,
               -eps])
        cube([eps + chamfer, eps + chamfer, block_h + eps]);

    // Cut along the 45-degree diagonal (in X+Y) through the extended block.
    // The diagonal runs from (low X, high Y) to (high X, low Y),
    // so the lower-left corner at A1 stays at full height.
    translate([corner_x - block_len/2 - eps, corner_y - block_wid/2 - eps,
               -eps])
        rotate([90, 0, 0])
            linear_extrude(height = eps + chamfer)
                polygon(points = [
                    [0, eps + chamfer],
                    [eps + chamfer, 0]
                ]);
}
