```openscad
// 10-well comb for casting agarose gels
// Print orientation: flat on bed, teeth pointing down (-Y)

// ---------- Parameters ----------
plate_t      = 1.5;   // plate thickness (Z)
spine_len    = 70;    // spine length along X
spine_w      = 12;    // spine width along Y
tooth_count  = 10;    // number of teeth
tooth_w      = 5.0;   // tooth width along X
tooth_len    = 12;    // tooth length along Y
tooth_h      = 12;    // tooth height from top edge of spine (Z)
gap          = 1.5;   // gap between neighbouring teeth (X)
eps          = 0.5;   // cutter overshoot past faces

$fn = 64;

// ---------- Derived ----------
pitch   = tooth_w + gap;                 // centre-to-centre spacing
x_start = -(tooth_count - 1) * pitch / 2; // first tooth centre (row centred)

// ---------- Model ----------
difference() {
    union() {
        // Spine (back bar)
        translate([-spine_len / 2, 0, 0])
            cube([spine_len, spine_w, plate_t]);

        // Teeth hanging in -Y direction
        for (i = [0 : tooth_count - 1])
            translate([x_start + i * pitch - tooth_w / 2,
                       -tooth_len,
                       plate_t])
                cube([tooth_w, tooth_len, tooth_h]);
    }

    // Clean up overlapping tooth / spine faces (coplanar removal)
    for (i = [0 : tooth_count - 1])
        translate([x_start + i * pitch - tooth_w / 2 - eps,
                   -tooth_len + eps,
                   plate_t + eps])
            cube([tooth_w + 2 * eps, tooth_len - 2 * eps, tooth_h + eps]);
}
```
