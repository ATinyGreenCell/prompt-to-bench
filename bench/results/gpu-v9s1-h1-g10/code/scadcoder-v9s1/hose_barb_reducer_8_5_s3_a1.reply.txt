```openscad
// Hose-barb reducer: 8 mm ID -> 5 mm ID
// Print orientation: standing upright along Z

$fn = 96;

// ---- Dimensions (mm) ----
id_8      = 8;     // 8 mm tubing inner diameter
id_5      = 5;     // 5 mm tubing inner diameter
bore      = 3;     // through-bore diameter
barb_h_8  = 10;    // height of each 8 mm barb
od_8_bot  = 7.5;   // outer diameter of 8 mm barb at bottom
od_8_top  = 9.5;   // outer diameter of 8 mm barb at top
collar_d  = 12;    // diameter of cylindrical collar
barb_h_5  = 8;     // height of each 5 mm barb
od_5_bot  = 6.5;   // outer diameter of 5 mm barb at bottom
od_5_top  = 5.0;   // outer diameter of 5 mm barb at top
z_20      = 20;    // start of collar
z_25      = 25;    // end of collar
z_41      = 41;    // end of part

eps = 0.5;         // cutter overshoot

// ---- Model ----
difference() {
    union() {
        // 8 mm barbs (z = 0 to 20)
        for (s = [-1, 1])
            translate([0, 0, s > 0 ? 10 : 0])
                linear_extrude(height = 10)
                    polygon([
                        [od_8_bot / 2, 0],
                        [od_8_top / 2, 10],
                        [od_8_bot / 2, 20]
                    ]);
        // collar (z = 20 to 25)
        translate([0, 0, z_20])
            cylinder(d = collar_d, h = 5);
        // 5 mm barbs (z = 25 to 41)
        for (s = [-1, 1])
            translate([0, 0, s > 0 ? 33 : 25])
                linear_extrude(height = 8)
                    polygon([
                        [od_5_bot / 2, 0],
                        [od_5_top / 2, 8],
                        [od_5_bot / 2, 16]
                    ]);
    }
    // through-bore
    translate([0, 0, -eps])
        cylinder(d = bore, h = z_41 + 2 * eps);
}
```
