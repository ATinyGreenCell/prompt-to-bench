```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor (e.g. syringe pump)
// Print orientation: base plate on bed, motor plate building upward (+Z)

// ---------------- Parameters ----------------
base_x0      = -25;   // base plate, X start
base_x1      = 25;    // base plate, X end
base_y0      = 0;     // base plate, Y start
base_y1      = 45;    // base plate, Y end
base_z0      = 0;     // base plate, Z start
base_z1      = 5;     // base plate, Z end

hole_x0      = -17;   // M4 hole 1, X
hole_y0      = 30;    // M4 hole 1, Y
hole_x1      = 17;    // M4 hole 2, X
hole_y1      = 30;    // M4 hole 2, Y
hole_d       = 4.5;   // M4 hole diameter

motor_x0     = -25;   // motor plate, X start
motor_x1     = 25;    // motor plate, X end
motor_y0     = 0;     // motor plate, Y start
motor_y1     = 5;     // motor plate, Y end
motor_z0     = 0;     // motor plate, Z start
motor_z1     = 50;    // motor plate, Z end

boss_d       = 23;    // motor boss hole diameter
boss_x       = 0;     // boss hole centre, X
boss_z       = 28;    // boss hole centre, Z
screw_d      = 3.4;   // M4 screw hole diameter
screw_pitch  = 31;    // screw hole pattern size (square)

gusset_t     = 5;     // gusset thickness (X)
gusset_leg   = 20;    // gusset leg length (Y and Z)
gusset_x0    = -25;   // left gusset, X start
gusset_x1    = -20;   // left gusset, X end

eps          = 0.5;   // cutter overshoot past faces

$fn = 96;

// ---------------- Model ----------------
union() {
    // Base plate with M4 through-holes
    difference() {
        translate([base_x0, base_y0, base_z0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);

        for (h = [0, 1]) {
            translate([hole_x0, hole_y0, base_z0 + h * (base_z1 - base_z0)])
                cylinder(d = hole_d, h = base_z1 - base_z0 + eps);
            translate([hole_x1, hole_y1, base_z0 + h * (base_z1 - base_z0)])
                cylinder(d = hole_d, h = base_z1 - base_z0 + eps);
        }
    }

    // Vertical motor plate with boss hole and M4 screw holes
    difference() {
        translate([motor_x0, motor_y0, motor_z0])
            cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1 - motor_z0]);

        // Boss hole along Y
        translate([boss_x, motor_y0 - eps, boss_z])
            cylinder(d = boss_d, h = motor_y1 - motor_y0 + 2 * eps);

        // Screw holes on 31 mm square pattern
        for (s = [0, 1]) {
            for (dx = [-1, 1]) {
                for (dy = [-1, 1]) {
                    translate([boss_x + dx * (screw_pitch / 2),
                               motor_y0 - eps + dy * (screw_pitch / 2),
                               motor_z0 - eps])
                        cylinder(d = screw_d, h = motor_z1 - motor_z0 + 2 * eps);
                }
            }
        }
    }

    // Left triangular gusset (flush with left edge)
    translate([gusset_x0, gusset_y0, gusset_z0])
        linear_extrude(height = gusset_t)
            polygon(points = [
                [0, 0],
                [gusset_leg, 0],
                [0, gusset_leg]
            ]);

    // Right triangular gusset (flush with right edge)
    translate([gusset_x1, gusset_y0, gusset_z0])
        linear_extrude(height = gusset_t)
            polygon(points = [
                [0, 0],
                [gusset_leg, 0],
                [0, gusset_leg]
            ]);
}
```

> **Note:** The gusset coordinates
