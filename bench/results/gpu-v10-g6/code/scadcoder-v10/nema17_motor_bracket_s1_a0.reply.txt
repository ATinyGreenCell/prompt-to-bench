```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor (e.g. syringe pump)
// Print orientation: base plate on the bed, motor plate building upward.

// ---------------- Parameters ----------------
base_x0      = -25;   // base plate, x start
base_x1      = 25;    // base plate, x end
base_y0      = 0;     // base plate, y start
base_y1      = 45;    // base plate, y end
base_z0      = 0;     // base plate, z start
base_z1      = 5;     // base plate, z end

hole_x       = -17;   // M4 through-hole, x
hole_y       = 30;    // M4 through-hole, y
hole_d       = 4.5;   // M4 through-hole diameter

hole_x2      = 17;    // M4 through-hole, x (second)
hole_y2      = 30;    // M4 through-hole, y (second)

motor_x0     = -25;   // motor plate, x start
motor_x1     = 25;    // motor plate, x end
motor_y0     = 0;     // motor plate, y start
motor_y1     = 5;     // motor plate, y end
motor_z0     = 0;     // motor plate, z start
motor_z1     = 50;    // motor plate, z end

boss_d       = 23;    // motor centring boss hole diameter
boss_x       = 0;     // boss hole centre, x
boss_z       = 28;    // boss hole centre, z

screw_d      = 3.4;   // M4 screw hole diameter
screw_n      = 4;     // number of screw holes
screw_pitch  = 7;     // screw hole pattern pitch (square)

gusset_t     = 5;     // gusset thickness (plate + gusset)
gusset_x0    = -25;   // gusset, x start (left edge)
gusset_x1    = -20;   // gusset, x end (left edge)
gusset_y0    = 5;     // gusset, Y leg length (right-angle corner)
gusset_z0    = 5;     // gusset, Z leg length (right-angle corner)

eps          = 0.5;   // cutter overshoot past faces

$fn = 64;

// ---------------- Model ----------------
union() {
    // Base plate with two M4 through-holes
    difference() {
        translate([base_x0, base_y0, base_z0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);

        for (h = [hole, hole2]) {
            translate([h.x, h.y, -eps])
                cylinder(d = h.d, h = base_z1 - base_z0 + 2 * eps);
        }
    }

    // Vertical motor plate with boss hole and four M3 screw holes
    difference() {
        translate([motor_x0, motor_y0, motor_z0])
            cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1 - motor_z0]);

        // Boss hole along Y, centred at x = 0, z = 28
        translate([boss_x, -eps, boss_z])
            cylinder(d = boss_d, h = motor_y1 - motor_y0 + 2 * eps);

        // Four screw holes on a 31 mm square pattern
        for (i = [0 : screw_n - 1]) {
            rotate([0, 0, i < screw_n - 1 ? 90 : 0])
                translate([0, screw_pitch * i, 0])
                    cylinder(d = screw_d, h = motor_z1 - motor_z0 + eps);
        }
    }

    // Two triangular gussets, flush with the left and right edges
    for (gx = [gusset_x0, gusset_x1]) {
        translate([gx, gusset_y0, gusset_z0])
            rotate([90, 0, 0])
                linear_extrude(height = gusset_t)
                    polygon(points = [
                        [0, 0],
                        [gusset_x1 - gx, 0],
                        [0, gusset_z1 - gusset_z0]
                    ]);
    }
}
```
