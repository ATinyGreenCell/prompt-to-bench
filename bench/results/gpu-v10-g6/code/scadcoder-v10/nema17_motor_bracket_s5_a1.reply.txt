```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor (syringe pump)
// Print orientation: base plate on bed, motor plate building up (+Z)

$fn = 96;

// ---- Base plate ----
base_x0 = -25; base_x1 = 25;   // X span of base plate
base_y0 = 0;   base_y1 = 45;  // Y span of base plate
base_z0 = 0;   base_z1 = 5;   // base plate thickness

// ---- Motor plate ----
motor_x0 = -25; motor_x1 = 25; // X span of motor plate
motor_y0 = 0;   motor_y1 = 5;  // Y depth of motor plate
motor_z0 = 0;   motor_z1 = 50; // motor plate height

// ---- Through-holes ----
hole_d     = 4.5;              // M4 hole diameter
hole_x     = [-17, 17];        // hole centre X positions
hole_y     = 30;               // hole centre Y position

// ---- Boss hole (motor centring) ----
boss_d     = 23;               // boss hole diameter
boss_x     = 0;                // boss hole centre X
boss_z     = 28;               // boss hole centre height

// ---- Screw holes for boss ----
screw_d    = 3.4;              // M3 screw hole diameter
screw_n    = 4;                // number of screw holes
screw_r    = 31 / 2;           // square pattern half-size (15.5 mm)

// ---- Gussets ----
gusset_t   = 5;                // gusset thickness
gusset_a   = 20;               // leg length along +Y
gusset_b   = 20;               // leg length along +Z
gusset_x   = [-25, 20];        // gusset X start (left, right)

// ---- Misc ----
eps        = 0.5;              // cutter overshoot past faces

union() {
    // Base plate with M4 holes
    difference() {
        translate([base_x0, base_y0, base_z0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1 - base_z0]);
        for (x = hole_x)
            translate([x, hole_y, base_z0 - eps])
                cylinder(d = hole_d, h = base_z1 - base_z0 + 2 * eps);
    }

    // Motor plate with boss hole and screw holes
    difference() {
        translate([motor_x0, motor_y0, motor_z0])
            cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1 - motor_z0]);
        // Boss hole along Y
        translate([boss_x, motor_y0 - eps, boss_z])
            cylinder(d = boss_d, h = motor_y1 - motor_y0 + 2 * eps);
        // Screw holes on square pattern
        for (i = [0 : screw_n - 1]) {
            a = screw_r * (i < 2 ? 1 : -1);
            b = screw_r * (i % 2 ? 1 : -1);
            translate([boss_x + b, motor_y0 - eps, motor_z0 - eps])
                cylinder(d = screw_d, h = motor_z1 - motor_z0 + 2 * eps);
        }
    }

    // Triangular gussets flush with plate edges
    for (gx = gusset_x)
        translate([gx, 0, 0])
            hull() {
                translate([0, gusset_a, 0]) cube([gusset_t, 1, 1]);     // bottom right
                translate([0, gusset_a, gusset_b]) cube([gusset_t, 1, 1]); // top right
                translate([gusset_t, gusset_a, gusset_b]) cube([1, 1, 1]);   // top left
            }
}
```
