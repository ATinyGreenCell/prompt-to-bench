```openscad
// Snap-on Clip for Lab Stand Rod and Tubing
// Orientation: Flat on XY plane, building up in +Z.
// Units: Millimetres.

// --- Dimensions ---
// Rod Stand Dimensions
$rod_diameter = 12.7;
$rod_wall_thickness = 3;
$rod_outer_diameter = $rod_diameter + 2 * $rod_wall_thickness; // 18.7
$rod_ring_height = 10;

// Tubing Dimensions
$tubing_inner_diameter = 6;
$tubing_wall_thickness = 2;
$tubing_outer_diameter = $tubing_inner_diameter + 2 * $tubing_wall_thickness; // 10
$tubing_ring_height = 10;

// Bridge Dimensions
$bridge_width_x = 4; // From x=8 to x=12
$bridge_width_y = 6;
$bridge_height = 10;

// Cutout Dimensions (Opening widths)
$rod_cutout_width = 10;
$rod_cutout_y_limit = $rod_cutout_width / 2; // 5

$tubing_cutout_width = 4.5;
$tubing_cutout_y_limit = $tubing_cutout_width / 2; // 2.25

// Positioning
$rod_center_x = 0;
$rod_center_y = 0;

$tubing_center_x = 15;
$tubing_center_y = 0;

// --- Geometry Generation ---

// 1. Rod Ring with Cutout
// Outer ring
rod_outer = translate([$rod_center_x, $rod_center_y, 0])
    difference() {
        cylinder(h = $rod_ring_height, d = $rod_outer_diameter, center = true);
        // Cutout: x < 0 and -5 < y < 5
        // We extend the cutout slightly past the face to ensure clean removal
        translate([-($rod_outer_diameter/2) - 0.1, 0, 0])
            cube([
                $rod_outer_diameter + 0.2, 
                $rod_cutout_width + 0.2, 
                $rod_ring_height + 0.1
            ]);
    };

// 2. Tubing Ring with Cutout
// Outer ring
tubing_outer = translate([$tubing_center_x, $tubing_center_y, 0])
    difference() {
        cylinder(h = $tubing_ring_height, d = $tubing_outer_diameter, center = true);
        // Cutout: x > 15 and -2.25 < y < 2.25
        // Extend cutout past the face
        translate([($tubing_outer_diameter/2) + 0.1, 0, 0])
            cube([
                $tubing_outer_diameter + 0.2, 
                $tubing_cutout_width + 0.2, 
                $tubing_ring_height + 0.1
            ]);
    };

// 3. Solid Bridge
// Joins x=8 to x=12, width 6mm in Y
bridge = translate([10, 0, 0]) // Center of bridge at x=10
    cube([$bridge_width_x, $bridge_width_y, $bridge_height]);

// --- Final Assembly ---
$fn = 64;

difference() {
    union() {
        rod_outer;
        tubing_outer;
        bridge;
    }
}
```
