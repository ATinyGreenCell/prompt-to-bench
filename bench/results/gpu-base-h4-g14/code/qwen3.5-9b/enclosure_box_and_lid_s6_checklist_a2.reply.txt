```openscad
// Two-part enclosure for electronics board.
// Part 1: Box (left). Part 2: Lid (right, printed upside down).
// Both parts are centred on the XY plane at Z=0.
// A small Z-gap is added to ensure they print together but remain separable.

// --- Dimensions ---
// Box outer dimensions (X, Y, Z)
box_X = 70;
box_Y = 50;
box_Z = 30;

// Wall and floor thickness
wall_thick = 2;
floor_thick = 2;

// Gap between parts (X direction)
gap = 10;

// Lid dimensions
lid_plate_thick = 2;
lip_wall_thick = 1.5;
lip_height = 4;

// Lid inner size (to fit box with 0.2mm clearance)
// Box Inner = 70 - 4 = 66. Lid Inner = 66 - 0.4 = 65.6.
lip_inner_X = 65.6;
lip_inner_Y = 45.6;

// Lid outer size (calculated from inner + 2*wall)
lip_outer_X = lip_inner_X + 2 * lip_wall_thick;
lip_outer_Y = lip_inner_Y + 2 * lip_wall_thick;

// --- Part 1: Box ---
// Position: Centred at X = -gap/2, Y = 0, Z = 0.
// The box sits on Z=0.

module box() {
    translate([-box_X / 2, -box_Y / 2, 0]) {
        // Floor
        cube([box_X, box_Y, floor_thick]);
        
        // Walls (difference to cut out the inside)
        // We build the solid block and subtract the inner void.
        // Solid block:
        translate([0, 0, floor_thick]) cube([box_X, box_Y, box_Z - floor_thick]);
        
        // Inner void (cut out)
        // Void starts at floor_thick, goes up to box_Z.
        // Width: box_inner_X, Height: box_inner_Y.
        translate([0, 0, floor_thick]) cube([box_X - 2 * wall_thick, box_Y - 2 * wall_thick, box_Z - floor_thick]);
    }
}

// --- Part 2: Lid ---
// Position: Centred at X = gap/2, Y = 0, Z = 0.
// Printed upside down: Plate on bed, lip on top.
// Lid geometry:
// Plate: 70 x 50 x 2mm.
// Lip: Rectangular wall on top.
// Lip thickness: 1.5mm.
// Lip height: 4mm.
// Lip inner: 65.6 x 45.6 (to fit box).
// Lip outer: 68.6 x 48.6 (calculated).

module lid() {
    translate([-box_X / 2 + gap, -box_Y / 2, 0]) {
        // Plate
        cube([box_X, box_Y, lid_plate_thick]);
        
        // Lip
        // Lip is a hollow box on top of the plate.
        // Base of lip is at z = lid_plate_thick.
        // Top of lip is at z = lid_plate_thick + lip_height.
        // Outer dims: lip_outer_X, lip_outer_Y.
        // Inner dims: lip_inner_X, lip_inner_Y.
        
        // Solid block for the lip
        translate([0, 0, lid_plate_thick]) cube([lip_outer_X, lip_outer_Y, lip_height]);
        
        // Inner void for the lip
        translate([0, 0, lid_plate_thick]) cube([lip_inner_X, lip_inner_Y, lip_height]);
    }
}

// --- Assembly ---
// Add a small Z-gap (0.1mm) between the box top and lid bottom to ensure they are distinct parts
// while printing together.
// Box top is at Z = box_Z.
// Lid bottom is at Z = 0 + lid_plate_thick = 2.
// We need to lift the lid up by 0.1mm relative to the box, or lower the box.
// Since the box is fixed at Z=0, we will translate the lid up by 0.1mm.
// Wait, the lid sits on the bed (Z=0). The box sits on the bed (Z=0).
// If we print them together, they are fused.
// To make them separable, we must introduce a gap in the geometry.
// We can translate the lid UP by 0.1mm so there is a 0.1mm air gap between the box top and lid bottom.
// This means the lid will be floating 0.1mm above the bed in the model, but the box is on the bed.
// This creates a 0.1mm gap between the two parts.

difference() {
    union() {
        box();
        translate([0, 0, 0.1]) lid(); // Lift lid by 0.1mm to create a gap
    }
}
```
