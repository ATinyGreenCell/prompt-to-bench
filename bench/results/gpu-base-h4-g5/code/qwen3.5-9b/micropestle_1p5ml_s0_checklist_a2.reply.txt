handle_z_start = 0;
handle_z_end = 45;
cone_z_start = 45;
cone_z_end = 62;
tip_d = 3;
tip_center_z = 62;

groove_width = 2;
groove_depth = 1;
groove_z_centers = [10, 15, 20, 25];

difference() {
