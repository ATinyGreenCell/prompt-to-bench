```openscad
// Pestle for 1.5 mL microcentrifuge tube - printed standing upright.
// Dimensions in mm.

// Handle
h_dia  = 8;
h_len  = 45;

// Cone
c_d1   = 8;
c_d2   = 3;
c_len  = 17; // 62 - 45

// Tip
tip_r  = 1.5; // 3 mm diameter
tip_z  = 62;  // Center of sphere

// Grooves
n_grooves = 4;
g_w      = 2; // Width in Z
g_d      = 1; // Depth
g_z_list = [10, 15, 20, 25];

$fn = 64;

// Main body construction
difference() {
    union() {
        // 1. Cylindrical handle
        translate([0, 0, 0]) cylinder(h = h_len, d = h_dia);
        
        // 2. Cone tapering from handle to tip
        translate([0, 0, h_len]) 
            cylinder(h = c_len, d1 = c_d1, d2 = c_d2);
        
        // 3. Rounded tip (sphere centered at z = 62)
        translate([0, 0, tip_z]) sphere(r = tip_r);
    }
    
    // Cut grooves into the handle
    // Grooves are centered at specific Z positions
    for (z_c = g_z_list) {
        translate([0, 0, z_c - g_w / 2]) {
            // Create a groove: a cylinder with a smaller diameter than the handle
            // The groove is 2mm wide in Z and 1mm deep
            cylinder(h = g_w, d = h_dia - 2 * g_d);
        }
    }
}
```
