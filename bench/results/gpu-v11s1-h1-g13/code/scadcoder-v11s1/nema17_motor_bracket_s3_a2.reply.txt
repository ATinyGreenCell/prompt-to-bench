```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor (e.g. syringe pump)
// Print orientation: base plate on bed, motor plate builds upward (+Z)

$fn = 64;

// ---- Base plate ----
base_x0 = -25;      // plate x start
base_x1 = 25;       // plate x end
base_y0 = 0;        // plate y start
base_y1 = 45;       // plate y end
base_z1 = 5;        // plate z end (thickness)

// ---- Motor plate ----
motor_x0 = -25;     // plate x start
motor_x1 = 25;      // plate x end
motor_y0 = 0;       // plate y start
motor_y1 = 5;       // plate y end (thickness)
motor_z1 = 50;      // plate z end (height)

// ---- Mount holes ----
hole_d4 = 4.5;      // M4 through-hole diameter
hole_x4 = [-17, 17];// M4 hole x positions
hole_y4 = 30;       // M4 hole y position

// ---- Boss hole ----
boss_d  = 23;       // motor centring boss hole diameter
boss_x  = 0;        // boss hole x centre
boss_z  = 28;       // boss hole z centre

// ---- Screw holes ----
screw_d = 3.4;      // M3 screw hole diameter
screw_n = 4;        // number of screw holes
screw_a = 31;       // square pattern side length

// ---- Gussets ----
gusset_t = 5;       // gusset thickness (X)
gusset_l = 20;      // gusset leg lengths (Y and Z)

// ---- Misc ----
eps = 0.5;          // cutter overshoot past faces

difference() {
    union() {
        // Base plate
        translate([base_x0, base_y0, 0])
            cube([base_x1 - base_x0, base_y1 - base_y0, base_z1]);

        // Motor plate
        translate([motor_x0, motor_y0, 0])
            cube([motor_x1 - motor_x0, motor_y1 - motor_y0, motor_z1]);

        // Gussets, flush with left and right edges
        for (sx = [-1, 1])
            translate([sx * (25 - gusset_t), 0, 0])
                rotate([90, 0, 0])
                    linear_extrude(height = gusset_t)
                        polygon([[gusset_l, gusset_l], [0, gusset_l], [gusset_l, 0]]);
    }

    // Base plate M4 through-holes
    for (x = hole_x4)
        translate([x, hole_y4, -eps])
            cylinder(d = hole_d4, h = base_z1 + 2 * eps);

    // Motor plate boss hole (along Y)
    translate([boss_x, 0, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_y1 + 2 * eps);

    // M3 screw holes on square pattern
    for (i = [0 : screw_n - 1]) {
        x = (i % 2 - 0.5) * screw_a;
        z = (i / 2 - 0.25) * screw_a;
        translate([x, motor_y0 - eps, z])
            cylinder(d = screw_d, h = motor_z1 + 2 * eps);
    }
}
```
