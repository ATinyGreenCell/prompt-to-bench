```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor (syringe pump)
// Print orientation: base plate on bed, motor plate builds upward (+Z)

$fn = 64;

// ---- Base plate ----
base_x0 = -25;      // plate x start
base_x1 = 25;       // plate x end
base_y1 = 45;       // plate y end (y0 = 0)
base_z1 = 5;        // plate thickness (z0 = 0)
hole_d  = 4.5;      // M4 through-hole diameter
hole_x  = [-17, 17];// through-hole x positions
hole_y  = 30;       // through-hole y position

// ---- Motor plate ----
motor_x1 = 25;      // plate x end (x0 = -25)
motor_y1 = 5;       // plate thickness (y0 = 0)
motor_z1 = 50;      // plate height (z0 = 0)

// ---- Boss hole (motor centring) ----
boss_d   = 23;      // hole diameter
boss_x   = 0;       // hole centre x
boss_z   = 28;      // hole centre z
eps      = 0.5;     // cutter overshoot past faces

// ---- Screw holes ----
screw_d  = 3.4;     // M3 hole diameter
screw_n  = 4;       // number of holes
screw_p  = 31;      // square pattern side length
screw_x0 = -screw_p/2; // pattern centre x
screw_y0 = -screw_p/2; // pattern centre y

// ---- Gussets ----
gusset_t     = 5;   // gusset thickness (X)
gusset_leg   = 20;  // right triangle leg length (Y and Z)
gusset_x0    = -25; // left gusset x start
gusset_x2    = -20; // right gusset x start / left end
gusset_x3    = 25;  // right gusset x end

difference() {
    union() {
        // Base plate
        translate([base_x0, 0, 0])
            cube([base_x1 - base_x0, base_y1, base_z1]);

        // Motor plate
        translate([motor_x0, 0, 0])
            cube([motor_x1 - motor_x0, motor_y1, motor_z1]);

        // Gussets, flush with left and right edges
        for (sx = [-1, 1])
            translate([sx * gusset_x0, 0, 0])
                rotate([90, 0, 0])
                    linear_extrude(height = gusset_t)
                        polygon([
                            [gusset_leg, gusset_leg],
                            [0, gusset_leg],
                            [0, 0]
                        ]);
    }

    // Base plate through-holes
    for (x = hole_x)
        translate([x, hole_y, -eps])
            cylinder(d = hole_d, h = base_z1 + 2 * eps);

    // Boss hole (along Y)
    translate([boss_x, 0, boss_z])
        rotate([0, 90, 0])
            cylinder(d = boss_d, h = motor_y1 + 2 * eps, center = true);

    // Screw holes on square pattern
    for (i = [0 : screw_n - 1]) {
        x = screw_x0 + (i % 2) * screw_p;
        y = screw_y0 + (i / 2) * screw_p;
        translate([x, y, 0])
            cylinder(d = screw_d, h = motor_z1 + 2 * eps);
    }
}
```
