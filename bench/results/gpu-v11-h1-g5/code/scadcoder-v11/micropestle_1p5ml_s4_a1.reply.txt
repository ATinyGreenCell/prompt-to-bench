// Tissue pestle for 1.5 mL microcentrifuge tube
// Print orientation: standing upright on the bed (z = 0)

$fn = 96;

// ---- Dimensions (mm) ----
handle_d      = 8;    // handle diameter
handle_h      = 45;   // handle height (z = 0 to 45)
cone_d_top    = 8;    // cone diameter at top (z = 45)
cone_d_bot    = 3;    // cone diameter at bottom (z = 62)
tip_d         = 3;    // sphere diameter
tip_z         = 62;   // sphere centre height
groove_count  = 4;    // circumferential grooves
groove_w      = 2;    // groove width (Z)
groove_depth  = 1;    // groove depth into handle
groove_z      = [10, 15, 20, 25]; // groove centre heights

// ---- Model ----
difference() {
    union() {
        // Handle
        cylinder(d = handle_d, h = handle_h);
        // Cone (slightly overlapping handle for a clean union)
        translate([0, 0, handle_h - 0.01])
            cylinder(d1 = cone_d_top, d2 = cone_d_bot, h = tip_z - handle_h + 0.01);
        // Rounded tip
        translate([0, 0, tip_z])
            sphere(d = tip_d);
    }
    // Grip grooves
    for (z = groove_z)
        translate([0, 0, z - groove_w / 2])
            cylinder(d = handle_d + 2, h = groove_w, depth = groove_depth);
}
