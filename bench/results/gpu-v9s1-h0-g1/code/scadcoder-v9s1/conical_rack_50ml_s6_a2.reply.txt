// Rack for six 50 mL conical (Falcon-type) tubes
// Print orientation: plate on bed, walls rising to z = 70
// After printing: flip over, stand on walls, tubes through holes

// ---- Dimensions (mm) ----
plate_x      = 130;   // plate length (X)
plate_y      = 90;    // plate width (Y)
plate_t      = 4;     // plate thickness (Z)

hole_d       = 30.5;  // tube hole diameter
hole_nx      = 3;     // holes along X
hole_ny      = 2;     // holes along Y
hole_pitch   = 40;    // centre-to-centre spacing (X and Y)

wall_len     = 130;   // wall length along X
wall_t       = 4;     // wall thickness along Y
wall_h       = 70;    // wall height above plate

eps          = 0.5;   // cutter overshoot

$fn = 96;

wall_top_z = plate_t + wall_h;

difference() {
    union() {
        // Plate
        translate([-plate_x/2, -plate_y/2, 0])
            cube([plate_x, plate_y, plate_t]);

        // Walls (overlap into plate for a clean union)
        for (s = [-1, 1])
            translate([0, s*(plate_y/2 - wall_t), -eps])
                cube([wall_len, wall_t, wall_top_z + 2*eps]);
    }

    // Holes
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([(i - (hole_nx - 1)/2) * hole_pitch,
                       (j - (hole_ny - 1)/2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2*eps);
}
