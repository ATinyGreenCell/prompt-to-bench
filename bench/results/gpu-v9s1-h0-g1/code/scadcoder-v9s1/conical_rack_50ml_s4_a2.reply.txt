```openscad
// Six-way conical tube rack (Falcon-style tubes)
// Print orientation: flat plate on the bed, two end walls rising to z = 70.
// After printing, flip the part over. The rack then stands on the walls
// and the tubes pass through the holes.

// ---------------- Parameters ----------------
plate_x      = 130;   // plate length along X (mm)
plate_y      = 90;    // plate width along Y (mm)
plate_t      = 4;     // plate thickness (mm), z = 0..4

hole_d       = 30.5;  // tube hole diameter (mm)
hole_nx      = 3;     // holes per row along X
hole_ny      = 2;     // rows along Y
hole_pitch   = 40;    // centre-to-centre spacing, X and Y (mm)

wall_h       = 70;    // end wall height above the plate (mm)

eps          = 0.5;   // cutter overshoot past faces (mm)
$fn          = 96;    // curve resolution

// ---------------- Model ----------------
difference() {
    // Base plate
    translate([-plate_x/2, -plate_y/2, 0])
        cube([plate_x, plate_y, plate_t]);

    // Two solid walls along the long edges, flush with the plate
    for (s = [-1, 1])
        translate([s > 0 ? plate_x/2 - 1 : -plate_x/2,
                   s * (plate_y/2 - 1), 0])
            cube([plate_x, plate_y/2 - 1, wall_h]);

    // Tube through-holes in a 2 x 3 grid, centred on the plate
    for (i = [0 : hole_nx - 1])
        for (j = [0 : hole_ny - 1])
            translate([(i - (hole_nx - 1)/2) * hole_pitch,
                       (j - (hole_ny - 1)/2) * hole_pitch,
                       -eps])
                cylinder(d = hole_d, h = plate_t + 2*eps);
}
```
