```openscad
// Adapter sleeve - a collar with a sleeve body that lines a 1.5 mL rack hole to hold a 0.2 mL
// PCR tube, with a through bore. NOVEL general part (a generic tube adapter).
//

top_d     = 13;
body_d    = 10.8;
collar_t  = 1.5;
body_h    = 21.5 - 1.5;
bore_d    = 6.2;

$fn = 72;

collar_d = top_d - 0.5;   // 0.25 mm clearance so a tube end does not bind at the top

difference() {
    union() {
        cylinder(h = collar_t, d = collar_d);                 // collar on the bed
        cylinder(h = body_h, d = body_d);                     // sleeve body above
    }
    translate([0, 0, -1]) cylinder(h = collar_t + body_h + 2, d = bore_d);   // through bore
}
```
