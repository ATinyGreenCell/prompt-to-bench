// Replacement knob for a hotplate stirrer - a cylinder with a D-shaped blind bore and a
// pointer groove, and with a ring of grip grooves around the outside. NOVEL general part.
//

bottom   = 30;
height   = 15;
shaft_d  = 6.2;
flat     = 4.7;
grooves  = 18;
pointer  = 1.5;

$fn = 64;

difference() {
    cylinder(h = height, d = bottom);
    translate([0, 0, -1]) intersection() {
        cylinder(h = height + 2, d = shaft_d);
        translate([0, -bottom / 2, 0]) rotate([0, 90, 0]) cylinder(h = bottom, d = flat);
    }
    for (i = [0 : grooves - 1])
        rotate([0, 0, i * 360 / grooves]) translate([bottom / 2, 0, 0]) cylinder(h = height, d = 2, $fn = 16);
    translate([0, 0, height - 1]) cylinder(h = 2, d = bottom);
    translate([0, 0, height - 1]) difference() {
        cylinder(h = 2, d = bottom);
        translate([0, -bottom / 2, 0]) rotate([0, 90, 0]) cylinder(h = bottom, d = flat);
    }
    translate([0, -bottom / 2, height - 1]) rotate([0, 0, -90]) linear_extrude(bottom) circle(d = pointer);
}
