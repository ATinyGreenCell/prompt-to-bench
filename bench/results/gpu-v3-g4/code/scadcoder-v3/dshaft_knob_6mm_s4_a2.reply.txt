```openscad
// Replacement knob for a hotplate stirrer with a 6 mm D-shaft
// Print orientation: side facing hotplate down on the bed (z = 0)

$fn = 96;

// ---------------- Knob body ----------------
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob total height

// ---------------- D-shaped blind bore ----------------
bore_d        = 6.2;   // bore circle diameter (clearance for 6 mm shaft)
bore_flat     = 4.7;   // distance from flat to opposite side of circle
bore_depth    = 12;    // blind bore depth from bottom face

// ---------------- Grip grooves ----------------
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // groove diameter (half-cylinder centred on outer surface)

// ---------------- Pointer groove ----------------
pointer_w     = 1.5;   // pointer groove width
pointer_d     = 1;     // pointer groove depth from top face

// ---------------- Helpers ----------------
eps           = 0.5;   // overshoot for clean boolean cuts

// Flat position measured from bore centre (+X side)
flat_x = bore_flat - bore_d / 2;

module d_bore() {
    translate([0, 0, -eps])
        linear_extrude(height = bore_depth + eps)
            intersection() {
                circle(d = bore_d);
                // keep everything from the far side up to the flat
                translate([-bore_d / 2 - 1, -bore_d / 2 - 1])
                    square([flat_x + bore_d / 2 + 1, bore_d + 2]);
            }
}

module grip_grooves() {
    for (i = [0 : groove_count - 1])
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps])
                cylinder(d = groove_d, h = knob_h + 2 * eps, $fn = 48);
}

module pointer_groove() {
    // runs from centre to outer edge on the same side as the flat (+X)
    translate([0, 0, knob_h - eps])
        linear_extrude(height = eps + 1)
            polygon([[-pointer_w / 2, 0], [pointer_w / 2, 0],
                     [pointer_w / 2, pointer_d], [-pointer_w / 2, pointer_d]]);
}

difference() {
    cylinder(d = knob_d, h = knob_h);
    d_bore();
    grip_grooves();
    pointer_groove();
}
```
