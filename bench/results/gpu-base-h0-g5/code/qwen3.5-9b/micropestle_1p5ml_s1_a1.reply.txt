// --- Dimensions ---
// Handle
d_handle = 8;

// Cone
d_cone_top = 8;
d_cone_bottom = 3;
h_cone = 17; // 62 - 45

z_tip_center = 62;

n_grooves = 4;
w_groove = 2; // Width in Z
d_groove = 1; // Depth (radial)
z_groove_centers = [10, 15, 20, 25];

// --- Main Body Construction ---
difference() {

    union() {

            d = d_handle,
            center = false

        translate([0, 0, 45]) {
