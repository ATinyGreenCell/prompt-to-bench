```openscad
// L-shaped mounting bracket for NEMA 17 stepper motor (e.g. syringe pump)
// Print orientation: base plate on the bed (z = 0), vertical plate builds up along +Z

$fn = 64;

// ---------------- Base plate ----------------
base_x_min   = -25;   // base plate X start
base_x_max   = 25;    // base plate X end
base_y_min   = 0;     // base plate Y start (back edge)
base_y_max   = 45;    // base plate Y end (front edge)
base_thick   = 5;     // base plate thickness (Z)

// ---------------- Vertical motor plate ----------------
vert_x_min   = -25;   // motor plate X start
vert_x_max   = 25;    // motor plate X end
vert_y_min   = 0;     // motor plate Y start
vert_y_max   = 5;     // motor plate Y end (thickness)
vert_height  = 50;    // motor plate height (Z)

// ---------------- Boss hole ----------------
boss_d       = 23;    // motor centring boss hole diameter
boss_x       = 0;     // boss hole X centre
boss_z       = 28;    // boss hole Z centre

// ---------------- Screw holes ----------------
screw_d      = 3.4;   // M3 screw hole diameter
pitch        = 31;    // square pitch for screw holes
screw_x      = 0;     // screw hole pattern X centre
screw_z      = boss_z; // screw hole pattern Z centre

// ---------------- Gussets ----------------
gusset_x_l   = -25;   // left gusset X start
gusset_x_r   = -20;   // left gusset X end
gusset_y     = 20;    // gusset leg length along +Y
gusset_z     = 20;    // gusset leg length along +Z
gusset_t     = 5;     // gusset thickness (X)

gusset_x_p   = 20;    // right gusset X start
gusset_x_e   = 25;    // right gusset X end

eps          = 0.5;   // cutter overshoot

// ---------------- Derived ----------------
width = base_x_max - base_x_min;

difference() {
    union() {
        // Base plate
        translate([base_x_min, base_y_min, 0])
            cube([width, base_y_max - base_y_min, base_thick]);

        // Vertical motor plate
        translate([vert_x_min, vert_y_min, 0])
            cube([width, vert_y_max - vert_y_min, vert_height]);

        // Left gusset (right-angle corner at y = 5, z = 5)
        translate([gusset_x_l, 5, 5])
            cube([gusset_x_r - gusset_x_l, gusset_y, gusset_z]);

        // Right gusset
        translate([gusset_x_p, 5, 5])
            cube([gusset_x_e - gusset_x_p, gusset_y, gusset_z]);
    }

    // Vertical through-holes in base plate (M4)
    for (sx = [-1, 1])
        translate([sx * 17, 30, -eps])
            cylinder(d = 4.5, h = base_thick + 2 * eps);

    // Boss hole through motor plate (along Y)
    translate([boss_x, vert_y_min - eps, boss_z])
        cylinder(d = boss_d, h = (vert_y_max - vert_y_min) + 2 * eps);

    // M3 screw holes on square pattern centred on boss hole
    for (ox = [-pitch / 2, pitch / 2])
        for (oz = [-pitch / 2, pitch / 2])
            translate([screw_x + ox, vert_y_min - eps, screw_z + oz])
                cylinder(d = screw_d, h = (vert_y_max - vert_y_min) + 2 * eps);
}
```
