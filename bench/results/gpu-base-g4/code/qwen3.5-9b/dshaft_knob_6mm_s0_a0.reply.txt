// Function to create a D-shape cylinder (Circle with one side cut flat)
// Returns a union of a cylinder and a cut plane
module d_shaft_bore(diameter, depth, flat_side = 1) {
    $fn = 64;
    r = diameter / 2;
    
    // Main cylinder part
    difference() {
        cylinder(h = depth, d = diameter, center = [0, 0, -depth/2]);
        
        // Cut the flat side
        // The flat side is at y = -r (if flat_side=1) or y = r (if flat_side=-1)
        // We cut a plane slightly past the edge to ensure clean removal
        cut_plane = flat_side * (r + 0.1);
        
        translate([0, cut_plane, -depth/2]) {
            cube([diameter + 2, 10, depth + 2], center = true);
        }
    }
}
