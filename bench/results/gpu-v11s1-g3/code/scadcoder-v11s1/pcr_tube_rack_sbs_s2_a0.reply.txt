```openscad
// 96-well PCR tube rack (0.2 mL) - microplate footprint
// Print orientation: base on XY plane, builds up +Z

// ---------- Parameters ----------
block_len    = 127.76;   // block length along X (mm)
block_wid    = 85.48;    // block width along Y (mm)
block_h      = 20;       // block height along Z (mm)

hole_d       = 6.2;      // through-hole diameter (mm)
hole_count_r = 8;        // number of rows along Y
hole_count_c = 12;       // number of columns along X
hole_pitch   = 9.0;      // centre-to-centre spacing (mm)

a1_x         = 14.38;    // first hole offset from left edge (mm)
a1_y         = 11.24;    // first hole offset from back edge (mm)

chamfer_leg  = 5;        // 45-degree chamfer leg length at A1 corner (mm)
eps          = 0.5;      // cutter overshoot past faces (mm)

$fn = 96;

// ---------- Model ----------
difference() {
    // Main block, centred in X/Y, resting on z = 0
    translate([-block_len/2, -block_wid/2, 0])
        cube([block_len, block_wid, block_h]);

    // 96 through-holes
    for (i = [0 : hole_count_c - 1])
        for (j = [0 : hole_count_r - 1])
            translate([a1_x + i * hole_pitch - block_len/2,
                       a1_y + j * hole_pitch - block_wid/2,
                       -eps])
                cylinder(d = hole_d, h = block_h + 2*eps);

    // 45-degree chamfer off the vertical edge at the A1 corner
    translate([a1_x - block_len/2, a1_y - block_wid/2, 0])
        linear_extrude(height = block_h)
            polygon([[-chamfer_leg, -chamfer_leg],
                     [ chamfer_leg, -chamfer_leg],
                     [ chamfer_leg,  chamfer_leg]]);
}
```
