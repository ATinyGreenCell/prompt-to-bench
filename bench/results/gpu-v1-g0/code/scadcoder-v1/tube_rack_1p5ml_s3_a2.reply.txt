```openscad
// Bench rack for 1.5 mL microcentrifuge tubes - a solid rectangular block with a regular square
// grid of vertical pockets, open at the top and closed at the bottom. NOVEL general part.
//

cols   = 6;
rows   = 4;
pitch  = 16;
d      = 11.2;
depth  = 25;
block_h = 30;

$fn = 48;

p  = max(pitch, d + 2 * 1.5);          // keep a wall between neighbouring holes
Lx = (cols - 1) * p + d + 2 * 1.5;      // X size: holes plus wall and edge walls
Ly = (rows - 1) * p + d + 2 * 1.5;      // Y size: holes plus wall and edge walls

difference() {
    cube([Lx, Ly, block_h]);
    for (i = [0 : cols - 1], j = [0 : rows - 1])
        translate([d / 2 + i * p, d / 2 + j * p, 0])
            cylinder(h = block_h - depth + 1, d = d);   // pocket open at the top
}
```
