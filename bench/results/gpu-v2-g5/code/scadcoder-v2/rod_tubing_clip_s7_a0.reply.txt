```openscad
// Snap-on clip - a flat (10 mm tall) part that snaps onto a round lab stand rod and holds a
// piece of round tubing next to it, with a bridge joining the two rings. NOVEL general part.
//

stand_d = 12.7;
stand_w = 3;
tubing_d = 6;
tubing_w = 2;
bridge_w = 6;
height  = 10;

$fn = 48;

ring_d = stand_d + stand_w;
open_x = -10;

difference() {
    union() {
        ring(ring_d, stand_w, open_x);
        ring(tubing_d + tubing_w, tubing_w, 15);
        translate([8, -bridge_w / 2, 0]) cube([bridge_w, bridge_w, height]);
    }
    translate([open_x - 1, -5, -1]) cube([10, 10, height]);          // rod opening
    translate([15, -2.25, -1]) cube([4.5, 4.5, height]);            // tubing opening
}

// Round ring with a rectangular cut out of one side
ring(d, w, open_x) {
    t = d - 2 * w;
    difference() {
        cylinder(h = height, d = d);
        translate([open_x, -5, -1]) cube([10, 10, height]);
        translate([open_x, 5, -1]) cube([10, 10, height]);
    }
}
```
