```openscad
// Two-part enclosure for a small electronics board (e.g., Arduino-based logger)
// Print orientation: box standing on the bed, lid plate flat on the bed.

$fn = 64;

// ---- Dimensions (mm) ----
box_x      = 70;    // box outer length (X)
box_y      = 50;    // box outer width (Y)
box_z      = 30;    // box outer height (Z)
box_wall   = 2;     // box wall thickness
box_floor  = 2;     // box floor thickness

gap        = 10;    // clear gap between the two parts

lip_x      = 65.6;  // lip outer length (X)
lip_y      = 45.6;  // lip outer width (Y)
lip_t      = 1.5;   // lip wall thickness
lip_h      = 4;     // lip height above plate

eps        = 0.5;   // cutter overshoot

// ---- Derived ----
box_inner_x = box_x - 2*box_wall;
box_inner_y = box_y - 2*box_wall;
plate_t     = 2;    // lid plate thickness

// ---- Model ----
union() {
    // Box: open-top container
    difference() {
        translate([-box_x/2, -box_y/2, 0])
            cube([box_x, box_y, box_z]);
        translate([-box_inner_x/2, -box_inner_y/2, box_z - box_wall])
            cube([box_inner_x, box_inner_y, box_wall + eps]);
    }

    // Lid: plate with a rectangular lip that slides into the box opening
    union() {
        translate([(box_x - lip_x)/2 + gap, -lip_y/2, 0])
            cube([lip_x, lip_y, plate_t]);          // base plate
        translate([(box_x - lip_x)/2 + gap, -lip_y/2, plate_t])
            cube([lip_x, lip_y, lip_t]);            // lip (overlaps plate for a clean union)
    }
}
```
