```openscad
// Two-part enclosure for a small electronics board (e.g. Arduino-based temperature logger)
// Print orientation: box standing on the bed, lid printed upside down on the +X side

// ---------- Parameters ----------
box_x      = 70;    // box outer size in X
box_y      = 50;    // box outer size in Y
box_z      = 30;    // box outer size in Z
box_wall   = 2;     // box wall thickness
box_floor  = 2;     // box floor thickness

gap        = 10;    // gap between box and lid

lip_x      = 65.6;  // lip outer size in X
lip_y      = 45.6;  // lip outer size in Y
lip_h      = 4;     // lip height above plate
lip_wall   = 1.5;   // lip wall thickness

eps        = 0.5;   // cutter overshoot

$fn = 64;

// ---------- Derived ----------
box_inner_x = box_x - 2*box_wall;
box_inner_y = box_y - 2*box_wall;
plate_t     = 2;    // plate thickness

// ---------- Model ----------
difference() {
    union() {
        // Box (open top)
        cube([box_x, box_y, box_z - box_floor]);
        translate([0, 0, box_z - box_floor])
            cube([box_inner_x, box_inner_y, box_wall]);

        // Lid: plate with centred lip
        union() {
            translate([box_x/2 - gap - lip_x/2, -lip_y/2, 0])
                cube([lip_x, lip_y, plate_t]);
            translate([box_x/2 - gap - lip_x/2, -lip_y/2, plate_t])
                cube([lip_x, lip_y, lip_h]);
        }
    }

    // Lip cutter, sunk into the plate so the wall is exactly lip_wall thick
    translate([box_x/2 - gap - lip_x/2 - box_wall, -lip_y/2 - box_wall, plate_t - lip_wall])
        cube([lip_x + 2*box_wall, lip_y + 2*box_wall, lip_wall + eps]);
}
```
