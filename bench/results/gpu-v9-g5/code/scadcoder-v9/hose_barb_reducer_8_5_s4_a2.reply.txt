```openscad
// Straight hose-barb reducer: 8 mm ID tubing end with two stacked barbs,
// a 12 mm collar, and a second 5 mm ID tubing end with two stacked barbs.
// A 3 mm bore runs straight through the full length.
// Print orientation: standing upright along +Z.

$fn = 96;

// ---- Dimensions (mm) ----
tube_a_id   = 8;    // first tubing inner diameter
tube_b_id   = 5;    // second tubing inner diameter
bore        = 3;    // through-bore diameter

barb_a_h    = 10;   // height of each 8 mm tubing barb
barb_a_od   = 7.5;  // outer diameter of each 8 mm tubing barb
barb_a_n    = 2;    // number of 8 mm tubing barbs (stacked)

collar_d    = 12;   // outer diameter of the collar
collar_h    = 5;    // height of the collar (z = 20 to 25)

barb_b_h    = 8;    // height of each 5 mm tubing barb
barb_b_od   = 6.5;  // outer diameter of each 5 mm tubing barb
barb_b_n    = 2;    // number of 5 mm tubing barbs (stacked)

overlap     = 0.01; // small overlap so adjacent shapes fuse cleanly

// ---- Model ----
difference() {
    union() {
        // 8 mm tubing barbs: z = 0 to 20
        for (i = [0 : barb_a_n - 1])
            translate([0, 0, i * barb_a_h])
                cylinder(h = barb_a_h, d = barb_a_od);

        // collar: z = 20 to 25
        translate([0, 0, collar_h * 0.01])
            cylinder(h = collar_h + overlap, d = collar_d);

        // 5 mm tubing barbs: z = 25 to 41
        for (i = [0 : barb_b_n - 1])
            translate([0, 0, 25 + i * barb_b_h])
                cylinder(h = barb_b_h, d = barb_b_od);
    }

    // through-bore
    translate([0, 0, -overlap])
        cylinder(h = 41 + 2 * overlap, d = bore);
}
```
