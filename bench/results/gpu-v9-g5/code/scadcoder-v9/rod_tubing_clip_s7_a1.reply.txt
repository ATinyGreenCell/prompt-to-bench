```openscad
// Snap-on tubing clip for 12.7 mm lab stand rod
// Print orientation: flat on bed, extrusion along +Z

$fn = 96;

// ---- Dimensions (mm) ----
rod_id      = 12.7;   // stand rod diameter
rod_wall    = 3;      // rod ring wall thickness
ring_h      = 10;     // extrusion height for rod ring

tub_id      = 6;      // tubing ring inner diameter
tub_wall    = 2;      // tubing ring wall thickness
tub_x       = 15;     // tubing ring centre X
tub_ring_h  = 10;     // extrusion height for tubing ring

bridge_w    = 6;      // solid bridge width (Y)
bridge_x0   = 8;      // solid bridge start (X)
bridge_x1   = 12;     // solid bridge end (X)

eps         = 0.5;    // cutter overshoot past faces

// ---- Derived ----
rod_od = rod_id + 2 * rod_wall;
tub_od = tub_id + 2 * tub_wall;

module rod_ring() {
    difference() {
        translate([0, 0, 0])
            cylinder(d = rod_od, h = ring_h);
        // opening on -X side: x < 0, -5 < y < 5
        translate([-1, -5, -eps])
            cube([1 + eps, 10, eps]);
    }
}

module tub_ring() {
    difference() {
        translate([tub_x, 0, 0])
            cylinder(d = tub_od, h = tub_ring_h);
        // opening on +X side: x > 15, -2.25 < y < 2.25
        translate([tub_x + eps, -2.25, 0])
            cube([eps, 4.5, tub_ring_h]);
    }
}

module solid_bridge() {
    // joins the two rings without blocking either hole
    translate([bridge_x0, -bridge_w / 2, 0])
        cube([bridge_x1 - bridge_x0, bridge_w, ring_h]);
}

union() {
    rod_ring();
    tub_ring();
    solid_bridge();
}
```
