```openscad
// Snap-on tubing clip for a 12.7 mm lab stand rod
// Print orientation: flat on bed, extrusion along +Z

$fn = 96;

// ---- Rod ring ----
rod_id   = 12.7;   // rod inner diameter
rod_wall = 3;      // ring wall thickness
rod_od   = rod_id + 2 * rod_wall;  // 18.7 mm outer diameter
rod_open = 10;     // width of opening cut from -X side

// ---- Tubing ring ----
tube_id  = 6;      // tubing inner diameter
tube_wall = 2;     // ring wall thickness
tube_od  = tube_id + 2 * tube_wall;  // 10 mm outer diameter
tube_x   = 15;     // ring centre X
tube_open = 4.5;   // width of opening cut from +X side

// ---- Bridge ----
bridge_w = 6;      // bridge width along Y
bridge_x0 = 8;     // bridge start X
bridge_x1 = 12;    // bridge end X

// ---- Cut clearance ----
eps = 0.5;         // overshoot for clean boolean cuts

difference() {
    union() {
        // Rod ring
        cylinder(d = rod_od, h = 10);
        translate([-rod_open / 2, -rod_open / 2, 0])
            cube([rod_open, rod_open, 10]);

        // Tubing ring
        translate([tube_x, 0, 0])
            cylinder(d = tube_od, h = 10);
        translate([tube_x + tube_open / 2, -tube_open / 2, 0])
            cube([tube_open, tube_open, 10]);

        // Solid bridge
        translate([bridge_x0, -bridge_w / 2, 0])
            cube([bridge_x1 - bridge_x0, bridge_w, 10]);
    }

    // Cut out the stand rod opening (x < 0, -5 < y < 5)
    translate([-rod_open / 2 - eps, -rod_open / 2, -eps])
        cube([rod_open + 2 * eps, rod_open + 2 * eps, 10 + 2 * eps]);

    // Cut out the tubing opening (x > 15, -2.25 < y < 2.25)
    translate([tube_x + tube_open / 2 - eps, -tube_open / 2, -eps])
        cube([tube_open + 2 * eps, tube_open + 2 * eps, 10 + 2 * eps]);
}
```
