```openscad
// Configuration and Dimensions
// All dimensions in millimetres (mm)

// Box Dimensions (Outer)
$box_outer_x = 70;
$box_outer_y = 50;
$box_outer_z = 30;

// Wall and Floor Thickness
$wall_thickness = 2;
$floor_thickness = 2;

// Lid Dimensions
$plate_thickness = 2;
$lip_thickness = 1.5;
$lip_height = 4;

// Lip Outer Size (to fit inside box opening)
$lip_outer_x = 65.6;
$lip_outer_y = 45.6;

// Clearance
$clearance = 0.2;

// Gap between parts for printing together
$gap = 10;

// Resolution for curved shapes
$fn = 64;

// --- Part 1: The Box ---
// Position: Centred on origin in X, Y. Z starts at 0.
// Outer box is 70x50x30.
// Inner cavity calculation:
// Inner X = 70 - 2*2 = 66
// Inner Y = 50 - 2*2 = 46
// Inner Z = 30 - 2 (floor) = 28 (open top)

translate([0, 0, 0])
difference() {
    // Outer Shell
    cube([
        $box_outer_x, 
        $box_outer_y, 
        $box_outer_z
    ]);
    
    // Cut out the interior
    // We extend the cut slightly past the faces to ensure clean separation
    translate([
        $wall_thickness + 0.05, 
        $wall_thickness + 0.05, 
        $floor_thickness + 0.05
    ])
    cube([
        $box_outer_x - 2*$wall_thickness - 0.1, 
        $box_outer_y - 2*$wall_thickness - 0.1, 
        $box_outer_z - $floor_thickness - 0.1
    ]);
}

// --- Part 2: The Lid ---
// Position: On the +X side of the box, with a 10mm gap.
// Box ends at X = 70. Lid starts at X = 70 + 10 = 80.
// Lid is printed upside down (plate on bed).
// Lid is centered on the bed in Y, but shifted in X.
// Lid Center X = 80 + (70/2) = 115.
// Lid Center Y = 0.

translate([115, 0, 0])
difference() {
    // 1. The Base Plate (70 x 50 x 2)
    cube([
        $box_outer_x, 
        $box_outer_y, 
        $plate_thickness
    ]);
    
    // 2. The Lip Structure
    // The lip sits ON TOP of the plate (Z = 2 to Z = 6).
    // It has an outer dimension of 65.6 x 45.6.
    // It has an inner hole to slide into the box.
    // Box inner opening is 66 x 46.
    // Lip outer is 65.6 x 45.6.
    // Lip inner needs to be (66 - 0.4) x (46 - 0.4) = 65.2 x 45.6 to allow 0.2mm clearance.
    
    translate([0, 0, $plate_thickness])
    difference() {
        // Solid Lip Block (Outer dimensions)
        cube([
            $lip_outer_x, 
            $lip_outer_y, 
            $lip_height + $lip_thickness
        ]);
        
        // Cut out the slot for the board
        // Extend cut slightly past faces
        translate([
            ($lip_outer_x - ($box_outer_x - 2*$wall_thickness - $clearance*2)) / 2 + 0.05,
            ($lip_outer_y - ($box_outer_y - 2*$wall_thickness - $clearance*2)) / 2 + 0.05,
            $lip_thickness + 0.05
        ])
        cube([
            $box_outer_x - 2*$wall_thickness - $clearance*2 - 0.1,
            $box_outer_y - 2*$wall_thickness - $clearance*2 - 0.1,
            $lip_height + $lip_thickness - 0.1
        ]);
    }
}
```
