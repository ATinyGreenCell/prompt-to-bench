// PCR tube adapter sleeve: fits a 0.2 mL tube in a 1.5 mL rack hole
// Print orientation: collar on bed (z = 0), sleeve body rises to top

// ---------------- Parameters ----------------
collar_d      = 13;    // collar outer diameter (on bed)
collar_t      = 1.5;   // collar thickness (z = 0 to 1.5)
sleeve_d      = 10.8;  // sleeve body outer diameter
sleeve_h      = 20;    // sleeve body height above collar (z = 1.5 to 21.5)
bore_d        = 6.2;   // through-bore diameter
eps           = 0.5;   // cutter overshoot past faces
$fn           = 96;    // curve resolution

// Derived
sleeve_top_z  = collar_t + sleeve_h;   // 21.5
total_h       = collar_t + sleeve_h;   // 21.5

// ---------------- Model ----------------
difference() {
    union() {
        // Collar (z = 0 to 1.5)
        cylinder(d = collar_d, h = collar_t);
        // Sleeve body (z = 1.5 to 21.5)
        translate([0, 0, collar_t])
            cylinder(d = sleeve_d, h = sleeve_h);
    }
    // Through bore (extends past both ends)
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = total_h + 2 * eps);
}
