```openscad
// Replacement knob for hotplate stirrer with D-shaft
// Print orientation: bottom face (with D-bore) on bed, z = 0

// ---------- Parameters ----------
knob_d        = 30;    // knob outer diameter
knob_h        = 15;    // knob height
shaft_d       = 6;     // D-shaft diameter
bore_depth    = 12;    // blind bore depth from bottom face
bore_flat     = 4.7;   // flat-to-flat width of D-bore (4.7 mm)
bore_d        = bore_flat + shaft_d; // 6.2 mm circle with one side cut flat
groove_count  = 18;    // number of vertical grip grooves
groove_d      = 2;     // groove diameter (half-cylinder)
pointer_w     = 1.5;   // pointer groove width at centre
pointer_d     = pointer_w + 1; // pointer groove diameter at outer edge
pointer_depth = 1;     // pointer groove depth from top face
eps           = 0.5;   // overcut so cutting faces are not coplanar

$fn = 96;

// ---------- Model ----------
difference() {
    // Main knob body
    cylinder(d = knob_d, h = knob_h);

    // D-shaped blind bore from bottom face
    translate([0, 0, -eps])
        cylinder(d = bore_d, h = bore_depth + 2 * eps, $fn = 64);

    // Vertical grip grooves around the outside
    for (i = [0 : groove_count - 1]) {
        rotate([0, 0, i * 360 / groove_count])
            translate([knob_d / 2, 0, -eps / 2])
                cylinder(d = groove_d, h = knob_h + eps, $fn = 64);
    }

    // Pointer groove on top face, same side as D-flat
    // (placed at -X so it lies on the flat side of the D-bore)
    translate([-knob_d / 2, 0, knob_h - pointer_depth])
        linear_extrude(height = pointer_depth + 0.01)
            circle(d = pointer_d, $fn = 64);
}
```
