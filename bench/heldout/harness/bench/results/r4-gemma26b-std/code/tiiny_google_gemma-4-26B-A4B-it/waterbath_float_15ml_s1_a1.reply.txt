// --- Dimensions ---
plate_thickness = 3;            // Thickness of the hexagonal plate
hex_across_flats = 90;           // Distance between parallel sides of the plate
hex_skirt_across_flats = 84;     // Distance between parallel sides of the skirt
skirt_wall_thickness = 2;        // Thickness of the hexagonal skirt wall
skirt_height_total = 20;         // Total height of the part (Z=0 to Z=20)

hole_diameter = 17.5;            // Diameter of the conical tube holes
hole_spacing_diameter = 54;      // Diameter of the circle on which holes are placed
sleeve_outer_diameter = 21.5;    // Outside diameter of the sleeves
sleeve_height_total = 20;        // Total height of the sleeves (Z=0 to Z=20)

// --- Constants ---
$fn = 64;

// Helper function to calculate the radius of a regular hexagon from across-flats distance
// Radius (to vertex) = (across_flats / 2) / cos(30)
function hex_radius(across_flats) = (across_flats / 2) / cos(30);

// --- Construction Logic ---
// The user wants the part printed "upside down" relative to use.
// The plate is at Z=0 to 3. The features (sleeves and skirt) rise from the plate to Z=20.
// This means the "bottom" of the part (the plate) is the base on the bed.
// To ensure the part is a single solid and the holes are correctly subtracted:

union() {
    // 1. The Hexagonal Plate (Base)
    // We use a difference to create the holes in the plate.
    difference() {
        // The solid plate
        // To get flats parallel to X, we rotate the hexagon by 30 degrees.
        // A standard cylinder(fn=6) has vertices on the X-axis. 
        // Rotating 30 deg puts the flat side parallel to X.
        rotate([0, 0, 30])
        cylinder(h = plate_thickness, r = hex_radius(hex_across_flats), $fn = 6);

        // Subtract the 7 holes (through the plate)
        // Center hole
        translate([0, 0, -0.01])
        cylinder(h = plate_thickness + 0.02, d = hole_diameter);
        
        // 6 Peripheral holes
        for (a = [0 : 60 : 300]) {
            rotate([0, 0, a])
            translate([hole_spacing_diameter / 2, 0, -0.01])
            cylinder(h = plate_thickness + 0.02, d = hole_diameter);
        }
    }

    // 2. The Sleeves and Skirt
    // These rise from the plate (Z=3) to Z=20.
    // However, to ensure a watertight solid, we model them from Z=0 to Z=20 
    // and let the union/difference handle the overlap.
    
    // The Sleeves
    // Note: The user said "rises from the plate to z=20". 
    // If the plate is 3mm thick, the sleeve height is 17mm.
    // To avoid gaps, we model them from Z=0 to Z=20.
    // Since they are "sleeves" for the holes, they must be hollow.
    // We will use a union of the solid cylinders and then subtract the holes 
    // from the entire assembly to ensure the holes go through the sleeves too.
    
    // Let's use a different approach for a cleaner model:
    // Total Part = (Plate + Sleeves + Skirt) - Holes
}

// --- RE-STRUCTURED MODEL FOR ACCURACY ---
// We will build the solid mass first, then subtract the holes.

difference() {
    // THE SOLID MASS
    union() {
        // The Plate
        rotate([0, 0, 30])
        cylinder(h = plate_thickness, r = hex_radius(hex_across_flats), $fn = 6);

        // The Sleeves (Solid cylinders)
        // They rise from the plate (Z=3) to Z=20.
        // To make them "sleeves", we add them as solid and subtract holes later.
        cylinder(h = skirt_height_total - plate_thickness, d = sleeve_outer_diameter); // Center sleeve
        for (a = [0 : 60 : 300]) {
            rotate([0, 0, a])
            translate([hole_spacing_diameter / 2, 0, plate_thickness])
            cylinder(h = skirt_height_total - plate_thickness, d = sleeve_outer_diameter);
        }

        // The Hexagonal Skirt
        // Rises from plate (Z=3) to Z=20.
        // Outer dimension: 84mm across flats.
        // Wall thickness: 2mm.
        // We model the skirt as a solid hexagonal prism and then subtract the inner hex.
        // But wait, the skirt is a feature that rises from the plate.
        // To make it a single part, we add the outer hex and subtract the inner hex.
        
        // Outer Skirt Shell
        rotate([0, 0, 30])
        cylinder(h = skirt_height_total - plate_thickness, r = hex_radius(hex_skirt_across_flats), $fn = 6);
        
        // Inner Skirt Void (to create the 2mm wall)
        // The skirt is 84mm across flats. Inner is 84 - 4 = 80mm across flats.
        // We subtract this from the union.
        // To ensure it doesn't cut the plate, we translate it to Z=plate_thickness.
        translate([0, 0, plate_thickness])
        rotate([0, 0, 30])
        cylinder(h = skirt_height_total - plate_thickness + 0.01, r = hex_radius(hex_skirt_across_flats - 2*skirt_wall_thickness), $fn = 6);
    }

    // THE HOLES (Subtracted from the whole assembly)
    // Center hole
    translate([0, 0, -0.01])
    cylinder(h = plate_thickness + skirt_height_total + 0.02, d = hole_diameter);
    
    // 6 Peripheral holes
    for (a = [0 : 60 : 300]) {
        rotate([0, 0, a])
        translate([hole_spacing_diameter / 2, 0, -0.01])
        cylinder(h = plate_thickness + skirt_height_total + 0.02, d = hole_diameter);
    }
}

// 1. Union(Plate, Sleeves, Outer_Skirt_Shell)
// 2. Subtract(Inner_Skirt_Void)
// 3. Subtract(Holes)
// This results in a plate with 7 holes, 6 sleeves around them, and a hexagonal wall.
