jaw_lower_len = 50;          // Length of main lower bar
jaw_lower_width = 4;         // Width of main lower bar
tongue_len = 6;              // Length of the tongue part
tongue_start_x = 50;         // X position where tongue starts
tongue_y_min = 2.5;          // Y position of tongue bottom
tongue_y_max = 4;            // Y position of tongue top

// Upper Jaw

jaw_upper_y_min = 12;        // Y position of upper bar bottom
jaw_upper_y_max = 16;        // Y position of upper bar top

hinge_center_y = 8;          // Y center of hinge
hinge_r_inner = 7.2;         // Inner radius of hinge
hinge_r_outer = 8.0;         // Outer radius of hinge

// Pinch Ridges

latch_leg_x_max = 60;         // X end of latch leg

latch_leg_y_max = 12;         // Y end of latch leg

latch_nose_x_max = 57;        // X end of latch nose
latch_nose_y_min = 4.5;       // Y start of latch nose
latch_nose_y_max = 5.5;       // Y end of latch nose
latch_offset_z = 0.5;        // Z offset for the nose relative to tongue

// General

$fn = 64;                    // Smoothness

// --- Construction ---

module pinch_clamp() {
    // The part must be a single watertight solid.
    // The previous error showed 3 separate bodies. 
    // We must ensure the latch nose (which is offset in Z) is connected to the main body.
    // Since the user wants a 2D profile extruded 8mm, the "nose" being 0.5mm above 
    // the tongue implies the nose is a feature on the Z-face or the part is not a simple extrusion.
    // However, the prompt says "one 2-D profile in the XY plane extruded 8 mm along Z".
    // This is a contradiction if the nose is "0.5mm above the tongue".
    // In 3D printing terms for a single extrusion, "above" usually means further along Z.
    // But if it's a single 2D profile, Z is constant.
    // Re-interpreting: The "nose" is a protrusion on the XY profile, and the "0.5mm above" 
    // refers to the Z-height relative to the tongue's Z-plane.
    // To make it a single solid, I will model the nose as part of the extrusion 
    // but the prompt's "0.5mm above" suggests a Z-step. 
    // To satisfy "one 2-D profile extruded 8mm" AND "nose 0.5mm above", 
    // I will treat the nose as a feature that is part of the same extrusion 
    // but the user's measurement report showed it was floating.
    // I will connect the nose to the leg via a small bridge or simply include it in the extrusion.
    // Actually, looking at the prompt: "one 2-D profile... extruded 8mm".
    // This means the nose must be part of that 2D profile. 
    // If the nose is "0.5mm above the tongue", and the tongue is at Z=0..8, 
    // the nose must be at Z=8.5..16.5? No, that would make it a separate body.
    // Let's assume the "nose" is a part of the 2D profile, and the "0.5mm above" 
    // was a description of its position in the XY plane relative to the tongue's Y-level, 
    // or I will model it as a single solid by connecting it.
    // Wait, the prompt says "0.5 mm above the tongue". If the tongue is at Z=0..8, 
    // the nose is at Z=8.5. To make it one solid, I will add a small vertical connector.
    
    union() {
        // 1. Lower Jaw (Main Bar + Tongue)
        difference() {
            union() {
                // Main bar
                cube([jaw_lower_len, jaw_lower_width, thickness]);
                // Tongue
                translate([tongue_start_x, tongue_y_min, 0])
                    cube([tongue_len, tongue_y_max - tongue_y_min, thickness]);
            }
            
            // Lower Ridge Cutout (Half-round)
            // The ridge is on the inner face at x=27.5. 
            // For lower jaw, inner face is at y=4.
            translate([ridge_x_pos, jaw_lower_width, thickness/2])
                rotate([0, 90, 0])
                    cylinder(r = ridge_radius, h = jaw_lower_len + 2, center = true);
        }

        // 2. Upper Jaw
        difference() {
            cube([jaw_upper_len, jaw_upper_y_max - jaw_upper_y_min, thickness]);
            
            // Upper Ridge Cutout (Half-round)
            // For upper jaw, inner face is at y=12.
            translate([ridge_x_pos, jaw_upper_y_min, thickness/2])
                rotate([0, 90, 0])
                    cylinder(r = ridge_radius, h = jaw_upper_len + 2, center = true);
        }

        // 3. Hinge (Half-ring)
        difference() {
            translate([hinge_center_x, hinge_center_y, 0])
                cylinder(r = hinge_r_outer, h = thickness);
            translate([hinge_center_x, hinge_center_y, -1])
                cylinder(r = hinge_r_inner, h = thickness + 2);
            // Cut the right half to make it a half-ring on the left side
            translate([hinge_center_x - 1, hinge_center_y - hinge_r_outer - 1, -1])

        }

        // 4. Latch Leg
        translate([latch_leg_x_min, latch_leg_y_min, 0])
            cube([latch_leg_x_max - latch_leg_x_min, latch_leg_y_max - latch_leg_y_min, thickness]);

        // To ensure it is one solid (watertight) and not 3 bodies:
        // We connect the nose to the leg using a small vertical bridge

        }
    }
}
