module hinge_half_ring() {
    // A half-ring on the x < 0 side, centered at (0, 8)
    // We create the ring using a difference of two cylinders, 
    // then use a cube to cut it so only the x < 0 part remains.
    difference() {
        // Outer cylinder
        translate([hinge_center_x, hinge_center_y, 0])
            cylinder(h = thickness, r = hinge_r_outer);
        
        // Inner cylinder
        translate([hinge_center_x, hinge_center_y, -1])
            cylinder(h = thickness + 2, r = hinge_r_inner);
            
        // Cut the right side (x > 0) to leave only the half-ring on the left
        translate([hinge_center_x - 0.01, hinge_center_y - hinge_r_outer - 1, -1])
            cube([hinge_r_outer + 1, hinge_r_outer * 2 + 2, thickness + 2]);
    }
}
