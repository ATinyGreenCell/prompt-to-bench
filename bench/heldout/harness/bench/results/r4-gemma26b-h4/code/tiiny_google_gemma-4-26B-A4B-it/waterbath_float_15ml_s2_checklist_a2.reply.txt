// Generates a regular hexagon with flats parallel to the X axis.
// We use a polygon to ensure exact control over the geometry.
module hex_shape(across_flats, height) {
    // Apothem (distance from center to flat) is across_flats / 2
    // Circumradius (distance from center to vertex) is apothem / cos(30)
    r_vertex = (across_flats / 2) / cos(30);
    
    // To make flats parallel to X, we rotate the polygon by 30 degrees
    // or define vertices such that the midpoints of sides are on the Y axis.
    // A standard hexagon has vertices at 0, 60, 120... 
    // Rotating by 30 degrees puts the flats at 0, 60, 120... (parallel to Y)
    // To get flats parallel to X, we rotate by 0 or 60.
    // Let's use the standard: vertices at 30, 90, 150, 210, 270, 330.
    // This puts the flats at 0, 60, 120... (parallel to Y).
    // To get flats parallel to X, we rotate the whole thing by 90 or 30.
    // The prompt says "two flats parallel to the X axis".
    // This means the midpoints of two sides are on the Y axis.
    // This occurs when vertices are at 0, 60, 120...
    
    rotate([0, 0, 0])
    linear_extrude(height = height)
    for (i = [0 : 5]) {
        rotate([0, 0, i * 60])
        polygon(points = [[0, 0], [r_vertex, 0], [r_vertex * cos(60), r_vertex * sin(60)]]);
    }
    // Note: The above is a triangle-based approach. Let's use a cleaner polygon.
}

// Robust Hexagon module
module regular_hex(across_flats, height) {
    r_vertex = (across_flats / 2) / cos(30);
    // Vertices for flats parallel to X: 0, 60, 120... (This makes flats parallel to Y)
    // Vertices for flats parallel to X: 30, 90, 150... (This makes flats parallel to X)
    // Wait, if vertices are at 30, 90, 150, then the side between 30 and 150 is vertical (parallel to Y).
    // If vertices are at 0, 60, 120, then the side between 60 and 120 is horizontal (parallel to X).
    // Let's use vertices at 0, 60, 120... to get flats parallel to X.
    
    linear_extrude(height = height)
    for (i = [0 : 5]) {
        rotate([0, 0, i * 60])
        polygon(points = [[0, 0], [r_vertex, 0], [r_vertex * cos(60), r_vertex * sin(60)]]);
    }
    // Actually, the simplest way to define a hexagon with flats parallel to X:
    // The vertices are at 30, 90, 150, 210, 270, 330.
}

module hex_solid(across_flats, height) {

}

module correct_hex(across_flats, height) {
    r_vertex = (across_flats / 2) / cos(30);
    linear_extrude(height = height)
    polygon(points = [
        [r_vertex * cos(30), r_vertex * sin(30)],
        [r_vertex * cos(90), r_vertex * sin(90)],
        [r_vertex * cos(150), r_vertex * sin(150)],
        [r_vertex * cos(210), r_vertex * sin(210)],
        [r_vertex * cos(270), r_vertex * sin(270)],
        [r_vertex * cos(330), r_vertex * sin(330)]
    ]);
}
