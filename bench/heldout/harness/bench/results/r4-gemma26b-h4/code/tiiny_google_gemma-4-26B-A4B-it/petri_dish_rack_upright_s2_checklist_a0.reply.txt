$fn = 128;

// Dividers are at x = [-100, -80, ..., 100]
// This means the distance between divider centers is 20mm.
// Slot width = 20mm - 3mm = 17mm.

    translate([-L/2, -W/2, 0])
        cube([L, W, H_base]);
}

module divider() {
    // Dividers rise from z=0 to z=60
    // They are 3mm thick in X, 80mm wide in Y
    translate([-divider_t/2, -W/2, 0])

}
