// Carrier for four 50 mL conical tubes, printed as two parts side by side.
// Part 1: base block + two end posts (with through holes for the handle bar).
// Part 2: square handle bar, printed lying flat on the bed.

$fn = 64;

// ---- Part 1: base block ----
base_x = 90;   // base block width along X
base_y = 90;   // base block width along Y
base_z = 35;   // base block height

seat_d = 30.5;  // blind seat diameter
seat_depth = 30; // blind seat depth (leaves 5 mm floor)
seat_x = 21;    // |x| of seat centres
seat_y = 24;    // |y| of seat centres

// ---- Part 1: end posts ----
post_t = 10;   // post thickness along X
post_w = 16;   // post width along Y
post_h = 160;  // post height (top at z = 160)
post_hole = 10.4; // square through-hole side, along X
post_hole_z = 150; // hole centre height

// ---- Part 2: handle bar ----
bar_t = 10;    // bar thickness (X)
bar_w = 10;    // bar width (Y)
bar_l = 100;   // bar length (Z, lying flat on bed)
bar_y = 60;    // bar centre Y (spans 55..65, 10 mm gap to base +Y face)

// ================= PART 1 =================
module part1() { difference(
    union(
        // base block, centred on origin
        translate([-base_x/2, -base_y/2, 0])
            cube([base_x, base_y, base_z]);

        // two end posts, flush with the +/-X end faces, centred on y = 0
        for (sx = [-1, 1])
            translate([sx * (base_x/2 - post_t/2), -post_w/2, 0])
                cube([post_t, post_w, post_h]);
    ),

    union(
        // four blind tube seats (5 mm floor)
        for (sx = [-1, 1])
            for (sy = [-1, 1])
                translate([sx * seat_x, sy * seat_y, 0])
                    cylinder(h = seat_depth + 0.01, d = seat_d);

        // square through-holes in each post, along X, centred at z = 150
        for (sx = [-1, 1])
            translate([sx * (base_x/2 - post_t/2) - 0.01, -post_hole/2, post_hole_z - post_hole/2])
                cube([post_t + 0.02, post_hole, post_hole]);
    )
); }

// ================= PART 2 (handle bar) =================
// Printed lying flat on the bed along X, centred on x = 0, at y = 55..65.
module part2() { translate([-bar_l/2, bar_y - bar_w/2, 0])
    cube([bar_l, bar_w, bar_t]); }

// ================= LAYOUT (side by side, both on the bed) =================
// Part 1 occupies x in [-45,45], y in [-45,45].
// Part 2 occupies x in [-50,50], y in [55,65].
// Shift part 2 in +X so the two printed parts do not overlap.
module part2_print() { translate([100, 0, 0]) part2(); }

// Final assembly: both parts, each resting on the XY plane at z = 0.
union() {
    part1();
    part2_print();
}
