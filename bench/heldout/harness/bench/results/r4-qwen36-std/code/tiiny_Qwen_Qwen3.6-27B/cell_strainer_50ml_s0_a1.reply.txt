```openscad
// Coarse Cell Strainer for 50 mL Conical Tube
// Designed to hang in the mouth of a 50 mL tube.

// --- Dimensions ---
// Cup dimensions
cup_outer_dia = 25;          // Outer diameter of the main cup body
cup_wall = 1.2;              // Wall thickness
cup_bore = cup_outer_dia - 2 * cup_wall; // Bore diameter (22.6 mm)
cup_height = 32;             // Total height of the cup structure
floor_thick = 1.2;           // Thickness of the floor

// Collar dimensions
collar_start_z = 26;         // Z height where the flare begins
collar_end_z = 30;           // Z height where the flare ends (rim starts)
collar_top_dia = 33;         // Outer diameter of the rim
collar_bot_dia = cup_outer_dia; // Outer diameter at the start of the flare

// Hole pattern
hole_dia = 1.0;              // Diameter of the filter holes
hole_spacing = 2;            // Grid spacing for holes
hole_radius_limit = 10;      // Max distance from center for a hole to be included

// Vent notches
vent_width = 3;              // Width of the vent notches
vent_start_z = 25;           // Z height where vents start
vent_offset = 12.5;          // Distance from axis where vent cut begins (half of collar_bot_dia)

// Rendering quality
$fn = 64;

// --- Main Structure ---
difference() {
    union() {
        // 1. Main Cup Body (Hollow Cylinder)
        // Outer wall
        cylinder(h = cup_height, r = cup_outer_dia / 2, center = false);
        
        // 2. Collar (Flared Rim)
        // The flare is a cone from z=26 to z=30
        // Bottom radius at z=26 is cup_outer_dia/2 = 12.5
        // Top radius at z=30 is collar_top_dia/2 = 16.5
        // Height of cone = 4
        translate([0, 0, collar_start_z])
            cylinder(h = collar_end_z - collar_start_z, 
                     r1 = collar_bot_dia / 2, 
                     r2 = collar_top_dia / 2, 
                     center = false);
        
        // The Rim (Cylinder from z=30 to z=32)
        translate([0, 0, collar_end_z])
            cylinder(h = cup_height - collar_end_z, 
                     r = collar_top_dia / 2, 
                     center = false);
    }

    // --- Cuts ---

    // 1. Hollow out the cup interior
    // We cut from z=floor_thick up to z=cup_height to leave the floor intact
    // Extend slightly past top to ensure clean cut
    translate([0, 0, floor_thick])
        cylinder(h = cup_height - floor_thick + 1, 
                 r = cup_bore / 2, 
                 center = false);

    // 2. Floor Holes
    // Grid of 1.0mm holes, 2mm spacing, centered, within 10mm radius
    for (x = [-10 : 2 : 10]) {
        for (y = [-10 : 2 : 10]) {
            // Check if the point is within the 10mm radius limit
            if (sqrt(x*x + y*y) <= hole_radius_limit) {
                translate([x, y, 0])
                    cylinder(h = floor_thick + 0.1, // Extend slightly through floor
                             r = hole_dia / 2, 
                             center = false);
            }
        }
    }

    // 3. Vent Notches
    // Two 3mm wide notches on +X and -X sides
    // They remove material beyond 12.5mm from axis (i.e., from the outer edge of the cup body)
    // From z=25 upwards
    
    // +X Vent
    translate([vent_offset, -20, vent_start_z])
        cube([vent_width, 40, cup_height - vent_start_z + 1], center = false);
        
    // -X Vent
    translate([-vent_offset - vent_width, -20, vent_start_z])
        cube([vent_width, 40, cup_height - vent_start_z + 1], center = false);
}
```
