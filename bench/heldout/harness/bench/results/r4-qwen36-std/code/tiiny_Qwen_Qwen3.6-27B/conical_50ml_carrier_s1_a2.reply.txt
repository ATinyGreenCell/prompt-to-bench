```openscad
// Dimensions for Part 1: Base Block
base_x = 90;
base_y = 90;
base_z = 35;

// Dimensions for Part 1: Tube Seats
seat_dia = 30.5;
seat_depth = 30;
seat_floor = 5; // base_z - seat_depth
seat_x_pos = 21;
seat_y_pos = 24;

// Dimensions for Part 1: Posts
post_x = 10;
post_y = 16;
post_z = 160;
post_hole_x = 10.4;
post_hole_y = 10.4;
post_hole_z = 150;

// Dimensions for Part 2: Handle
handle_x = 100;
handle_y = 10;
handle_z = 10;
handle_y_start = 55;
handle_y_end = 65;

// Global settings
$fn = 64;

// --- Part 1: Base Block ---
// Centered on origin in X and Y, resting on z=0
// The base block is 90x90x35, centered at (0,0,0) in XY, so it spans X: -45..45, Y: -45..45, Z: 0..35
difference() {
    // Main base block
    translate([0, 0, 0])
        cube([base_x, base_y, base_z], center = true);
    
    // Four blind seats for 50 mL tubes
    // Seats are 30.5mm dia, 30mm deep. Floor is 5mm thick (35-30=5).
    // Centers at x = -21, 21 and y = -24, 24.
    for (sx = [-seat_x_pos, seat_x_pos]) {
        for (sy = [-seat_y_pos, seat_y_pos]) {
            translate([sx, sy, 0]) {
                cylinder(r = seat_dia / 2, h = seat_depth + 1, center = false);
            }
        }
    }
    
    // Two posts
    // Posts are 10mm thick (X) and 16mm wide (Y).
    // They stand on the base at its two X ends, flush with the end faces.
    // Base X ends are at -45 and 45.
    // Post X center should be at -45 + 5 = -40 and 45 - 5 = 40.
    // Posts are centred on y = 0.
    // Posts rise to z = 160. Since base is 0..35, posts start at z=0? 
    // "stand on the base... and rise to z=160". This implies the post top is at z=160.
    // The post height is 160. It sits on the base (z=0). So it spans z=0 to z=160.
    for (px = [-base_x/2 + post_x/2, base_x/2 - post_x/2]) {
        translate([px, -post_y/2, 0]) {
            cube([post_x, post_y, post_z], center = false);
            
            // Square hole through post along X axis
            // Hole is 10.4 x 10.4 mm.
            // Centred at z = 150.
            // Centred on Y (post is centered on Y=0, so hole is centered on Y=0).
            // Hole goes straight through along X.
            translate([post_x/2, -post_hole_y/2, post_hole_z - post_hole_y/2]) {
                cube([post_x + 1, post_hole_y, post_hole_y + 1], center = false);
            }
        }
    }
}

// --- Part 2: Handle ---
// Square bar 10 x 10 mm and 100 mm long.
// Lying on the bed along X and centred on x = 0.
// Spans y = 55 to 65.
// Resting on z=0.
translate([-handle_x/2, handle_y_start, 0]) {
    cube([handle_x, handle_y, handle_z], center = false);
}
```
