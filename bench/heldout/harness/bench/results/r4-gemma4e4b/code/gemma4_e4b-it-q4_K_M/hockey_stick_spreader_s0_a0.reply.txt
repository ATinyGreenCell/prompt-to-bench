```openscad
// L-shaped (hockey-stick) cell spreader for plating bacteria

// --- Dimensions ---

// Handle dimensions
handle_length = 150;    // Length of the main bar (X direction)
handle_width = 6;       // Width of the handle
handle_thickness = 4;   // Thickness of the handle
handle_x_start = 0;     // X start for handle placement
handle_y_min = -3;      // Y min for handle placement
handle_y_max = 3;       // Y max for handle placement

// Spreading foot dimensions
foot_length = 40;       // Length of the spreading foot
foot_width = 5;         // Width of the spreading foot
foot_thickness = 2.5;   // Thickness of the spreading foot
angle_degrees = 120;    // Angle between handle and foot (counter-clockwise from +X)

// Boss dimensions
boss_diameter = 6;      // Diameter of the central boss
boss_height = 4;        // Height of the central boss

// Hole dimensions
hole_diameter = 4;      // Diameter of the hanging hole
hole_x_position = 143;  // X position where the hole is cut in the handle

// --- Constants ---
$fn = 64;

// --- Geometry Construction ---

module handle() {
    cube([handle_length, handle_width, handle_thickness], center = false);
}

module foot() {
    // Calculate rotation matrix for the foot placement
    angle_rad = angle_degrees * (PI / 180);
    
    // The foot's centerline starts at the origin and points 120 degrees CCW from +X.
    // We define the foot centered around its own local axes, then rotate it.
    translate([foot_length/2, 0, foot_thickness/2]) {
        rotate([0, 0, angle_degrees]) {
            cube([foot_length, foot_width, foot_thickness], center = true);
        }
    }
}

module boss() {
    cylinder(h = boss_height, r = boss_diameter / 2, center = true);
}

module handle_hole() {
    // Cut a hole through the handle at x=143, y=0.
    translate([hole_x_position, 0, -1]) { // Extend slightly below Z=0 for clean cut
        cylinder(h = handle_thickness + 2, r = hole_diameter / 2);
    }
}

// --- Assembly ---

difference() {
    // 1. Base structure (Handle and Foot)
    union() {
        // Handle: Centered at origin for simplicity of placement relative to the boss/foot junction
        translate([handle_x_start, -handle_width/2, 0]) {
            handle();
        }

        // Foot: Rotated around the origin (where it meets the handle)
        // We need to translate the foot so its center aligns with the origin after rotation.
        // Since we defined the foot module centered on its own local axes, and we want one end at the origin, 
        // we place it such that its geometric center is offset by half its length along the rotated axis.
        translate([0, 0, 0]) { // Start placement at origin
            rotate([0, 0, angle_degrees]) {
                // The foot needs to be positioned so one end (the junction) is at the origin.
                // Since we defined it centered on its own axes, we shift it back by half its length along the rotated axis.
                translate([-foot_length/2, 0, 0]) {
                    cube([foot_length, foot_width, foot_thickness], center = true);
                }
            }
        }
    }

    // 2. Subtractions (Hole and Boss removal)
    union() {
        // A. Hole in the handle
        handle_hole();

        // B. Cut out space for the boss to ensure clean union at the junction, though the boss is added later.
        // We don't strictly need this if we just add the boss on top of the union, but it ensures the base material is continuous.
    }
}

// 3. Boss addition (Placed on top of the structure)
translate([0, 0, boss_height / 2]) {
    boss();
}
```
