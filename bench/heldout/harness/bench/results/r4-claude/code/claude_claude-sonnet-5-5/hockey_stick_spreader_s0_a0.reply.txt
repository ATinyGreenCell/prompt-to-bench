```openscad
// Hockey-stick cell spreader, printed flat

$fn = 64;

// Handle
handle_len   = 150;   // handle length along +X
handle_w     = 6;     // handle width (Y)
handle_t     = 4;     // handle thickness (Z)

// Foot
foot_len     = 40;    // foot length
foot_w       = 5;     // foot width
foot_t       = 2.5;   // foot thickness (Z)
foot_angle   = 120;   // foot direction, CCW from +X

// Boss at origin
boss_d       = 6;     // boss diameter
boss_h       = 4;     // boss height

// Hanging hole
hole_d       = 4;     // hole diameter
hole_x       = 143;   // hole X position
hole_y       = 0;     // hole Y position

eps = 0.5;            // cutter overshoot

difference() {
    union() {
        // handle bar
        translate([0, -handle_w/2, 0])
            cube([handle_len, handle_w, handle_t]);

        // foot bar, centre line from origin at foot_angle
        rotate([0, 0, foot_angle])
            translate([0, -foot_w/2, 0])
                cube([foot_len, foot_w, foot_t]);

        // joining boss
        cylinder(d = boss_d, h = boss_h);
    }

    // hanging hole through the handle
    translate([hole_x, hole_y, -eps])
        cylinder(d = hole_d, h = handle_t + 2*eps);
}
```
